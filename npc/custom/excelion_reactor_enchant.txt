//==================================================
// Excelion / Reactor custom enchanter
// Supports:
// 15128 - Excelion Suit
// 20773 - Excelion Wing
// 28941 - Excelion Shield
// 22103 - Excelion Leg
//==================================================

moc_para01,38,36,4	script	Reactor Master	4_SCR_AT_ROBOTS,{
	disable_items;

	mes "[Reactor Master]";
	mes "Eu trabalho com reactors para alguns equipamentos especificos.";
	mes "Tambem posso montar reactors e vender equipamentos Excelion.";
	next;

	switch(select("Criar Encanto:Colocar Encanto:Comprar Equipamento Excelion:Cancelar")) {
	case 1:
		callsub L_RandomReactor;
		end;
	case 2:
		break;
	case 3:
		mes "[Reactor Master]";
		mes "Cada equipamento custa 5 " + getitemname(6962) + " e 10 " + getitemname(6961) + ".";
		next;
		if (select("Abrir troca:Cancelar") == 2)
			close;
		close2;
		callshop "barter_eden_excelion";
		end;
	default:
		close;
	}

	.@menu$ = "";
	.@count = 0;
	setarray .@scan_parts[0],
		EQI_ARMOR, EQI_GARMENT, EQI_HAND_L,
		EQI_HEAD_TOP, EQI_HEAD_MID, EQI_HEAD_LOW,
		EQI_HAND_R, EQI_SHOES, EQI_ACC_L, EQI_ACC_R;

	for (.@i = 0; .@i < getarraysize(.@scan_parts); ++.@i) {
		.@equip_id = getequipid(.@scan_parts[.@i]);
		if (!callsub(L_IsSupportedEquip, .@equip_id))
			continue;

		.@menu$ += getequipname(.@scan_parts[.@i]) + ":";
		.@part_list[.@count] = .@scan_parts[.@i];
		++.@count;
	}

	if (.@count == 0) {
		mes "[Reactor Master]";
		mes "Nao encontrei nenhum dos itens suportados equipado.";
		mes "Itens: 15128, 20773, 28941 e 22103.";
		close;
	}

	mes "[Reactor Master]";
	mes "Escolha o equipamento:";
	next;
	.@sel = select(.@menu$) - 1;
	.@part = .@part_list[.@sel];
	.@equip_id = getequipid(.@part);
	.@profile = callsub(L_GetProfile, .@equip_id);

	switch(select("Encantar:Remover reactors:Cancelar")) {
	case 1:
		break;
	case 2:
		callsub L_RemoveReactors, .@part;
		end;
	default:
		close;
	}

	setarray .@list[0],
		4970,	// Reactor_P_FIRE
		4971,	// Reactor_P_WATER
		4972,	// Reactor_P_GROUND
		4973,	// Reactor_P_WIND
		4974,	// Reactor_T_FIRE
		4975,	// Reactor_T_WATER
		4976,	// Reactor_T_GROUND
		4977,	// Reactor_T_WIND
		4978,	// Reactor_Cure_101
		4979,	// Reactor_Cure_102
		4980,	// Reactor_Cure_201
		4981,	// Reactor_Cure_202
		4982,	// Reactor_A_STR
		4983,	// Reactor_A_INT
		4984,	// Reactor_A_DEF
		4985,	// Reactor_A_AVOI
		4986,	// Reactor_A_ATK
		4987,	// Reactor_A_MATK
		4988,	// Reactor_A_MHP
		4989,	// Reactor_A_MSP
		4990,	// Reactor_A_FROZ
		4991;	// Reactor_A_ASPD

	setarray .@card[0],
		getequipcardid(.@part,0),
		getequipcardid(.@part,1),
		getequipcardid(.@part,2),
		getequipcardid(.@part,3);
	.@equip_refine = getequiprefinerycnt(.@part);

	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) ||
		callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@equip_refine)) {
		mes "[Reactor Master]";
		mes "O equipamento mudou durante a conversa.";
		close;
	}

	for (.@i = 1; .@i <= 3; ++.@i) {
		if (.@card[.@i] > 0 && !callsub(L_IsReactorEnchant, .@card[.@i])) {
			mes "[Reactor Master]";
			mes "Esse item ja possui outro tipo de encantamento nos slots ocultos.";
			mes "Remova-os antes de usar este sistema.";
			close;
		}
	}

	.@menu$ = "";
	.@menu_count = 0;
	for (.@i = 0; .@i < getarraysize(.@list); ++.@i) {
		.@req = .@list[.@i];
		.@enchant_id = .@list[.@i];
		.@limit = callsub(L_GetEnchantLimit, .@profile, .@enchant_id);
		if (.@limit < 1)
			continue;

		.@first_slot_only = callsub(L_IsFirstSlotOnlyEnchant, .@enchant_id);
		.@same = 0;
		for (.@j = 1; .@j <= 3; ++.@j)
			if (.@card[.@j] == .@enchant_id)
				++.@same;

		if (.@first_slot_only && .@card[1] != 0)
			.@state$ = "primeiro slot ocupado";
		else if (.@same >= .@limit)
			.@state$ = "limite atingido";
		else if (countitem(.@req) < 1)
			.@state$ = "sem material";
		else
			.@state$ = "";

		if (.@state$ != "")
			.@menu$ += "^777777" + getitemname(.@req) + " [" + .@state$ + "]^000000:";
		else
			.@menu$ += getitemname(.@req) + ":";

		.@req_list[.@menu_count] = .@req;
		.@enchant_list[.@menu_count] = .@enchant_id;
		++.@menu_count;
	}

	if (.@menu_count == 0) {
		mes "[Reactor Master]";
		mes "Esse item nao aceita nenhum reactor por este sistema.";
		close;
	}

	mes "[Reactor Master]";
	mes "Escolha o reactor:";
	next;
	.@sel = select(.@menu$ + "Cancelar") - 1;
	if (.@sel >= .@menu_count)
		close;

	.@req = .@req_list[.@sel];
	.@enchant_id = .@enchant_list[.@sel];
	.@limit = callsub(L_GetEnchantLimit, .@profile, .@enchant_id);
	.@first_slot_only = callsub(L_IsFirstSlotOnlyEnchant, .@enchant_id);
	.@same = 0;
	for (.@i = 1; .@i <= 3; ++.@i)
		if (.@card[.@i] == .@enchant_id)
			++.@same;

	if (countitem(.@req) < 1) {
		mes "[Reactor Master]";
		mes "Voce nao possui o material necessario.";
		close;
	}
	if (.@limit < 1) {
		mes "[Reactor Master]";
		mes "Esse reactor nao e compativel com o item selecionado.";
		close;
	}
	if (.@first_slot_only && .@card[1] != 0) {
		mes "[Reactor Master]";
		mes "Esse reactor precisa entrar primeiro no slot oculto inicial.";
		close;
	}
	if (.@same >= .@limit) {
		mes "[Reactor Master]";
		mes "Esse reactor ja atingiu o limite permitido nesse item.";
		close;
	}

	if (.@card[1] == 0)
		.@target_slot = 1;
	else if (.@card[2] == 0)
		.@target_slot = 2;
	else if (.@card[3] == 0)
		.@target_slot = 3;
	else {
		mes "[Reactor Master]";
		mes "Os tres slots ocultos ja estao ocupados.";
		close;
	}
	if (.@first_slot_only && .@target_slot != 1) {
		mes "[Reactor Master]";
		mes "Esse reactor e exclusivo do slot 2.";
		close;
	}

	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) ||
		callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@equip_refine))
		close;

	delitem .@req, 1;
	setitemcard(.@part, .@target_slot, .@enchant_id);
	specialeffect2 EF_REPAIRWEAPON;

	mes "[Reactor Master]";
	mes "Encantamento aplicado em " + getequipname(.@part) + ".";
	close;

L_RemoveReactors:
	.@part = getarg(0);
	setarray .@card[0],
		getequipcardid(.@part,0),
		getequipcardid(.@part,1),
		getequipcardid(.@part,2),
		getequipcardid(.@part,3);

	.@menu$ = "";
	.@count = 0;
	for (.@i = 1; .@i <= 3; ++.@i) {
		if (!callsub(L_IsReactorEnchant, .@card[.@i]))
			continue;
		.@menu$ += "Slot " + (.@i + 1) + " (" + getitemname(.@card[.@i]) + "):";
		.@slot_map[.@count] = .@i;
		++.@count;
	}

	if (.@count == 0) {
		mes "[Reactor Master]";
		mes "Esse equipamento nao possui reactors para remover.";
		close;
	}

	mes "[Reactor Master]";
	mes "Escolha o que deseja remover:";
	next;
	.@sel = select(.@menu$ + "Remover todos:Cancelar") - 1;
	if (.@sel == .@count) {
		for (.@i = 1; .@i <= 3; ++.@i)
			if (callsub(L_IsReactorEnchant, .@card[.@i]))
				setitemcard(.@part, .@i, 0);

		mes "[Reactor Master]";
		mes "Todos os reactors foram removidos.";
		close;
	}
	if (.@sel > .@count)
		close;

	setitemcard(.@part, .@slot_map[.@sel], 0);
	mes "[Reactor Master]";
	mes "Reactor removido com sucesso.";
	close;

L_IsSupportedEquip:
	.@id = getarg(0);
	if (.@id == 15128 || .@id == 20773 || .@id == 28941 || .@id == 22103)
		return 1;
	return 0;

L_IsReactorEnchant:
	.@id = getarg(0);
	if (.@id >= 4970 && .@id <= 4991)
		return 1;
	return 0;

L_IsFirstSlotOnlyEnchant:
	.@id = getarg(0);
	switch (.@id) {
	case 4970:
	case 4971:
	case 4972:
	case 4973:
	case 4982:
	case 4983:
	case 4984:
	case 4985:
	case 4986:
	case 4987:
	case 4988:
	case 4989:
	case 4991:
		return 1;
	default:
		return 0;
	}

L_GetProfile:
	.@id = getarg(0);
	switch (.@id) {
	case 15128:
		return 1; // armor profile
	case 20773:
		return 2; // garment profile
	case 28941:
	case 22103:
		return 3; // generic reactor profile
	default:
		return 0;
	}

L_GetEnchantLimit:
	.@profile = getarg(0);
	.@id = getarg(1);

	if (.@profile == 1) {
		switch (.@id) {
		case 4970:
		case 4971:
		case 4972:
		case 4973:
		case 4982:
		case 4983:
		case 4984:
		case 4985:
		case 4986:
		case 4987:
		case 4988:
		case 4989:
		case 4990:
		case 4991:
			return 3;
		case 4978:
		case 4979:
		case 4980:
		case 4981:
			return 3;
		default:
			return 0;
		}
	}

	if (.@profile == 2) {
		switch (.@id) {
		case 4974:
		case 4975:
		case 4976:
		case 4977:
		case 4978:
		case 4979:
		case 4980:
		case 4981:
		case 4984:
		case 4986:
		case 4987:
		case 4988:
		case 4989:
			return 3;
		case 4985:
		case 4991:
			return 3;
		default:
			return 0;
		}
	}

	if (.@profile == 3) {
		switch (.@id) {
		case 4982:
		case 4983:
		case 4984:
		case 4985:
		case 4986:
		case 4987:
		case 4988:
		case 4989:
		case 4991:
			return 3;
		case 4970:
		case 4971:
		case 4972:
		case 4973:
		case 4974:
		case 4975:
		case 4976:
		case 4977:
		case 4978:
		case 4979:
		case 4980:
		case 4981:
		case 4990:
			return 3;
		default:
			return 0;
		}
	}

	return 0;

L_RandomReactor:
	mes "[Reactor Master]";
	mes "Com 5 " + getitemname(6962) + " eu monto um reactor aleatorio.";
	mes "A chance total de sair um reactor e de 60%.";
	next;

	if (countitem(6962) < 5) {
		mes "[Reactor Master]";
		mes "Voce precisa de 5 " + getitemname(6962) + ".";
		close;
	}

	if (select("Montar reactor:Cancelar") == 2)
		close;

	delitem 6962,5;
	.@rnd = rand(1,10000);
	.@fail = 0;

	if (.@rnd <= 150) .@item = 4970;			// 1.50% E-Fogo
	else if (.@rnd <= 300) .@item = 4971;		// 1.50% E-Agua
	else if (.@rnd <= 450) .@item = 4972;		// 1.50% E-Terra
	else if (.@rnd <= 600) .@item = 4973;		// 1.50% E-Vento
	else if (.@rnd <= 700) .@item = 4990;		// 1.00% I-Gelo
	else if (.@rnd <= 750) .@item = 4982;		// 0.50% A-FOR
	else if (.@rnd <= 800) .@item = 4983;		// 0.50% A-INT
	else if (.@rnd <= 1200) .@item = 4984;		// 4.00% A-DEF
	else if (.@rnd <= 1350) .@item = 4986;		// 1.50% A-ATQ
	else if (.@rnd <= 1500) .@item = 4987;		// 1.50% A-ATQM
	else if (.@rnd <= 1900) .@item = 4988;		// 4.00% A-HP
	else if (.@rnd <= 2300) .@item = 4989;		// 4.00% A-SP
	else if (.@rnd <= 2700) .@item = 4985;		// 4.00% A-ESQV
	else if (.@rnd <= 3000) .@item = 4991;		// 3.00% A-VdA
	else if (.@rnd <= 3400) .@item = 4974;		// 4.00% R-Fogo
	else if (.@rnd <= 3800) .@item = 4975;		// 4.00% R-Agua
	else if (.@rnd <= 4200) .@item = 4976;		// 4.00% R-Terra
	else if (.@rnd <= 4600) .@item = 4977;		// 4.00% R-Vento
	else if (.@rnd <= 5000) .@item = 4978;		// 4.00% C-Corpo
	else if (.@rnd <= 5200) .@item = 4979;		// 2.00% C-Alma
	else if (.@rnd <= 5700) .@item = 4980;		// 5.00% C-Fisico
	else if (.@rnd <= 6000) .@item = 4981;		// 3.00% C-Mental
	else {
		.@fail = 1;
		.@rnd2 = rand(1,13000);
		if (.@rnd2 <= 1000) .@item = 999;		// Steel
		else if (.@rnd2 <= 3000) .@item = 998;	// Iron
		else if (.@rnd2 <= 8000) .@item = 1002;	// Iron_Ore
		else .@item = 7054;					// Brigan
	}

	getitem .@item,1;
	specialeffect EF_SPELLBREAKER;

	mes "[Reactor Master]";
	if (.@fail)
		mes "A montagem falhou. Pelo menos sobrou " + getitemname(.@item) + ".";
	else
		mes "Voce recebeu " + getitemname(.@item) + ".";
	close;

}
