//----------------------------------------------
// Sub-NPC Stylist (versao gratuita)
//----------------------------------------------
S_FreeStylist:
	set .@max_cloth_color, getbattleflag("max_cloth_color");
	if (.@max_cloth_color < 1)
		set .@max_cloth_color, 1;

	setarray .@Styles[1],
		.@max_cloth_color,
		getbattleflag("max_hair_style"),
		getbattleflag("max_hair_color");
	setarray .@Look[1],
		LOOK_CLOTHES_COLOR,
		LOOK_HAIR,
		LOOK_HAIR_COLOR;
	set .@s, select(" ~ Cor da Roupa: ~ Estilo de Cabelo: ~ Cor do Cabelo");
	if (.@s == 1 && (checkriding() || checkdragon() || checkwug() || checkmadogear()) && .@Styles[.@s] > 8) {
		set .@Styles[.@s], 8;
		message strcharinfo(0),"Cores de roupa acima de #8 foram ocultadas enquanto voce estiver montado.";
	}
	set .@Revert, getlook(.@Look[.@s]);
	set .@Style,1;
	set .@Dir,1;

	while(1) {
		if (.@Style < 1 || .@Style > .@Styles[.@s])
			set .@Style, 1;

		set .@Style, callsub(S_ApplyValidStyle, .@Look[.@s], .@Style, .@Styles[.@s], .@Dir);
		message strcharinfo(0),"Este estilo e #"+.@Style+".";

		set .@menu$, " ~ Proximo (^0055FF"+((.@Style!=.@Styles[.@s])?.@Style+1:1)+"^000000): ~ Anterior (^0055FF"+((.@Style!=1)?.@Style-1:.@Styles[.@s])+"^000000): ~ Pular para...: ~ Voltar para a Original (^0055FF"+.@Revert+"^000000): ~ Sair";
		
		switch(select(.@menu$)) {
		case 1:
			set .@Dir, 1;
			set .@Style, .@Style + 1;
			if (.@Style > .@Styles[.@s])
				set .@Style, 1;
			break;
		case 2:
			set .@Dir, -1;
			set .@Style, .@Style - 1;
			if (.@Style < 1)
				set .@Style, .@Styles[.@s];
			break;
		case 3:
			set .@Dir, 1;
			message strcharinfo(0),"Escolha um estilo entre 1 - "+.@Styles[.@s]+"."; input .@Style,1,.@Styles[.@s];
			if (!.@Style)
				set .@Style, rand(1,.@Styles[.@s]);
			break;
		case 4:
			set .@Dir, 1;
			set .@Style, callsub(S_ApplyValidStyle, .@Look[.@s], .@Revert, .@Styles[.@s], 1);
			break;
		case 5:
			close;
		}
	}

S_ApplyValidStyle:
	set .@look, getarg(0);
	set .@style, getarg(1);
	set .@max, getarg(2);
	set .@dir, getarg(3);

	for (set .@i, 0; .@i < .@max; set .@i, .@i + 1) {
		if (.@style < 1)
			set .@style, .@max;
		if (.@style > .@max)
			set .@style, 1;

		setlook .@look, .@style;
		if (getlook(.@look) == .@style)
			return .@style;

		if (.@dir < 0)
			set .@style, .@style - 1;
		else
			set .@style, .@style + 1;
	}
	return getlook(.@look);
	end;
}
