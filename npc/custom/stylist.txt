-	script	Estilista	122,{
	set .@max_cloth_color, getbattleflag("max_cloth_color");
	if (.@max_cloth_color > 4)
		set .@max_cloth_color, 4;

	setarray .@Styles[1],
		.@max_cloth_color,
		getbattleflag("max_hair_style"),
		getbattleflag("max_hair_color");
	setarray .@Look[1],
		LOOK_CLOTHES_COLOR,
		LOOK_HAIR,
		LOOK_HAIR_COLOR;
	set .@s, select(" ~ Cor da Roupa: ~ Estilo de Cabelo: ~ Cor do Cabelo");
	set .@Revert, getlook(.@Look[.@s]);
	set .@Style,1;

	while(1) {
		if (.@Style < 1 || .@Style > .@Styles[.@s])
			set .@Style, 1;

		setlook .@Look[.@s], .@Style;
		message strcharinfo(0),"Este estilo é #"+.@Style+".";

		set .@menu$, " ~ Próximo (^0055FF"+((.@Style!=.@Styles[.@s])?.@Style+1:1)+"^000000): ~ Anterior (^0055FF"+((.@Style!=1)?.@Style-1:.@Styles[.@s])+"^000000): ~ Pular para...: ~ Voltar para a Original (^0055FF"+.@Revert+"^000000)";
		
		switch(select(.@menu$)) {
		case 1:
			set .@Style, .@Style + 1;
			if (.@Style > .@Styles[.@s])
				set .@Style, 1;
			break;
		case 2:
			set .@Style, .@Style - 1;
			if (.@Style < 1)
				set .@Style, .@Styles[.@s];
			break;
		case 3:
			message strcharinfo(0),"Escolha um estilo entre 1 - "+.@Styles[.@s]+".";
			input .@Style,1,.@Styles[.@s];
			if (!.@Style)
				set .@Style, rand(1,.@Styles[.@s]);
			break;
		case 4:
			set .@Style, .@Revert;
			setlook .@Look[.@s], .@Revert;
			break;
		}
	}
}
