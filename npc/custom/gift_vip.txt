-	script	Receber VIP	4_M_MANAGER,{
	mes "[VIP]";
	
	// Ja recebeu?
	if (#Reward_Vip) {
		mes "Voce ja recebeu o ^00FF00VIP^000000 nesta conta.";
		mes "Cada conta pode receber apenas uma vez.";
		close;
	}

	// Checar peso antes de entregar
	if (!checkweight(9616,1)) {
		mes "Voce esta com peso ou itens demais no inventario.";
		mes "Guarde alguns itens e tente novamente.";
		close;
	}

	// Entregar VIP
	getitem 9616,1;
	set #Reward_Vip, 1;

	mes "^00FF00VIP entregue com sucesso!^000000";
	mes "Aproveite seu beneficio.";
	close;

OnPCLoginEvent:
	if (#Reward_Vip)
		end;

	if (!checkweight(9616,1)) {
		dispbottom "[VIP] Seu inventario esta cheio. Libere peso/espaco e relogue para receber seu VIP.";
		end;
	}

	getitem 9616,1;
	set #Reward_Vip, 1;
	dispbottom "[VIP] Beneficio VIP entregue automaticamente. Aproveite!";
	end;
}
