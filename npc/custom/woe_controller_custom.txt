prontera,156,197,4	script	Informacoes de Guerra	111,{
	mes "[WoE - Informacoes]";
	.Castle$ = "prtg_cas03";

    if (agitcheck())
        mes "Status: ^00FF00ATIVA^000000";
    else
        mes "Status: ^FF0000INATIVA^000000";

    mes "Castelo: ^0055FF"+getcastlename(.Castle$)+"^000000";

    next;
    switch(select("~ Ir ao Castelo:~ Ver Dono:~ Sair")) {

		case 1:
            if (getcharid(2) == 0) {
                mes "Voce precisa estar em uma guild para entrar no castelo.";
                close;
            }
            
			warp .Castle$, 164, 174;
			close;

		case 2:
			mes "[Vencedor da Semana]";
			if ($WOE_WEEK_WIN_GID > 0 && $WOE_WEEK_WIN_GNAME$ != "") {
				mes "A guild vencedora da ultima WoE foi ^00AAFF"+$WOE_WEEK_WIN_GNAME$+"^000000.";
				if ($WOE_WEEK_CASTLE$ != "")
					mes "Castelo conquistado: ^777777"+getcastlename($WOE_WEEK_CASTLE$)+"^000000.";
			} else
				mes "Ainda nao ha uma guild vencedora registrada para esta semana.";
			next;
			doevent "WOE_CONTROL#Custom::OnMenu";
			end;

		case 3:
			close;
    }
OnAgitStart:
    while(agitcheck() || agitcheck2() || agitcheck3()) {
        sleep 500;
        specialeffect EF_BEGINSPELL6;
    }
	end;
}

//============================================================
//   SCRIPT PRINCIPAL
//============================================================
-	script	WOE_CONTROL#Custom	-1,{

//------------------------------------------------------------
//  CONFIGURACAO GERAL
//------------------------------------------------------------
OnInit:
    // Nome do castelo (1 castelo)
    .Castle$ = "prtg_cas03";
    donpcevent "Bandeira do Castelo#prontera::OnRefresh";
	setmapflag .Castle$, mf_loadevent;
	setmapflag .Castle$, mf_gvg_castle;
    MapRespawnGuildID .Castle$, GetCastleData(.Castle$,CD_GUILD_ID), 6;

    // Warp para o castelo
    .MapX  = 17;

    .MapY  = 221;

    end;

OnAgitInit:
    if (gettime(DT_DAYOFWEEK) != SUNDAY) end;

    if (gettime(DT_HOUR) == 22 && gettime(DT_MINUTE) >= 30) {
        if (!$WOE_CUSTOM_ACTIVE && !agitcheck())
            donpcevent strnpcinfo(3)+"::OnStartWoE";
        end;
    }

    if ($WOE_CUSTOM_ACTIVE && gettime(DT_HOUR) >= 23)
        donpcevent strnpcinfo(3)+"::OnEndWoE";
    end;

OnPCLoadMapEvent:
    if (strcharinfo(3) != .Castle$) end;
    if (!agitcheck()) end;

    deltimer strnpcinfo(3)+"::OnKvmPointMinute";
    addtimer 60000, strnpcinfo(3)+"::OnKvmPointMinute";

    .@gid = getcharid(2);
    .@cid = getcharid(0);
    .@aid = getcharid(3);

    if (.@gid <= 0) end;

    woe_streak_combo = 0;
    woe_streak_total = 0;
    woe_lastkill_time = 0;

    if (inarray($WOE_ENTRY_CID, .@cid) >= 0) end;

    $WOE_ENTRY_CID[getarraysize($WOE_ENTRY_CID)] = .@cid;
    $WOE_ENTRY_AID[getarraysize($WOE_ENTRY_AID)] = .@aid;
    $WOE_ENTRY_GID[getarraysize($WOE_ENTRY_GID)] = .@gid;
    $WOE_ENTRY_COUNT[.@gid] = $WOE_ENTRY_COUNT[.@gid] + 1;
    if (inarray($WOE_ENTRY_LIST, .@gid) < 0)
        $WOE_ENTRY_LIST[getarraysize($WOE_ENTRY_LIST)] = .@gid;
    end;

OnKvmPointMinute:
    if (strcharinfo(3) != .Castle$) end;
    if (!agitcheck()) end;

    kvm_point = kvm_point + 1;
    dispbottom "[WoE] Voce ganhou 1 ponto KVM por permanecer no castelo. Total: " + kvm_point + ".";

    addtimer 60000, strnpcinfo(3)+"::OnKvmPointMinute";
    end;

//============================================================
//   CONTROLE DE HORARIO (a cada minuto)
//============================================================
OnClock1800:
OnClock1830:
OnClock1900:
OnClock1930:
OnClock2000:
OnClock2030:
OnClock2100:
OnClock2130:
    if (gettime(DT_DAYOFWEEK) != SUNDAY) end;
    if ($WOE_CUSTOM_ACTIVE || agitcheck()) end;

    .@minutes_left = (22 - gettime(DT_HOUR)) * 60 - gettime(DT_MINUTE);
    if (.@minutes_left <= 0) end;

    if (.@minutes_left >= 60)
        .@time_left$ = (.@minutes_left / 60) + " hora(s)";
    else
        .@time_left$ = .@minutes_left + " minuto(s)";

    announce "[WoE] A Guerra do Emperium acontece hoje as 22:00! Prepare sua guild.", bc_all;
    end;

OnClock2200:

    .@day  = gettime(DT_DAYOFWEEK);

    // Inicio
    .woe_day   = SUNDAY;   // Domingo
    if (.@day != .woe_day) end;
    if ($WOE_CUSTOM_ACTIVE || agitcheck()) end;

OnStartWoE:

    // Resetar dono do castelo ao carregar script
	setcastledata(.Castle$, 1, 0); // 0 = sem guild
	setcastledata(.Castle$, 2, 0); // Remove economia
	setcastledata(.Castle$, 3, 0); // Remove defesa
    deletearray $WOE_ENTRY_CID;
    deletearray $WOE_ENTRY_AID;
    deletearray $WOE_ENTRY_GID;
    deletearray $WOE_ENTRY_LIST;
    deletearray $WOE_ENTRY_COUNT;
    deletearray $WOE_TIME;
    deletearray $WOE_LIST;
    $WOE_WIN_TIME = 0;

    $WOE_CUSTOM_ACTIVE = 1;
    $WOE_CUSTOM_ENDED = 0;
    $WOE_END_SUMMARY_ANNOUNCED = 0;
    callfunc("ClearCastle", .Castle$, .@gid);

    AgitStart;
    if (agitcheck())
        GvgOn .Castle$;
        sleep 200;
        donpcevent "WOE_CONTROL#Custom::OnEVENT";
    end;

    // Fim
OnClock2300:
    .@day = gettime(DT_DAYOFWEEK);
    .woe_day = SUNDAY; // Domingo
    if (.@day != .woe_day) end;

OnEndWoE:
    if ($WOE_CUSTOM_ENDED) end;
    if (!agitcheck() && !$WOE_CUSTOM_ACTIVE) end;

    $WOE_CUSTOM_ENDED = 1;
    $WOE_CUSTOM_ACTIVE = 0;

    .@winner = getcastledata(.Castle$, 1);
    .@best_gid = 0;
    .@best_time = 0;
    for (.@i = 0; .@i < getarraysize($WOE_LIST); .@i++) {
        .@gid = $WOE_LIST[.@i];
        if ($WOE_TIME[.@gid] > .@best_time) {
            .@best_time = $WOE_TIME[.@gid];
            .@best_gid = .@gid;
        }
    }

    announce "[WoE] A Guerra do Emperium terminou!", bc_all|bc_woe;

    if (.@winner > 0)
        announce "[WoE] A guilda "+getguildname(.@winner)+" conquistou o castelo "+getcastlename(.Castle$)+"!", bc_all|bc_woe;
    else
        announce "[WoE] Nenhuma guild conquistou o castelo "+getcastlename(.Castle$)+".", bc_all|bc_woe;

    if (.@best_gid > 0 && .@best_time > 0)
        announce "[WoE] A guilda "+getguildname(.@best_gid)+" ficou mais tempo com o castelo ("+.@best_time+" minutos)!", bc_all|bc_woe;
    else
        announce "[WoE] Nenhuma guild pontuou tempo de dominio suficiente.", bc_all|bc_woe;

    $WOE_END_SUMMARY_ANNOUNCED = 1;

    killmonster .Castle$, strnpcinfo(3)+"::OnEmpDie";

    stopnpctimer;

    AgitEnd;
    sleep 100; // aguarda atualizacao do estado
    donpcevent strnpcinfo(3) + "::OnReward";

    end;
// ======================================================================
// SPAWN DO EMPERIUM
// ======================================================================
OnEVENT:
    // remove o anterior	
    killmonster .Castle$, strnpcinfo(3)+"::OnEmpDie";

    monster .Castle$, .MapX, .MapY, "Emperium", 1905, 1, strnpcinfo(3)+"::OnEmpDie";

    announce "[WoE] A Guerra do Emperium comecou!", bc_all|bc_woe;
    end;

// ======================================================================
// EVENTO DE MORTE DO EMPERIUM
// ======================================================================
OnEmpDie:
    .@gid = getcharid(2);
    .@gname$ = getguildname(.@gid);

	setcastledata .Castle$, 1, .@gid;
    initnpctimer;

    announce "[WoE] "+.@gname$+" quebrou o Emperium em "+.Castle$+"!", bc_all|bc_woe;

	sleep 1000;
	// Expulsar todos que nao sao da guild vencedora
	callfunc("ClearCastle", .Castle$, .@gid);

    killmonster .Castle$, strnpcinfo(3)+"::OnEmpDie";
    sleep 7000;

    guardian .Castle$, 17, 221,
        "Emperium", 1905,
        strnpcinfo(3)+"::OnEmpDie", 0;

    end;

// A cada minuto
OnTimer60000:

	// WoE ativa?
	if (!agitcheck()) end;

	.@gid = getcastledata(.Castle$, 1); // guild dona atual
    debugmes "Pontos adicionado a guild " + getguildname(.@gid) + " Total: " + $WOE_TIME[.@gid];

	if (.@gid > 0) {
		// Soma 1 minuto ao dono atual
		$WOE_TIME[.@gid] = $WOE_TIME[.@gid] + 1;

		// Se ainda nao existe na lista de participantes, adiciona
		if (inarray($WOE_LIST, .@gid) < 0) {
			$WOE_LIST[getarraysize($WOE_LIST)] = .@gid;
		}
	}

	initnpctimer;

	end;

//============================================================
//   KILLS / STREAKS DA WOE
//============================================================
OnPCKillEvent:
    if (strcharinfo(3) != .Castle$) end;
    if (!agitcheck()) end;

    .@killer_aid = getcharid(3);
    .@victim_aid = killedrid;
    .@now = gettimetick(2);

    if (.@killer_aid == .@victim_aid) end;

    .@killer$ = strcharinfo(0);
    .@victim$ = rid2name(.@victim_aid);
    if (.@victim$ == "")
        .@victim$ = "um jogador";

    announce "[WoE] "+.@killer$+" matou "+.@victim$+"!", bc_map, "0xff0000";

    //---------------------------------
    // DOUBLE KILL / TRIPLE KILL
    //---------------------------------
    if ((.@now - woe_lastkill_time) <= 10)
        woe_streak_combo++;
    else
        woe_streak_combo = 1;

    //---------------------------------
    // STREAK CONTINUO SEM MORRER
    //---------------------------------
    woe_streak_total++;

    //----------------------------------------------
    // KILL COMBO (varias kills em pouco tempo)
    //----------------------------------------------
    .@played_combo = 0;
    if (woe_streak_combo == 2) {
        announce "[WoE] "+.@killer$+" fez DOUBLE KILL!", bc_map, "0xff0000";
        soundeffectall "kills/double_kill.wav", 0, strcharinfo(3);
        .@played_combo = 1;
    }
    else if (woe_streak_combo == 3) {
        announce "[WoE] "+.@killer$+" fez TRIPLE KILL!", bc_map, "0xff0000";
        soundeffectall "kills/triple_kill.wav", 0, strcharinfo(3);
        .@played_combo = 1;
    }
    else if (woe_streak_combo == 4) {
        announce "[WoE] "+.@killer$+" fez MANIAC!!", bc_map, "0xff0000";
        soundeffectall "kills/maniac.wav", 0, strcharinfo(3);
        .@played_combo = 1;
    }
    else if (woe_streak_combo >= 5) {
        announce "[WoE] "+.@killer$+" fez SAVAGE!!!", bc_map, "0xff0000";
        soundeffectall "kills/savage.wav", 0, strcharinfo(3);
        .@played_combo = 1;
    }

    //----------------------------------------------
    // STREAKS CONTINUOS (sem morrer)
    //----------------------------------------------
    if (!.@played_combo) {
        switch (woe_streak_total) {

            case 3:
                announce "[WoE] "+.@killer$+" esta em KILLING SPREE!", bc_map, "0xff0000";
                soundeffectall "kills/killing_spree.wav", 0, strcharinfo(3);
                break;

            case 4:
                announce "[WoE] "+.@killer$+" esta em MEGA KILL!", bc_map, "0xff0000";
                soundeffectall "kills/mega_kill.wav", 0, strcharinfo(3);
                break;

            case 5:
                announce "[WoE] "+.@killer$+" esta UNSTOPPABLE!!", bc_map, "0xff0000";
                soundeffectall "kills/unstoppable.wav", 0, strcharinfo(3);
                break;

            case 6:
                announce "[WoE] "+.@killer$+" fez MONSTER KILL!!!", bc_map, "0xff0000";
                soundeffectall "kills/monster_kill.wav", 0, strcharinfo(3);
                break;

            case 7:
                announce "[WoE] "+.@killer$+" esta GODLIKE!!!", bc_map, "0xff0000";
                soundeffectall "kills/godlike.wav", 0, strcharinfo(3);
                break;

            case 8:
                announce "[WoE] "+.@killer$+" esta LEGENDARY!!!!", bc_map, "0xff0000";
                soundeffectall "kills/legendary.wav", 0, strcharinfo(3);
                break;

            default:
                if (woe_streak_total > 8) {
                    announce "[WoE] "+.@killer$+" esta LEGENDARY!!!!", bc_map, "0xff0000";
                    soundeffectall "kills/legendary.wav", 0, strcharinfo(3);
                }
                break;
        }
    }

    woe_lastkill_time = .@now;
    callfunc("F_BattlePassRegisterWeekly", "woe");
    end;

OnPCDieEvent:
    if (strcharinfo(3) != .Castle$) end;
    if (!agitcheck()) end;

    if (woe_streak_total >= 3) {
        announce "[WoE] "+strcharinfo(0)+" foi ELIMINADO APOS UMA SEQUENCIA DE "+woe_streak_total+" KILLS!!", bc_map, "0xff0000";
        soundeffectall "kills/shutdown.wav", 0, strcharinfo(3);
    }

    woe_streak_total = 0;
    woe_streak_combo = 0;
    woe_lastkill_time = 0;
    callfunc("F_BattlePassRegisterWeekly", "woe");
    end;
//============================================================
//   RECOMPENSA SIMPLIFICADA
//============================================================
OnReward:

    .@winner = getcastledata(.Castle$, 1);
    .@sender$ = "Sistema de WoE";
    setarray .@refine[0], 0;
    setarray .@bound[0], 0;
    setarray .@card0[0], 0;
    setarray .@card1[0], 0;
    setarray .@card2[0], 0;
    setarray .@card3[0], 0;

    if (.@winner) {
        $WOE_WEEK_WIN_GID = .@winner;
        $WOE_WEEK_WIN_GNAME$ = getguildname(.@winner);
        $WOE_WEEK_CASTLE$ = .Castle$;
        donpcevent "Bandeira do Castelo#prontera::OnRefresh";
        debugmes "Vencedor da WOE: " + .@winner;
        if (!$WOE_END_SUMMARY_ANNOUNCED)
            announce "[WoE] A guilda "+getguildname(.@winner)+" Conquistou o castelo ("+.Castle$+")!", bc_all|bc_woe;
    } else {
        if (!$WOE_END_SUMMARY_ANNOUNCED)
            announce "[WoE] A WoE terminou sem guild dona do castelo.", bc_all|bc_woe;
    }

    //-----------------------------------------
    // 2. PREMIO DE DEFESA (MAIOR PONTUACAO)
    //-----------------------------------------
    .@best_gid = 0;
    .@best_time = 0;
    .@second_gid = 0;
    .@second_time = 0;

    for (.@i = 0; .@i < getarraysize($WOE_LIST); .@i++) {
        .@gid = $WOE_LIST[.@i];

        if ($WOE_TIME[.@gid] > .@best_time) {
            .@second_time = .@best_time;
            .@second_gid = .@best_gid;
            .@best_time = $WOE_TIME[.@gid];
            .@best_gid  = .@gid;
        } else if ($WOE_TIME[.@gid] > .@second_time) {
            .@second_time = $WOE_TIME[.@gid];
            .@second_gid  = .@gid;
        }
    }

    $WOE_WIN_TIME = .@best_gid;

    if (.@best_gid > 0 && .@best_time > 0) {
        if (!$WOE_END_SUMMARY_ANNOUNCED)
            announce "[WoE] A guilda "+getguildname(.@best_gid)+" dominou o castelo por mais tempo ("+.@best_time+" minutos)!", bc_all|bc_woe;
    } else {
        if (!$WOE_END_SUMMARY_ANNOUNCED)
            announce "[WoE] Nenhuma guilda pontuou tempo de dominio suficiente para premio de defesa!", bc_all|bc_woe;
    }

    //-----------------------------------------
    // 3. PREMIO DE SEGUNDO LUGAR
    //-----------------------------------------
    if (.@second_gid == .@winner) {
        .@second_gid = 0;
        .@second_time = 0;
        for (.@i = 0; .@i < getarraysize($WOE_LIST); .@i++) {
            .@gid = $WOE_LIST[.@i];
            if (.@gid == .@winner)
                continue;
            if ($WOE_TIME[.@gid] > .@second_time) {
                .@second_time = $WOE_TIME[.@gid];
                .@second_gid = .@gid;
            }
        }
    }

    if (.@second_gid > 0 && .@second_time > 0) {
        announce "[WoE] A guilda "+getguildname(.@second_gid)+" ficou em segundo lugar na WoE!", bc_all|bc_woe;
    }

    .@winner_coin_reward = 200;
    .@best_coin_reward = 200;
    .@second_coin_reward = 100;

    //-----------------------------------------
    // 1. PREMIOS PARA TODOS QUE PARTICIPARAM
    //-----------------------------------------
    for (.@i = 0; .@i < getarraysize($WOE_ENTRY_CID); .@i++) {
        .@participant_gid = $WOE_ENTRY_GID[.@i];

        .@title$ = "Premio de Participacao (WoE)";
        .@body$  = "Voce participou da WoE. Esta caixa contem os premios de participacao.";
        setarray .@item_id[0], 14002;
        setarray .@amount[0], 1;

        mail $WOE_ENTRY_CID[.@i], .@sender$, .@title$, .@body$, 0,
            .@item_id, .@amount, .@refine, .@bound,
            .@card0, .@card1, .@card2, .@card3;

        if (.@participant_gid == .@winner && .@winner > 0) {
            .@title$ = "Premio de Conquista (WoE)";
            .@body$  = "Sua guild conquistou o castelo na WoE. Este premio foi enviado diretamente para voce.";
            setarray .@item_id[0], 9647;
            setarray .@amount[0], .@winner_coin_reward;

            mail $WOE_ENTRY_CID[.@i], .@sender$, .@title$, .@body$, 0,
                .@item_id, .@amount, .@refine, .@bound,
                .@card0, .@card1, .@card2, .@card3;
        }

        if (.@participant_gid == .@best_gid && .@best_gid > 0 && .@best_time > 0) {
            .@title$ = "Premio de Defesa (WoE)";
            .@body$  = "Sua guild manteve o dominio por mais tempo na WoE. Este premio foi enviado diretamente para voce.";
            setarray .@item_id[0], 9646;
            setarray .@amount[0], .@best_coin_reward;

            mail $WOE_ENTRY_CID[.@i], .@sender$, .@title$, .@body$, 0,
                .@item_id, .@amount, .@refine, .@bound,
                .@card0, .@card1, .@card2, .@card3;
        }

        if (.@participant_gid == .@second_gid && .@second_gid > 0 && .@second_time > 0) {
            .@title$ = "Premio de Segundo Lugar (WoE)";
            .@body$  = "Sua guild ficou em segundo lugar no tempo de dominio da WoE. Este premio foi enviado diretamente para voce.";
            setarray .@item_id[0], 9646;
            setarray .@amount[0], .@second_coin_reward;

            mail $WOE_ENTRY_CID[.@i], .@sender$, .@title$, .@body$, 0,
                .@item_id, .@amount, .@refine, .@bound,
                .@card0, .@card1, .@card2, .@card3;
        }

        if (attachrid($WOE_ENTRY_AID[.@i])) {
            dispbottom "[WoE] Premio de participacao: 1x Veteran Siege Supply Box foi enviado ao seu email.";
            dispbottom "[WoE] A caixa contem: Ticket de Minerios x10.";

            if (.@participant_gid == .@winner && .@winner_coin_reward > 0)
                dispbottom "[WoE] Premio de conquista: "+.@winner_coin_reward+"x Moeda Cheffenia (Hard) foi enviado ao seu email.";
            if (.@participant_gid == .@best_gid && .@best_coin_reward > 0)
                dispbottom "[WoE] Premio de defesa: "+.@best_coin_reward+"x Moeda Cheffenia (Normal) foi enviado ao seu email.";
            if (.@participant_gid == .@second_gid && .@second_coin_reward > 0)
                dispbottom "[WoE] Premio de segundo lugar: "+.@second_coin_reward+"x Moeda Cheffenia (Normal) foi enviado ao seu email.";

            detachrid();
        }
    }

    // reset para proxima WoE
    deletearray $WOE_TIME;
    deletearray $WOE_LIST;
    deletearray $WOE_ENTRY_CID;
    deletearray $WOE_ENTRY_AID;
    deletearray $WOE_ENTRY_GID;
    deletearray $WOE_ENTRY_LIST;
    deletearray $WOE_ENTRY_COUNT;
    $WOE_WIN_TIME = 0;
    $WOE_CUSTOM_ACTIVE = 0;
end;
}

//============================================================
// Expulsar TODOS que nao sao da guild vencedora
//============================================================
function	script	ClearCastle	{	

	.@castle$ = getarg(0);       // mapa do castelo
	.@owner_gid = getarg(1);     // guild vencedora

	// pega todos os players do castelo
	getmapunits(BL_PC, .@castle$, .@list);

	for (.@i = 0; .@i < getarraysize(.@list); .@i++) {

		if (attachrid(.@list[.@i])) {

			// se nao e da guild dona ? expulsa
			if (getcharid(2) != .@owner_gid) {
				warp "SavePoint", 0, 0;
			}

			detachrid();
		}
	}

	return;
}
