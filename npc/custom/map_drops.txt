function	script	CustomMapDrop	{
    .@map$ = getarg(0);       // nome do mapa
    .@items = getarg(1); 
    .@rates = getarg(2); 
    
    if (strcharinfo(3) == .@map$ || strcharinfo(3) == instance_mapname(.@map$)) { // se o player estiver no mapa certo
        // Usa o drop final do Honey no MI como referencia de bonus.
        // Dividido por 2 para que o valor base sem bonus resulte em 100,
        // mantendo uma taxa enviada como 20 = 20% real.
        .@bonus = max(get_drop_bonus(1018, 518) / 2, 100);

		.@drop_final = .@rates * .@bonus;
		if (rand(10000) < .@drop_final) {
			partylootitem .@items, 1, .@drop_final;
		}
        
    }

    return;
}

// Drop global da Cheffenia:
// bossnia_01 e bossnia_02: Album Mini Boss (1001466) com chance dobrada: 1/50000 = 0,002%.
// bossnia_03 e bossnia_04: Album MVP (1001467) com chance padrao: 1/100000 = 0,001%.
function	script	BossniaMapDrop	{
	.@map$ = strcharinfo(3);

	if (.@map$ == "bossnia_01" || .@map$ == "bossnia_02") {
		.@item_id = 1001466;
		.@rand_max = 5000000;
	} else if (.@map$ == "bossnia_03" || .@map$ == "bossnia_04") {
		.@item_id = 1001467;
		.@rand_max = 10000000;
	} else
		return;

	// Usa 1% (100/10000) como referencia para obter o multiplicador pessoal
	// do killer e conserva a precisao da taxa base.
	.@adjusted_rate = get_adjusted_drop_rate(killedrid, 100);
	if (rand(.@rand_max) < .@adjusted_rate)
		partylootitem .@item_id, 1, 1;

	return;
}

// Drop das Joias Etereas (1000325~~1000328) do Errante da Fenda (31900):
// Taxa original: 1/10000 (0,01%). Reduzida pela metade: 1/20000 (0,005%).
// Usa 1% (100/10000) com get_adjusted_drop_rate para preservar os bonus pessoais do jogador.
function	script	FendaMapDrop	{
	if (killedrid != 31900)
		return;

	.@adjusted_rate = get_adjusted_drop_rate(31900, 100);
	setarray .@etel_items[0], 1000325, 1000326, 1000327, 1000328;
	for (.@i = 0; .@i < 4; ++.@i) {
		if (rand(2000000) < .@adjusted_rate)
			partylootitem .@etel_items[.@i], 1, .@adjusted_rate;
	}
	return;
}

// Cartas do Bio5 Hard em lhz_dun_n3 e lhz_dun_n4.
// Mapdrop 5 = 5/100000 (0,005%, ou 1/20000) antes dos bonus pessoais.
// A referencia de 1% preserva Goma de Mascar, VIP, bonus por raca/classe,
// bonus de renascimento e a penalidade de nivel calculados pelo map-server.
function	script	Bio5HardCardMapDrop	{
	.@map$ = strcharinfo(3);
	if (.@map$ != "lhz_dun_n3" && .@map$ != "lhz_dun_n4")
		return;

	switch (killedrid) {
	case 25000: .@card_id = 4674; break; // GuillotineCross_Card
	case 25001: .@card_id = 4675; break; // Archbishop_Card
	case 25002: .@card_id = 4678; break; // Warlock_Card
	case 25003: .@card_id = 4676; break; // Ranger_Card
	case 25004: .@card_id = 4677; break; // Mechanic_Card
	case 25005: .@card_id = 4679; break; // RuneKnight_Card
	case 25006: .@card_id = 4680; break; // RoyalGuard_Card
	case 25007: .@card_id = 4681; break; // Genetic_Card
	case 25008: .@card_id = 4671; break; // Sorcerer_Card
	case 25009: .@card_id = 4672; break; // Sura_Card
	case 25010: .@card_id = 4682; break; // ShadowChaser_Card
	case 25011: .@card_id = 4673; break; // Minstrel_Card
	case 25012: .@card_id = 4683; break; // Wanderer_Card
	default:
		return;
	}

	.@mapdrop_rate = 5;
	.@bonus_reference = get_adjusted_drop_rate(killedrid, 100);
	.@adjusted_rate = .@mapdrop_rate * .@bonus_reference;

	// Denominador 100000 * 100 para conservar fracoes abaixo de 1/10000.
	if (rand(10000000) < .@adjusted_rate)
		partylootitem .@card_id, 1, max(1, .@adjusted_rate / 1000);

	return;
}

// Um unico evento distribui as mortes para os sistemas customizados.
// Assim eles ocupam apenas uma vaga na fila de eventos do jogador.
-	script	GlobalKillDispatcher	-1,{
OnNPCKillEvent:
	callfunc("F_ContadorEdenKill", killedrid);
	callfunc("F_BattlePassSync");
	callfunc("BossniaMapDrop");
	callfunc("FendaMapDrop");
	callfunc("F_ContadorPoringKill", killedrid);
	callfunc("F_MapCustomDropsKill", killedrid);
	callfunc("F_MVP_PointsRegister", killedrid);
	callfunc("F_MVP_KillsRegister", killedrid);
	end;
}
