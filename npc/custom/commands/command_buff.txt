-	script	BuffVIP	-1,{

OnInit:
	bindatcmd "buff", strnpcinfo(3)+"::OnAtcommand", 0, 99;
	bindatcmd "buffs", strnpcinfo(3)+"::OnAtcommand", 0, 99;
	end;

OnAtcommand:
	if (getgroupid() > 0 || @BOT_AI_ACTIVE) {
		.@Delay = 5;
		.@time = 600000;

		// Verifica tempo restante
		if (@last_buff_time + .@delay > gettimetick(2)) {
			.@remaining = (@last_buff_time + .@delay) - gettimetick(2);
			message strcharinfo(0), "Voce podera usar novamente em " + .@remaining + " segundo(s).";
			end;
		}

		// Atualiza o tempo do ultimo uso
		@last_buff_time = gettimetick(2);
		repairall;
		doevent "Healer::OnBuffOnly";
		end;
	}
	
	message strcharinfo(0), "Este comando e exclusivo para jogadores VIP.";
	end;
}
