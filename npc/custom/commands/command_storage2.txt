-	script	Storage2Command	-1,{

OnInit:
	bindatcmd "storage2", strnpcinfo(3) + "::OnAtcommand";
	end;

OnAtcommand:
	if (#Storage2Slots < 100) {
		dispbottom "[Storage 2] Voce ainda nao desbloqueou este armazem.";
		dispbottom "[Storage 2] Use um Cupom de Expansao do Armazem para libera-lo.";
		end;
	}

	openstorage2 1;
	end;
}
