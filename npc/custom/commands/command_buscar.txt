OnAtcommand:
	if (.@atcmd_numparameters > 0)
		.@item_id = atoi(.@atcmd_parameters$[0]);
	else {
		mes "[Busca de Itens]";
		mes "Digite o ID do item que deseja procurar:";
		input .@item_id, 1, 1000000;
	}

	.@source_count = getitemsource(.@item_id, .@source_names$, .@source_npcs$, .@source_types, .@source_maps$);
	if (.@source_count < 0) {
		mes "[Busca de Itens]";
		mes "O ID informado nao pertence a um item valido.";
		close;
	}

	// Alguns itens possuem dezenas de fontes. Os lacos abaixo sao finitos,
	// mas podem ultrapassar o limite padrao de comandos do interpretador.
	freeloop(1);

	// Filtra, mapeia e agrupa as fontes em uma unica passagem.
	.@raw_source_count = .@source_count;
	.@source_count = 0;
	for (.@i = 0; .@i < .@raw_source_count; ++.@i) {
		// Fontes oficiais duplicadas ou indisponiveis nao aparecem no @buscar.
		if (.@source_npcs$[.@i] == "shop_sp_cor" ||
			.@source_npcs$[.@i] == "barter_rgsr_in_1" ||
			compare(.@source_npcs$[.@i], "shop_ext171_"))
			continue;

		// As lojas Grace sao divididas por classe, mas pertencem ao mesmo NPC.
		if (compare(.@source_npcs$[.@i], "barter_grace_JOB_")) {
			.@source_names$[.@i] = "Operador OS - Equipamentos Grace";
			.@source_maps$[.@i] = "Eden";
		} else {
			for (.@j = 0; .@j < .source_map_count; ++.@j) {
				if (.@source_npcs$[.@i] == .source_ids$[.@j]) {
					.@source_names$[.@i] = .source_labels$[.@j];
					.@source_maps$[.@i] = "Menu de Lojas (@lojas)";
					break;
				}
			}
			for (.@j = 0; .@j < .physical_source_count; ++.@j) {
				if (.@source_npcs$[.@i] == .physical_source_ids$[.@j]) {
					.@source_names$[.@i] = .physical_npc_names$[.@j];
					.@source_maps$[.@i] = .physical_map_names$[.@j];
					break;
				}
			}
		}

		// A primeira fonte de cada NPC, tipo e mapa e mantida.
		.@duplicate = 0;
		for (.@j = 0; .@j < .@source_count; ++.@j) {
			if (.@source_names$[.@i] != .@source_names$[.@j] ||
				.@source_types[.@i] != .@source_types[.@j] ||
				.@source_maps$[.@i] != .@source_maps$[.@j])
				continue;
			.@duplicate = 1;
			break;
		}
		if (.@duplicate)
			continue;

		if (.@source_count != .@i) {
			.@source_names$[.@source_count] = .@source_names$[.@i];
			.@source_npcs$[.@source_count] = .@source_npcs$[.@i];
			.@source_types[.@source_count] = .@source_types[.@i];
			.@source_maps$[.@source_count] = .@source_maps$[.@i];
		}
		++.@source_count;
	}
	// Acrescenta caixas que entregam o item diretamente ou por item group.
	.@box_count = getitemboxsource(.@item_id, .@box_ids);
	for (.@i = 0; .@i < .@box_count; ++.@i) {
		.@box_name$ = getitemclientname(.@box_ids[.@i], .client_iteminfo$);
		if (.@box_name$ == "")
			.@box_name$ = getitemclientname(.@box_ids[.@i], .client_iteminfo_custom$);
		if (.@box_name$ == "")
			.@box_name$ = getitemclientname(.@box_ids[.@i], .client_iteminfo_fallback$);
		if (.@box_name$ == "")
			.@box_name$ = getitemclientname(.@box_ids[.@i], .client_iteminfo_custom_fb$);
		if (.@box_name$ == "")
			.@box_name$ = getitemname(.@box_ids[.@i]);
		if (.@box_name$ == "" || .@box_name$ == "null")
			.@box_name$ = replacestr(getiteminfo(.@box_ids[.@i], ITEMINFO_AEGISNAME), "_", " ");

		.@source_names$[.@source_count] = .@box_name$;
		.@source_npcs$[.@source_count] = "";
		.@source_types[.@source_count] = 3;
		.@source_maps$[.@source_count] = "";
		.@source_box_ids[.@source_count] = .@box_ids[.@i];
		++.@source_count;
	}
	freeloop(0);

	mes "[Busca de Itens]";
	// Usa o mesmo nome e a mesma codificacao da tabela do cliente.
	.@item_name$ = getitemclientname(.@item_id, .client_iteminfo$);
	if (.@item_name$ == "")
		.@item_name$ = getitemclientname(.@item_id, .client_iteminfo_custom$);
	if (.@item_name$ == "")
		.@item_name$ = getitemclientname(.@item_id, .client_iteminfo_fallback$);
	if (.@item_name$ == "")
		.@item_name$ = getitemclientname(.@item_id, .client_iteminfo_custom_fb$);
	if (.@item_name$ == "")
		.@item_name$ = getitemname(.@item_id);
	if (.@item_name$ == "" || .@item_name$ == "null")
		.@item_name$ = replacestr(getiteminfo(.@item_id, ITEMINFO_AEGISNAME), "_", " ");
	mes "Item: ^0055FF" + .@item_name$ + "^000000 (ID " + .@item_id + ")";
	if (.@source_count == 0) {
		mes "Ainda nao encontrei este item em lojas de NPC, trocas barter ou caixas.";
		mes "Drops, instancias e lojas de jogadores serao adicionados nas proximas etapas.";
		close;
	}

	mes "Este item pode ser obtido nas fontes abaixo. Escolha uma para ver os detalhes ou abrir o menu.";
	.@page = 0;
	.@per_page = 8;

	while (true) {
		.@start = .@page * .@per_page;
		.@end = min(.@start + .@per_page, .@source_count);
		.@menu$ = "";

		for (.@i = .@start; .@i < .@end; ++.@i) {
			.@kind$ = (.@source_types[.@i] == 3 ? "Caixa" : (.@source_types[.@i] == 2 ? "Troca" : "Loja"));
			.@where$ = (.@source_maps$[.@i] == "" ? "" : " - " + .@source_maps$[.@i]);
			.@menu$ += .@source_names$[.@i] + " [" + .@kind$ + "]" + .@where$ + ":";
		}

		.@shown = .@end - .@start;
		if (.@page > 0)
			.@menu$ += "< Pagina anterior:";
		if (.@end < .@source_count)
			.@menu$ += "Proxima pagina >:";
		.@menu$ += "Cancelar";

		.@choice = select(.@menu$);
		if (.@choice <= .@shown) {
			.@selected = .@start + .@choice - 1;
			break;
		}

		.@navigation = .@shown + 1;
		if (.@page > 0) {
			if (.@choice == .@navigation) {
				--.@page;
				continue;
			}
			++.@navigation;
		}
		if (.@end < .@source_count && .@choice == .@navigation) {
			++.@page;
			continue;
		}
		close;
	}

	mes "[Busca de Itens]";
	if (.@source_types[.@selected] == 3) {
		mes "Este item pode ser obtido ao abrir a caixa ^0055FF" + .@source_names$[.@selected] + "^000000 (ID " + .@source_box_ids[.@selected] + ").";
		mes "A recompensa da caixa pode ser direta ou fazer parte de um grupo de itens.";
		close;
	}

	mes "Este item pode ser obtido no NPC ^0055FF" + .@source_names$[.@selected] + "^000000.";
	if (.@source_maps$[.@selected] != "")
		mes "Local: ^00AA00" + .@source_maps$[.@selected] + "^000000.";
	if (select("Abrir o menu deste NPC:Cancelar") == 2)
		close;

	.@shop_name$ = .@source_npcs$[.@selected];
	close2;
	callshop .@shop_name$, 1;
	end;
}
