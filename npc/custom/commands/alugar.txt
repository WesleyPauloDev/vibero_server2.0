-	script	Alugar	-1,{

OnInit:
	bindatcmd "alugar", strnpcinfo(3)+"::OnAtcommand";
	end;

OnAtcommand:
	callfunc("alugar_function");
	end;
}

function	script	alugar_function	{
	// 1. Ramo dos Mercadores (Mecanico, Meister, Ferreiro, Bioquimico, Biolo, Mercador)
	if (BaseClass == Job_Merchant) {
		// Mecanico / Meister: Madogear 01, Madogear 02 ou Carrinho
		if (Class == Job_Mechanic || Class == Job_Mechanic_T || Class == Job_Mechanic2 || Class == Job_Mechanic_T2 || Class == Job_Meister || Class == Job_Meister2 || getskilllv("NC_MADOLICENCE") > 0) {
			if (getskilllv("NC_MADOLICENCE") == 0) skill "NC_MADOLICENCE", 5, 1;
			if (getskilllv("MC_PUSHCART") == 0) skill "MC_PUSHCART", 10, 1;

			switch (select(" ~ Madogear 01: ~ Madogear 02: ~ Carrinho")) {
			case 1:
				if (checkcart()) setcart 0;
				setmadogear true;
				break;
			case 2:
				if (checkcart()) setcart 0;
				setmadogear true, MADO_SUIT;
				break;
			case 3:
				if (checkmadogear()) setmadogear false;
				if (!checkcart()) setcart;
				break;
			}
			specialeffect2 EF_TEIHIT3;
			return;
		}

		// Outras classes de Mercador (Ferreiro, Mestre-Ferreiro, Alquimista, Criador, Bioquimico, Biolo)
		if (getskilllv("MC_PUSHCART") == 0) skill "MC_PUSHCART", 10, 1;
		if (!checkcart() && !checkmadogear()) setcart;
		specialeffect2 EF_TEIHIT3;
		return;
	}

	// 2. Ramo dos Arqueiros (Sentinela, Windhawk, Cacador, Sniper, Bardo, Odalisca)
	if (BaseClass == Job_Archer) {
		// Sentinela / Windhawk: Lobo (Warg) ou Falcao
		if (Class == Job_Ranger || Class == Job_Ranger_T || Class == Job_Ranger2 || Class == Job_Ranger_T2 || Class == Job_Windhawk || Class == Job_Windhawk2 || getskilllv("RA_WUGMASTERY") > 0 || getskilllv("RA_WUGRIDER") > 0) {
			if (getskilllv("RA_WUGMASTERY") == 0) skill "RA_WUGMASTERY", 1, 1;
			if (getskilllv("RA_WUGRIDER") == 0) skill "RA_WUGRIDER", 1, 1;
			if (getskilllv("HT_FALCON") == 0) skill "HT_FALCON", 1, 1;

			switch (select(" ~ Lobo (Warg): ~ Falcao")) {
			case 1:
				if (checkfalcon()) setoption OPTION_FALCON, 0;
				if (!checkwug()) setoption OPTION_WUG, 1;
				if (countitem(6124) == 0) getitem 6124, 1; // Flauta do Lobo (Wolf's Flute)
				break;
			case 2:
				if (checkwug()) setoption OPTION_WUG|OPTION_WUGRIDER, 0;
				if (!checkfalcon()) setfalcon;
				break;
			}
			specialeffect2 EF_TEIHIT3;
			return;
		}

		// Cacador / Sniper: Falcao
		if (Class == Job_Hunter || Class == Job_Hunter_T || Class == Job_Sniper || getskilllv("HT_FALCON") > 0) {
			if (getskilllv("HT_FALCON") == 0) skill "HT_FALCON", 1, 1;
			if (!checkfalcon() && !checkwug()) setfalcon;
			specialeffect2 EF_TEIHIT3;
			return;
		}
	}

	// 3. Ramo dos Espadachins (Cavaleiro Runico, Dragon Knight, Guardiao Real, Imperial Guard, Paladino, Cavaleiro)
	if (BaseClass == Job_Swordman) {
		// Cavaleiro Runico / Dragon Knight: Dragao
		if (Class == Job_Rune_Knight || Class == Job_Rune_Knight_T || Class == Job_Rune_Knight2 || Class == Job_Rune_Knight_T2 || Class == Job_Dragon_Knight || Class == Job_Dragon_Knight2 || getskilllv("RK_DRAGONTRAINING") > 0) {
			if (getskilllv("KN_RIDING") == 0) skill "KN_RIDING", 1, 1;
			if (getskilllv("RK_DRAGONTRAINING") == 0) skill "RK_DRAGONTRAINING", 5, 1;
			if (!checkdragon()) setdragon 1;
			specialeffect2 EF_TEIHIT3;
			return;
		}

		// Guardiao Real / Imperial Guard / Templario / Paladino / Cavaleiro / Lorde: Grifo / Peco
		if (Class == Job_Knight || Class == Job_Lord_Knight || Class == Job_Crusader || Class == Job_Paladin || Class >= Job_Royal_Guard || Class == Job_Imperial_Guard || Class == Job_Imperial_Guard2 || getskilllv("KN_RIDING") > 0) {
			if (getskilllv("KN_RIDING") == 0) skill "KN_RIDING", 1, 1;
			if (!checkriding() && !checkdragon()) setriding;
			specialeffect2 EF_TEIHIT3;
			return;
		}
	}

	message strcharinfo(0), "Voce nao atende aos requisitos para alugar.";
	return;
}
