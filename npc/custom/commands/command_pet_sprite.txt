-	script	PetSpriteCommand	-1,{

OnInit:
	bindatcmd "petsprite", strnpcinfo(3)+"::OnAtcommand";
	bindatcmd "petvisual", strnpcinfo(3)+"::OnAtcommand";
	end;

OnPCLoginEvent:
	addtimer 3000, strnpcinfo(3)+"::OnApplySprite";
	end;

OnApplySprite:
	.@pet_id = getpetinfo(PETINFO_ID);
	if (.@pet_id <= 0)
		end;

	.@mob_id = getd("#PET_SPRITE_" + .@pet_id);
	if (.@mob_id > 0)
		setpetsprite(.@mob_id);
	end;

OnAtcommand:
	if (getpetinfo(PETINFO_ID) <= 0) {
		mes "[Pet Stylist]";
		mes "Voce precisa estar com um pet ativo para trocar o sprite.";
		close;
	}

	mes "[Pet Stylist]";
	mes "Digite o Mob ID do sprite que deseja usar no seu pet.";
	mes "A troca e apenas visual; os bonus do pet original continuam.";
	next;

	input .@mob_id;

	if (.@mob_id <= 0) {
		mes "[Pet Stylist]";
		mes "Mob ID invalido.";
		close;
	}

	mes "[Pet Stylist]";
	mes "Deseja alterar o sprite do seu pet para o Mob ID ^0055FF" + .@mob_id + "^000000?";
	next;

	if (select("Sim:Nao") == 2)
		close;

	if (!setpetsprite(.@mob_id)) {
		mes "[Pet Stylist]";
		mes "Nao foi possivel alterar o sprite.";
		mes "Confira se o Mob ID existe no mob_db.yml.";
		close;
	}

	setd("#PET_SPRITE_" + getpetinfo(PETINFO_ID), .@mob_id);

	specialeffect2 EF_HO_UP;
	mes "[Pet Stylist]";
	mes "Pronto. O sprite do seu pet foi alterado.";
	mes "Os bonus e a classe real do pet original foram mantidos.";
	close;
}
