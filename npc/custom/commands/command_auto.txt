// ======================================================================
// Sistema de Caca Automatica para Jogadores (@auto / @manual / @auto setup)
// ======================================================================
// - Comando @auto para iniciar a caca automatica no mapa atual do jogador
// - Comando @manual para desativar e pausar o tempo
// - Menu @auto setup para configurar Amuleto de Ziegfried (7621) e Modo XP
// - 30 minutos diarios gratuitos: na virada do dia, se < 30 reseta para 30, se >= 30 mantem
// - Pergaminhos de Tempo (+30 minutos) acumulativos
// - Pausa o tempo imediatamente ao morrer
// - Se configurado Ziegfried e possuir 7621 no inventario, ressuscita e continua
// - Modo XP: ao atingir BaseLevel e JobLevel maximos da classe atual, teleporta para a cidade
// - Guarda loots/drops no armazem automaticamente ao ficar pesado (>= 70%)
// - Usa @buff automaticamente e renova periodicamente
// - Usa Asa de Mosca (601) quando nao houver monstros por perto
// Arquivo: npc/custom/commands/command_auto.txt
// ======================================================================

function	script	F_AutoCheckDaily	{
	.@today = gettime(DT_YYYYMMDD);
	if (#AutoLastDay != .@today) {
		#AutoLastDay = .@today;
		if (#AutoTimeSecs < 1800) {
			#AutoTimeSecs = 1800; // 30 minutos gratuitos
		}
	}
	return;
}

function	script	F_AddAutoTime	{
	.@mins = getarg(0, 30);
	callfunc "F_AutoCheckDaily";
	#AutoTimeSecs += (.@mins * 60);
	specialeffect2 EF_INCAGILITY;
	.@total_m = #AutoTimeSecs / 60;
	.@total_s = #AutoTimeSecs % 60;
	dispbottom "==================================================", 0x00FFFF;
	dispbottom "[Auto]: +" + .@mins + " minutos adicionados ao seu saldo de @auto!", 0x55FF55;
	dispbottom "[Auto]: Tempo total acumulado: " + .@total_m + "m " + (.@total_s < 10 ? "0" : "") + .@total_s + "s.", 0xFFFF00;
	dispbottom "==================================================", 0x00FFFF;
	return;
}

-	script	AutoPlayerSystem	-1,{

OnInit:
	bindatcmd("auto", "AutoPlayerSystem::OnAutoCommand", 0, 99);
	bindatcmd("manual", "AutoPlayerSystem::OnManualCommand", 0, 99);
	end;

// ======================================================================
// COMANDO: @manual (Desativacao instantanea)
// ======================================================================
OnManualCommand:

		dispbottom "[Auto]: Este comando e exclusivo para jogadores VIP (Grupo 5).", 0xFF5555;
		end;
	}
	if (!@AUTO_ACTIVE) {
		dispbottom "[Auto]: O modo automatico nao esta ativo no momento.", 0xFFAAAA;

// COMANDO: @auto [setup | config | status | on | off]
// ======================================================================
OnAutoCommand:

		dispbottom "[Auto]: Este comando e exclusivo para jogadores VIP (Grupo 5).", 0xFF5555;
		end;
	}
	.@arg$ = strtolower(.@atcmd_parameters$[0]);

	if (.@arg$ == "setup" || .@arg$ == "config" || .@arg$ == "ajustes") {
		callsub OnOpenSetupMenu;
		end;
	}

	if (.@arg$ == "off" || .@arg$ == "stop" || .@arg$ == "parar") {
		if (@AUTO_ACTIVE) {
			callsub OnStopAuto, "[Auto]: Modo automatico desativado. Tempo restante pausado.";
		} else {
			dispbottom "[Auto]: O modo automatico nao esta ativo.", 0xFFAAAA;
		}
		end;
	}

	callfunc "F_AutoCheckDaily";

	if (.@arg$ == "status" || .@arg$ == "tempo" || .@arg$ == "time") {
		.@rm_m = #AutoTimeSecs / 60;
		.@rm_s = #AutoTimeSecs % 60;
		dispbottom "==================================================", 0x00FFFF;
		dispbottom "   STATUS DO MODO AUTOMATICO (@auto)", 0xFFFF00;
		dispbottom "==================================================", 0x00FFFF;
		dispbottom "Status: " + (@AUTO_ACTIVE ? "^00FF00[ATIVO]^000000" : "^FF0000[INATIVO]^000000"), 0xFFFFFF;
		dispbottom "Tempo Restante: " + .@rm_m + "m " + (.@rm_s < 10 ? "0" : "") + .@rm_s + "s", 0xFFFFFF;

		end;
	}

	// Se ja estiver ativo, avisa e mostra tempo
	if (@AUTO_ACTIVE) {
		.@rm_m = #AutoTimeSecs / 60;
		.@rm_s = #AutoTimeSecs % 60;

		dispbottom "[Auto]: Tempo restante: " + .@rm_m + "m " + (.@rm_s < 10 ? "0" : "") + .@rm_s + "s. Use '@manual' para desligar.", 0xFFFF00;
		end;
	}

	// Verificacoes previas de seguranca
	if (Hp <= 0 || Dead) {
		dispbottom "[Auto]: Voce nao pode ativar o modo automatico enquanto estiver morto.", 0xFF5555;
		end;
	}
	if (checkvending() || checkchatting()) {
		dispbottom "[Auto]: Feche sua loja ou sala de chat antes de ativar o modo automatico.", 0xFF5555;
		end;
	}

	if (#AutoTimeSecs <= 0) {
		dispbottom "[Auto]: Voce nao possui tempo de @auto restante hoje.", 0xFF5555;
		dispbottom "[Auto]: O tempo diario gratuito e de 30 minutos. Use Pergaminhos de Tempo (+30m) para recarregar!", 0xFFFF00;
		end;
	}

	.@cur_map$ = strcharinfo(3);
	if (getmapflag(.@cur_map$, MF_TOWN) || .@cur_map$ == "prontera" || .@cur_map$ == "morocc" || .@cur_map$ == "geffen" || .@cur_map$ == "payon" || .@cur_map$ == "alberta" || .@cur_map$ == "izlude" || .@cur_map$ == "aldebaran" || .@cur_map$ == "yuno" || .@cur_map$ == "amatsu" || .@cur_map$ == "gonryun" || .@cur_map$ == "louyang" || .@cur_map$ == "ayothaya" || .@cur_map$ == "einbroch" || .@cur_map$ == "einbech" || .@cur_map$ == "lighthalzen" || .@cur_map$ == "hugel" || .@cur_map$ == "rachel" || .@cur_map$ == "veins" || .@cur_map$ == "jawaii" || .@cur_map$ == "moscovia" || .@cur_map$ == "mid_camp" || .@cur_map$ == "manuk" || .@cur_map$ == "splendide" || .@cur_map$ == "brasilis" || .@cur_map$ == "dicastes01" || .@cur_map$ == "mora" || .@cur_map$ == "dewata" || .@cur_map$ == "malangdo" || .@cur_map$ == "malaya" || .@cur_map$ == "eclage" || .@cur_map$ == "lasagna") {
		dispbottom "==================================================", 0xFF5555;
		dispbottom "[Auto]: O modo automatico nao pode ser utilizado dentro de cidades!", 0xFF5555;
		dispbottom "==================================================", 0xFF5555;
		end;
	}

	stand;
	@AUTO_ACTIVE = 1;
	@ab_tp_quiz_cooldown_until = gettimetick(2) + 86400;
	@ab_tp_quiz_active = 0;
	@ab_tp_quiz_timeout = 0;
	setpcblock PCBLOCK_ALL|PCBLOCK_IMMUNE, false;
	@AUTO_CONSECUTIVE_WING_TELEPORTS = 0;
	@AUTO_MAP$ = strcharinfo(3);
	@AUTO_LAST_TICK = gettimetick(2);
	@AUTO_LAST_BUFF = gettimetick(2);
	callsub OnApplyBaseBuffs;
	if ((BaseClass == Job_Acolyte || Class == Job_Acolyte || Class == Job_Acolyte_High || BaseJob == Job_Acolyte || BaseJob == Job_Acolyte_High) && getskilllv("AL_HOLYLIGHT") <= 0) {
		skill "AL_HOLYLIGHT", 1, 0;
	}
	if ((BaseClass == Job_Mage || Class == Job_Mage || Class == Job_Mage_High || BaseJob == Job_Mage || BaseJob == Job_Mage_High) && getskilllv("MG_ENERGYCOAT") <= 0) {
		skill "MG_ENERGYCOAT", 1, 0;
	}
	if ((BaseClass == Job_Archer || Class == Job_Archer || Class == Job_Archer_High || BaseJob == Job_Archer || BaseJob == Job_Archer_High) && getskilllv("HT_BLITZBEAT") > 0) {
		if (getskilllv("HT_FALCON") <= 0) {
			skill "HT_FALCON", 1, 0;
		}
		if (!checkfalcon()) {
			setfalcon 1;
		}
	}
	if (!checkcart() && !checkmadogear() && getskilllv("MC_PUSHCART") > 0) {
		setcart;
	}
	if (getstatus(SC_DEFENDER)) {
		sc_end SC_DEFENDER;
	}
	if (getstatus(SC_AUTOSPELL)) {
		sc_end SC_AUTOSPELL;
	}
	if (getstatus(SC_HIDING)) {
		sc_end SC_HIDING;
	}
	@AUTO_CURRENT_TARGET = 0;
	@AUTO_TARGET_TRACK_ID = 0;
	@AUTO_TARGET_LAST_HP = 0;
	@AUTO_TARGET_PROGRESS_TICK = 0;
	@AUTO_BLOCKED_TARGET = 0;
	@AUTO_BLOCKED_UNTIL = 0;
	deletearray @AUTO_BLOCKED_MOB[0], 8;
	deletearray @AUTO_BLOCKED_TIME[0], 8;
	@AUTO_BLOCKED_INDEX = 0;
	@AUTO_LAST_TX = 0;
	@AUTO_LAST_TY = 0;
	@AUTO_STUCK_X = 0;
	@AUTO_STUCK_Y = 0;
	@AUTO_STUCK_TICKS = 0;
	@AUTO_SKILL_ROT = 0;
	@AUTO_LAST_HEAVENDRIVE = 0;
	@AUTO_LAST_TELEPORT = 0;
	@AUTO_CONSECUTIVE_NO_DAMAGE = 0;

	.@rm_m = #AutoTimeSecs / 60;
	.@rm_s = #AutoTimeSecs % 60;
	dispbottom "==================================================", 0x00FFFF;
	dispbottom "   MODO AUTOMATICO ATIVADO COM SUCESSO!", 0x55FF55;
	dispbottom "==================================================", 0x00FFFF;
	dispbottom "Mapa de Caca: " + @AUTO_MAP$, 0xFFFFFF;
	dispbottom "Tempo Restante: " + .@rm_m + "m " + (.@rm_s < 10 ? "0" : "") + .@rm_s + "s", 0xFFFF00;
	dispbottom "Para desligar a qualquer momento, digite: @manual", 0xFFFFFF;
	dispbottom "Para abrir as configuracoes, digite: @auto setup", 0xFFFFFF;
	dispbottom "==================================================", 0x00FFFF;

	deltimer "AutoPlayerSystem::OnStepTimer";
	addtimer 500, "AutoPlayerSystem::OnStepTimer";
	end;

// ======================================================================
// MENU INTERATIVO: @auto setup
// ======================================================================
OnOpenSetupMenu:
	callfunc "F_AutoCheckDaily";
	.@rm_m = #AutoTimeSecs / 60;
	.@rm_s = #AutoTimeSecs % 60;

	mes "[^0000FFConfiguracoes do Modo Auto^000000]";
	mes "Personalize as opcoes do seu modo automatico:";
	mes "----------------------------------------";
	mes "Tempo Restante: ^008800" + .@rm_m + "m " + (.@rm_s < 10 ? "0" : "") + .@rm_s + "s^000000";

	mes "Armazenamento no Armazem: ^00AA00[Automatico ao atingir 70% peso]^000000";

		"Amuleto de Ziegfried (" + (#AutoUseZiegfried ? "^00AA00Ligado^000000" : "^FF0000Desligado^000000") + ")",
		"Modo de Foco (" + (#AutoFarmMode ? "Foco XP" : "Farm Livre") + ")",

		"Fechar Menu"
	)) {

		#AutoUseZiegfried = !#AutoUseZiegfried;
		dispbottom "[Auto]: Usar Amuleto de Ziegfried ao morrer agora esta: " + (#AutoUseZiegfried ? "^00FF00ATIVADO^000000" : "^FF0000DESATIVADO^000000"), 0xFFFF00;
		goto OnOpenSetupMenu;

	case 2:
		#AutoFarmMode = !#AutoFarmMode;
		if (#AutoFarmMode) {
			dispbottom "[Auto]: Modo de Foco alterado para: ^00FFFFFoco XP^000000 (Ao atingir Base e Job maximos da classe, teleporta para a cidade e desliga o tempo).", 0x55FF55;
		} else {
			dispbottom "[Auto]: Modo de Foco alterado para: ^00FF00Farm Livre^000000 (Continua cacando mesmo no nivel maximo ate usar @manual ou o tempo esgotar).", 0x55FF55;
		}
		goto OnOpenSetupMenu;

	case 3:
		mes "[^0000FFTempo Restante^000000]";
		mes "Seu saldo atual de modo automatico:";
		mes "^00AA00" + .@rm_m + " minuto(s) e " + .@rm_s + " segundo(s).^000000";
		mes " ";
		mes "Todos os dias voce recebe 30 minutos gratuitos.";
		mes "Pergaminhos de Tempo (+30m) acumulam e nao sao perdidos na virada do dia.";
		next;
		goto OnOpenSetupMenu;

	case 4:
		mes "[^0000FFComo Funciona o Modo Auto^000000]";
		mes "1. Va ate o mapa onde deseja cacar.";
		mes "2. Configure seu ^008800@autopot^000000, ^008800@autoloot^000000 e ^008800@restock^000000.";
		mes "3. Digite ^0000FF@auto^000000 para comecar a cacada.";
		mes "4. O bot usara as habilidades ofensivas que voce tiver aprendido na sua arvore de skills.";
		mes "5. O bot usa @buff e Asa de Mosca (601) quando nao houver monstros por perto.";
		mes "6. Ao ficar com peso >= 70%, seus drops coletados sao guardados no armazem.";
		mes "7. Ao morrer, o tempo pausa. Se Ziegfried estiver ativo e voce tiver o item 7621, ele revive e continua.";
		mes "8. Para desligar a qualquer momento, digite ^FF0000@manual^000000.";
		next;
		goto OnOpenSetupMenu;

	case 5:
		close;
	}
	end;

// ======================================================================
// SUB-ROTINA: Finalizacao Segura do Modo Auto
// ======================================================================
OnStopAuto:
	.@reason$ = getarg(0, "");
	@AUTO_ACTIVE = 0;
	@AUTO_CONSECUTIVE_WING_TELEPORTS = 0;
	@ab_tp_quiz_cooldown_until = 0;
	@ab_tp_quiz_active = 0;
	@ab_tp_quiz_timeout = 0;
	setpcblock PCBLOCK_ALL|PCBLOCK_IMMUNE, false;
	@AUTO_CURRENT_TARGET = 0;
	@AUTO_CURRENT_ATTACK_TARGET = 0;
	@AUTO_TARGET_TRACK_ID = 0;
	@AUTO_TARGET_LAST_HP = 0;
	@AUTO_TARGET_PROGRESS_TICK = 0;
	@AUTO_BLOCKED_TARGET = 0;
	@AUTO_BLOCKED_UNTIL = 0;
	deletearray @AUTO_BLOCKED_MOB[0], 8;
	deletearray @AUTO_BLOCKED_TIME[0], 8;
	@AUTO_BLOCKED_INDEX = 0;
	@AUTO_LAST_TX = 0;
	@AUTO_LAST_TY = 0;
	@AUTO_STUCK_TICKS = 0;
	@AUTO_CONSECUTIVE_NO_DAMAGE = 0;
	@AUTO_ROGUE_STEP = 0;
	@BUFF_HOMUN_RES = 0;
	@LAST_HOMUN_PITCHER_TICK = 0;
	deltimer "AutoPlayerSystem::OnStepTimer";
	unitstopwalk getcharid(3), USW_FORCE_STOP;
	unitstopattack getcharid(3);
	if (getstatus(SC_HIDING)) {
		sc_end SC_HIDING;
	}

	if (.@reason$ != "") {
		dispbottom .@reason$, 0xFFFF00;
	}
	.@rm_m = #AutoTimeSecs / 60;
	.@rm_s = #AutoTimeSecs % 60;
	dispbottom "[Auto]: Tempo restante atual: " + .@rm_m + "m " + (.@rm_s < 10 ? "0" : "") + .@rm_s + "s.", 0x00FFFF;
	return;

// ======================================================================
// EVENTO: Morte do Jogador (OnPCDieEvent)
// ======================================================================
OnPCDieEvent:
	if (!@AUTO_ACTIVE) end;

	// Pausa imediata do timer
	deltimer "AutoPlayerSystem::OnStepTimer";

	// Atualiza e deduz o tempo decorrido ate o instante exato da morte
	.@now = gettimetick(2);
	.@diff = .@now - @AUTO_LAST_TICK;
	if (.@diff > 0) {
		#AutoTimeSecs -= .@diff;
		if (#AutoTimeSecs < 0) #AutoTimeSecs = 0;
		@AUTO_LAST_TICK = .@now;
	}

	// Verifica se a opcao de usar Ziegfried esta ativada e se tem o item no inventario
	if (#AutoUseZiegfried && countitem(7621) > 0 && #AutoTimeSecs > 0) {
		delitem 7621, 1;
		specialeffect2 EF_RESURRECTION;
		recovery 0, getcharid(0);
		stand;
		percentheal 100, 100;
		dispbottom "==================================================", 0x55FF55;
		dispbottom "[Auto]: Amuleto de Ziegfried utilizado com sucesso!", 0x55FF55;
		dispbottom "[Auto]: Personagem ressuscitado. Retomando modo automatico...", 0xFFFF00;
		dispbottom "==================================================", 0x55FF55;

		@AUTO_CURRENT_TARGET = 0;
		@AUTO_TARGET_TRACK_ID = 0;
		@AUTO_TARGET_LAST_HP = 0;
		@AUTO_TARGET_PROGRESS_TICK = 0;
		@AUTO_LAST_TX = 0;
		@AUTO_LAST_TY = 0;
		@AUTO_STUCK_TICKS = 0;
		@AUTO_LAST_TICK = gettimetick(2);

		addtimer 1500, "AutoPlayerSystem::OnStepTimer";
		end;
	}

	// Sem Ziegfried: desliga o modo automatico e pausa o tempo
	callsub OnStopAuto, "[Auto]: Voce foi derrotado. Modo automatico encerrado e contagem de tempo pausada.";
	end;

// ======================================================================
// EVENTOS: Troca de Mapa / Logout
// ======================================================================
OnPCLoadMapEvent:
	if (@AUTO_ACTIVE && strcharinfo(3) != @AUTO_MAP$) {
		callsub OnStopAuto, "[Auto]: Voce mudou de mapa. Modo automatico encerrado e tempo pausado.";
	}
	end;

OnPCLogoutEvent:
	if (@AUTO_ACTIVE) {
		.@now = gettimetick(2);
		.@diff = .@now - @AUTO_LAST_TICK;
		if (.@diff > 0) {
			#AutoTimeSecs -= .@diff;
			if (#AutoTimeSecs < 0) #AutoTimeSecs = 0;
		}
		@AUTO_ACTIVE = 0;
		deltimer "AutoPlayerSystem::OnStepTimer";
	}
	end;

// ======================================================================
// LOOP PRINCIPAL DE INTELIGENCIA ARTIFICIAL E COMBATE
// ======================================================================
OnStepTimer:
	if (!@AUTO_ACTIVE) end;

	// Se estiver morto, interrompe
	if (Hp <= 0 || Dead) {
		callsub OnStopAuto, "[Auto]: Personagem incapacitado. Modo automatico pausado.";
		end;
	}

	// Garante que o jogador esta em pe
	stand;

	// Verifica se permanece no mapa configurado ou entrou em cidade
	getmapxy(.@cur_map$, .@cx, .@cy, BL_PC);
	if (.@cur_map$ != @AUTO_MAP$ || getmapflag(.@cur_map$, MF_TOWN)) {
		callsub OnStopAuto, "[Auto]: Voce esta em uma cidade ou saiu do mapa configurado. Modo automatico encerrado.";
		end;
	}

	// Atualizacao do Tempo Diario
	.@now = gettimetick(2);
	@ab_tp_quiz_cooldown_until = .@now + 86400;
	@ab_tp_quiz_active = 0;
	@ab_tp_quiz_timeout = 0;
	.@diff = .@now - @AUTO_LAST_TICK;
	if (.@diff > 0) {
		#AutoTimeSecs -= .@diff;
		@AUTO_LAST_TICK = .@now;
	}

	// Verifica se o tempo esgotou
	if (#AutoTimeSecs <= 0) {
		#AutoTimeSecs = 0;
		callsub OnStopAuto, "[Auto]: Seu tempo diario de modo automatico esgotou! Utilize um Pergaminho de Tempo para continuar.";
		end;
	}

	// ==================================================================
	// Checagem periodica e renovacao direta dos buffs basicos
	// ==================================================================
	if (!getstatus(SC_BLESSING) || !getstatus(SC_INCREASEAGI) || (.@now - @AUTO_LAST_BUFF) >= 300) {
		@AUTO_LAST_BUFF = .@now;
		callsub OnApplyBaseBuffs;
	}

	// ==================================================================
	// Reconhecimento e ativacao de habilidades de suporte / auto-buffs
	// Usa buffs proprios com delay de 1 minuto ou duracao do status
	// ==================================================================
	.@buff_used = callsub(OnCheckSupportBuffs);
	if (.@buff_used) {
		addtimer 700, "AutoPlayerSystem::OnStepTimer";
		end;
	}

	// ==================================================================
	// Foco em XP: Se atingiu nivel Base e Job maximos da classe atual
	// ==================================================================
	if (#AutoFarmMode == 1 && is_maxbaselv() && is_maxjoblv()) {
		dispbottom "==================================================", 0x00FFFF;
		dispbottom "[Auto]: Nivel Base e Classe maximos da classe atingidos!", 0x55FF55;
		dispbottom "[Auto]: Retornando para a cidade e encerrando o modo automatico.", 0xFFFF00;
		dispbottom "==================================================", 0x00FFFF;
		callsub OnStopAuto, "";
		warp "SavePoint", 0, 0;
		end;
	}

	// ==================================================================
	// Gestao de Peso: Guarda drops acumulados no Armazem ao atingir >= 70%
	// Nao guarda itens/pocoes configurados no @autopot
	// ==================================================================
	if ((Weight * 100 / MaxWeight) >= 70) {
		.@ap_hp = (getd("@autopot_hp_ID") ? getd("@autopot_hp_ID") : autopot_hp_id);
		.@ap_sp = (getd("@autopot_sp_ID") ? getd("@autopot_sp_ID") : autopot_sp_id);
		.@stored_items = autostore_loot(70, .@ap_hp, .@ap_sp);
		if (.@stored_items > 0) {
			dispbottom "[Auto]: Peso elevado! " + .@stored_items + " item(ns) de drop foram guardados com seguranca no armazem.", 0x00FFFF;
		}
	}

	// ==================================================================
	// Verificacao de Capacidade de Combate (Recursos Essenciais)
	// Se nao puder atacar, retorna a cidade, exibe a causa e desliga
	// ==================================================================
	// 1. Mercador sem Zeny para usar Mammonita (custo de 10 Zeny)
	if ((BaseClass == Job_Merchant || Class == Job_Merchant || Class == Job_Merchant_High || BaseJob == Job_Merchant || BaseJob == Job_Merchant_High) && getskilllv("MC_MAMMONITE") > 0) {
		.@mammo_cost = 10;
		if (Zeny < .@mammo_cost) {
			.@has_alt_skill = 0;
			if (getskilllv("NC_ARMSCANNON") > 0 && Sp >= 30) .@has_alt_skill = 1;
			if (getskilllv("GN_CARTCANNON") > 0 && Sp >= 30) .@has_alt_skill = 1;
			if (checkcart()) {
				if (getskilllv("WS_CARTTERMINATION") > 0 && Sp >= 15) .@has_alt_skill = 1;
				if (getskilllv("MC_CARTREVOLUTION") > 0 && Sp >= 12) .@has_alt_skill = 1;
			}
			if (!.@has_alt_skill) {
				dispbottom "==================================================", 0xFF5555;
				dispbottom "[Auto]: Seu Zeny esgotou e voce nao pode usar Mammonita!", 0xFF5555;
				dispbottom "[Auto]: Retornando para a cidade e encerrando o modo automatico.", 0xFFFF00;
				dispbottom "==================================================", 0xFF5555;
				callsub OnStopAuto, "";
				warp "SavePoint", 0, 0;
				end;
			}
		}
	}

	// 2. Arqueiros / Atiradores sem municao (flechas ou balas)
	if (BaseClass == Job_Archer || Class == Job_Archer || Class == Job_Archer_High || BaseJob == Job_Archer || BaseJob == Job_Archer_High || Class == Job_Gunslinger || Class == Job_Rebellion) {
		if (getequipid(EQI_HAND_R) > 0 && getequipid(EQI_AMMO) <= 0) {
			dispbottom "==================================================", 0xFF5555;
			dispbottom "[Auto]: Suas municoes (flechas/balas) acabaram!", 0xFF5555;
			dispbottom "[Auto]: Retornando para a cidade e encerrando o modo automatico.", 0xFFFF00;
			dispbottom "==================================================", 0xFF5555;
			callsub OnStopAuto, "";
			warp "SavePoint", 0, 0;
			end;
		}
	}

	// 3. Arma Quebrada
	if (getstatus(SC_BROKENWEAPON)) {
		dispbottom "==================================================", 0xFF5555;
		dispbottom "[Auto]: Sua arma esta quebrada! Impossibilitado de atacar.", 0xFF5555;
		dispbottom "[Auto]: Retornando para a cidade e encerrando o modo automatico.", 0xFFFF00;
		dispbottom "==================================================", 0xFF5555;
		callsub OnStopAuto, "";
		warp "SavePoint", 0, 0;
		end;
	}

	// 4. Impossibilidade de causar dano aos monstros
	if (@AUTO_CONSECUTIVE_NO_DAMAGE >= 3) {
		dispbottom "==================================================", 0xFF5555;
		dispbottom "[Auto]: Voce nao esta conseguindo atacar ou causar dano aos monstros!", 0xFF5555;
		dispbottom "[Auto]: Retornando para a cidade e encerrando o modo automatico.", 0xFFFF00;
		dispbottom "==================================================", 0xFF5555;
		callsub OnStopAuto, "";
		warp "SavePoint", 0, 0;
		end;
	}

	// Libera alvo bloqueado por parede/obstaculo apos timeout
	if (@AUTO_BLOCKED_TARGET > 0 && .@now >= @AUTO_BLOCKED_UNTIL) {
		@AUTO_BLOCKED_TARGET = 0;
		@AUTO_BLOCKED_UNTIL = 0;
	}

	// ==================================================================
	// Deteccao de MVP nas Proximidades -> Teleporta para nao morrer
	// ==================================================================
	.@num_mobs_around = getareaunits(BL_MOB, @AUTO_MAP$, .@cx - 14, .@cy - 14, .@cx + 14, .@cy + 14, .@mob_check_list[0]);
	.@found_mvp = 0;
	for (.@k = 0; .@k < .@num_mobs_around; ++.@k) {
		.@chk_mid = .@mob_check_list[.@k];
		if (!unitexists(.@chk_mid)) continue;
		getunitdata .@chk_mid, .@chk_data;
		.@chk_class = .@chk_data[UMOB_CLASS];
		if ((.@chk_data[UMOB_MODE] & MD_MVP) || getmonsterinfo(.@chk_class, MOB_MVPEXP) > 0 || (getmonsterinfo(.@chk_class, MOB_MODE) & MD_MVP)) {
			.@found_mvp = .@chk_mid;
			break;
		}
	}

	if (.@found_mvp > 0) {
		dispbottom "==================================================", 0xFFAA00;
		dispbottom "[Auto]: MVP detectado nas proximidades! Teleportando para nao morrer...", 0xFFAA00;
		dispbottom "==================================================", 0xFFAA00;
		@AUTO_CURRENT_TARGET = 0;
		@AUTO_TARGET_TRACK_ID = 0;
		@AUTO_TARGET_LAST_HP = 0;
		@AUTO_TARGET_PROGRESS_TICK = 0;
		@AUTO_LAST_TX = 0;
		@AUTO_LAST_TY = 0;
		@AUTO_STUCK_TICKS = 0;
		unitstopattack getcharid(3);
		unitstopwalk getcharid(3), USW_FORCE_STOP;

		// Se o mapa bloquear teleporte, retorna a cidade para salvar o personagem
		if (getmapflag(@AUTO_MAP$, MF_NOTELEPORT) || getmapflag(@AUTO_MAP$, MF_NOWARP)) {
			dispbottom "[Auto]: Teleporte bloqueado neste mapa! Retornando para a cidade...", 0xFF5555;
			callsub OnStopAuto, "";
			warp "SavePoint", 0, 0;
			end;
		}

		if (countitem(601) > 0) {
			delitem 601, 1;
		} else if (getskilllv("AL_TELEPORT") > 0 && Sp >= 9) {
			Sp -= 9;
		}
		warp @AUTO_MAP$, 0, 0;
		addtimer 1000, "AutoPlayerSystem::OnStepTimer";
		end;
	}

	// 1. Validacao do Alvo Atual
	if (@AUTO_CURRENT_TARGET > 0) {
		@AUTO_CONSECUTIVE_WING_TELEPORTS = 0;
		if (!unitexists(@AUTO_CURRENT_TARGET)) {
			@AUTO_CURRENT_TARGET = 0;
			@AUTO_CURRENT_ATTACK_TARGET = 0;
			@AUTO_TARGET_TRACK_ID = 0;
			@AUTO_TARGET_LAST_HP = 0;
			@AUTO_TARGET_PROGRESS_TICK = 0;
			@AUTO_LAST_TX = 0;
			@AUTO_LAST_TY = 0;
		} else {
			getunitdata @AUTO_CURRENT_TARGET, .@cur_data;
			.@cur_class = .@cur_data[UMOB_CLASS];
			.@cur_target_id = .@cur_data[UMOB_TARGETID];
			.@d = distance(.@cx, .@cy, .@cur_data[UMOB_X], .@cur_data[UMOB_Y]);

			// Se morreu, planta/cogumelo, se afastou mais de 14 celulas ou foi engatado por outro jogador (Anti-KS):
			if (.@cur_data[UMOB_HP] <= 0 || (.@cur_class >= 1078 && .@cur_class <= 1085) || (.@cur_class >= 1930 && .@cur_class <= 1933) || .@cur_class == 1064 ||
			    .@d > 14 || (.@cur_data[UMOB_HP] < .@cur_data[UMOB_MAXHP] && .@cur_target_id > 0 && .@cur_target_id != getcharid(3))) {
				@AUTO_CURRENT_TARGET = 0;
				@AUTO_CURRENT_ATTACK_TARGET = 0;
				@AUTO_TARGET_TRACK_ID = 0;
				@AUTO_TARGET_LAST_HP = 0;
				@AUTO_TARGET_PROGRESS_TICK = 0;
				@AUTO_LAST_TX = 0;
				@AUTO_LAST_TY = 0;
			} else {
				.@mx = .@cur_data[UMOB_X];
				.@my = .@cur_data[UMOB_Y];

				// Detecta parede / linha de visao bloqueada se o HP nao mudar em tempo habil
				// Para corpo-a-corpo (alcance 1), concede 6s para evitar falsos positivos de miss/ASPD baixa
				if (@AUTO_TARGET_TRACK_ID == @AUTO_CURRENT_TARGET) {
					.@w_temp_id = getequipid(EQI_HAND_R);
					.@w_temp_r = 1;
					if (.@w_temp_id > 0) {
						.@w_temp_r = getiteminfo(.@w_temp_id, ITEMINFO_RANGE);
						if (.@w_temp_r <= 0) .@w_temp_r = 1;
					}
					.@max_wait = (.@d <= .@w_temp_r ? 6 : 4);

					if (.@cur_data[UMOB_HP] < @AUTO_TARGET_LAST_HP) {
						@AUTO_TARGET_LAST_HP = .@cur_data[UMOB_HP];
						@AUTO_TARGET_PROGRESS_TICK = .@now;
						@AUTO_CONSECUTIVE_NO_DAMAGE = 0;
					} else if ((.@now - @AUTO_TARGET_PROGRESS_TICK) >= .@max_wait) {
						// Alvo atras de obstaculo ou inatingivel: bloqueia e troca imediatamente
						@AUTO_BLOCKED_INDEX = (@AUTO_BLOCKED_INDEX + 1) % 8;
						@AUTO_BLOCKED_MOB[@AUTO_BLOCKED_INDEX] = @AUTO_CURRENT_TARGET;
						@AUTO_BLOCKED_TIME[@AUTO_BLOCKED_INDEX] = .@now + 25;
						@AUTO_CURRENT_TARGET = 0;
						@AUTO_CURRENT_ATTACK_TARGET = 0;
						@AUTO_TARGET_TRACK_ID = 0;
						@AUTO_TARGET_LAST_HP = 0;
						@AUTO_TARGET_PROGRESS_TICK = 0;
						@AUTO_LAST_TX = 0;
						@AUTO_LAST_TY = 0;
						@AUTO_STUCK_TICKS = 0;
						unitstopattack getcharid(3);
						unitstopwalk getcharid(3), USW_FORCE_STOP;
						unitwalk getcharid(3), .@cx + rand(-3, 3), .@cy + rand(-3, 3);
						addtimer 400, "AutoPlayerSystem::OnStepTimer";
						end;
					} else if (.@cur_data[UMOB_HP] > @AUTO_TARGET_LAST_HP) {
						@AUTO_TARGET_LAST_HP = .@cur_data[UMOB_HP];
					}
				}
			}
		}
	}

	// 2. Busca por Novo Monstro Valido em Raio de ate 14 celulas
	if (@AUTO_CURRENT_TARGET == 0) {
		.@num_mobs = getareaunits(BL_MOB, @AUTO_MAP$, .@cx - 14, .@cy - 14, .@cx + 14, .@cy + 14, .@mob_list[0]);
		.@best_dist = 999;
		.@best_mob = 0;

		for (.@i = 0; .@i < .@num_mobs; ++.@i) {
			.@mid = .@mob_list[.@i];
			if (!unitexists(.@mid)) continue;

			// Ignora alvos recentemente bloqueados por obstaculos/paredes
			.@is_blocked = 0;
			for (.@b = 0; .@b < 8; ++.@b) {
				if (@AUTO_BLOCKED_MOB[.@b] == .@mid && .@now < @AUTO_BLOCKED_TIME[.@b]) {
					.@is_blocked = 1;
					break;
				}
			}
			if (.@is_blocked) continue;

			getunitdata .@mid, .@t_data;
			.@m_class = .@t_data[UMOB_CLASS];
			// Ignora plantas, cogumelos e MVPs
			if ((.@m_class >= 1078 && .@m_class <= 1085) || (.@m_class >= 1930 && .@m_class <= 1933) || .@m_class == 1064)
				continue;
			if ((.@t_data[UMOB_MODE] & MD_MVP) || getmonsterinfo(.@m_class, MOB_MVPEXP) > 0 || (getmonsterinfo(.@m_class, MOB_MODE) & MD_MVP))
				continue;

			// Anti-KS: nao ataca monstro que ja esta sofrendo dano de outro jogador
			.@m_target = .@t_data[UMOB_TARGETID];
			if (.@t_data[UMOB_HP] < .@t_data[UMOB_MAXHP] && .@m_target > 0 && .@m_target != getcharid(3))
				continue;

			if (.@t_data[UMOB_HP] > 0) {
				.@d = distance(.@cx, .@cy, .@t_data[UMOB_X], .@t_data[UMOB_Y]);
				if (.@d < .@best_dist) {
					.@best_dist = .@d;
					.@best_mob = .@mid;
					.@mx = .@t_data[UMOB_X];
					.@my = .@t_data[UMOB_Y];
				}
			}
		}

		if (.@best_mob > 0) {
			@AUTO_CURRENT_TARGET = .@best_mob;
			@AUTO_CONSECUTIVE_WING_TELEPORTS = 0;
		}
	}

	// 3. Nenhum Monstro Encontrado na Tela -> Usa Asa de Mosca (601) se o mapa permitir, ou explora caminhando
	if (@AUTO_CURRENT_TARGET == 0) {
		@AUTO_ROGUE_STEP = 0;
		if (getstatus(SC_HIDING)) {
			sc_end SC_HIDING;
		}
		// Se possuir Asa de Mosca (601) no inventario e o teleporte nao for bloqueado no mapa:
		if (countitem(601) > 0 && !getmapflag(@AUTO_MAP$, MF_NOTELEPORT) && !getmapflag(@AUTO_MAP$, MF_NOWARP)) {
			if ((.@now - @AUTO_LAST_TELEPORT) >= 1) {
				@AUTO_LAST_TELEPORT = .@now;
				@AUTO_CONSECUTIVE_WING_TELEPORTS++;

				// Cancela modo automatico se usar 10 asas de mosca seguidas sem encontrar monstros
				if (@AUTO_CONSECUTIVE_WING_TELEPORTS >= 10) {
					dispbottom "==================================================", 0xFF5555;
					dispbottom "[Auto]: 10 asas de mosca utilizadas sem encontrar monstros!", 0xFF5555;
					dispbottom "[Auto]: Este mapa nao possui monstros suficientes para o modo automatico.", 0xFFFF00;
					dispbottom "[Auto]: Retornando para a cidade e encerrando o modo automatico.", 0xFFFF00;
					dispbottom "==================================================", 0xFF5555;
					callsub OnStopAuto, "";
					warp "SavePoint", 0, 0;
					end;
				}

				delitem 601, 1;
				@AUTO_LAST_TX = 0;
				@AUTO_LAST_TY = 0;
				@AUTO_STUCK_TICKS = 0;
				unitstopattack getcharid(3);
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				warp @AUTO_MAP$, 0, 0;
				addtimer 1000, "AutoPlayerSystem::OnStepTimer";
				end;
			}
		}

		// Se nao tiver Asa de Mosca (601), mapa bloquear teleporte ou aguardar intervalo: caminha e explora ao redor
		if (@AUTO_LAST_TX == 0 || @AUTO_LAST_TY == 0 || (@AUTO_STUCK_X == .@cx && @AUTO_STUCK_Y == .@cy)) {
			@AUTO_LAST_TX = .@cx + rand(-8, 8);
			@AUTO_LAST_TY = .@cy + rand(-8, 8);
			unitstopwalk getcharid(3), USW_FORCE_STOP;
			unitwalk getcharid(3), @AUTO_LAST_TX, @AUTO_LAST_TY;
		}
		@AUTO_STUCK_X = .@cx;
		@AUTO_STUCK_Y = .@cy;
		addtimer 600, "AutoPlayerSystem::OnStepTimer";
		end;
	}

	// 4. Monstro Encontrado -> Aproximacao e Combate
	.@dist = distance(.@cx, .@cy, .@mx, .@my);
	.@atk_range = callsub(OnGetCombatRange);

	// Se estiver fora de alcance, caminha ate o alvo
	if (.@dist > .@atk_range) {
		// Anti-travamento se o jogador nao conseguir se mover ate o monstro
		if (@AUTO_STUCK_X == .@cx && @AUTO_STUCK_Y == .@cy) {
			@AUTO_STUCK_TICKS++;
			if (@AUTO_STUCK_TICKS >= 3) {
				// Travou tentando alcancar alvo (ex: rio, abismo, cerca): bloqueia alvo e desvia
				@AUTO_BLOCKED_INDEX = (@AUTO_BLOCKED_INDEX + 1) % 8;
				@AUTO_BLOCKED_MOB[@AUTO_BLOCKED_INDEX] = @AUTO_CURRENT_TARGET;
				@AUTO_BLOCKED_TIME[@AUTO_BLOCKED_INDEX] = .@now + 25;
				@AUTO_CURRENT_TARGET = 0;
				@AUTO_LAST_TX = 0;
				@AUTO_LAST_TY = 0;
				@AUTO_STUCK_TICKS = 0;
				unitstopattack getcharid(3);
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				unitwalk getcharid(3), .@cx + rand(-3, 3), .@cy + rand(-3, 3);
				addtimer 400, "AutoPlayerSystem::OnStepTimer";
				end;
			}
		} else {
			@AUTO_STUCK_TICKS = 0;
		}

		if (@AUTO_LAST_TX != .@mx || @AUTO_LAST_TY != .@my) {
			if (getstatus(SC_HIDING)) {
				sc_end SC_HIDING;
			}
			@AUTO_LAST_TX = .@mx;
			@AUTO_LAST_TY = .@my;
			unitstopwalk getcharid(3), USW_FORCE_STOP;
			.@walk_ok = unitwalk(getcharid(3), .@mx, .@my);
			if (!.@walk_ok) {
				// Monstro inalcancavel por caminho direto: bloqueia na hora e desvia
				@AUTO_BLOCKED_INDEX = (@AUTO_BLOCKED_INDEX + 1) % 8;
				@AUTO_BLOCKED_MOB[@AUTO_BLOCKED_INDEX] = @AUTO_CURRENT_TARGET;
				@AUTO_BLOCKED_TIME[@AUTO_BLOCKED_INDEX] = .@now + 25;
				@AUTO_CURRENT_TARGET = 0;
				@AUTO_LAST_TX = 0;
				@AUTO_LAST_TY = 0;
				@AUTO_STUCK_TICKS = 0;
				unitstopattack getcharid(3);
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				unitwalk getcharid(3), .@cx + rand(-3, 3), .@cy + rand(-3, 3);
				addtimer 400, "AutoPlayerSystem::OnStepTimer";
				end;
			}
		}

		@AUTO_STUCK_X = .@cx;
		@AUTO_STUCK_Y = .@cy;
		addtimer 400, "AutoPlayerSystem::OnStepTimer";
		end;
	}

	// Chegou no alcance de ataque! Para de andar
	if (@AUTO_LAST_TX != 0 || @AUTO_LAST_TY != 0) {
		@AUTO_LAST_TX = 0;
		@AUTO_LAST_TY = 0;
		unitstopwalk getcharid(3), USW_FORCE_STOP;
	}
	@AUTO_STUCK_TICKS = 0;
	@AUTO_STUCK_X = .@cx;
	@AUTO_STUCK_Y = .@cy;

	// Inicia tracking do HP para monitoramento de travamento
	if (@AUTO_TARGET_TRACK_ID != @AUTO_CURRENT_TARGET) {
		getunitdata @AUTO_CURRENT_TARGET, .@atk_data;
		@AUTO_TARGET_TRACK_ID = @AUTO_CURRENT_TARGET;
		@AUTO_TARGET_LAST_HP = .@atk_data[UMOB_HP];
		@AUTO_TARGET_PROGRESS_TICK = .@now;
	}

	// Engata ataque fisico basico apenas se estiver no alcance da arma equipada
	.@w_id = getequipid(EQI_HAND_R);
	.@w_range = 1;
	if (.@w_id > 0) {
		.@w_range = getiteminfo(.@w_id, ITEMINFO_RANGE);
		if (.@w_range <= 0) .@w_range = 1;
	}
	if (!getstatus(SC_HIDING)) {
		if (.@dist <= .@w_range || .@w_range >= .@atk_range) {
			if (@AUTO_CURRENT_ATTACK_TARGET != @AUTO_CURRENT_TARGET) {
				@AUTO_CURRENT_ATTACK_TARGET = @AUTO_CURRENT_TARGET;
				unitattack getcharid(3), @AUTO_CURRENT_TARGET, 1;
			}
		}
	}
	@AUTO_SKILL_ROT++;

	// ==================================================================
	// ROTACAO DE HABILIDADES NATIVAS DA ARVORE DO JOGADOR
	// ==================================================================
	.@cur_aspd = readparam(bAspd);
	if (.@cur_aspd < 100) .@cur_aspd = 100;
	if (.@cur_aspd > 195) .@cur_aspd = 195;
	.@step_delay = (200 - .@cur_aspd) * 15;
	if (.@step_delay < 350) .@step_delay = 350;

	// 1. ESPADACHIM E EVOLUCOES (Inclui Espadachim T / Swordman High)
	if (BaseClass == Job_Swordman || Class == Job_Swordman || Class == Job_Swordman_High || BaseJob == Job_Swordman || BaseJob == Job_Swordman_High) {
		// Cavaleiro Runico: Se Sopro de Gelo estiver no lvl 10, usa exclusivamente Sopro do Dragao e Sopro de Gelo
		if (getskilllv("RK_DRAGONBREATH_WATER") >= 10) {
			if (Sp >= 30) {
				if ((@AUTO_SKILL_ROT % 2 == 0) && getskilllv("RK_DRAGONBREATH") > 0) {
					unitskillusepos getcharid(3), "RK_DRAGONBREATH", getskilllv("RK_DRAGONBREATH"), .@mx, .@my;
					.@step_delay = 850;
				} else {
					unitskillusepos getcharid(3), "RK_DRAGONBREATH_WATER", getskilllv("RK_DRAGONBREATH_WATER"), .@mx, .@my;
					.@step_delay = 850;
				}
			}
		}
		// Cavaleiro Runico Fisico:
		// 1. Prioridade: Impacto Flamejante (alcance seguro <= 3 celulas)
		// 2. Cooldown de Impacto: Onda de Choque
		// 3. Nem Impacto nem Onda disponivel: Vento Cortante (alcance seguro <= 2 celulas)
		else if (getskilllv("RK_IGNITIONBREAK") > 0 && Sp >= 35 && .@dist <= 3 && .@now >= @CD_RK_IGNITIONBREAK) {
			@CD_RK_IGNITIONBREAK = .@now + 2;
			unitskilluseid getcharid(3), "RK_IGNITIONBREAK", getskilllv("RK_IGNITIONBREAK"), getcharid(3), 0, 0, 0, 0;
			.@step_delay = 900;
		} else if (getskilllv("RK_SONICWAVE") > 0 && Sp >= (30 + getskilllv("RK_SONICWAVE") * 3) && .@dist <= 8 && .@now >= @CD_RK_SONICWAVE && (.@now < @CD_RK_IGNITIONBREAK || .@dist > 3 || getskilllv("RK_IGNITIONBREAK") <= 0)) {
			@CD_RK_SONICWAVE = .@now + 2;
			unitskilluseid getcharid(3), "RK_SONICWAVE", getskilllv("RK_SONICWAVE"), @AUTO_CURRENT_TARGET, 0, 0, 0, 0;
			.@step_delay = 550;
		} else if (getskilllv("RK_WINDCUTTER") > 0 && Sp >= 20 && .@dist <= 2) {
			unitskilluseid getcharid(3), "RK_WINDCUTTER", getskilllv("RK_WINDCUTTER"), getcharid(3), 0, 0, 0, 0;
			.@step_delay = 500;
		} else if (getskilllv("RK_DRAGONBREATH") > 0 && Sp >= 30) {
			unitskillusepos getcharid(3), "RK_DRAGONBREATH", getskilllv("RK_DRAGONBREATH"), .@mx, .@my;
			.@step_delay = 900;
		} else if (getskilllv("LG_BANISHINGPOINT") > 0 && Sp >= 20) {
			unitskilluseid getcharid(3), "LG_BANISHINGPOINT", getskilllv("LG_BANISHINGPOINT"), @AUTO_CURRENT_TARGET, 0, 0, 0, 0;
			.@step_delay = 450;
		} else if (getskilllv("LG_CANNONSPEAR") > 0 && Sp >= 30) {
			unitskilluseid getcharid(3), "LG_CANNONSPEAR", getskilllv("LG_CANNONSPEAR"), @AUTO_CURRENT_TARGET, 0, 0, 0, 0;
			.@step_delay = 700;
		} else if (getskilllv("LK_SPIRALPIERCE") > 0 && Sp >= 20) {
			unitskilluseid getcharid(3), "LK_SPIRALPIERCE", getskilllv("LK_SPIRALPIERCE"), @AUTO_CURRENT_TARGET, 0, 0, 0, 0;
			.@step_delay = 800;
		} else if (getskilllv("PA_SHIELDCHAIN") > 0 && Sp >= (25 + getskilllv("PA_SHIELDCHAIN") * 3) && getequipid(EQI_HAND_L) > 0) {
			unitskilluseid getcharid(3), "PA_SHIELDCHAIN", getskilllv("PA_SHIELDCHAIN"), @AUTO_CURRENT_TARGET, 0, 0, 0, 0;
			.@step_delay = 800;
		} else if (getskilllv("CR_HOLYCROSS") > 0 && Sp >= 15) {
			unitskilluseid getcharid(3), "CR_HOLYCROSS", getskilllv("CR_HOLYCROSS"), @AUTO_CURRENT_TARGET, 0, 0, 0, 0;
			.@step_delay = 500;
		} else if (getskilllv("KN_BOWLINGBASH") > 0 && Sp >= 15) {
			unitskilluseid getcharid(3), "KN_BOWLINGBASH", getskilllv("KN_BOWLINGBASH"), @AUTO_CURRENT_TARGET, 0, 0, 0, 0;
			.@step_delay = 600;
		} else if (getskilllv("SM_MAGNUM") > 0 && Sp >= 30 && (@AUTO_SKILL_ROT % 3 == 0)) {
			unitskillusepos getcharid(3), "SM_MAGNUM", getskilllv("SM_MAGNUM"), .@cx, .@cy;
			.@step_delay = 650;
		} else if (getskilllv("SM_BASH") > 0 && Sp >= 8) {
			unitskilluseid getcharid(3), "SM_BASH", getskilllv("SM_BASH"), @AUTO_CURRENT_TARGET, 0, 0, 0, 0;
			.@step_delay = 450;
		}
	}
	// 2. MAGO E EVOLUCOES (Inclui Mago T / Mage High)
	else if (BaseClass == Job_Mage || Class == Job_Mage || Class == Job_Mage_High || BaseJob == Job_Mage || BaseJob == Job_Mage_High) {
		// Arquimago e evolucoes: Mantem Amplificacao Mistica ativa antes de conjurar magias ofensivas
		if (getskilllv("HW_MAGICPOWER") > 0 && Sp >= 40 && !getstatus(SC_MAGICPOWER) && .@now >= @BUFF_HW_MAGICPOWER + 2) {
			@BUFF_HW_MAGICPOWER = .@now;
			unitskilluseid getcharid(3), "HW_MAGICPOWER", getskilllv("HW_MAGICPOWER"), getcharid(3), 0, 0, 0, 0;
			.@step_delay = 450;
			addtimer .@step_delay, "AutoPlayerSystem::OnStepTimer";
