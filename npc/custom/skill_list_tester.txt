// NPC de teste defensivo.
// Fluxo simplificado:
// - Debuff opcional
// - Dano fisico ou magico
// - Fisico usa skill neutra fixa
// - Magico escolhe a skill pelo elemento
// - Mantem filtros de tamanho, raca e modo do mob

prontera,156,278,4	script	Resistence Tester	627,{
	mes "[Skill Tester]";
	mes "Vamos configurar um atacante para testar sua defesa.";
	next;

	.@reuse_last = 0;
	if (SkillTest_Last_Valid) {
		mes "[Skill Tester]";
		mes "Deseja reutilizar a ultima configuracao?";
		next;
		.@reuse_last = (select("Reutilizar ultima","Configurar de novo") == 1);
	}

	.@debuff_mode = 0;
	.@debuff_skill$ = "";
	.@debuff_label$ = "Nenhum";
	.@debuff_lv = 0;
	.@damage_enabled = 1;
	.@d_idx = 0;

	setarray .@debuff_label_list$[0],
		"Dispel",
		"Congelamento",
		"Stun",
		"Petrificacao",
		"Silencio",
		"Cegueira",
		"Maldicao";

	setarray .@debuff_skill_list$[0],
		"SA_DISPELL",
		"MG_FROSTDIVER",
		"NPC_STUNATTACK",
		"MG_STONECURSE",
		"NPC_SILENCEATTACK",
		"NPC_BLINDATTACK",
		"NPC_CURSEATTACK";

	setarray .@debuff_lv_list[0],
		5,10,5,10,5,5,5;

	.@skill_label$ = "Nenhum";
	.@skill_name$ = "";
	.@skill_id = 0;
	.@skill_lv = 0;
	.@real_skill_lv = 0;
	.@skill_cap = 5;
	.@damage_type = 0;
	.@mob_ele_name$ = "Neutro";
	.@mob_ele = ELE_NEUTRAL;
	.@e_idx = 0;

	setarray .@ele_name$,
		"Neutro","Agua","Terra","Fogo","Vento",
		"Veneno","Sombrio","Sagrado","Fantasma","Morto-Vivo";
	setarray .@ele_value,
		ELE_NEUTRAL,ELE_WATER,ELE_EARTH,ELE_FIRE,ELE_WIND,
		ELE_POISON,ELE_DARK,ELE_HOLY,ELE_GHOST,ELE_UNDEAD;

	setarray .@size_name$,
		"Pequeno","Medio","Grande";
	setarray .@size_value,
		0,1,2;

	setarray .@race_name$,
		"Amorfo","Morto-Vivo","Bruto","Planta","Inseto",
		"Peixe","Demonio","Humanoide","Anjo","Dragao";
	setarray .@race_value,
		RC_FORMLESS,RC_UNDEAD,RC_BRUTE,RC_PLANT,RC_INSECT,
		RC_FISH,RC_DEMON,RC_DEMIHUMAN,RC_ANGEL,RC_DRAGON;

	if (.@reuse_last) {
		.@debuff_mode = SkillTest_Last_DebuffMode;
		.@d_idx = SkillTest_Last_DebuffIdx;
		.@damage_type = SkillTest_Last_DamageType;
		.@skill_lv = SkillTest_Last_SkillLv;
		.@mode_sel = SkillTest_Last_ModeSel;
		.@s_idx = SkillTest_Last_SizeIdx;
		.@r_idx = SkillTest_Last_RaceIdx;
		.@e_idx = SkillTest_Last_EleIdx;
	} else {
		.@debuff_mode = select("Sem debuff","Somente debuff","Debuff com dano") - 1;
	}

	if (.@debuff_mode == 1)
		.@damage_enabled = 0;

	if (.@debuff_mode > 0) {
		if (!.@reuse_last) {
			mes "[Skill Tester]";
			mes "Escolha o debuff.";
			next;

			.@d_idx = select(
				"Dispel",
				"Congelamento",
				"Stun",
				"Petrificacao",
				"Silencio",
				"Cegueira",
				"Maldicao"
			) - 1;
		}

		.@debuff_label$ = .@debuff_label_list$[.@d_idx];
		.@debuff_skill$ = .@debuff_skill_list$[.@d_idx];
		.@debuff_lv = .@debuff_lv_list[.@d_idx];
	}

	if (.@damage_enabled) {
		if (!.@reuse_last) {
			mes "[Skill Tester]";
			mes "Escolha o tipo de dano.";
			next;

			.@damage_type = select("Fisico","Magico") - 1;
		}
		.@skill_max_lv = 5;

		if (!.@reuse_last) {
			mes "[Skill Tester]";
			mes "Escolha a intensidade do teste.";
			next;

			.@lv_menu$ = "";
			for (.@i = 1; .@i <= .@skill_max_lv; ++.@i)
				.@lv_menu$ += "Lv " + .@i + ":";

			.@skill_lv = select(.@lv_menu$);
		}

		if (.@damage_type == 0) {
			.@skill_label$ = "Earthquake";
			.@skill_name$ = "NPC_EARTHQUAKE";
			.@skill_id = 653;
			.@skill_cap = 10;
		} else {
			.@skill_id = 0;
			if (!.@reuse_last) {
				mes "[Skill Tester]";
				mes "Escolha o elemento da skill magica.";
				next;

				.@e_idx = select(
					"Neutro",
					"Agua",
					"Terra",
					"Fogo",
					"Vento",
					"Veneno",
					"Sombrio",
					"Sagrado",
					"Fantasma",
					"Morto-Vivo"
				) - 1;
			}

			.@mob_ele_name$ = .@ele_name$[.@e_idx];
			.@mob_ele = .@ele_value[.@e_idx];
			.@skill_cap = 5;

			switch (.@mob_ele) {
				case ELE_NEUTRAL:
					.@skill_label$ = "Drain Life";
					.@skill_name$ = "WL_DRAINLIFE";
					.@skill_cap = 5;
					break;
				case ELE_WATER:
					.@skill_label$ = "Jack Frost";
					.@skill_name$ = "WL_JACKFROST";
					.@skill_cap = 5;
					break;
				case ELE_EARTH:
					.@skill_label$ = "Earth Strain";
					.@skill_name$ = "";
					.@skill_id = 2216;
					.@skill_cap = 5;
					break;
				case ELE_FIRE:
					.@skill_label$ = "Crimson Rock";
					.@skill_name$ = "WL_CRIMSONROCK";
					.@skill_cap = 5;
					break;
				case ELE_WIND:
					.@skill_label$ = "Chain Lightning";
					.@skill_name$ = "WL_CHAINLIGHTNING";
					.@skill_cap = 5;
					break;
				case ELE_POISON:
					.@skill_label$ = "Poison Buster";
					.@skill_name$ = "SO_POISON_BUSTER";
					.@skill_cap = 5;
					break;
				case ELE_HOLY:
					.@skill_label$ = "Adoramus";
					.@skill_name$ = "AB_ADORAMUS";
					.@skill_cap = 10;
					break;
				case ELE_GHOST:
					.@skill_label$ = "Soul Expansion";
					.@skill_name$ = "WL_SOULEXPANSION";
					.@skill_cap = 5;
					break;
				case ELE_DARK:
					.@skill_label$ = "Curse Explosion";
					.@skill_name$ = "SP_CURSEEXPLOSION";
					.@skill_id = 0;
					.@skill_cap = 10;
					break;
				case ELE_UNDEAD:
					.@skill_label$ = "Undead Attack";
					.@skill_name$ = "NPC_UNDEADATTACK";
					.@skill_cap = 5;
					break;
				default:
					.@skill_label$ = "Soul Expansion";
					.@skill_name$ = "WL_SOULEXPANSION";
					.@skill_cap = 5;
					break;	
			}
		}

		.@real_skill_lv = ((.@skill_lv * .@skill_cap) + 4) / 5;
		if (.@real_skill_lv < 1)
			.@real_skill_lv = 1;
		if (.@real_skill_lv > .@skill_cap)
			.@real_skill_lv = .@skill_cap;
	}

	if (!.@reuse_last) {
		mes "[Skill Tester]";
		mes "O atacante sera mob normal ou MVP?";
		next;

		.@mode_sel = select("Mob normal","MVP") - 1;
	}
	.@mob_mode = MD_NONE;
	.@mob_mode_name$ = "Mob normal";
	if (.@mode_sel) {
		.@mob_mode |= MD_MVP|MD_STATUSIMMUNE|MD_KNOCKBACKIMMUNE|MD_DETECTOR;
		.@mob_mode_name$ = "MVP";
	}

	if (!.@reuse_last) {
		mes "[Skill Tester]";
		mes "Escolha o tamanho do atacante.";
		next;

		.@s_idx = select("Pequeno","Medio","Grande") - 1;
	}

	.@mob_size_name$ = .@size_name$[.@s_idx];
	.@mob_size = .@size_value[.@s_idx];

	if (!.@reuse_last) {
		mes "[Skill Tester]";
		mes "Escolha a raca do atacante.";
		next;

		.@r_idx = select(
			"Amorfo",
			"Morto-Vivo",
			"Bruto",
			"Planta",
			"Inseto",
			"Peixe",
			"Demonio",
			"Humanoide",
			"Anjo",
			"Dragao"
		) - 1;
	}

	.@mob_race_name$ = .@race_name$[.@r_idx];
	.@mob_race = .@race_value[.@r_idx];

	mes "[Resumo]";
	mes "Debuff   : " + .@debuff_label$;
	if (.@damage_enabled)
		mes "Dano     : " + .@skill_label$ + " Lv " + .@real_skill_lv;
	else
		mes "Dano     : Nenhum";
	mes "Tipo     : " + .@mob_mode_name$;
	mes "Tamanho  : " + .@mob_size_name$;
	mes "Raca     : " + .@mob_race_name$;
	if (.@damage_enabled && .@damage_type == 1)
		mes "Elemento : " + .@mob_ele_name$ + " 4";
	else
		mes "Elemento : Neutro";
	next;

	if (select("Executar teste","Cancelar") == 2)
		close;

	SkillTest_Last_Valid = 1;
	SkillTest_Last_DebuffMode = .@debuff_mode;
	SkillTest_Last_DebuffIdx = .@d_idx;
	SkillTest_Last_DamageType = .@damage_type;
	SkillTest_Last_SkillLv = .@skill_lv;
	SkillTest_Last_ModeSel = .@mode_sel;
	SkillTest_Last_SizeIdx = .@s_idx;
	SkillTest_Last_RaceIdx = .@r_idx;
	SkillTest_Last_EleIdx = .@e_idx;

	getmapxy(.@map$, .@x, .@y, BL_PC);
	.@target_aid = getcharid(3);

	monster .@map$, .@x, .@y, "Atacante de Teste", .mob_base_id, 1;
	.@gid = $@mobid[0];

	if (!.@gid) {
		dispbottom "Nao foi possivel criar o atacante de teste.";
		close;
	}

	setunitdata .@gid, UMOB_SIZE, .@mob_size;
	setunitdata .@gid, UMOB_RACE, .@mob_race;
	setunitdata .@gid, UMOB_MODE, .@mob_mode;
	setunitdata .@gid, UMOB_DMGIMMUNE, 1;
	setunitdata .@gid, UMOB_SCOPTION, OPTION_INVISIBLE;
	setunitdata .@gid, UMOB_LOOKDIR, 5;
	if (.@damage_enabled && .@damage_type == 1) {
		setunitdata .@gid, UMOB_ELETYPE, .@mob_ele;
		setunitdata .@gid, UMOB_ELELEVEL, 4;
	} else {
		setunitdata .@gid, UMOB_ELETYPE, ELE_NEUTRAL;
		setunitdata .@gid, UMOB_ELELEVEL, 4;
	}

	close2;

	if (!attachrid(.@target_aid)) {
		unitkill .@gid;
		end;
	}

	if (.@debuff_skill$ != "") {
		unitskilluseid .@gid, .@debuff_skill$, .@debuff_lv, .@target_aid, -100, 0, 0, 1;
		sleep 200;
	}

	if (.@damage_enabled) {
		if (.@skill_id == 653 || .@skill_id == 2216)
			unitskillusepos .@gid, .@skill_id, .@real_skill_lv, .@x, .@y, -100, 0, 0, 1;
		else if (.@skill_id)
			unitskilluseid .@gid, .@skill_id, .@real_skill_lv, .@target_aid, -100, 0, 0, 1;
		else
			unitskilluseid .@gid, .@skill_name$, .@real_skill_lv, .@target_aid, -100, 0, 0, 1;
		sleep 1200;
	}

	unitkill .@gid;
	end;

OnInit:
	.mob_base_id = 20273;
	end;
}
