function	script	F_Angelical_IsTarget	{
	.@item_id = getarg(0);
	if (.@item_id >= 15397 && .@item_id <= 15402) return 1;
	if (.@item_id == 20947 || .@item_id == 22209) return 1;
	return 0;
}

function	script	F_Angelical_ItemGroup	{
	.@item_id = getarg(0);
	if (.@item_id >= 15397 && .@item_id <= 15402) return 1;
	if (.@item_id == 20947 || .@item_id == 22209) return 2;
	return 0;
}

function	script	F_Angelical_Roll	{
	.@pool = getarg(0);
	deletearray .@item_ids[0], getarraysize(.@item_ids);
	deletearray .@rates[0], getarraysize(.@rates);

	switch (.@pool) {
		case 1:
			setarray .@item_ids[0],4705,4706,4707,4764,4765,29241,4762,29238,4794,4902,4786,4787,4867,4900,4801,4802;
			for (.@i = 0; .@i < getarraysize(.@item_ids); .@i++)
				.@rates[.@i] = 100;
			break;

		case 2:
			setarray .@item_ids[0],4706,4820,4812,4835,4843,4802,29238,29241,4903,4790,4900,4873,4707,4821,4826,4836,4844,4881,310076,310077,310078,310080,310079,310081;
			setarray .@rates[0],952,595,595,595,595,202,178,142,119,119,119,119,95,59,59,59,59,23,11,11,11,11,11,11;
			break;

		case 3:
			setarray .@item_ids[0],4707,4900,4802,29238,29241,4903,4790,4820,4812,4835,4873,4843,4821,4826,4836,4844,4881,310076,310077,310078,310080,310079,310081;
			setarray .@rates[0],692,461,461,461,415,346,346,300,300,300,300,300,230,230,230,230,230,115,115,115,115,115,115;
			break;

		case 4:
			setarray .@item_ids[0],4706,4707,4765,29241,4762,29238,4902,29026,4788,4789,4867,4900,4801,4802;
			for (.@i = 0; .@i < getarraysize(.@item_ids); .@i++)
				.@rates[.@i] = 100;
			break;

		case 5:
			setarray .@item_ids[0],4707,4765,29241,4762,29238,4902,29026,4788,4789,4867,4900,4801,4802;
			for (.@i = 0; .@i < getarraysize(.@item_ids); .@i++)
				.@rates[.@i] = 100;
			break;

		default:
			return 0;
	}

	.@total = 0;
	for (.@i = 0; .@i < getarraysize(.@item_ids); .@i++)
		.@total += .@rates[.@i];

	.@roll = rand(.@total);
	.@acc = 0;
	for (.@i = 0; .@i < getarraysize(.@item_ids); .@i++) {
		.@acc += .@rates[.@i];
		if (.@roll < .@acc)
			return .@item_ids[.@i];
	}

	return .@item_ids[0];
}
