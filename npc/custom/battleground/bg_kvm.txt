-	script	KvM01_CMD_Queue	-1,{
OnInit:
	bindatcmd "bg", "KvM01_CMD_Queue::OnJoin";

	disablenpc "Croix";
	disablenpc "Guillaume";	

	// Recupera o estado apos @reloadscript dentro do horario do evento.
	if (callsub(S_IsBGOpenNow)) {
		.bg_status = 1;
		enablenpc "Croix";
		enablenpc "Guillaume";
	}
	else
		.bg_status = 0;
	
	end;

OnClock1200:
OnClock1500:
OnClock1800:
	if (!callsub(S_IsBGOpenNow)) end;
	if (.bg_status) end;

	.bg_status = 1;
	enablenpc "Croix";
	enablenpc "Guillaume";	
	announce "Battleground aberto! Use @bg.", bc_all, 0xFF0000;
	end;

OnClock1300:
OnClock1900:
OnClock2000:
	if (callsub(S_IsBGOpenNow)) end;
	if (.bg_status == 0) end;

	announce "Battleground encerrado.", bc_all;
	.bg_status = 0;
	$@KvM01_CMD_Ready = 0;
	for (.@i = 0; .@i < getarraysize($@KvM01_CMD_CID); .@i++) {
		if (!isloggedin($@KvM01_CMD_AID[.@i], $@KvM01_CMD_CID[.@i]))
			continue;
		if (attachrid($@KvM01_CMD_AID[.@i])) {
			kvm_cmd_queue = 0;
			detachrid;
		}
	}
	deletearray $@KvM01_CMD_CID;
	deletearray $@KvM01_CMD_AID;
	deletearray $@KvM01_CMD_TEAM;
	disablenpc "Croix";
	disablenpc "Guillaume";	
	end;

S_IsBGOpenNow:
	.@day = gettime(DT_DAYOFWEEK);
	.@hour = gettime(DT_HOUR);

	// Sabado e domingo: 15:00 as 20:00.
	if ((.@day == 0 || .@day == 6) && .@hour >= 15 && .@hour < 20)
		return 1;

	// Segunda a sexta: 12:00 as 13:00 e 18:00 as 19:00.
	if (.@day >= 1 && .@day <= 5) {
		if ((.@hour >= 12 && .@hour < 13) || (.@hour >= 18 && .@hour < 19))
			return 1;
	}

	return 0;
OnJoin:
    if (.bg_status == 0) {
        dispbottom "O Battleground esta fechado.";
        end;
    }

	if ($@KvM01BG) {
		dispbottom "O Battleground ja esta em andamento. Aguarde a proxima rodada.";
		end;
	}

	if (getcharid(4)) {
		dispbottom "Voce ja esta em um Battleground.";
		end;
	}

	.@cid = getcharid(0);
	.@aid = getcharid(3);

	if (inarray($@KvM01_CMD_CID, .@cid) >= 0) {
		dispbottom "Voce ja esta na fila do Battleground.";
		end;
	}

	callsub S_CleanupQueue;

	.@my_ip$ = getcharip();
	for (.@i = 0; .@i < getarraysize($@KvM01_CMD_CID); .@i++) {
		if (!isloggedin($@KvM01_CMD_AID[.@i], $@KvM01_CMD_CID[.@i]))
			continue;

		.@other_ip$ = getcharip($@KvM01_CMD_AID[.@i]);
		if (.@other_ip$ != "" && .@other_ip$ == .@my_ip$) {
			dispbottom "Entrada bloqueada: ja existe uma conta com o mesmo IP na fila do KVM.";
			end;
		}
	}

	.@team1 = callsub(S_CountQueue, 1);
	.@team2 = callsub(S_CountQueue, 2);

	if (.@team1 >= 5 && .@team2 >= 5) {
		dispbottom "A fila do Battleground esta cheia. Aguarde a proxima rodada.";
		end;
	}

	if (.@team1 <= .@team2)
		.@team = 1;
	else
		.@team = 2;

	if (.@team == 1 && .@team1 >= 5)
		.@team = 2;
	else if (.@team == 2 && .@team2 >= 5)
		.@team = 1;

	.@idx = getarraysize($@KvM01_CMD_CID);
	$@KvM01_CMD_CID[.@idx] = .@cid;
	$@KvM01_CMD_AID[.@idx] = .@aid;
	$@KvM01_CMD_TEAM[.@idx] = .@team;
	kvm_cmd_queue = .@team;

	if (.@team == 1)
		dispbottom "Voce entrou na fila KVM pelo time Guillaume. Continue jogando, voce sera chamado quando iniciar.";
	else
		dispbottom "Voce entrou na fila KVM pelo time Croix. Continue jogando, voce sera chamado quando iniciar.";

	announce "[BG] "+strcharinfo(0)+" entrou na fila do Battleground. Use @bg para participar!", bc_all, 0x00FF00;

	callsub S_CheckStart;
	end;

OnPCLogoutEvent:
	if (!kvm_cmd_queue) end;
	callsub S_RemoveFromQueue, getcharid(0);
	end;

S_CheckStart:
	callsub S_CleanupQueue;

	.@my_ip$ = getcharip();
	for (.@i = 0; .@i < getarraysize($@KvM01_CMD_CID); .@i++) {
		if (!isloggedin($@KvM01_CMD_AID[.@i], $@KvM01_CMD_CID[.@i]))
			continue;

		.@other_ip$ = getcharip($@KvM01_CMD_AID[.@i]);
		if (.@other_ip$ != "" && .@other_ip$ == .@my_ip$) {
			dispbottom "Entrada bloqueada: ja existe uma conta com o mesmo IP na fila do KVM.";
			end;
		}
	}

	.@team1 = callsub(S_CountQueue, 1);
	.@team2 = callsub(S_CountQueue, 2);

	if (.@team1 < 1 || .@team2 < 1)
		return;

	if (.@team1 >= 5 && .@team2 >= 5) {
		announce "KVM 5x5 alcancado! A batalha comecara agora!", bc_all, 0x00FF00;
		callsub S_StartCommandQueue;
		return;
	}

	if (!$@KvM01_CMD_Ready) {
		$@KvM01_CMD_Ready = 1;
		announce "Minimo do KVM atingido! A batalha comecara em 30 segundos. Use @bg para entrar na fila.", bc_all, 0x00FF00;
		stopnpctimer;
		initnpctimer;
	}
	return;

OnTimer30000:
	stopnpctimer;
	$@KvM01_CMD_Ready = 0;
	callsub S_CleanupQueue;

	.@my_ip$ = getcharip();
	for (.@i = 0; .@i < getarraysize($@KvM01_CMD_CID); .@i++) {
		if (!isloggedin($@KvM01_CMD_AID[.@i], $@KvM01_CMD_CID[.@i]))
			continue;

		.@other_ip$ = getcharip($@KvM01_CMD_AID[.@i]);
		if (.@other_ip$ != "" && .@other_ip$ == .@my_ip$) {
			dispbottom "Entrada bloqueada: ja existe uma conta com o mesmo IP na fila do KVM.";
			end;
		}
	}

	.@team1 = callsub(S_CountQueue, 1);
	.@team2 = callsub(S_CountQueue, 2);

	if (.@team1 < 1 || .@team2 < 1) {
		announce "KVM cancelado: jogadores insuficientes. Use @bg para entrar na fila.", bc_all, 0xFF0000;
		end;
	}

	callsub S_StartCommandQueue;
	end;

S_StartCommandQueue:
	if ($@KvM01BG)
		return;

	$@KvM01BG = 1;
	$@KvM01_CMD_Ready = 0;
	$@KvM01BG_Ready = 0;

	$@KvM01BG_id1 = bg_create("bat_c01",146,55,"KvM01_BG::OnGuillaumeQuit","KvM01_BG::OnGuillaumeDie");
	$@KvM01BG_id2 = bg_create("bat_c01",53,128,"KvM01_BG::OnCroixQuit","KvM01_BG::OnCroixDie");

	for (.@i = 0; .@i < getarraysize($@KvM01_CMD_CID); .@i++) {
		if (!isloggedin($@KvM01_CMD_AID[.@i], $@KvM01_CMD_CID[.@i]))
			continue;

		if (!attachrid($@KvM01_CMD_AID[.@i]))
			continue;

		kvm_bg_streak_total = 0;
		kvm_bg_streak_combo = 0;
		kvm_bg_lastkill_time = 0;

		if ($@KvM01_CMD_TEAM[.@i] == 1) {
			Bat_Team = 1;
			kvm_cmd_queue = 0;
			bg_join $@KvM01BG_id1, "bat_c01", 53, 128;
		}
		else {
			Bat_Team = 2;
			kvm_cmd_queue = 0;
			bg_join $@KvM01BG_id2, "bat_c01", 146, 55;
		}

		dispbottom "Sua batalha KVM comecou!";
		detachrid;
	}

	deletearray $@KvM01_CMD_CID;
	deletearray $@KvM01_CMD_AID;
	deletearray $@KvM01_CMD_TEAM;

	donpcevent "KvM01_BG::OnStart";
	return;

S_CountQueue:
	.@team = getarg(0);
	.@count = 0;

	for (.@i = 0; .@i < getarraysize($@KvM01_CMD_CID); .@i++) {
		if ($@KvM01_CMD_TEAM[.@i] != .@team)
			continue;
		if (!isloggedin($@KvM01_CMD_AID[.@i], $@KvM01_CMD_CID[.@i]))
			continue;
		.@count++;
	}

	return .@count;

S_CleanupQueue:
	for (.@i = getarraysize($@KvM01_CMD_CID) - 1; .@i >= 0; .@i--) {
		if (isloggedin($@KvM01_CMD_AID[.@i], $@KvM01_CMD_CID[.@i]))
			continue;

		deletearray $@KvM01_CMD_CID[.@i], 1;
		deletearray $@KvM01_CMD_AID[.@i], 1;
		deletearray $@KvM01_CMD_TEAM[.@i], 1;
	}
	return;

S_RemoveFromQueue:
	.@cid = getarg(0);
	.@idx = inarray($@KvM01_CMD_CID, .@cid);

	if (.@idx >= 0) {
		deletearray $@KvM01_CMD_CID[.@idx], 1;
		deletearray $@KvM01_CMD_AID[.@idx], 1;
		deletearray $@KvM01_CMD_TEAM[.@idx], 1;
		kvm_cmd_queue = 0;
	}
	return;
}

-	script	KvM01_BG	-1,{
	end;

OnPCKillEvent:
	if (strcharinfo(3) != "bat_c01") end;
	if ($@KvM01BG != 2) end;

	.@killer_aid = getcharid(3);
	.@victim_aid = killedrid;
	.@now = gettimetick(2);

	if (.@killer_aid == .@victim_aid) end;

	.@killer$ = strcharinfo(0);
	.@victim$ = rid2name(.@victim_aid);
	if (.@victim$ == "")
		.@victim$ = "um jogador";

	announce "[BG] "+.@killer$+" matou "+.@victim$+"!", bc_all, 0xFF0000;

	if ((.@now - kvm_bg_lastkill_time) <= 10)
		kvm_bg_streak_combo++;
	else
		kvm_bg_streak_combo = 1;

	kvm_bg_streak_total++;

	.@played_combo = 0;
	if (kvm_bg_streak_combo == 2) {
		announce "[BG] "+.@killer$+" fez DOUBLE KILL!", bc_all, 0xFF0000;
		soundeffectall "kills/double_kill.wav", 0, strcharinfo(3);
		.@played_combo = 1;
	}
	else if (kvm_bg_streak_combo == 3) {
		announce "[BG] "+.@killer$+" fez TRIPLE KILL!", bc_all, 0xFF0000;
		soundeffectall "kills/triple_kill.wav", 0, strcharinfo(3);
		.@played_combo = 1;
	}
	else if (kvm_bg_streak_combo == 4) {
		announce "[BG] "+.@killer$+" fez MANIAC!!", bc_all, 0xFF0000;
		soundeffectall "kills/maniac.wav", 0, strcharinfo(3);
		.@played_combo = 1;
	}
	else if (kvm_bg_streak_combo >= 5) {
		announce "[BG] "+.@killer$+" fez SAVAGE!!!", bc_all, 0xFF0000;
		soundeffectall "kills/savage.wav", 0, strcharinfo(3);
		.@played_combo = 1;
	}

	if (!.@played_combo) {
		switch (kvm_bg_streak_total) {
			case 3:
				announce "[BG] "+.@killer$+" esta em KILLING SPREE!", bc_all, 0xFF0000;
				soundeffectall "kills/killing_spree.wav", 0, strcharinfo(3);
				break;
			case 4:
				announce "[BG] "+.@killer$+" esta em MEGA KILL!", bc_all, 0xFF0000;
				soundeffectall "kills/mega_kill.wav", 0, strcharinfo(3);
				break;
			case 5:
				announce "[BG] "+.@killer$+" esta UNSTOPPABLE!!", bc_all, 0xFF0000;
				soundeffectall "kills/unstoppable.wav", 0, strcharinfo(3);
				break;
			case 6:
				announce "[BG] "+.@killer$+" fez MONSTER KILL!!!", bc_all, 0xFF0000;
				soundeffectall "kills/monster_kill.wav", 0, strcharinfo(3);
				break;
			case 7:
				announce "[BG] "+.@killer$+" esta GODLIKE!!!", bc_all, 0xFF0000;
				soundeffectall "kills/godlike.wav", 0, strcharinfo(3);
				break;
			default:
				if (kvm_bg_streak_total >= 8) {
					announce "[BG] "+.@killer$+" esta LEGENDARY!!!!", bc_all, 0xFF0000;
					soundeffectall "kills/legendary.wav", 0, strcharinfo(3);
				}
				break;
		}
	}

	kvm_bg_lastkill_time = .@now;

	if (attachrid(.@victim_aid)) {
		if (kvm_bg_streak_total >= 3) {
			announce "[BG] "+strcharinfo(0)+" foi ELIMINADO APOS UMA SEQUENCIA DE "+kvm_bg_streak_total+" KILLS!!", bc_all, 0xFF0000;
			soundeffectall "kills/shutdown.wav", 0, "bat_c01";
		}
		kvm_bg_streak_total = 0;
		kvm_bg_streak_combo = 0;
		kvm_bg_lastkill_time = 0;
		detachrid;
	}
	attachrid(.@killer_aid);
	end;

OnInit:
	if (getbattleflag("feature.bgqueue"))
		bg_unbook "bat_c01";
		getmapunits(BL_PC, "bat_c01", .@units);
	for (.@i = 0; .@i < getarraysize(.@units); .@i++) {

	setwall "bat_c01",54,122,6,7,0,"batc01wall_a";

OnGuillaumeQuit:

function	script	BG_KVM_Shop_Process	{

	// =============================
	// CONFIGURACAO DA LOJA
	// =============================
	// Itens e precos em pontos KVM/BG
	setarray .@shop_items[0], 400024, 400025, 400026, 400027, 400028, 400029, 5170, 300040, 9593, 9199, 9198, 12596, 12210, 14545, 14536, 12375, 12710, 9675, 9676, 9682, 9683;
	setarray .@shop_prices[0],   100,    100,    100,    100,    100,    100,  100,    100,  100,  100,   50,    10,    50,   200,    10,    20,    20,   50,  100,    1,    1; // preco por item em pontos

	// =============================
	// PARAMETROS DA FUNCAO
	// =============================
	.@id  = getarg(0); // ID do item comprado
	.@qtd = getarg(1); // Quantidade comprada

	// =============================
	// PROCURA O ITEM NA TABELA
	// =============================
	.@found = 0;
	for (.@i = 0; .@i < getarraysize(.@shop_items); .@i++) {
		if (.@id == .@shop_items[.@i]) {
			.@found = 1;
			.@unit_price = .@shop_prices[.@i];
			break;
		}
	}

	if (!.@found) {
		dispbottom "[BG] Erro: item nao esta registrado na loja.";
		return;
	}

	// =============================
	// CALCULA O CUSTO TOTAL
	// =============================
	.@total_cost = .@unit_price * .@qtd;

	// =============================
	// VERIFICA PONTOS
	// =============================
	if (kvm_point < .@total_cost) {
		dispbottom "Pontos KVM/BG insuficientes.";
		return;
	}

	// =============================
	// DESCONTA E ENTREGA
	// =============================
	kvm_point -= .@total_cost;
	getitem .@id, .@qtd;
	logshop "SCRIPT_SHOP_BUY", "BG_Bravura_Shop", .@id, .@qtd, 0, 0, 0, "kvm_point", .@total_cost;

	dispbottom "[BG] -" + .@total_cost + " ponto(s) KVM/BG. Restam: " + kvm_point + ".";
	return;
}
