// Shop normal com desconto desativado pela flag "no".
// A cobranca em pontos fica no script abaixo.
-	shop	Vote_Shop	-1,no,400012:20,12210:4,12223:3,12224:3,100231:3,9615:5,14533:4,7776:25,19637:20,20141:20,20538:20,20313:20,5674:50

function	script	Vote_Shop_Process	{
	setarray .@shop_items[0], 400012, 12210, 12223, 12224, 100231, 9615, 14533, 7776, 19637, 20141, 20538, 20313, 5674;
	setarray .@shop_prices[0],    20,     4,     3,     3,     3,    5,     4,   25,    20,    20,    20,    20,   50;

	.@id = getarg(0);
	.@qtd = getarg(1);

	.@found = 0;
	for (.@i = 0; .@i < getarraysize(.@shop_items); .@i++) {
		if (.@id == .@shop_items[.@i]) {
			.@found = 1;
			.@unit_price = .@shop_prices[.@i];
			break;
		}
	}

	if (!.@found) {
		dispbottom "[Votos] Erro: item nao esta registrado na loja.";
		return;
	}

	.@total_cost = .@unit_price * .@qtd;
	if (#VOTE_POINT < .@total_cost) {
		dispbottom "Pontos de Votos insuficiente.";
		return;
	}

	#VOTE_POINT -= .@total_cost;
	getitem .@id, .@qtd;
	logshop "SCRIPT_SHOP_BUY", "Vote_Shop", .@id, .@qtd, 0, 0, 0, "#VOTE_POINT", .@total_cost;
	dispbottom "[Votos] -" + .@total_cost + " ponto(s) de votos. Restam: " + #VOTE_POINT + ".";
	return;
}
