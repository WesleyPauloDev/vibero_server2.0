function	script	F_Anubis_Recycle_Costume_Enchant	{
	.@coin_id = 9650; // Anubis_Coin
	.@coin_amount = 25;

	setarray .@enchant_id[0],
		25445,25446,25447,25448,25449,25450,25451,25452,25453,25454,25455,25456,
		29047,29056,29156,29157,29158,29358,29359,29360,29361,

		29477,29478,29479,29480,29481,29482,29483,29484,29485,29486,29487,29488,
		29513,29514,29515,29516,29517,29518,29519,29520,29521,29522,29523,29524,
		29611,29612,29613,29614,29615,29616,29617,29618,29619,29620,29621,
		29651,29652,29653,29654,29655,29656,29657,29658,29659,29660,29661,29662,29663,
		29664,29665,29666,29667,29668,29669,29670,29671,
		310000,310001,310002,310003,310004,310005,310006,310007,310008,310009,310010,310011,
		310180,310181,310182,310183,310184,310185,310186,310187,310188,310189,310190,310191,
		310257,310258,310259,310260,310261,310262,310263,310264,310265,310266,310267,310268,
		310325,310326,310327,310328,310329,310330,
		310660,310661,310662,310663,310664,310665,310666,310667,
		1000610,1000611,1000612,1000613,1000614,1000615,1000616,1000617,1000618,
		1000666,1000667,1000668,1000669,1000670,1000671,1000672,1000673,1000674,
		1000857,1000858,1000859,1000860,1000861,1000862,1000863,1000864,1000865,
		1000968,1000969,1000970,1000971,1000972,1000973,1000974,1000975,1000976,
		1001044,1001045,1001046,1001047,1001048,1001049,1001050,1001051,1001052,
		1001159,1001160,1001161,1001162,1001163,1001164,1001165,1001166,1001167;

	while (true) {
		.@menu$ = "";
		.@total = 0;

		for (.@i = 0; .@i < getarraysize(.@enchant_id); .@i++) {
			.@count = countitem(.@enchant_id[.@i]);
			if (.@count <= 0)
				continue;

			.@slot[.@total] = .@i;
			.@menu$ += getitemname(.@enchant_id[.@i]) + " (x" + .@count + ") -> " + .@coin_amount + " " + getitemname(.@coin_id) + ":";
			.@total++;
		}

		mes "[Reciclagem Anubis]";
		if (!.@total) {
			mes "Voce nao possui encantamentos visuais das caixas para reciclar.";
			close;
		}

		mes "Escolha a pedra de encantamento visual que deseja reciclar.";
		mes "Cada item reciclado rende " + .@coin_amount + "x " + getitemname(.@coin_id) + ".";
		next;

		.@choice = select(.@menu$ + "Cancelar") - 1;
		if (.@choice >= .@total)
			close;

		.@item_id = .@enchant_id[.@slot[.@choice]];
		if (countitem(.@item_id) < 1) {
			mes "[Reciclagem Anubis]";
			mes "Esse item nao esta mais no seu inventario.";
			close;
		}
		if (!checkweight(.@coin_id, .@coin_amount)) {
			mes "[Reciclagem Anubis]";
			mes "Voce esta com peso alto. Libere peso e tente novamente.";
			close;
		}

		delitem .@item_id, 1;
		getitem .@coin_id, .@coin_amount;
		logshop "SCRIPT_EXCHANGE", "AnubisEnchantRecycle", .@coin_id, .@coin_amount, 0, .@item_id, 1, "", 0;

		mes "[Reciclagem Anubis]";
		mes "Voce reciclou 1x " + getitemname(.@item_id) + ".";
		mes "Recebeu " + .@coin_amount + "x " + getitemname(.@coin_id) + ".";
		next;

		if (select("Continuar reciclando:Encerrar") == 2)
			close;
		clear;
	}
	end;
}

function	script	PetEggSellMenu	{
	.@pet_ticket_id = 9199;
	.@total = 0;
	// Ovos da pet_egg_barter_shop e seus valores em Pet_Ticket.
	setarray .@egg_id[0],
		9031, // Spring_Rabbit_Egg
		9032, // Knife_Goblin_Egg
		9035, // Red_Deleter_Egg
		9036, // Diabolic_Egg
		9037, // Wanderer_Egg
		9096, // Cat_O_Nine_Tail_Egg
		9090, // Littleisis_Egg
		9070, // Savage_Egg
		9087, // High_Orc_Egg
		9091, // Choco_Egg
		9107, // Ancient_Mummy_Egg
		9089, // Ammut_Egg
		9071, // Grand_Peco_Egg
		9093, // Hyegun_Egg
		9105, // Hodremlin_Egg
		9108, // Xm_Teddybear_Egg
		9092, // Eggring_Egg
		9094, // Leaf_Lunatic_Egg
		9200, // S_Nydhog_Egg
		9201, // VALKYRIE_N_Egg
		9202, // EDDGA_Egg
		9203, // OSIRIS_N_Egg
		9204, // PHREEONI_N_Egg
		9205; // DARK_LORD_N_Egg
	setarray .@ticket_value[0],
		1,1,1,1,1,1,
		5,5,5,5,5,5,
		20,20,20,20,20,20,
		100,100,100,100,100,100;
	for (.@i = 0; .@i < getarraysize(.@egg_id); .@i++) {
		if (countitem(.@egg_id[.@i]) > 0) {
			.@slot[.@total] = .@i;
			.@menu$ += getitemname(.@egg_id[.@i]) + " (x" + countitem(.@egg_id[.@i]) + ") -> " + .@ticket_value[.@i] + " Pet Ticket:";
			.@total++;
		}
	}
	mes "[Loja de Pets]";
	if (!.@total) {
		mes "Voce nao possui nenhum ovo de pet desta loja no inventario.";
		close;
	}
	.@choice = select(.@menu$) - 1;
	.@selected = .@slot[.@choice];
	.@egg = .@egg_id[.@selected];
	.@tickets = .@ticket_value[.@selected];
	if (countitem(.@egg) < 1) {
		mes "Esse ovo nao esta mais no seu inventario.";
		close;
	}
	if (!checkweight(.@pet_ticket_id, .@tickets)) {
		mes "Voce esta com peso alto. Libere peso e tente novamente.";
		close;
	}
	delitem .@egg, 1;
	getitem .@pet_ticket_id, .@tickets;
	logshop "SCRIPT_EXCHANGE", "PetEggSellMenu", .@pet_ticket_id, .@tickets, 0, .@egg, 1, "", 0;
	mes "Voce vendeu 1x " + getitemname(.@egg) + ".";
	mes "Recebeu " + .@tickets + "x " + getitemname(.@pet_ticket_id) + ".";
	close;
}
