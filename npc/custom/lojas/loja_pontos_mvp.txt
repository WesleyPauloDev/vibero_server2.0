function	script	MVP_Shop_Process	{

	// =============================
	// CONFIGURAÇÃO DA LOJA
	// =============================
	// Itens e preços em Pontos MVP
	setarray .@shop_items[0], 12411, 12412, 9617, 7621, 6396, 23436, 23926, 9592, 24661, 24662, 24663, 24664, 24665, 24666, 24667, 24668, 24669, 24670, 24671, 24672, 24673, 24674, 24675, 24676, 24677, 24678, 24679, 24680, 24681, 24682, 24683;
	setarray .@shop_prices[0], 20,   30,   50,   5,    30,   50,    100,   100,  50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50; // preço por item em pontos

	// =============================
	// PARÂMETROS DA FUNÇÃO
	// =============================
	.@id  = getarg(0); // ID do item comprado
	.@qtd = getarg(1); // Quantidade comprada

	// =============================
	// PROCURA O ITEM NA TABELA
	// =============================
	.@found = 0;
	for (.@i = 0; .@i < getarraysize(.@shop_items); .@i++) {
		if (.@id == .@shop_items[.@i]) {
			.@found = 1;
			.@unit_price = .@shop_prices[.@i];
			break;
		}
	}

	if (!.@found) {
		dispbottom "[Pontos MVP] Erro: item não está registrado na loja.";
		return;
	}

	// =============================
	// CALCULA O CUSTO TOTAL
	// =============================
	.@total_cost = .@unit_price * .@qtd;

	// =============================
	// VERIFICA PONTOS
	// =============================
	if (#mvp_points < .@total_cost) {
		dispbottom "Pontos MVP insuficiente.";
        return;
	}

	// =============================
	// DESCONTA E ENTREGA
	// =============================
	#mvp_points -= .@total_cost;
	getitem .@id, .@qtd;
	logshop "SCRIPT_SHOP_BUY", "MVP_Shop", .@id, .@qtd, 0, 0, 0, "#mvp_points", .@total_cost;

	dispbottom "[Pontos MVP] -" + .@total_cost + " ponto(s) de MVP. Restam: " + #mvp_points + ".";
	return;
}
