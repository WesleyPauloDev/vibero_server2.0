function	script	MVP_Shop_Process	{

	// =============================
	// CONFIGURAÇÃO DA LOJA
	// =============================
	// Itens e preços em Pontos MVP
	setarray .@shop_items[0], 7621, 12411, 12412, 6396, 9617, 23436, 24661, 24662, 24663, 24664, 24665, 24666, 24667, 24668, 24669, 24670, 24671, 24672, 24673, 24674, 24675, 24676, 24677, 24678, 24679, 24680, 24681, 24682, 24683, 24734, 24735, 24736, 24737, 24746, 24417, 24418, 24419, 24420, 24421, 9592, 23927, 24386, 24326, 24327;
	setarray .@shop_prices[0],   5,    20,    30,   30,   50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,    50,  100,   100,    50,    50,    50; // preço por item em pontos

	// =============================
	// PARÂMETROS DA FUNÇÃO
	// =============================
	.@id  = getarg(0); // ID do item comprado
	.@qtd = getarg(1); // Quantidade comprada

	// =============================
	// PROCURA O ITEM NA TABELA
	// =============================
	.@found = 0;
	for (.@i = 0; .@i < getarraysize(.@shop_items); .@i++) {
		if (.@id == .@shop_items[.@i]) {
			.@found = 1;
			.@unit_price = .@shop_prices[.@i];
			break;
		}
	}

	if (!.@found) {
		dispbottom "[Pontos MVP] Erro: item não está registrado na loja.";
		return;
	}

	// =============================
	// CALCULA O CUSTO TOTAL
	// =============================
	.@total_cost = .@unit_price * .@qtd;

	// =============================
	// VERIFICA PONTOS
	// =============================
	if (#mvp_points < .@total_cost) {
		dispbottom "Pontos MVP insuficiente.";
        return;
	}

	// =============================
	// DESCONTA E ENTREGA
	// =============================
	#mvp_points -= .@total_cost;
	getitem .@id, .@qtd;
	logshop "SCRIPT_SHOP_BUY", "MVP_Shop", .@id, .@qtd, 0, 0, 0, "#mvp_points", .@total_cost;

	dispbottom "[Pontos MVP] -" + .@total_cost + " ponto(s) de MVP. Restam: " + #mvp_points + ".";
	return;
}
