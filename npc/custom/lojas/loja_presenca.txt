-	script	Loja de Presenca	-1,{
OnTalk:
	mes "[Vendedor]";
	mes "Confira meus produtos!";
	npcshopattach "Presence_Shop";
	dispbottom "Pontos de Presenca: " + #presence_points;
	callshop "Presence_Shop",1;
	end;

OnBuyItem:
	for (.@i = 0; .@i < getarraysize(@bought_nameid); .@i++) {
		callfunc "Presence_Shop_Process", @bought_nameid[.@i], @bought_quantity[.@i];
	}
	end;
}
// Shop normal com desconto desativado pela flag "no".
// A cobranca em pontos fica no script abaixo.
-	shop	Presence_Shop	-1,no,9670:20,9671:20,12214:20,9678:30,9673:50,9674:50,9677:50,9642:100

function	script	Presence_Shop_Process	{
	setarray .@shop_items[0], 9670, 9671, 12214, 9678, 9673, 9674, 9677, 9642;
	setarray .@shop_prices[0],  20,   20,    20,   30,  50,   50,   50,   100;

	.@id = getarg(0);
	.@qtd = getarg(1);

	.@found = 0;
	for (.@i = 0; .@i < getarraysize(.@shop_items); .@i++) {
		if (.@id == .@shop_items[.@i]) {
			.@found = 1;
			.@unit_price = .@shop_prices[.@i];
			break;
		}
	}

	if (!.@found) {
		dispbottom "[Presenca] Erro: item nao esta registrado na loja.";
		return;
	}

	.@total_cost = .@unit_price * .@qtd;
	if (#presence_points < .@total_cost) {
		dispbottom "Pontos de Presenca insuficiente.";
		return;
	}

	#presence_points -= .@total_cost;
	getitem .@id, .@qtd;
	dispbottom "[Presenca] -" + .@total_cost + " ponto(s) de presenca. Restam: " + #presence_points + ".";
	return;
}
