function	script	F_InstanceRouletteLog	{
	.@event_type$ = getarg(0);
	.@category$ = getarg(1);
	.@instance_name$ = getarg(2, "");
	.@reward_item_id = getarg(3, 0);
	.@reward_amount = getarg(4, 0);
	.@guaranteed_item_id = getarg(5, 0);
	.@guaranteed_amount = getarg(6, 0);
	.@rare_item_id = getarg(7, 0);
	.@rare_amount = getarg(8, 0);
	.@period_expr$ = "DATE_FORMAT(CURDATE(), '%Y-%m-%d')";

	if (.@event_type$ == "reward" || .@event_type$ == "bonus") {
		.@already_logged = 0;
		query_logsql(
			"SELECT COUNT(*) FROM `instance_roulette_log` " +
			"WHERE `account_id` = " + getcharid(3) + " " +
			"AND `event_type` = '" + escape_sql(.@event_type$) + "' " +
			"AND `category` = '" + escape_sql(.@category$) + "' " +
			"AND `period_key` = " + .@period_expr$,
			.@already_logged
		);

		if (.@already_logged > 0) {
			logmes "[INSTANCE_ROULETTE_DUPLICATE_BLOCK] acc=" + getcharid(3) + " char=" + getcharid(0) + " event=" + .@event_type$ + " category=" + .@category$;
			return 0;
		}
	}

	query_logsql(
		"INSERT INTO `instance_roulette_log` " +
		"(`account_id`, `char_id`, `char_name`, `event_type`, `category`, `instance_name`, `period_key`, " +
		"`reward_item_id`, `reward_item_name`, `reward_amount`, `guaranteed_item_id`, `guaranteed_item_name`, " +
		"`guaranteed_amount`, `rare_item_id`, `rare_item_name`, `rare_amount`, `map`, `created_at`) VALUES (" +
		getcharid(3) + ", " +
		getcharid(0) + ", '" +
		escape_sql(strcharinfo(0)) + "', '" +
		escape_sql(.@event_type$) + "', '" +
		escape_sql(.@category$) + "', '" +
		escape_sql(.@instance_name$) + "', " +
		.@period_expr$ + ", " +
		.@reward_item_id + ", '" +
		((.@reward_item_id > 0) ? escape_sql(getitemname(.@reward_item_id)) : "") + "', " +
		.@reward_amount + ", " +
		.@guaranteed_item_id + ", '" +
		((.@guaranteed_item_id > 0) ? escape_sql(getitemname(.@guaranteed_item_id)) : "") + "', " +
		.@guaranteed_amount + ", " +
		.@rare_item_id + ", '" +
		((.@rare_item_id > 0) ? escape_sql(getitemname(.@rare_item_id)) : "") + "', " +
		.@rare_amount + ", '" +
		escape_sql(strcharinfo(3)) + "', NOW())"
	);

	return 1;
}
