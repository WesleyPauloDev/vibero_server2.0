        "Assalto ao Dirigivel",

        "Tumulo do Monarca",
        "Charleston em Crise",

        "Torre do Demônio",

        "Sarah e Fenrir",
        "Jardim de Vermuth",
        "Fazenda de Pitayas",
        "Laboratório Wolfchev",
        "Templo do Rei Demônio",
        "Ilha de Bios",

	// Reset diário simples

	// set #inst60_status,2;
	// set #inst120_status,2;
	// set #inst175_status,0;
	if (#inst_cooldown <= gettimetick(2)) {
		set #inst60_status,0;
		set #inst120_status,0;
		set #inst175_status,0;
		set #inst_cooldown, gettimetick(2) + callfunc("get_daily_reset");

	if (#recs == 3) {

		mes "Parabéns! Você Conclui Todas Missões da Roleta hoje.";
		getitem 12412,1;
		#recs = 4;
		close;
	}
	if (#recs == 4) {
		mes "[Roleta das Instâncias]";
		mes "Você já conclui todas missões de hoje.";
		close;
	}

	mes "Uma de cada categoria de nível.";
	next;

	mes "Escolha a categoria de missão que deseja roletar:";
	next;

	if (#inst60_status < 2)      set .@lv30_60$, "^FF0000[Pendente]^000000";
	else if (#inst60_status == 2)	set .@lv30_60$, "^0000FF[Resgatar]^000000";	
	else                       set .@lv30_60$, "^00FF00[OK]^000000";

	if (.@sel == 5) close;

	// ===============================
	// CHECAGEM DE NÍVEL MÍNIMO
	// ===============================
	switch (.@sel) {
		case 1:
			if (#inst60_status == 3) {

			set .@cat$, "60";
			setarray .@inst$, .inst_60$;

				goto L_Rewards;

			}

				mes "Você precisa ser pelo menos nível ^FF000060^000000 para essa categoria.";

			}
			set .@cat$, "120";
			setarray .@inst$, .inst_120$;
			break;

				goto L_Rewards;

			}

	.@var$ = ".inst_" + .@cat$ + "$";
	.@sorteada$ = getd(.@var$+"["+.@final+"]");
    .@inst_name$ = "#inst" + .@cat$ + "_name$";
    .@inst_status$ = "#inst" + .@cat$ + "_status";
    setd(.@inst_name$, .@sorteada$);
    setd(.@inst_status$, 1);
	mes "Sua missão de instância é: ^FFAA00" + getd("#inst"+.@cat$+"_name$") + "^000000!";
	mes "Complete-a e volte aqui para sua recompensa.";
	close;

L_InfoRewards:
	mes "[Roleta das Instâncias]";
	mes "Ao completar a missão sorteada, você recebe:";
	mes "- ^00AA002 unidades^000000 de 1 item aleatório da categoria.";
	mes "- ^00AA001x " + getitemname(40111) + "^000000 garantido.";
	mes "- ^FF99005%^000000 de chance de receber ^00AA001x " + getitemname(9592) + "^000000.";
	next;
	mes "[Lv 30~60]";

	mes "- " + getitemname(12210) + " (12210)";
	mes "- " + getitemname(7621) + " (7621)";

	mes "[Lv 60~120]";
	mes "Sorteia 1 destes, com chances iguais:";
	mes "- " + getitemname(22818) + " (22818)";
	mes "- " + getitemname(12223) + " (12223)";

	mes "- " + getitemname(9609) + " (9609)";

	mes "- " + getitemname(12214) + " (12214)";
