function	script	F_ShadowGuaranteedRefineTicket	{
	disable_items;

	.@ticket_id = getarg(0);
	setarray .@target_slots[0], EQI_SHADOW_ARMOR, EQI_SHADOW_WEAPON, EQI_SHADOW_SHIELD, EQI_SHADOW_SHOES, EQI_SHADOW_ACC_R, EQI_SHADOW_ACC_L;

	.@menu$ = "";
	.@count = 0;

	for (.@i = 0; .@i < getarraysize(.@target_slots); ++.@i) {
		.@slot = .@target_slots[.@i];
		if (!getequipisequiped(.@slot))
			continue;

		.@refine = getequiprefinerycnt(.@slot);
		if (.@refine < 7 || .@refine >= 10)
			continue;

		.@menu$ += getequipname(.@slot) + " [+" + .@refine + " -> +" + (.@refine + 1) + "]" + ":";
		.@slot_list[.@count] = .@slot;
		++.@count;
	}

	mes "[Ticket de Refino Shadow]";
	if (.@count == 0) {
		mes "Equipe um equipamento Shadow com refino entre +7 e +9 para usar este ticket.";
		close;
	}

	mes "Escolha o Shadow que deseja refinar.";
	.@pick = select(.@menu$) - 1;
	.@part = .@slot_list[.@pick];

	.@equip_id = getequipid(.@part);
	.@refine = getequiprefinerycnt(.@part);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	if (!getequipisequiped(.@part)) {
		mes "Nenhum equipamento esta equipado neste slot.";
		close;
	}

	if (.@refine < 7) {
		mes "Este ticket so pode ser usado em equipamentos Shadow +7 ou superior.";
		close;
	}

	if (.@refine >= 10) {
		mes "Este equipamento ja esta com refino +10 ou superior.";
		close;
	}

	mes "Deseja refinar " + getitemname(.@equip_id) + " de +" + .@refine + " para +" + (.@refine + 1) + "?";
	if (select("Sim:Cancelar") == 2) {
		mes "Operacao cancelada.";
		close;
	}

	if (countitem(.@ticket_id) < 1) {
		mes "Voce nao possui o ticket necessario.";
		close;
	}

	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refine) ||
		callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3])) {
		mes "O equipamento foi alterado durante o processo.";
		close;
	}

	delitem .@ticket_id,1;
	successrefitem .@part;
	specialeffect2 EF_REFINEOK;

	mes "O equipamento foi refinado com sucesso para +" + (.@refine + 1) + ".";
	close;
}
