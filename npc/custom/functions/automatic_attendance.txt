-	script	Automatic_Attendance	-1,{

OnInit:
	.ponto_intervalo = 600000;	// 10 minutos = 600.000 ms
	end;

OnPCLoginEvent:
	// Inicia o contador quando o jogador entra
	addtimer .ponto_intervalo, strnpcinfo(3) + "::OnGivePoints";
	end;

OnGivePoints:
	set .@aid, getcharid(3);

	// Se o jogador nao estiver online, encerra
	if (!isloggedin(.@aid)) end;

	// Se o jogador estiver em autotrade (@autotrade), nao recebe pontos
	if (checkvending() & 2) {
		addtimer .ponto_intervalo, strnpcinfo(3) + "::OnGivePoints";
		end;
	}

	// Da o ponto de presenca
	set .@pontos, #presence_points + 1;
	set #presence_points, .@pontos;

	dispbottom "[Presenca] Voce ganhou 1 ponto de presenca! Total: " + .@pontos + " ponto(s).";

	// Reagendar o timer
	addtimer .ponto_intervalo, strnpcinfo(3) + "::OnGivePoints";
	end;
}
