-	script	Reputation_Login	-1,{

OnPCLoginEvent:
	
	// Inicia o contador quando o jogador entra
	if (#reput_reborn)  {
		add_reputation_points(7, -10000);
		add_reputation_points(7, #reput_reborn * 100);
		set #exp_final, exp_base_custom - (exp_base_custom * #reput_reborn / 100);
		set #drop_final, drop_base + (#reput_reborn * 2);
	} else {
		set #exp_final, exp_base_custom;
		set #drop_final, drop_base;
	}
	set #hp_final, 0;
	set #sp_final, 0;
	set #allstats_final, 0;

	setarray .RepIDs[0], 
		1, // Instance
		2, // Votos
		3, // Cheffenia
		4, // Cheffenia Hard
		5, // Bio5
		6, // Bio5 Hard
		8; // Dominador de Classe

	setarray .RepVars$[0],
		"#reput_instance",
		"#reput_votos",
		"#reput_cheffenia",
		"#reput_cheffenia_hard",
		"#reput_bio5",
		"#reput_bio5_hard",
		"#reput_class_dominator";

	for (.@i = 0; .@i < getarraysize(.RepIDs); .@i++) {

		.@rep_id = .RepIDs[.@i];
		.@var$   = .RepVars$[.@i];

		if (getd(.@var$)) {
			add_reputation_points(.@rep_id, -10000);
			add_reputation_points(.@rep_id, getd(.@var$) * 100);
		}
	}
	
	// Instância + 20% EXP por Nível
	.@reput_instance = (get_reputation_points(1) / 1000) * 20;
	set #reput_instance, .@reput_instance;

	set #exp_final, #exp_final + (exp_base_custom * .@reput_instance / 100);

	// Votos + 5 em todos os status
	.@reput_votos = (get_reputation_points(2) / 1000) * 5;
	set #reput_votos, .@reput_votos;

	#allstats_final += .@reput_votos;

	// Cheffenia Normal + 10% HP/SP por Nível
	.@reput_cheffenia = (get_reputation_points(3) / 1000) * 10;
	set #reput_cheffenia, .@reput_cheffenia;
	
	set #hp_final, #hp_final + .@reput_cheffenia;
	set #sp_final, #sp_final + .@reput_cheffenia;
	
	// Cheffenia Hard + 5 em todos os status
	.@reput_cheffenia_hard = (get_reputation_points(4) / 1000) * 5;
	set #reput_cheffenia_hard, .@reput_cheffenia_hard;

	#allstats_final += .@reput_cheffenia_hard;

	// Bio5 Normal + 10 HP/SP por Nível
	.@reput_bio5 = (get_reputation_points(5) / 1000) * 10;
	set #reput_bio5, .@reput_bio5;

	set #hp_final, #hp_final + .@reput_bio5;
	set #sp_final, #sp_final + .@reput_bio5;

	// Bio5 Hard + 5 em todos os status
	.@reput_bio5_hard = (get_reputation_points(6) / 1000) * 5;
	set #reput_bio5_hard, .@reput_bio5_hard;

	#allstats_final += .@reput_bio5_hard;

	// Dominador de Classes + 20% Drop/+20% XP por Nível
	.@reput_class_dominator = (get_reputation_points(8) / 1000) * 20;
	set #reput_class_dominator, .@reput_class_dominator;

	set #drop_final, #drop_final + .@reput_class_dominator;
	set #exp_final, #exp_final + (exp_base_custom * .@reput_class_dominator / 100);

	// Aplicação dos Bonus
	set_reborn_exp(#exp_final);
	set_reborn_drop(#drop_final);

	set_bonus_hpsp(getcharid(0), #hp_final, #sp_final);

	set_bonus_allstats(getcharid(0),  #allstats_final);
	end;
}
