function	script	F_InstanceEntry	{

	// ==============================
	// Parametros esperados:
	// ==============================
	// getarg(0) -> Nome exato da instancia no DB (string)
	// getarg(1) -> Nome do NPC (string)
	// getarg(2) -> Nome da variavel de cooldown (string de variavel de conta)
	// getarg(3) -> Nome da variavel de status (string de variavel de conta)
	// getarg(4) -> Level minimo (numero)
	// getarg(5) -> Nome amigavel da instancia (string opcional)
	// getarg(6) -> Quest de caca iniciada ao entrar (numero opcional)
	// ==============================

	.@db_name$ = getarg(0);
	.@npc_name$ = getarg(1);
	.@cooldown_var$ = getarg(2);
	.@status_var$ = getarg(3);
	.@min_level = getarg(4);
	.@display_name$ = getarg(5, .@db_name$);
	.@hunt_quest = getarg(6, 0);

    .@cooldown_name$ = "#cooldown_" + .@cooldown_var$;
    .@status_name$ = "#" + .@status_var$ + "_STATUS";
	.@shared_vdec = (.@status_var$ == "VillaDeception" || .@status_var$ == "VillaDeceptionAdvanced");
	if (.@status_var$ == "VillaDeception") {
		.@shared_cooldown_name$ = "#cooldown_vdec_adv";
		.@shared_status_name$ = "#VillaDeceptionAdvanced_STATUS";
	}
	else if (.@status_var$ == "VillaDeceptionAdvanced") {
		.@shared_cooldown_name$ = "#cooldown_vdec";
		.@shared_status_name$ = "#VillaDeception_STATUS";
	}

    // debugmes ".@db_name$: " + .@db_name$;
    // debugmes ".@npc_name$: " + .@npc_name$;
    // debugmes ".@cooldown_name$: " + getd(.@cooldown_name$);
    // debugmes ".@status_name$: " + getd(.@status_name$);
    // debugmes ".@min_level: " + .@min_level;

	// Checa jogador anexado
	if (!isloggedin(getcharid(3))) end;

	// Salva ID do jogador
	.@player_rid = getattachedrid();
	attachrid(.@player_rid);
	sleep2 100;
    
	// ============================
	// 1. Checa Level minimo
	// ============================
	if (BaseLevel < .@min_level) {
		mes "[" + .@npc_name$ + "]";
		mes "Uau~ Voce ainda e muito inexperiente!";
		mes "Volte quando atingir o nivel base ^FF5555" + .@min_level + "^000000.";
		close;
	}

	// ============================
	// 2. Checa se esta em grupo
	// ============================
	if (getcharid(1) < 1) {
		mes "[" + .@npc_name$ + "]";
		mes "Voce precisa estar em um grupo para entrar na instancia.";
		close;
	}

	// ============================
	// 3. Reseta cooldown se expirou
	// ============================
	if (getd(.@cooldown_name$) < gettimetick(2) && getd(.@status_name$) != 0) {
        setd(.@status_name$, 0);
    }
	if (.@shared_vdec && getd(.@shared_cooldown_name$) < gettimetick(2) && getd(.@shared_status_name$) != 0) {
		setd(.@shared_status_name$, 0);
	}

	// ============================
	// 4. Se ja completou hoje
	// ============================
	if (getd(.@status_name$) == 3 || (.@shared_vdec && getd(.@shared_status_name$) == 3)) {
		mes "[" + .@npc_name$ + "]";
		mes "Voce nao pode mais entrar nesta instancia hoje.";
		mes "Volte amanha apos o reset diario.";
		close;
	}

	.@live_iid = instance_id(IM_PARTY);
	.@same_live_instance = (.@live_iid > 0 && instance_live_info(ILI_NAME, .@live_iid) == .@db_name$);
	if ((getd(.@status_name$) > 0 || (.@shared_vdec && getd(.@shared_status_name$) > 0)) && !.@same_live_instance) {
		mes "[" + .@npc_name$ + "]";
		mes "Voce nao pode mais entrar nesta instancia hoje.";
		mes "Volte amanha apos o reset diario.";
		close;
	}

	// =========================
	// BLOQUEIO POR PARTY + IP
	// =========================
	if (!callfunc("F_CheckInstanceIPPermission")) {
		mes "[" + .@npc_name$ + "]";
		mes "Jogadores com mesmo IP so podem participar de mais de uma ";
		mes "instancia simultaneamente, se estiverem em grupos diferentes.";
		close;
	}

	// ============================
	// 5. Dialogo inicial
	// ============================
	mes "[" + .@npc_name$ + "]";
	mes "Ola, aventureiro! Deseja criar ou entrar na instancia ^0055FF" + .@display_name$ + "^000000?";
	next;

	if (getd(.@status_name$) > 0 || .@same_live_instance)
		.@menu = select("Entrar na instancia", "Cancelar") + 1;
	else
		.@menu = select("Criar instancia", "Entrar na instancia", "Cancelar");

	switch(.@menu) {

		// ============================
		// 6. Criar instancia
		// ============================
		case 1:
			if (!is_party_leader()) {
				mes "Somente o lider do grupo pode criar a instancia.";
				close;
			}
			if (getd(.@status_name$) == 0) {
				.@iid = instance_create(.@db_name$);
				if (.@iid >= 0) {
					mes "[" + .@npc_name$ + "]";
					mes "A instancia esta se abrindo...";
					announce "[Instancias] " + strcharinfo(0) + " criou " + .@display_name$ + " para o grupo " + getpartyname(getcharid(1)) + ".", bc_all, "0xFF9900";
					callfunc("F_InstanceLogStart", .@status_var$, .@display_name$);

					setd(.@cooldown_name$, gettimetick(2) + callfunc("get_daily_reset"));
					setd(.@status_name$, 1);
					if (.@shared_vdec) {
						setd(.@shared_cooldown_name$, getd(.@cooldown_name$));
						setd(.@shared_status_name$, 1);
					}
				} else {
					mes "Falha ao criar a instancia.";
				}
			} else {
				mes "A instancia ja foi criada. Use ^0000FF'Entrar na instancia'^000000 para entrar.";
			}
			close;

		// ============================
		// 7. Entrar na instancia
		// ============================
		case 2:
			switch(instance_enter(.@db_name$)) {
				case IE_OK:
					setd(.@cooldown_name$, gettimetick(2) + callfunc("get_daily_reset"));
					setd(.@status_name$, 2);
					if (.@shared_vdec) {
						setd(.@shared_cooldown_name$, getd(.@cooldown_name$));
						setd(.@shared_status_name$, 2);
					}
					if (.@hunt_quest > 0) {
						if (isbegin_quest(.@hunt_quest))
							erasequest .@hunt_quest;
						setquest .@hunt_quest;
					}
					callfunc("F_InstanceLogEnter", .@status_var$, .@display_name$);
					close2;
					end;
				case IE_NOINSTANCE:
					if (getd(.@status_name$) == 0) {
						mes "A instancia ainda nao foi criada.";
						close;
					}
					mes "A instancia nao esta disponivel no momento.";
					mes "Sua tentativa de hoje ja foi registrada, entao sera preciso aguardar o proximo reset.";
					close;
				case IE_NOMEMBER:
					mes "Voce nao faz parte de um grupo valido para esta instancia.";
					close;
				default:
					mes "Ocorreu um erro desconhecido ao tentar entrar.";
					close;
			}
			close;

		// ============================
		// 8. Cancelar
		// ============================
		case 3:
			mes "[" + .@npc_name$ + "]";
			mes "Tudo bem, volte quando estiver pronto.";
			close;
	}
	return;
}

function	script	F_InstanceProgressTouch	{
	.@instance_code$ = getarg(0);
	.@instance_name$ = callfunc("F_InstanceHistoryCode", getarg(1, .@instance_code$));
	query_sql("INSERT INTO `instance_monitor_member_state` (`account_id`, `instance_code`, `instance_name`, `tried_count`, `completed_count`, `last_try_at`) VALUES ("
		+ getcharid(3) + ", '"
		+ escape_sql(.@instance_code$) + "', '"
		+ escape_sql(.@instance_name$) + "', 1, 0, NOW()) ON DUPLICATE KEY UPDATE `instance_name` = VALUES(`instance_name`), `tried_count` = `tried_count` + 1, `last_try_at` = NOW()");
	return;
}

function	script	F_InstanceProgressComplete	{
	.@account_id = getarg(0);
	.@instance_code$ = getarg(1);
	.@instance_name$ = callfunc("F_InstanceHistoryCode", getarg(2, .@instance_code$));
	query_sql("INSERT INTO `instance_monitor_member_state` (`account_id`, `instance_code`, `instance_name`, `tried_count`, `completed_count`, `last_try_at`, `last_completed_at`) VALUES ("
		+ .@account_id + ", '"
		+ escape_sql(.@instance_code$) + "', '"
		+ escape_sql(.@instance_name$) + "', 1, 1, NOW(), NOW()) ON DUPLICATE KEY UPDATE `instance_name` = VALUES(`instance_name`), `tried_count` = IF(`tried_count` < 1, 1, `tried_count`), `completed_count` = `completed_count` + 1, `last_completed_at` = NOW(), `last_try_at` = COALESCE(`last_try_at`, NOW())");
	return;
}

function	script	F_InstanceEnsureAttemptState	{
	.@instance_code$ = getarg(0);
	.@instance_name$ = callfunc("F_InstanceHistoryCode", getarg(1, .@instance_code$));
	.@tried = 0;
	query_sql("SELECT COUNT(*) FROM `instance_monitor_member_state` WHERE `account_id` = "
		+ getcharid(3) + " AND `instance_code` = '" + escape_sql(.@instance_code$) + "'", .@tried);
	if (.@tried < 1)
		callfunc("F_InstanceProgressTouch", .@instance_code$, .@instance_name$);
	return;
}

function	script	F_InstanceRouletteComplete	{
	.@finish_name$ = getarg(0, "");
	.@finish_match$ = getarg(1, .@finish_name$);
	.@finish_code$ = callfunc("F_InstanceHistoryCode", .@finish_name$);
	.@match_code$ = callfunc("F_InstanceHistoryCode", .@finish_match$);

	for (.@i = 0; .@i < 3; .@i++) {
		if (.@i == 0) {
			.@roulette_name_var$ = "#inst60_name$";
			.@roulette_status_var$ = "#inst60_status";
		}
		else if (.@i == 1) {
			.@roulette_name_var$ = "#inst120_name$";
			.@roulette_status_var$ = "#inst120_status";
		}
		else {
			.@roulette_name_var$ = "#inst175_name$";
			.@roulette_status_var$ = "#inst175_status";
		}

		.@roulette_name$ = getd(.@roulette_name_var$);
		if (.@roulette_name$ == "")
			continue;

		.@roulette_code$ = callfunc("F_InstanceHistoryCode", .@roulette_name$);
		if (.@roulette_name$ == .@finish_name$ || .@roulette_name$ == .@finish_match$ || .@roulette_code$ == .@finish_code$ || .@roulette_code$ == .@match_code$) {
			if (getd(.@roulette_status_var$) == 1)
				setd(.@roulette_status_var$, 2);
		}
	}
	return;
}
function	script	F_InstanceGetOpenLogId	{
	.@instance_name$ = callfunc("F_InstanceHistoryCode", getarg(0, ""));
	.@party_id = getcharid(1);
	.@log_id = 0;
	if (.@party_id < 1)
		return 0;
	query_sql("SELECT `id` FROM `instance_monitor_log` WHERE `party_id` = "
		+ .@party_id + " AND `status` = 'OPEN' ORDER BY (`instance_name` = '"
		+ escape_sql(.@instance_name$) + "') DESC, `id` DESC LIMIT 1", .@log_id);
	return .@log_id;
}

function	script	F_InstanceLogStart	{
	.@instance_code$ = getarg(0);
	.@instance_name$ = callfunc("F_InstanceHistoryCode", getarg(1, .@instance_code$));
	.@party_id = getcharid(1);
	if (.@party_id < 1)
		return;
	query_sql("INSERT INTO `instance_monitor_log` (`instance_name`, `party_id`, `party_name`, `created_by_account_id`, `created_by_char_id`, `created_by_name`, `status`, `started_at`) VALUES ('"
		+ escape_sql(.@instance_name$) + "', "
		+ .@party_id + ", '"
		+ escape_sql(getpartyname(.@party_id)) + "', "
		+ getcharid(3) + ", "
		+ getcharid(0) + ", '"
		+ escape_sql(strcharinfo(0)) + "', 'OPEN', NOW())");
	callfunc("F_InstanceProgressTouch", .@instance_code$, .@instance_name$);
	query_sql("INSERT INTO `instance_monitor_member_log` (`instance_log_id`, `instance_code`, `instance_name`, `party_id`, `account_id`, `char_id`, `char_name`, `entered_at`, `completed`) VALUES ("
		+ callfunc("F_InstanceGetOpenLogId", .@instance_name$) + ", '"
		+ escape_sql(.@instance_code$) + "', '"
		+ escape_sql(.@instance_name$) + "', "
		+ .@party_id + ", "
		+ getcharid(3) + ", "
		+ getcharid(0) + ", '"
		+ escape_sql(strcharinfo(0)) + "', NOW(), 0) ON DUPLICATE KEY UPDATE `char_name` = VALUES(`char_name`), `entered_at` = COALESCE(`entered_at`, VALUES(`entered_at`))");
	return;
}

function	script	F_InstanceLogEnter	{
	.@instance_code$ = getarg(0);
	.@instance_name$ = callfunc("F_InstanceHistoryCode", getarg(1, .@instance_code$));
	.@party_id = getcharid(1);
	.@log_id = callfunc("F_InstanceGetOpenLogId", .@instance_name$);
	if (.@party_id < 1 || .@log_id < 1)
		return;
	callfunc("F_InstanceProgressTouch", .@instance_code$, .@instance_name$);
	query_sql("INSERT INTO `instance_monitor_member_log` (`instance_log_id`, `instance_code`, `instance_name`, `party_id`, `account_id`, `char_id`, `char_name`, `entered_at`, `completed`) VALUES ("
		+ .@log_id + ", '"
		+ escape_sql(.@instance_code$) + "', '"
		+ escape_sql(.@instance_name$) + "', "
		+ .@party_id + ", "
		+ getcharid(3) + ", "

function	script	F_InstanceLogFinish	{
	.@instance_name$ = callfunc("F_InstanceHistoryCode", getarg(0));
	.@instance_code$ = callfunc("F_InstanceHistoryCode", .@instance_name$);
	.@party_id = getcharid(1);
	.@log_id = 0;
	if (.@party_id < 1)
		return;
	.@log_id = callfunc("F_InstanceGetOpenLogId", .@instance_name$);
	if (.@log_id < 1)
		return;
	getpartymember(.@party_id, 1);
	getpartymember(.@party_id, 2);
	for (.@i = 0; .@i < $@partymembercount; .@i++) {
		if (!isloggedin($@partymemberaid[.@i], $@partymembercid[.@i]))
			continue;
		query_sql("UPDATE `instance_monitor_member_log` SET `completed` = 1, `completed_at` = NOW() WHERE `instance_log_id` = "
			+ .@log_id + " AND `account_id` = " + $@partymemberaid[.@i] + " AND `entered_at` IS NOT NULL");
		callfunc("F_InstanceProgressComplete", $@partymemberaid[.@i], .@instance_code$, .@instance_name$);
	}
	query_sql("UPDATE `instance_monitor_log` SET `status` = 'COMPLETED', `completed_at` = NOW(), `finished_by_account_id` = "
		+ getcharid(3) + ", `finished_by_char_id` = "
		+ getcharid(0) + ", `finished_by_name` = '"
		+ escape_sql(strcharinfo(0)) + "' WHERE `id` = " + .@log_id);
	return;
}

function	script	F_InstanceFinishBase	{
	.@instance_name$ = getarg(0);
	.@status_var$ = getarg(1);
	.@coin_amount = getarg(2, 10);
	.@ticket_amount = getarg(3, 2);
	.@roulette_name_var$ = getarg(4, "");
	.@roulette_status_var$ = getarg(5, "");
	.@rep_id = getarg(6, 1);
	.@rep_amount = getarg(7, 10);
	.@rep_achievement = getarg(8, 110002);
	.@rep_cap = getarg(9, 3000);
	.@final_status = getarg(10, 3);
	.@refresh_weekly = getarg(11, 0);
	.@roulette_match_name$ = getarg(12, .@instance_name$);
	.@announce_finish = getarg(13, 1);

	if (.@status_var$ == "")
		return 0;
	if (getd(.@status_var$) != 2)
		return 0;
	if ((.@coin_amount > 0 && checkweight(9608, .@coin_amount) == 0) || (.@ticket_amount > 0 && checkweight(9198, .@ticket_amount) == 0))
		return -1;
	if (.@rep_amount > 0)
		callfunc("F_AddRep", .@rep_id, .@rep_amount, .@rep_achievement, .@rep_cap);
	callfunc("F_DiaryRefreshDaily");
	if (.@refresh_weekly)
		callfunc("F_DiaryRefreshWeekly");
	if (.@announce_finish)
		callfunc("F_InstanceFinishAnnounce", .@instance_name$);
	if (.@roulette_name_var$ != "" && .@roulette_status_var$ != "") {
		.@roulette_name_value$ = getd(.@roulette_name_var$);
		if (.@roulette_name_value$ == .@roulette_match_name$)
			setd(.@roulette_status_var$, 2);
	}
	callfunc("F_InstanceRouletteComplete", .@instance_name$, .@roulette_match_name$);
	if (.@coin_amount > 0)
		getitem 9608, .@coin_amount;
	if (.@ticket_amount > 0)
		getitem 9198, .@ticket_amount;
	setd(.@status_var$, .@final_status);
	return 1;
}
function	script	F_InstanceMenuHistoryState	{
	.@instance_code$ = getarg(0);
	.@completed = 0;
	.@entries = 0;
	.@where$ = "`instance_code` = '" + escape_sql(.@instance_code$) + "'";
	if (.@instance_code$ == "GhostPalace")
		.@where$ = .@where$ + " OR `instance_code` IN ('Ghost Palace','Palacio das Magoas') OR `instance_name` IN ('Ghost Palace','Palacio das Magoas')";
	else if (.@instance_code$ == "TombRemorse")
		.@where$ = .@where$ + " OR `instance_code` IN ('Tomb of Remorse','Tumba do Remorso') OR `instance_name` IN ('Tomb of Remorse','Tumba do Remorso')";
	query_sql("SELECT COUNT(*) FROM `instance_monitor_member_state` WHERE `account_id` = "
		+ getcharid(3) + " AND (" + .@where$ + ") AND `completed_count` > 0 AND DATE(`last_completed_at`) = CURDATE()", .@completed);
	if (.@completed > 0)
		return 2;
	query_sql("SELECT COUNT(*) FROM `instance_monitor_member_state` WHERE `account_id` = "
		+ getcharid(3) + " AND (" + .@where$ + ") AND DATE(`last_try_at`) = CURDATE()", .@entries);
	if (.@entries > 0)
		return 1;
	query_sql("SELECT COALESCE(MAX(`completed_count`), 0), COALESCE(MAX(`tried_count`), 0) FROM `instance_monitor_member_state` WHERE `account_id` = "
		+ getcharid(3) + " AND (" + .@where$ + ")", .@completed, .@entries);
	if (.@completed > 0 || .@entries > 0)
		return 3;
	return 0;
}

function	script	F_InstanceMenuOption	{
	.@label$ = getarg(0);
	.@cooldown_var$ = getarg(1);
	.@status_var$ = getarg(2);
	.@status_name$ = "#" + .@status_var$ + "_STATUS";
	.@status = getd(.@status_name$);
	.@history = callfunc("F_InstanceMenuHistoryState", .@status_var$);

	if (.@history == 2 && .@status > 0)
		return "^6CFF6C" + .@label$ + "^000000";

	if (.@history == 2)
		return "^6CFF6C" + .@label$ + "^000000";

	if (.@history == 1)
		return "^FF5A5A" + .@label$ + "^000000";

	if (.@history == 3)
		return "^FFD166" + .@label$ + "^000000";

	if (.@status > 0)
		return "^FF5A5A" + .@label$ + "^000000";

	return "^777777" + .@label$ + "^000000";
}

		#VillaDeceptionAdvanced_STATUS = #VillaDeception_STATUS;
		#cooldown_vdec_adv = #cooldown_vdec;

		#VillaDeception_STATUS = #VillaDeceptionAdvanced_STATUS;
		#cooldown_vdec = #cooldown_vdec_adv;
	}
