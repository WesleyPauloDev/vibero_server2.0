//=====================================================
// Sistema de Rate de Fim de Semana
//=====================================================

-	script	Weekend_Rates	-1,{

OnInit:
	// 100 = 1x, 150 = 1.5x, 200 = 2x.
	.normal_base_exp_rate = 100;
	.normal_job_exp_rate = 100;
	.normal_drop_rate = 100;
	.normal_mvp_card_rate = 100;

	.weekend_base_exp_rate = 200;
	.weekend_job_exp_rate = 200;
	.weekend_drop_rate = 150;
	.weekend_mvp_card_rate = 500;
	.bonus_week_active = 0;

	bindatcmd "weekendrates", strnpcinfo(3)+"::OnAtcommand", 99, 99;
	bindatcmd "ratesferiado", strnpcinfo(3)+"::OnHolidayAtcommand", 99, 99;
	bindatcmd "mvpcard5x", strnpcinfo(3)+"::OnMvpCardAtcommand", 99, 99;
	callsub S_ApplyRates, 0;
	end;

OnEventWeekSync:
	callsub S_ApplyRates, 1;
	end;

OnClock0000:
OnClock0200:
OnClock1200:
OnClock2359:
	callsub S_ApplyRates, 1;
	end;

OnClock0015:
OnClock0215:
OnClock0415:
OnClock0615:
OnClock0815:
OnClock1015:
OnClock1215:
OnClock1415:
OnClock1615:
OnClock1815:
OnClock2015:
OnClock2215:
	callsub S_AnnounceActiveRates;
	end;

OnPCLoginEvent:
	addtimer 1000, strnpcinfo(3)+"::OnLoginNotice";
	end;

OnLoginNotice:
	if (callsub(S_IsWeekendEvent)) {
		dispbottom "[Evento] Rates de fim de semana ativas: Base EXP " + .weekend_base_exp_rate + "% / Job EXP " + .weekend_job_exp_rate + "% / Drop " + .weekend_drop_rate + "%.";
		if ($Weekend_Rates_MvpCard5x)
			dispbottom "[Evento] Drop de carta MVP 5x ativo ate domingo as 23:59.";
	}
	end;

OnAtcommand:
	callsub S_ApplyRates, 0;
	mes "[Weekend Rates]";
	if (.weekend_rates_active) {
		mes "Status: ^00AA00ativo^000000.";
		mes "Base EXP: " + .weekend_base_exp_rate + "%";
		mes "Job EXP: " + .weekend_job_exp_rate + "%";
		mes "Drop: " + .weekend_drop_rate + "%";
		mes "Carta MVP: " + .current_mvp_card_rate + "%";
	}
	else {
		mes "Status: ^777777inativo^000000.";
		mes "Base EXP: " + .normal_base_exp_rate + "%";
		mes "Job EXP: " + .normal_job_exp_rate + "%";
		mes "Drop: " + .normal_drop_rate + "%";
		mes "Carta MVP: " + .normal_mvp_card_rate + "%";
	}
	mes "Feriado antecipado: " + ($Weekend_Rates_Early ? "^00AA00ativo^000000" : "^777777inativo^000000") + ".";
	mes "Carta MVP 5x armada: " + ($Weekend_Rates_MvpCard5x ? "^00AA00sim^000000" : "^777777nao^000000") + ".";
	mes "Carta MVP 5x evento semanal: " + ($Event_MvpCard5x ? "^00AA00sim^000000" : "^777777nao^000000") + ".";
	mes "Comandos: @ratesferiado ativar/desativar";
	mes "Comandos: @mvpcard5x ativar/desativar";
	close;

OnHolidayAtcommand:
	.@cmd$ = (.@atcmd_numparameters > 0 ? strtolower(.@atcmd_parameters$[0]) : "ativar");
	if (.@cmd$ == "on" || .@cmd$ == "ativar" || .@cmd$ == "1") {
		$Weekend_Rates_Early = 1;
		$Weekend_Rates_MvpCard5x = 1;
		callsub S_ApplyRates, 1;
		announce "[Evento] Feriado: rates especiais e carta MVP 5x ativados ate domingo as 23:59.", bc_all, 0x00FF00;
		dispbottom "[Weekend Rates] Feriado ativado. Carta MVP 5x tambem foi armada.";
		end;
	}
	if (.@cmd$ == "off" || .@cmd$ == "desativar" || .@cmd$ == "0") {
		$Weekend_Rates_Early = 0;
		$Weekend_Rates_MvpCard5x = 0;
		callsub S_ApplyRates, 1;
		announce "[Evento] Feriado encerrado: rates normais restauradas.", bc_all, 0x00AAFF;
		dispbottom "[Weekend Rates] Feriado e carta MVP 5x desativados.";
		end;
	}
	dispbottom "[Weekend Rates] Use: @ratesferiado ativar ou @ratesferiado desativar.";
	end;

OnMvpCardAtcommand:
	.@cmd$ = (.@atcmd_numparameters > 0 ? strtolower(.@atcmd_parameters$[0]) : "ativar");
	if (.@cmd$ == "on" || .@cmd$ == "ativar" || .@cmd$ == "1") {
		$Weekend_Rates_MvpCard5x = 1;
		callsub S_ApplyRates, 1;
		dispbottom "[Weekend Rates] Carta MVP 5x armada. Ela fica ativa enquanto as rates de fim de semana/feriado estiverem ativas.";
		end;
	}
	if (.@cmd$ == "off" || .@cmd$ == "desativar" || .@cmd$ == "0") {
		$Weekend_Rates_MvpCard5x = 0;
		callsub S_ApplyRates, 1;
		dispbottom "[Weekend Rates] Carta MVP 5x desativada.";
		end;
	}
	dispbottom "[Weekend Rates] Use: @mvpcard5x ativar ou @mvpcard5x desativar.";
	end;

S_IsWeekendEvent:
	.@day = gettime(DT_DAYOFWEEK);
	.@hour = gettime(DT_HOUR);

	if (.@day == 0 && .@hour == 23 && gettime(DT_MINUTE) >= 59) {
		$Weekend_Rates_Early = 0;
		$Weekend_Rates_MvpCard5x = 0;
	}

	if (.bonus_week_active)
		return 1;

	if ($Weekend_Rates_Early)
		return 1;

	// Quinta, sexta e sabado inteiros.
	if (.@day >= 4 && .@day <= 6)
		return 1;

	// Domingo ate 23:58; as 23:59 o evento encerra.
	if (.@day == 0 && !(.@hour == 23 && gettime(DT_MINUTE) >= 59))
		return 1;

	return 0;

S_IsMvpCardEvent:
	if (!$Weekend_Rates_MvpCard5x && !$Event_MvpCard5x)
		return 0;
	if ($Event_MvpCard5x)
		return 1;
	if (!callsub(S_IsWeekendEvent))
		return 0;
	return 1;

S_ApplyRates:
	.@announce = getarg(0);
	.@old_state = .weekend_rates_active;
	.@weekend = callsub(S_IsWeekendEvent);
	.@mvp_card_event = callsub(S_IsMvpCardEvent);

	if (.@weekend) {
		.@base_exp_rate = .weekend_base_exp_rate;
		.@job_exp_rate = .weekend_job_exp_rate;
		.@drop_rate = .weekend_drop_rate;
	}
	else {
		.@base_exp_rate = .normal_base_exp_rate;
		.@job_exp_rate = .normal_job_exp_rate;
		.@drop_rate = .normal_drop_rate;
	}
	.@mvp_card_rate = (.@mvp_card_event ? .weekend_mvp_card_rate : .normal_mvp_card_rate);

	callsub S_SetRates, .@base_exp_rate, .@job_exp_rate, .@drop_rate, .@mvp_card_rate;
	.weekend_rates_active = .@weekend;
	.current_mvp_card_rate = .@mvp_card_rate;

	if (.@announce && .@old_state != .weekend_rates_active) {
		if (.weekend_rates_active)
			announce "[Evento] Fim de semana iniciado: rates especiais ativadas!", bc_all, 0x00FF00;
		else
			announce "[Evento] Fim de semana encerrado: rates normais restauradas.", bc_all, 0x00AAFF;
	}
	return;

S_SetRates:
	.@base_exp_rate = getarg(0);
	.@job_exp_rate = getarg(1);
	.@drop_rate = getarg(2);
	.@mvp_card_rate = getarg(3);

	setbattleflag("base_exp_rate", .@base_exp_rate);
	setbattleflag("job_exp_rate", .@job_exp_rate);

	setbattleflag("item_rate_common", .@drop_rate);
	setbattleflag("item_rate_common_boss", .@drop_rate);
	setbattleflag("item_rate_common_mvp", .@drop_rate);
	setbattleflag("item_rate_heal", .@drop_rate);
	setbattleflag("item_rate_heal_boss", .@drop_rate);
	setbattleflag("item_rate_heal_mvp", .@drop_rate);
	setbattleflag("item_rate_use", .@drop_rate);
	setbattleflag("item_rate_use_boss", .@drop_rate);
	setbattleflag("item_rate_use_mvp", .@drop_rate);
	setbattleflag("item_rate_equip", .@drop_rate);
	setbattleflag("item_rate_equip_boss", .@drop_rate);
	setbattleflag("item_rate_equip_mvp", .@drop_rate);
	setbattleflag("item_rate_card", .@drop_rate);
	setbattleflag("item_rate_card_boss", .@drop_rate);
	setbattleflag("item_rate_card_mvp", .@mvp_card_rate);
	setbattleflag("item_rate_mvp", .@drop_rate);
	setbattleflag("item_rate_adddrop", .@drop_rate);
	setbattleflag("item_rate_treasure", .@drop_rate, true);
	return;

S_AnnounceActiveRates:
	if (!callsub(S_IsWeekendEvent))
		return;

	announce "[Evento] Rates de fim de semana ativas: Base EXP " + .weekend_base_exp_rate + "% / Job EXP " + .weekend_job_exp_rate + "% / Drop " + .weekend_drop_rate + "%." + ($Weekend_Rates_MvpCard5x ? " Carta MVP 5x ativa." : ""), bc_all, 0x00FF00;
	return;
}
