function	script	F_BravuraRefineTicket	{
	disable_items;

	.@ticket_id = getarg(0);
	setarray .@target_slots[0], EQI_HEAD_TOP, EQI_ARMOR, EQI_GARMENT, EQI_HAND_L, EQI_SHOES;
	setarray .@target_ids[0], 5170, 400027, 400026, 400028, 400025;

	.@menu$ = "";
	.@count = 0;

	for (.@i = 0; .@i < getarraysize(.@target_slots); ++.@i) {
		.@slot = .@target_slots[.@i];
		if (!getequipisequiped(.@slot))
			continue;

		.@equip_id = getequipid(.@slot);
		if (.@equip_id != .@target_ids[.@i])
			continue;

		.@refine = getequiprefinerycnt(.@slot);
		if (.@refine >= 9)
			continue;

		.@menu$ += getequipname(.@slot) + " [+" + .@refine + " -> +9]:";
		.@slot_list[.@count] = .@slot;
		.@id_list[.@count] = .@equip_id;
		++.@count;
	}

	mes "[Ticket de Refino Bravura]";
	if (.@count == 0) {
		mes "Equipe uma Boina Alada, Traje, Manteau, Escudo ou Sapatos da Bravura com refino menor que +9.";
		close;
	}

	mes "Escolha o equipamento que deseja refinar para +9.";
	.@pick = select(.@menu$) - 1;
	.@part = .@slot_list[.@pick];
	.@expected_id = .@id_list[.@pick];

	.@equip_id = getequipid(.@part);
	.@refine = getequiprefinerycnt(.@part);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	if (.@equip_id != .@expected_id || inarray(.@target_ids[0], .@equip_id) == -1) {
		mes "Este ticket so pode ser usado nos equipamentos elegiveis da Bravura.";
		close;
	}

	if (.@refine >= 9) {
		mes "Este equipamento ja esta com refino +9 ou superior.";
		close;
	}

	mes "Deseja refinar " + getitemname(.@equip_id) + " para +9?";
	if (select("Sim:Cancelar") == 2) {
		mes "Operacao cancelada.";
		close;
	}

	if (countitem(.@ticket_id) < 1) {
		mes "Voce nao possui o ticket necessario.";
		close;
	}

	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refine) ||
		callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3])) {
		mes "O equipamento foi alterado durante o processo.";
		close;
	}

	delitem .@ticket_id,1;
	successrefitem .@part, 9 - .@refine;
	specialeffect2 EF_REFINEOK;

	mes "O equipamento foi refinado com sucesso para +9.";
	close;
}
