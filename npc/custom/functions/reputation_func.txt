function	script	F_AddRep	{

	// args:
	// 0: rep_id        (Id em reputation.yml)
	// 1: add_points    (quanto vai adicionar)
	// 2: ach_id        (achievement id, ex: 110002)
	// 3: max_points    (Maximum em reputation.yml)

	.@rep_id     = getarg(0);
	.@add_points = getarg(1);
	.@ach_id     = getarg(2);
	.@max_points = getarg(3);
	.@db_max = callfunc("F_GetRepMax", .@rep_id);
	if (.@db_max > .@max_points)
		.@max_points = .@db_max;

	// adiciona reputacao
	if (.@rep_id == 7) {
		set_reputation_points(.@rep_id, .@add_points);
	} else {
		add_reputation_points(.@rep_id, .@add_points);
	}

	// pega valor atual
	.@cur = get_reputation_points(.@rep_id);

	// Espelha reputacoes compartilhadas em variaveis de conta para leitura imediata.
	switch (.@rep_id) {
		case 1: set #reput_instance, .@cur; break;
		case 3: set #reput_cheffenia, .@cur; break;
		case 4: set #reput_cheffenia_hard, .@cur; break;
		case 5: set #reput_bio5, .@cur; break;
		case 6: set #reput_bio5_hard, .@cur; break;
		case 8: set #reput_class_dominator, .@cur; break;
	}

	// se chegou no maximo e ainda nao tem o achievement, completa
	if (.@cur >= .@max_points && !achievementexists(.@ach_id)) {
		achievementcomplete(.@ach_id);
	}

	return .@cur;
}

function	script	F_SyncRepAchievement	{

	// args:
	// 0: rep_id        (Id em reputation.yml)
	// 1: ach_id        (achievement id)
	// 2: max_points    (Maximum em reputation.yml)
	//
	// Reputacao por conta:
	// completa o achievement do personagem atual se a conta ja atingiu o maximo.
	// Excecao: Reborn (rep_id 7) continua sendo exclusivo do personagem.

	.@rep_id     = getarg(0);
	.@ach_id     = getarg(1);
	.@max_points = getarg(2);
	.@db_max = callfunc("F_GetRepMax", .@rep_id);
	if (.@db_max > .@max_points)
		.@max_points = .@db_max;

	if (.@rep_id == 7)
		return;

	if (get_reputation_points(.@rep_id) >= .@max_points && !achievementexists(.@ach_id)) {
		achievementcomplete(.@ach_id);
	}

	return;
}


function	script	F_GetRepMax	{
	switch (getarg(0)) {
		case 1:
			return 10000;
		case 2:
			return 10000;
		case 3:
		case 4:
		case 5:
		case 6:
			return 10000;
		case 7:
		case 8:
			return 5000;
	}
	return getarg(1, 0);
}
function	script	F_PartyNearbyKillReward	{

	// 0: map$               mapa valido para a recompensa
	// 1: item_id            item/moeda
	// 2: item_rate          chance em base 10000
	// 3: counter_var$       nome da variavel de contador por personagem
	// 4: threshold          qtd de kills para premio de reputacao
	// 5: rep_id             id da reputacao
	// 6: rep_amount         pontos por ciclo
	// 7: ach_id             achievement da reputacao
	// 8: rep_cap            limite maximo
	// 9: rep_message$       mensagem ao ganhar reputacao
	// 10: radius            distancia maxima entre killer e membro

	.@map$ = getarg(0);
	.@item_id = getarg(1, 0);
	.@item_rate = getarg(2, 0);
	.@counter_var$ = getarg(3, "");
	.@threshold = getarg(4, 5);
	.@rep_id = getarg(5, 0);
	.@rep_amount = getarg(6, 0);
	.@ach_id = getarg(7, 0);
	.@rep_cap = getarg(8, 0);
	.@rep_message$ = getarg(9, "");
	.@radius = getarg(10, 30);

	.@killer_aid = getcharid(3);
	.@party_id = getcharid(1);
	getmapxy(.@killer_map$,.@killer_x,.@killer_y,BL_PC);

	if (.@killer_map$ != .@map$)
		return 0;

	if (.@party_id < 1) {
		if (.@counter_var$ != "")
			setd(.@counter_var$, getd(.@counter_var$) + 1);
		if (.@counter_var$ != "" && getd(.@counter_var$) >= .@threshold) {
			setd(.@counter_var$, 0);
			if (.@rep_id > 0 && .@rep_amount > 0)
				callfunc("F_AddRep", .@rep_id, .@rep_amount, .@ach_id, .@rep_cap);
			if (.@rep_message$ != "")
				dispbottom .@rep_message$;
		}
		if (.@item_id > 0 && .@item_rate > 0)
			callfunc("CustomMapDrop", .@map$, .@item_id, .@item_rate);
		return 1;
	}

	getpartymember(.@party_id,1);
	getpartymember(.@party_id,2);
	for (.@i = 0; .@i < $@partymembercount; .@i++) {
		if (!isloggedin($@partymemberaid[.@i], $@partymembercid[.@i]))
			continue;
		if (!attachrid($@partymemberaid[.@i]))
			continue;

		getmapxy(.@member_map$,.@member_x,.@member_y,BL_PC);
		if (.@member_map$ != .@map$)
			continue;
		if (distance(.@killer_x,.@killer_y,.@member_x,.@member_y) > .@radius)
			continue;

		if (.@counter_var$ != "")
			setd(.@counter_var$, getd(.@counter_var$) + 1);
		if (.@counter_var$ != "" && getd(.@counter_var$) >= .@threshold) {
			setd(.@counter_var$, 0);
			if (.@rep_id > 0 && .@rep_amount > 0)
				callfunc("F_AddRep", .@rep_id, .@rep_amount, .@ach_id, .@rep_cap);
			if (.@rep_message$ != "")
				dispbottom .@rep_message$;
		}
	}

	attachrid(.@killer_aid);
	// Uma unica rolagem por monstro morto. A entrega respeita autoloot e as
	// opcoes nativas de divisao de itens do grupo.
	if (.@item_id > 0 && .@item_rate > 0)
		callfunc("CustomMapDrop", .@map$, .@item_id, .@item_rate);
	return 1;
}
