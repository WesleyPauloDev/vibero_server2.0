inter01,0,0,0,0	boss_monster	Guia da Morte	22404,1,10800000,300000,1

function	script	F_MapCustomDropsKill	{
		.@mob_id = getarg(0, killedrid);
		if (strcharinfo(3) != "inter01")
			return;

		// Exemplo de chamada da funcao
		// Mapa: prt_fild08
		// Itens: 607, 501
		// Rates: 1%, 1%
		callfunc("CustomMapDrop", "inter01", 607, 1);
        callfunc("CustomMapDrop", "inter01", 9611, 1);

		// Mapa: inter01
        // Bonus: +50% BaseExp e +20% JobExp
        callfunc("MapExpBonus", "inter01", 100, 20);


		if (.@mob_id == 21400 && #d_monstros == 1) {
			#d_monstros_count++;
			// quando chegar a 100, marca como completa
			if (#d_monstros_count >= 100) {
				#d_monstros = 2;
				dispbottom "[Diaria] Voce derrotou 100 Anubis! Fale com o Mural para receber a recompensa.";
			} else {
				dispbottom "[Diaria] Anubis mortos: " + #d_monstros_count + "/100";
			}
		}
		return;
}
