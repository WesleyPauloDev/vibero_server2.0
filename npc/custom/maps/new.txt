		}

		end;
}

// ----------------------- Quest do Iniciante -----------------------
function	script	F_ContadorPoringKill	{
		.@mob_id = getarg(0, killedrid);

		if (quest_poring_started == 1 && .@mob_id == 1002) {
			quest_poring_count++;
			if (quest_poring_count < 20)
			message strcharinfo(0), "Poring [ "+quest_poring_count+" / 20 ]";
			else {
				message strcharinfo(0), "Poring Completo";
				quest_poring_started = 2;
			}

		}
		return;
}

new_1-1,60,123,4	script	Quest do Iniciante	810,{

	// quest_poring_done = 0;
	// quest_poring_started = 1;
	// quest_poring_count = 0;
	if (quest_poring_done) {
		mes "[Quest Iniciante]";
		mes "Voce ja completou essa missao.";
		close;
	}

	if (quest_poring_started == 0) {
		mes "[Quest Iniciante]";
		mes "Bem-vindo a Area de Treinamento!";
		mes "Gostaria de me ajudar matando 20 Porings?";
		mes "";
		mes "[Recompensa]";
		mes "- 100 Pocoes Vermelhas";
		mes "- 100 Pocoes Azuis";
		mes "- 5000 de experiencia base";
		next;
		if (select("Sim:Nao") == 2) close;

		quest_poring_started = 1;
		quest_poring_count = 0;
		mes "[Quest Iniciante]";
		mes "Boa sorte! Volte quando tiver derrotado 20 Porings.";
		close;
	}

	if (quest_poring_started == 1 && quest_poring_count < 20) {
		mes "[Quest Iniciante]";
		mes "Voce ainda nao matou 20 Porings.";
		mes "Progresso: " + quest_poring_count + " / 20";
		close;
	}

	if (quest_poring_started == 2 && quest_poring_count >= 20) {
		mes "[Quest Iniciante]";
		mes "Parabens! Voce completou a missao.";
		mes "Aqui esta sua recompensa:";
		mes "- 100 Pocoes Vermelhas";
		mes "- 100 Pocoes Azuis";
		mes "- 5000 de experiencia base";
		getitem 501, 100; // Pocao Vermelha do iniciante
		getitem 505, 100; // Pocao Azul do iniciante
		getexp 5000,0;      // Recompensa de experiencia base
		quest_poring_done = 1;
		quest_poring_started = 0;
		quest_poring_count = 0;
		close;
	}
}
