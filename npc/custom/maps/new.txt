		next;
		
		if (select("Sim, me leve para Prontera.:Ainda não.") == 1) {
			mes "[Examinador]";
			mes "Boa sorte em sua aventura!";
			close2;
			warp "prontera", 156, 191; // Coordenadas centrais de Prontera
			end;
		} else {
			mes "[Examinador]";
			mes "Tudo bem, volte quando estiver pronto.";
			close;
		}

		mes "Você precisa atingir o nível 10 de classe para prosseguir.";
		close;
	}
	OnInit:
		waitingroom "Examinador",0;
		end;
}

new_1-1,60,123,4	script	Quest do Iniciante	810,{

	if (quest_poring_done) {
		mes "[Quest Iniciante]";
		mes "Você já completou essa missão.";
		close;
	}

	if (quest_poring_started == 0) {
		mes "[Quest Iniciante]";
		mes "Bem-vindo à Área de Treinamento!";
		mes "Gostaria de me ajudar matando 10 Porings?";
		mes "";
		mes "[Recompensa]";
		mes "- 100 Poções Vermelhas";
		mes "- 100 Poções Azuis";
		mes "- 5000 de experiência base";
		next;
		if (select("Sim:Não") == 2) close;

		quest_poring_started = 1;
		quest_poring_count = 0;
		mes "[Quest Iniciante]";
		mes "Boa sorte! Volte quando tiver derrotado 10 Porings.";
		close;
	}

	if (quest_poring_started == 1 && quest_poring_count < 10) {
		mes "[Quest Iniciante]";
		mes "Você ainda não matou 10 Porings.";
		mes "Progresso: " + quest_poring_count + " / 10";
		close;
	}

	if (quest_poring_started == 1 && quest_poring_count >= 10) {
		mes "[Quest Iniciante]";
		mes "Parabéns! Você completou a missão.";
		mes "Aqui está sua recompensa:";
		mes "- 100 Poções Vermelhas";
		mes "- 100 Poções Azuis";
		getitem 501, 100; // Poção Vermelha do iniciante
		getitem 505, 100; // Poção Azul do iniciante
		getexp 5000,0;      // Recompensa de experiência base
		quest_poring_done = 1;
		quest_poring_started = 0;
		quest_poring_count = 0;
		close;
	}
}
