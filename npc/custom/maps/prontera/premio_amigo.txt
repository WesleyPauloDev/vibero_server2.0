-	script	Premio do Amigo#base	-1,{

	mes "[Premio do Amigo]";
	switch(select("Quero saber sobre o premio!","Um amigo me indicou!","Resgatar premio por indicacao","Sair")) {
		case 1:
			mes "Uma caixa especial recebida ao indicar um amigo.";
			mes "Contem os seguintes itens para quem indicou:";
			mes "^0055FF10x Goma de Mascar^000000";
			mes "^0055FF10x Super Pocao Media de Vida^000000";
			mes "^0055FF10x Macaroons Sortidos^000000";
			mes "^0055FF10x Taca da Felicidade^000000";
			mes "^0055FF4x Diario de Aventuras^000000";
			next;
			mes "Ao informar o nome de quem te indicou,";
			mes "voce recebe ^0055FF1 unidade de cada item^000000.";
			mes "Quem te indicou podera resgatar o pacote maior neste NPC.";
			close;
		case 2:
			break;
		case 3:
			callsub S_ClaimReferralReward;
			close;
		default:
			close;
	}

	.@is_gm = (getgroupid() >= 99);

	if (!.@is_gm && BaseLevel < 20) {
		mes "E necessario mais experiencia no jogo.";
		mes "Volte quando alcancar o nivel 20.";
		close;
	}

	// Cada conta so pode informar quem indicou uma unica vez.
	if (!.@is_gm && #friend_ref_set) {
		mes "Voce ja nos informou quem te indicou.";
		close;
	}

	.@ip$ = getcharip();

	while (true) {
		mes "Digite o nick de quem te indicou:";
		input .@friend$;

		query_sql("SELECT char_id, account_id, name FROM `char` WHERE name = '"+escape_sql(.@friend$)+"'", .@friend_id, .@friend_aid, .@friend_name$);
		if (!getarraysize(.@friend_id)) {
			mes "Nick nao encontrado. Deseja tentar novamente?";
			if (select("Tentar novamente","Cancelar") == 1)
				next;
			mes "Indicacao cancelada.";
			break;
		}

		// Impedir auto-indicacao (mesma conta)
		if (!.@is_gm && .@friend_aid[0] == getcharid(2)) {
			mes "Voce nao pode indicar a si proprio.";
			break;
		}

		// Impedir indicacao de quem joga do mesmo IP
		query_sql("SELECT last_ip FROM `login` WHERE account_id = "+.@friend_aid[0], .@friend_ip$);
		if (!.@is_gm && getarraysize(.@friend_ip$) > 0 && .@friend_ip$[0] == .@ip$) {
			mes "Voce nao pode indicar alguem que joga do mesmo IP.";
			break;
		}

		setarray .@reward_item_id[0], 12210, 22818, 12223, 12224, 14545;
		setarray .@small_reward_amount[0], 1, 1, 1, 1, 1;
		if (!checkweight2(.@reward_item_id, .@small_reward_amount)) {
			mes "Voce precisa liberar espaco ou peso no inventario antes de receber seu premio.";
			close;
		}

		// Registra quem indicou para bloquear novas alteracoes nesta conta.
		#friend_ref_set = 1;
		#friend_ref_name$ = .@friend_name$[0];
		#friend_ref_time = gettimetick(2);

		// Deixa o premio maior pendente para o indicador resgatar no NPC.
		.@pending_var$ = "$friend_ref_pending_"+.@friend_id[0];
		.@last_ref_var$ = "$friend_ref_last_"+.@friend_id[0]+"$";
		setd .@pending_var$, getd(.@pending_var$) + 1;
		setd .@last_ref_var$, strcharinfo(0);

		// Premia imediatamente quem foi indicado com 1 unidade de cada item.
		for (.@i = 0; .@i < getarraysize(.@reward_item_id); .@i++)
			getitem .@reward_item_id[.@i], .@small_reward_amount[.@i];
		#friend_ref_reward_time = gettimetick(2);

		next;
		mes "[Premio do Amigo]";
		mes "Nick encontrado! Sua indicacao foi registrada.";
		mes "Voce recebeu seu premio agora.";
		mes "Seu amigo podera resgatar o premio maior neste NPC.";
		break;
	}
	close;

S_ClaimReferralReward:
	.@pending_var$ = "$friend_ref_pending_"+getcharid(0);
	.@last_ref_var$ = "$friend_ref_last_"+getcharid(0)+"$";
	.@pending = getd(.@pending_var$);

	if (.@pending <= 0) {
		mes "Voce nao possui premios de indicacao para resgatar.";
		return;
	}

	mes "Voce possui ^0055FF"+.@pending+"^000000 premio(s) de indicacao pendente(s).";
	if (getd(.@last_ref_var$) != "")
		mes "Ultimo indicado: ^0055FF"+getd(.@last_ref_var$)+"^000000.";
	next;

	setarray .@reward_item_id[0], 12210, 22818, 12223, 12224, 14545;
	setarray .@reward_item_amount[0], 10, 10, 10, 10, 4;
	if (!checkweight2(.@reward_item_id, .@reward_item_amount)) {
		mes "Voce precisa liberar espaco ou peso no inventario antes de receber seu premio.";
		return;
	}

	for (.@i = 0; .@i < getarraysize(.@reward_item_id); .@i++)
		getitem .@reward_item_id[.@i], .@reward_item_amount[.@i];
	#friend_ref_claim_time = gettimetick(2);

	setd .@pending_var$, .@pending - 1;
	mes "Premio de indicacao resgatado com sucesso.";
	if (.@pending - 1 > 0)
		mes "Ainda resta(m) ^0055FF"+(.@pending - 1)+"^000000 premio(s) para resgatar.";
	return;
}
