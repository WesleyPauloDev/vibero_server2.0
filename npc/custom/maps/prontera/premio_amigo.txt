-	script	Premio do Amigo#base	-1,{

	mes "[Prêmio do Amigo]";
	switch(select("Quero saber sobre o prêmio!","Um amigo me indicou!","Resgatar prêmio por indicação","Sair")) {
		case 1:
			mes "Uma caixa especial recebida ao indicar um amigo.";
			mes "Contém os seguintes itens para quem indicou:";
			mes "^0055FF10x Goma de Mascar^000000";
			mes "^0055FF10x Super Pocao Media de Vida^000000";
			mes "^0055FF10x Macaroons Sortidos^000000";
			mes "^0055FF10x Taça da Felicidade^000000";
			mes "^0055FF4x Diario de Aventuras^000000";
			next;
			mes "Ao informar o nome de quem te indicou,";
			mes "você recebe ^0055FF1 unidade de cada item^000000.";
			mes "Quem te indicou poderá resgatar o pacote maior neste NPC.";
			close;
		case 2:
			break;
		case 3:
			callsub S_ClaimReferralReward;
			close;
		default:
			close;
	}

	if (BaseLevel < 20) {
		mes "É necessário mais experiência no jogo.";
		mes "Volte quando alcançar o nível 20.";
		close;
	}

	// Cada conta só pode informar quem indicou uma única vez.
	if (#friend_ref_set) {
		mes "Você já nos informou quem te indicou.";
		close;
	}

	.@ip$ = getcharip();

	while (true) {
		mes "Digite o nick de quem te indicou:";
		input .@friend$;

		query_sql("SELECT char_id, account_id, name FROM `char` WHERE name = '"+escape_sql(.@friend$)+"'", .@friend_id, .@friend_aid, .@friend_name$);
		if (!getarraysize(.@friend_id)) {
			mes "Nick não encontrado. Deseja tentar novamente?";
			if (select("Tentar novamente","Cancelar") == 1)
				next;
			mes "Indicação cancelada.";
			break;
		}

		// Impedir auto-indicação (mesma conta)
		if (.@friend_aid[0] == getcharid(2)) {
			mes "Você não pode indicar a si próprio.";
			break;
		}

		// Impedir indicação de quem joga do mesmo IP
		query_sql("SELECT last_ip FROM `login` WHERE account_id = "+.@friend_aid[0], .@friend_ip$);
		if (getgroupid() < 99 && getarraysize(.@friend_ip$) > 0 && .@friend_ip$[0] == .@ip$) {
			mes "Você não pode indicar alguém que joga do mesmo IP.";
			break;
		}

		setarray .@reward_item_id[0], 12210, 22818, 12223, 12224, 14545;
		setarray .@small_reward_amount[0], 1, 1, 1, 1, 1;
		if (!checkweight2(.@reward_item_id, .@small_reward_amount)) {
			mes "Você precisa liberar espaço ou peso no inventário antes de receber seu prêmio.";
			close;
		}

		// Registra quem indicou para bloquear novas alterações nesta conta.
		#friend_ref_set = 1;
		#friend_ref_name$ = .@friend_name$[0];
		#friend_ref_time = gettimetick(2);

		// Deixa o prêmio maior pendente para o indicador resgatar no NPC.
		.@pending_var$ = "$friend_ref_pending_"+.@friend_id[0];
		.@last_ref_var$ = "$friend_ref_last_"+.@friend_id[0]+"$";
		setd .@pending_var$, getd(.@pending_var$) + 1;
		setd .@last_ref_var$, strcharinfo(0);

		// Premia imediatamente quem foi indicado com 1 unidade de cada item.
		for (.@i = 0; .@i < getarraysize(.@reward_item_id); .@i++)
			getitem .@reward_item_id[.@i], .@small_reward_amount[.@i];
		#friend_ref_reward_time = gettimetick(2);

		next;
		mes "[Prêmio do Amigo]";
		mes "Nick encontrado! Sua indicação foi registrada.";
		mes "Você recebeu seu prêmio agora.";
		mes "Seu amigo poderá resgatar o prêmio maior neste NPC.";
		break;
	}
	close;

S_ClaimReferralReward:
	.@pending_var$ = "$friend_ref_pending_"+getcharid(0);
	.@last_ref_var$ = "$friend_ref_last_"+getcharid(0)+"$";
	.@pending = getd(.@pending_var$);

	if (.@pending <= 0) {
		mes "Você não possui prêmios de indicação para resgatar.";
		return;
	}

	mes "Você possui ^0055FF"+.@pending+"^000000 prêmio(s) de indicação pendente(s).";
	if (getd(.@last_ref_var$) != "")
		mes "Último indicado: ^0055FF"+getd(.@last_ref_var$)+"^000000.";
	next;

	setarray .@reward_item_id[0], 12210, 22818, 12223, 12224, 14545;
	setarray .@reward_item_amount[0], 10, 10, 10, 10, 4;
	if (!checkweight2(.@reward_item_id, .@reward_item_amount)) {
		mes "Você precisa liberar espaço ou peso no inventário antes de receber seu prêmio.";
		return;
	}

	for (.@i = 0; .@i < getarraysize(.@reward_item_id); .@i++)
		getitem .@reward_item_id[.@i], .@reward_item_amount[.@i];
	#friend_ref_claim_time = gettimetick(2);

	setd .@pending_var$, .@pending - 1;
	mes "Prêmio de indicação resgatado com sucesso.";
	if (.@pending - 1 > 0)
		mes "Ainda resta(m) ^0055FF"+(.@pending - 1)+"^000000 prêmio(s) para resgatar.";
	return;
}
