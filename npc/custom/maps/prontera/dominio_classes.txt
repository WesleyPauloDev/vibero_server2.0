-	script	Dominio de Classes#base	-1,{
OnTalk:
	// #class_mastered_4096 = 0;
	// changesex;
	// sleep 1000;
	// end;
	// #classes_mastered = 4;
	// set_reputation_points(8, 0);
	// set #reput_class_dominator, 0;

	mes "[Dominio das Classes]";
	mes "Saudacoes, guerreiro.";
	mes "Eu sou o guardiao do ^FFCC00Dominio das Classes^000000.";
	mes "Aqui, cada vez que voce atingir o nivel maximo com uma classe diferente sendo rate 1X,";
	mes "e sendo classe transcendente ou classe expandida,";
	mes "sua reputacao perante os deuses aumentara!";
	next;

	mes "[Dominio das Classes]";
	mes "Cada classe nova aceita, em rate 1X e com classe trans ou expandidas,";
	mes "Expandidas sao: Super Aprendiz | Doram | Ninja | Justiceiro e Taekwon e evolucoes,";
	mes "lhe dara ^00FF00pontos de reputacao^000000.";
	mes "Ao alcancar certos niveis de reputacao, voce ganhara bonus especiais permanentes!";
	next;

	// Tabela de classes (3rd e transclasses)
	setarray .@classes_id[0],
		4060, // Rune Knight T
		4073, // Royal Guard T
		4065, // Guillotine Cross T
		4079, // Shadow Chaser T
		4063, // Arch Bishop T
		4077, // Sura T
		4061, // Warlock T
		4074, // Sorcerer T
		4062, // Ranger T
		4075, // Minstrel
		// 4076, // Wanderer
		4064, // Mechanic
		4078, // Genetic
		4211, // Kagerou
		// 4212, // Oboro
		4215, // Rebellion
		4218, // Summoner
		4239, // Star Emperor
		4240, // Soul Reaper
		4190; // Super Novice Expandido

	switch(select("Registrar minha classe:Ver progresso das classes:Sair")) {
		case 1:
			if (exp_base_custom > 100) {
				mes "[Dominio das Classes]";
				mes "Somente Rate 1X serao reconhecidos pelos deuses.";
				close;
			}

			.@class_id = Class;

			// Normaliza classes e evolucoes para os IDs canonicos da tabela
			if ((eaclass() & EAJ_BASEMASK) == EAJ_NINJA || BaseClass == Job_Ninja || .@class_id == 4211 || .@class_id == 4212 || .@class_id == 4304 || .@class_id == 4305 || .@class_id == JOB_SHINKIRO || .@class_id == JOB_SHIRANUI) {
				.@class_id = 4211; // Ninja canonico
			} else if ((eaclass() & EAJ_BASEMASK) == EAJ_SUMMONER || BaseClass == Job_Summoner || .@class_id == 4218 || .@class_id == 4308 || .@class_id == JOB_SPIRIT_HANDLER) {
				.@class_id = Job_Summoner; // Doram canonico (4218)
			} else if ((eaclass() & EAJ_BASEMASK) == EAJ_GUNSLINGER || .@class_id == 4215 || .@class_id == 4306 || .@class_id == JOB_NIGHT_WATCH) {
				.@class_id = 4215; // Rebellion / Gunslinger canonico
			} else if (.@class_id == 4239 || .@class_id == 4302 || .@class_id == JOB_SKY_EMPEROR) {
				.@class_id = 4239; // Star Emperor canonico
			} else if (.@class_id == 4240 || .@class_id == 4303 || .@class_id == JOB_SOUL_ASCETIC) {
				.@class_id = 4240; // Soul Reaper canonico
			} else if (.@class_id == 4190 || .@class_id == 4307 || .@class_id == JOB_HYPER_NOVICE) {
				.@class_id = 4190; // Super Novice Expandido canonico
			} else if (.@class_id == 4076 || .@class_id == 4097 || .@class_id == JOB_TROUVERE) {
				.@class_id = 4075; // Minstrel/Wanderer canonico
			}

			.@rate_event_active = (getbattleflag("base_exp_rate") >= 300 || getvariableofnpc(.weekend_rates_active, "Weekend_Rates") || $Event_Exp2x);
			.@required_base = (.@rate_event_active ? 175 : 200);
			.@required_job = 70;
			if (BaseLevel < .@required_base || JobLevel < .@required_job) {
				mes "[Dominio das Classes]";
				mes "Voce ainda nao alcancou o nivel necessario (" + .@required_base + "/" + .@required_job + ").";
				close;
			}

			if (getd("#class_mastered_" + .@class_id) || (.@class_id == 4211 && (getd("#class_mastered_4211") || getd("#class_mastered_4212") || getd("#class_mastered_4304") || getd("#class_mastered_4305")))) {
				mes "[Dominio das Classes]";
				mes "Voce ja alcancou o dominio desta classe.";
				close;
			}

			// Verifica se a classe atual e elegivel
			.@eligible = 0;
			for (.@i = 0; .@i < getarraysize(.@classes_id); .@i++) {
				if (.@class_id == .@classes_id[.@i]) { .@eligible = 1; break; }
			}
			if (!.@eligible) {
				mes "[Dominio das Classes]";
				mes "Voce nao pode receber esta conquista com a classe atual.";
				mes "Classes nao expandidas precisam ser do set transcendente.";
				close;
				end;
			}

			setd("#class_mastered_" + .@class_id, 1);
			switch (.@class_id) {
				case 4075:
					setd("#class_mastered_4076", 1);
					break;
				case 4211:
					setd("#class_mastered_4211", 1);
					setd("#class_mastered_4212", 1);
					setd("#class_mastered_4304", 1);
					setd("#class_mastered_4305", 1);
					break;
			}
			#classes_mastered++;
			specialeffect2 EF_JOBLVUP50;

			mes "[Dominio das Classes]";
			mes "Parabens! Voce atingiu o nivel maximo com uma nova classe!";
			mes "Total de classes dominadas: ^00FF00" + #classes_mastered + "/18^000000.";

			// Recompensa
			// add_reputation_points(8, 300);
			callfunc("F_AddRep", 8, 300, 110009, 5000);

			close;

		case 2:
			mes "Classes conquistadas: ^00FF00" + #classes_mastered + "^000000";
			mes "Classes restantes: ^FF0000" + (18 - #classes_mastered) + "^000000";
			next;

			setarray .@classes_name$[0],
				"Rune Knight Transcendente",
				"Royal Guard Transcendente",
				"Guillotine Cross Transcendente",
				"Shadow Chaser Transcendente",
				"Arch Bishop Transcendente",
				"Sura Transcendente",
				"Warlock Transcendente",
				"Sorcerer Transcendente",
				"Ranger Transcendente",
				"Minstrel/Wanderer",
				// "Minstrel/Wanderer",
				"Mechanic",
				"Genetic",
				"Ninja (Shinkiro/Shiranui)",
				// "Kagerou/Oboro",
				"Rebellion",
				"Doram (Spirit Handler)",
				"Star Emperor",
				"Soul Reaper",
				"Super Aprendiz Expandido";

			// Listar progresso
			mes "==== ^00FF00Classes Conquistadas^000000 ====";
			for (.@i = 0; .@i < getarraysize(.@classes_id); .@i++) {
				switch (.@classes_id[.@i]) {
					case 4211: // Ninja (Kagerou/Oboro/Shinkiro/Shiranui)
						if (getd("#class_mastered_4211") || getd("#class_mastered_4212") || getd("#class_mastered_4304") || getd("#class_mastered_4305"))
							mes "^00FF00- " + .@classes_name$[.@i] + "^000000";
						break;
					case 4075: // Minstrel/Wanderer
						if (getd("#class_mastered_4075") || getd("#class_mastered_4076"))
							mes "^00FF00- " + .@classes_name$[.@i] + "^000000";
						break;
					default:
						if (getd("#class_mastered_" + .@classes_id[.@i]))
							mes "^00FF00- " + .@classes_name$[.@i] + "^000000";
						break;
				}
			}

			mes "===== ^FF0000Classes Restantes^000000 =====";
			for (.@i = 0; .@i < getarraysize(.@classes_id); .@i++) {
				switch (.@classes_id[.@i]) {
					case 4211: // Ninja (Kagerou/Oboro/Shinkiro/Shiranui)
						if (!getd("#class_mastered_4211") && !getd("#class_mastered_4212") && !getd("#class_mastered_4304") && !getd("#class_mastered_4305"))
							mes "^FF0000- " + .@classes_name$[.@i] + "^000000";
						break;
					case 4075: // Minstrel/Wanderer
						if (!getd("#class_mastered_4075") && !getd("#class_mastered_4076"))
							mes "^FF0000- " + .@classes_name$[.@i] + "^000000";
						break;
					default:
						if (!getd("#class_mastered_" + .@classes_id[.@i]))
							mes "^FF0000- " + .@classes_name$[.@i] + "^000000";
						break;
				}
			}
			close;

		case 3:
			mes "[Dominio das Classes]";
			mes "Retorne quando estiver pronto para provar seu dominio.";
			close;
	}
	end;
}
