-	script	Dominio de Classes#base	-1,{
OnTalk:
	// #class_mastered_4096 = 0;
	// changesex;
	// sleep 1000;
	// end;
	// #classes_mastered = 4;

	mes "[Domínio das Classes]";
	mes "Saudações, guerreiro.";
	mes "Eu sou o guardião do ^FFCC00Domínio das Classes^000000.";
	mes "Aqui, cada vez que você atingir o nível máximo com uma classe diferente sendo rate 1X,";
	mes "sua reputação perante os deuses aumentará!";
	next;

	mes "[Domínio das Classes]";
	mes "Cada classe nova que atingir o nível máximo (175/70) sendo rate 1X, lhe dará ^00FF00pontos de reputação^000000.";
	mes "Ao alcançar certos níveis de reputação, você ganhará bônus especiais permanentes!";
	next;

	// Tabela de classes (3rd e transclasses)
	setarray .@classes_id[0],
		4060, // Rune Knight T
		4073, // Royal Guard T
		4065, // Guillotine Cross T
		4079, // Shadow Chaser T
		4063, // Arch Bishop T
		4077, // Sura T
		4061, // Warlock T
		4074, // Sorcerer T
		4062, // Ranger T
		4075, // Minstrel
		// 4076, // Wanderer
		4064, // Mechanic
		4078, // Genetic
		4211, // Kagerou
		// 4212, // Oboro
		4215, // Rebellion
		4239, // Star Emperor
		4240, // Soul Reaper
		4190; // Super Novice Expandido

	switch(select("Registrar minha classe:Ver progresso das classes:Sair")) {
		case 1:
			if (exp_base_custom > 100) {
				mes "[Domínio das Classes]";
				mes "Somente Rate 1X serão reconhecidos pelos deuses.";
				close;
			}

			if (BaseLevel < 175 || JobLevel < 70) {
				mes "[Domínio das Classes]";
			mes "Você ainda não alcançou o nível necessário (175/70).";
				close;
			}	

			.@class_id = Class;

			if (getd("#class_mastered_" + .@class_id)) {
				mes "[Domínio das Classes]";
				mes "Você já alcançou o domínio desta classe.";
				close;
			}

			// Tabela de equivalncia entre classes masculino/feminino
			switch (.@class_id) {
				// ?? Minstrel <-> Wanderer
				case 4075: // Minstrel
					setd("#class_mastered_4076", 1);
					break;
				case 4076: // Wanderer
					setd("#class_mastered_4075", 1);
					break;

				// ?? Ninja <-> Kunoichi (Kagerou / Oboro)
				case 4211: // Kagerou
					setd("#class_mastered_4212", 1);
					break;
				case 4212: // Oboro
					setd("#class_mastered_4211", 1);
					break;
			}
			
			switch (.@class_id) {
				// Minstrel/Wanderer -> 4075
				case 4076: .@class_id = 4075; break;

				// Kagerou/Oboro -> 4211
				case 4212: .@class_id = 4211; break;
			}

			// --- Verifica se a classe atual  elegvel (pertence  lista) ---
			.@eligible = 0;
			for (.@i = 0; .@i < getarraysize(.@classes_id); .@i++) {
				if (.@class_id == .@classes_id[.@i]) { .@eligible = 1; break; }
			}
			if (!.@eligible) {
				mes "[Domínio das Classes]";
				mes "Você não pode receber esta conquista com a classe atual.";
				close;
				end;
			}

			setd("#class_mastered_" + .@class_id, 1);
			#classes_mastered++;
			specialeffect2 EF_JOBLVUP50;			

			mes "[Domínio das Classes]";
			mes "Parabéns! Você atingiu o nível máximo com uma nova classe!";
			mes "Total de classes dominadas: ^00FF00" + #classes_mastered + "/17^000000.";

			// Recompensa
			// add_reputation_points(8, 300);
			callfunc("F_AddRep", 8, 300, 110009, 5000);

			close;

		case 2:
			mes "Classes conquistadas: ^00FF00" + #classes_mastered + "^000000";
			mes "Classes restantes: ^FF0000" + (17 - #classes_mastered) + "^000000";
			next;

			setarray .@classes_name$[0],
				"Rune Knight Transcendente",
				"Royal Guard Transcendente",
				"Guillotine Cross Transcendente",
				"Shadow Chaser Transcendente",
				"Arch Bishop Transcendente",
				"Sura Transcendente",
				"Warlock Transcendente",
				"Sorcerer Transcendente",
				"Ranger Transcendente",
				"Minstrel/Wanderer",
				// "Minstrel/Wanderer",
				"Mechanic",
				"Genetic",
				"Kagerou/Oboro",
				// "Kagerou/Oboro",
				"Rebellion",
				"Star Emperor",
				"Soul Reaper",
				"Super Aprendiz Expandido";

			// Listar progresso
			mes "==== ^00FF00Classes Conquistadas^000000 ====";
			for (.@i = 0; .@i < getarraysize(.@classes_id); .@i++) {
				// setd("#class_mastered_" + .@classes_id[.@i], 0);
				switch (.@classes_id[.@i]) {
					case 4211: // Kagerou/Oboro
						// setd("#class_mastered_4212", 0);
						if (getd("#class_mastered_4211") || getd("#class_mastered_4212"))
							mes "^00FF00- " + .@classes_name$[.@i] + "^000000";
						break;
					case 4075: // Minstrel/Wanderer
						// setd("#class_mastered_4076", 0);
						if (getd("#class_mastered_4075") || getd("#class_mastered_4076"))
							mes "^00FF00- " + .@classes_name$[.@i] + "^000000";
						break;
					default:
						if (getd("#class_mastered_" + .@classes_id[.@i]))
							mes "^00FF00- " + .@classes_name$[.@i] + "^000000";
						break;
				}
			}

			mes "===== ^FF0000Classes Restantes^000000 =====";
			for (.@i = 0; .@i < getarraysize(.@classes_id); .@i++) {
				switch (.@classes_id[.@i]) {
					case 4211: // Kagerou/Oboro
						if (!getd("#class_mastered_4211") && !getd("#class_mastered_4212"))
							mes "^FF0000- " + .@classes_name$[.@i] + "^000000";
						break;
					case 4075: // Minstrel/Wanderer
						if (!getd("#class_mastered_4075") && !getd("#class_mastered_4076"))
							mes "^FF0000- " + .@classes_name$[.@i] + "^000000";
						break;
					default:
						if (!getd("#class_mastered_" + .@classes_id[.@i]))
							mes "^FF0000- " + .@classes_name$[.@i] + "^000000";
						break;
				}
			}
			close;

		case 3:
			mes "[Domínio das Classes]";
			mes "Retorne quando estiver pronto para provar seu domínio.";
			close;
	}
	end;
}
