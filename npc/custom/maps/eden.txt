function	script	F_Loja_Eden_Menu	{
	mes "[Loja do Eden]";

	switch(select("Consumiveis:Armas:Equipamentos:Set Mora:Trocar Pocoes:Sair")) {
		case 1:
			callshop "consumable_eden"; // Loja de Consumiveis
			end;

		case 2:
			callshop "weapons_eden"; // Loja de armas
			end;

		case 3:
			callshop "equipments_eden"; // Loja de equipamentos
			end;
		
		case 4:
			mes "Aqui voce encontra os equipamentos do Set Mora.";
			switch(select("Fisico:Magico:Voltar")) {
				case 1:
					callshop "mora_set_physical_eden";
					end;
				case 2:
					callshop "mora_set_magical_eden";
					end;
				default:
					close;
			}
			end;
		
		case 5:
			mes "Aqui voce pode trocar suas Pocoes[Iniciante] por outras do mesmo tipo.";
			switch(select("Trocar Pocoes de HP em Pocoes de SP:Trocar Pocoes de SP em Pocoes de HP")) {
				case 1:
					mes "Quantas Pocoes deseja trocar?";
					input .@amount;
					if (.@amount <= 0) {
						mes "Quantidade invalida.";
						end;
					}
					if (countitem(9660) < .@amount) {
						mes "Voce nao possui Pocoes de HP suficientes.";
						end;
					}
					delitem 9660, .@amount; // Remove as pocoes de HP
					getitem 9661, .@amount; // Da as pocoes de SP
					logshop "SCRIPT_EXCHANGE", "eden_potion_swap", 9661, .@amount, 0, 9660, .@amount, "", 0;
					end;
				case 2:
					mes "Quantas Pocoes deseja trocar?";
					input .@amount;
					if (.@amount <= 0) {
						mes "Quantidade invalida.";
						end;
					}
					if (countitem(9661) < .@amount) {
						mes "Voce nao possui Pocoes de SP suficientes.";
						end;
					}
					delitem 9661, .@amount; // Remove as pocoes de SP
					getitem 9660, .@amount; // Da as pocoes de HP
					logshop "SCRIPT_EXCHANGE", "eden_potion_swap", 9660, .@amount, 0, 9661, .@amount, "", 0;
					end;
			}
			end;

		default:
			mes "Volte sempre!";
			close;
	}
	end;
}
