prontera,198,188,3	script	Guia de Criacao	4_M_ALCHE_A,{
	disable_items;

	mes "[Guia de Criacao]";
	mes "Posso mostrar receitas de criacao.";
	next;

	while (1) {
		switch(select("Reacao Alquimica:Comidas Mixed Cooking:Sair")) {
		case 1:
			callsub L_ReacaoMenu;
			break;
		case 2:
			callsub L_CookMenu;
			break;
		default:
			close;
		}
	}

L_ReacaoMenu:
	while (1) {
		switch(select("Ver receitas:Buscar por ID do produto:Como usar:Sair")) {
		case 1:
			callsub L_List;
			break;
		case 2:
			callsub L_Search;
			break;
		case 3:
			mes "[Guia de Criacao]";
			mes "Use Reacao Alquimica e selecione os materiais exatos no inventario.";
			mes "Nao coloque item a mais, item a menos ou quantidade diferente.";
			mes "Se a combinacao bater com uma receita, o item sera produzido.";
			mes "A quantidade final pode variar conforme a receita.";
			mes "As chances e quantidades finais seguem a base skill_changematerial_db.txt.";
			next;
			break;
		default:
			return;
		}
	}

L_List:
	.@page = 0;
	.@per_page = 12;
	.@total = getarraysize(.product);
	.@pages = (.@total + .@per_page - 1) / .@per_page;

	while (1) {
		.@start = .@page * .@per_page;
		.@end = .@start + .@per_page;
		if (.@end > .@total)
			.@end = .@total;

		.@menu$ = "";
		for (.@i = .@start; .@i < .@end; .@i++)
			.@menu$ += .product_name$[.@i] + " [" + .product[.@i] + "]:";
		.@menu$ += "Proxima pagina:Pagina anterior:Buscar por ID:Voltar";

		mes "[Guia de Criacao]";
		mes "Pagina " + (.@page + 1) + " de " + .@pages + ".";
		mes "Escolha um produto para ver os materiais.";
		next;

		.@pick = select(.@menu$);
		.@visible = .@end - .@start;

		if (.@pick <= .@visible) {
			callsub L_ShowRecipe, .@start + .@pick - 1;
			next;
		} else if (.@pick == .@visible + 1) {
			.@page++;
			if (.@page >= .@pages)
				.@page = 0;
		} else if (.@pick == .@visible + 2) {
			.@page--;
			if (.@page < 0)
				.@page = .@pages - 1;
		} else if (.@pick == .@visible + 3) {
			callsub L_Search;
		} else {
			return;
		}
	}

L_Search:
	mes "[Guia de Criacao]";
	mes "Digite o ID do produto final.";
	mes "Exemplo: 999 para Aco.";
	next;
	input .@id;

	.@found = 0;
	for (.@i = 0; .@i < getarraysize(.product); .@i++) {
		if (.product[.@i] == .@id) {
			callsub L_ShowRecipe, .@i;
			.@found++;
			next;
		}
	}

	if (!.@found) {
		mes "[Guia de Criacao]";
		mes "Nao encontrei receita para esse produto.";
		next;
	}
	return;

L_ShowRecipe:
	.@i = getarg(0);
	mes "[Guia de Criacao]";
	mes "^0000CD" + .product_name$[.@i] + "^000000 [" + .product[.@i] + "]";
	mes "Use exatamente os materiais abaixo.";
	if (.out_min[.@i] == .out_max[.@i])
		mes "Resultado: " + .out_min[.@i] + "x se der certo.";
	else
		mes "Resultado: de " + .out_min[.@i] + " a " + .out_max[.@i] + "x se der certo.";
	mes "Chance base: " + (.out_rate[.@i] / 10) + "%.";
	mes "Materiais:";
	callsub L_ShowMaterial, .mat1[.@i], .amt1[.@i], .mat1_name$[.@i];
	callsub L_ShowMaterial, .mat2[.@i], .amt2[.@i], .mat2_name$[.@i];
	callsub L_ShowMaterial, .mat3[.@i], .amt3[.@i], .mat3_name$[.@i];
	return;

L_ShowMaterial:
	.@item = getarg(0);
	.@amount = getarg(1);
	.@name$ = getarg(2);
	if (.@item > 0)
		mes " - " + .@amount + "x " + .@name$ + " [" + .@item + "]";
	return;

L_CookMenu:
	while (1) {
		switch(select("Ver receitas:Como funciona:Simular meu resultado:Voltar")) {

			callsub L_CookList;
			break;

			break;
		case 3:

	mes "Escolha uma comida para ver a receita.";
	next;

	.@pick = select(.@menu$);
	if (.@pick > getarraysize(.cook_product))
		return;

	callsub L_CookShowRecipe, .@pick - 1;
	next;
	return;

L_CookShowRecipe:
	.@i = getarg(0);
	mes "[Guia de Criacao]";
	mes "^0000CD" + .cook_product_name$[.@i] + "^000000 [" + .cook_product[.@i] + "]";
	mes "Efeito: " + .cook_effect$[.@i];
	mes "Livro necessario:";
	mes " - Livro de Culinaria Avancado [11022]";
	mes "O livro precisa estar no inventario e nao e consumido.";
	mes "Materiais consumidos:";
	callsub L_ShowMaterial, .cook_mat1[.@i], .cook_amt1[.@i], .cook_mat1_name$[.@i];
	callsub L_ShowMaterial, .cook_mat2[.@i], .cook_amt2[.@i], .cook_mat2_name$[.@i];
	callsub L_ShowMaterial, .cook_mat3[.@i], .cook_amt3[.@i], .cook_mat3_name$[.@i];
	callsub L_ShowMaterial, .cook_mat4[.@i], .cook_amt4[.@i], .cook_mat4_name$[.@i];
	callsub L_ShowMaterial, .cook_mat5[.@i], .cook_amt5[.@i], .cook_mat5_name$[.@i];
	callsub L_ShowMaterial, .cook_mat6[.@i], .cook_amt6[.@i], .cook_mat6_name$[.@i];
	mes "Use exatamente estes materiais.";
	mes "Lv1: tenta criar 1 unidade.";
	mes "Lv2: pode criar 5, 8, 10 ou 10~11 unidades.";
	return;

L_CookHelp:
	mes "[Guia de Criacao]";
	mes "Todas precisam do Livro de Culinaria Avancado no inventario.";

L_CookSimulate:
	.@base = JobLevel / 4 + readparam(bLuk) / 2 + readparam(bDex) / 3;
	if (.@base < 46)
		.@lv1 = 0;
	else if (.@base >= 165)
		.@lv1 = 100;
	else
		.@lv1 = ((.@base - 45) * 100) / 120;

	.@result$ = "";
	if (.@base <= 134)
		.@result$ = "falha";
	if (.@base >= 16 && .@base <= 154) {
		if (.@result$ != "")
			.@result$ += ", ";
		.@result$ += "5 comidas";
	}
	if (.@base >= 36 && .@base <= 174) {
		if (.@result$ != "")
			.@result$ += ", ";
		.@result$ += "8 comidas";
	}
	if (.@base >= 56 && .@base <= 194) {
		if (.@result$ != "")
			.@result$ += ", ";
		.@result$ += "10 comidas";
	}
	if (.@base >= 76) {
		if (.@result$ != "")
			.@result$ += ", ";
		.@result$ += "10 ou 11 comidas";
	}

	mes "[Guia de Criacao]";
	mes "Pontuacao atual: " + .@base + ".";
	mes "Quanto maior a pontuacao, melhor o resultado.";
	mes "Lv1: chance aproximada de " + .@lv1 + "% para criar 1 comida.";
	mes "Lv2: pode resultar em " + .@result$ + ".";
	mes "DEX, LUK e Job Level aumentam essa pontuacao.";
	next;
	return;
