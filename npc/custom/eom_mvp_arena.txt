//=============================================================
// Arena MVP EOM
// Instancia custom para uso de Galho Sangrento por grupo
//=============================================================

prontera,157,188,4	script	Arena MVP EOM	4_EP19_IWIN,{
	.@instance_name$ = "Arena MVP EOM";

	mes "[Arena MVP]";
	mes "^0055FFUse esta sala para farmar MVPs com Galho Sagrento^000000.";
	mes "Somente membros do seu grupo poderão entrar.";
	next;

	if (getcharid(1) < 1) {
		mes "[Arena MVP]";
		mes "Você precisa estar em um grupo.";
		close;
	}

	mes "[Arena MVP]";
	mes "Deseja entrar na sala do seu grupo?";
	if (select("Entrar agora:Cancelar") == 2)
		close;

	switch (instance_enter(.@instance_name$)) {
	case IE_OK:
		goto L_EnterArena;
	case IE_NOINSTANCE:
		break;
	default:
		mes "[Arena MVP]";
		mes "Não foi possível entrar na sala agora.";
		mes "Tente novamente em instantes.";
		close;
	}

	if (!is_party_leader()) {
		mes "[Arena MVP]";
		mes "O líder do grupo ainda não criou a sala.";
		mes "Peça para ele falar comigo primeiro.";
		close;
	}

	mes "[Arena MVP]";
	mes "Não existe uma sala aberta para o seu grupo.";
	mes "Deseja criar uma nova agora?";
	if (select("Criar sala:Cancelar") == 2)
		close;

	if (instance_create(.@instance_name$) < 0) {
		mes "[Arena MVP]";
		mes "Não foi possível criar a instância agora.";
		mes "Verifique se seu grupo já não possui uma instância ativa.";
		close;
	}

	if (instance_enter(.@instance_name$) != IE_OK) {
		mes "[Arena MVP]";
		mes "A instância foi criada, mas não foi possível entrar agora.";
		mes "Tente falar comigo novamente.";
		close;
	}

L_EnterArena:
	.@map$ = instance_mapname("gs_room");
	if (.@map$ != "") {
		removemapflag .@map$, mf_nobranch;
		removemapflag .@map$, MF_GVG;
		removemapflag .@map$, MF_GVG_NOPARTY;
		removemapflag .@map$, MF_GVG_CASTLE;
		removemapflag .@map$, MF_GVG_TE;
		removemapflag .@map$, MF_GVG_TE_CASTLE;
		removemapflag .@map$, MF_SNOW;
		removemapflag .@map$, MF_FOG;
		removemapflag .@map$, MF_SAKURA;
		removemapflag .@map$, MF_LEAVES;
		removemapflag .@map$, MF_CLOUDS;
		removemapflag .@map$, MF_CLOUDS2;
		removemapflag .@map$, MF_FIREWORKS;
		warp .@map$,50,50;
	}
	dispbottom "Você entrou na Arena MVP.";
	end;
}
