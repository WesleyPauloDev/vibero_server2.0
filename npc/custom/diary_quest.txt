function	script	F_DiaryRefreshDaily	{
	if (#cooldown_diario <= gettimetick(2)) {
		#d_monstros = 1;
		#d_monstros_count = 0;
		#instancia = 1;
		#voto = 1;
		#evento = 1;
		#todas = 1;
		#todas_count = 0;
		#cooldown_diario = gettimetick(2) + callfunc("get_daily_reset");
	}
	return;
}
function	script	F_DiaryRefreshWeekly	{
	if (#cooldown_semanal <= gettimetick(2)) {
		#torre = 1;
		#boss = 1;
		#todas_semanais = 1;
		#semanal_count = 0;
		#cooldown_semanal = gettimetick(2) + callfunc("get_weekly_reset");
	}
	return;
}

function	script	F_DiaryRewardLog	{
	.@type$ = getarg(0);
	.@key$ = getarg(1);
	.@name$ = getarg(2);
	.@item_id = getarg(3);
	.@amount = getarg(4);
	.@progress = getarg(5, 0);
	.@goal = getarg(6, 0);

	if (.@type$ == "weekly")
		.@period_expr$ = "DATE_FORMAT(DATE_SUB(CURDATE(), INTERVAL (DAYOFWEEK(CURDATE()) - 1) DAY), '%Y-%m-%d')";
	else
		.@period_expr$ = "DATE_FORMAT(CURDATE(), '%Y-%m-%d')";

	.@already_claimed = 0;
	query_logsql(
		"SELECT COUNT(*) FROM `diary_quest_reward_log` " +
		"WHERE `account_id` = " + getcharid(3) + " " +
		"AND `quest_type` = '" + escape_sql(.@type$) + "' " +
		"AND `quest_key` = '" + escape_sql(.@key$) + "' " +
		"AND `period_key` = " + .@period_expr$,
		.@already_claimed
	);

	if (.@already_claimed > 0) {
		logmes "[DIARY_QUEST_DUPLICATE_BLOCK] acc=" + getcharid(3) + " char=" + getcharid(0) + " quest=" + .@type$ + ":" + .@key$;
		return 0;
	}

	query_logsql(
		"INSERT INTO `diary_quest_reward_log` " +
		"(`account_id`, `char_id`, `char_name`, `quest_type`, `quest_key`, `quest_name`, `period_key`, " +
		"`reward_item_id`, `reward_item_name`, `reward_amount`, `progress_value`, `progress_goal`, `map`, `claimed_at`) VALUES (" +
		getcharid(3) + ", " +
		getcharid(0) + ", '" +
		escape_sql(strcharinfo(0)) + "', '" +
		escape_sql(.@type$) + "', '" +
		escape_sql(.@key$) + "', '" +
		escape_sql(.@name$) + "', " +
		.@period_expr$ + ", " +
		.@item_id + ", '" +
		escape_sql(getitemname(.@item_id)) + "', " +
		.@amount + ", " +
		.@progress + ", " +
		.@goal + ", '" +
		escape_sql(strcharinfo(3)) + "', NOW())"
	);

	return 1;
}
