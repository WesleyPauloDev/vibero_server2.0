prontera,164,159,4	script	Mestre de Grade	813,{

    mes "[Mestre de Grade]";
    mes "Olá aventureiro!";
    mes "Eu sou o responsável por aprimorar a *Grade* dos seus equipamentos.";
    next;

    switch(select("Gradear","Saber mais","Sair")) {

        // ================================
        // ? OPÇÃO 1 – Abrir a interface
        // ================================
        case 1:
            close2;
            atcommand("@enchantgradeui");
            end;

        // ================================
        // ? OPÇÃO 2 – Saber mais
        // ================================
        case 2:

            mes "[O que é Grade?]";
            mes "A Grade é um sistema especial de aprimoramento que melhora a qualidade do equipamento.";
            mes "Ela aumenta atributos como ataque, defesa e bônus adicionais, dependendo do tipo da peça.";
            next;

            mes "[Quais equipamentos podem receber Grade?]";
            mes "Apenas certos equipamentos são compatíveis com o sistema de Grade:";
            mes "• ^00AAFFArmaduras de nível 2^000000;";
            mes "• ^00AAFFArmas de nível 5^000000.";
            mes "Itens que não se enquadram nesses níveis não podem ser gradeados.";
            next;

            mes "[Requisitos mínimos]";
            mes "Para tentar aplicar Grade, o equipamento precisa estar refinado apartir:";
            mes "• ^FFAA00+9^000000 ^777777None^000000 | ^FFAA00+10^000000 Grade D e ^FFAA00+11^000000 Grade C e B.";
            mes "As chances de sucesso são baseadas no nível de refino:";
            mes "Por exemplo para deixar na Grade D";
            mes "• ^00AAFF+9 -> 10% de chance^000000;";
            mes "• ^00AAFF+10 -> 20% de chance^000000;";
            mes "• ^00AAFF+11 ou mais -> 70% de chance^000000.";
            next;

            mes "[Quais são os níveis de Grade?]";
            mes "Os níveis de Grade são:";
            mes " ^777777None^000000 | ^FFDD00D^000000 | ^FFC800C^000000 | ^FF9900B^000000 | ^FF5500A^000000";
            mes "Quanto mais alta a Grade:";
            mes "• Menores são as chances de Sucesso!";
            mes "• Melhores são os bônus!";
            next;

            mes "[Como funciona o aprimoramento?]";
            mes "• Em alguns casos de falha, o equipamento pode quebrar;";
            mes "• Catalisadores aumentam a chance de sucesso.";
            next;

            mes "[Quais materiais são necessários?]";
            mes "Você precisará de itens especiais como Etel Dust, Etel Topaz e outros materiais.";
            mes "Eles podem ser obtidos ou comprados na ^00AAFFLoja de Instância^000000.";
            next;

            mes "Deseja tentar aprimorar a Grade do seu equipamento agora?";
            if (select("Sim","Não") == 2) close;

            close2;
            atcommand("@enchantgradeui");
            end;

        // ================================
        // ? OPÇÃO 3 – Sair
        // ================================
        case 3:
            close;
    }

}
