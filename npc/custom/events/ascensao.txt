//============================================================
// Evento Ascensao
// - Entrada: 100.000 Zeny por acesso.
// - Limite: 4 horas diarias por conta.
// - MVP: liberado a cada 2.000 monstros, com 2 horas de intervalo.
//============================================================

function	script	F_AscensaoResetDaily	{
	.@today = gettime(DT_YYYYMMDD);
	if (#AscensaoDay != .@today) {
		#AscensaoDay = .@today;
		#AscensaoUsed = 0;
		if (#AscensaoStart)
			#AscensaoStart = gettimetick(2);
	}
	return;
}

function	script	F_AscensaoCommitTime	{
	callfunc "F_AscensaoResetDaily";
	.@now = gettimetick(2);
	if (#AscensaoStart > 0) {
		.@elapsed = .@now - #AscensaoStart;
		if (.@elapsed > 0)
			#AscensaoUsed += .@elapsed;
		#AscensaoStart = .@now;
	}
	if (#AscensaoUsed > 14400)
		#AscensaoUsed = 14400;
	return 14400 - #AscensaoUsed;
}

prontera,163,244,4	script	Guardiao da Ascensao	4_EP18_MARAM,{
	if (#AscensaoStart > 0) {
		.@remaining = callfunc("F_AscensaoCommitTime");
		#AscensaoStart = 0;
	} else {
		callfunc "F_AscensaoResetDaily";
		.@remaining = 14400 - #AscensaoUsed;
	}

	.@hours = .@remaining / 3600;
	.@minutes = (.@remaining % 3600) / 60;
	mes "[Guardiao da Ascensao]";
	mes "A passagem custa ^FF0000100.000 Zeny^000000 por entrada.";
	mes "Cada conta pode permanecer no mapa por ate quatro horas por dia.";
	mes "Tempo restante hoje: ^0055FF" + sprintf("%02d:%02d", .@hours, .@minutes) + "^000000.";
	if (.@remaining <= 0) {
		mes "Seu limite diario foi esgotado. Volte amanha.";
		close;
	}
	next;
	if (select("Entrar na Ascensao:Agora nao") == 2)
		close;
	if (Zeny < 100000) {
		mes "[Guardiao da Ascensao]";
		mes "Voce precisa de 100.000 Zeny para entrar.";
		close;
	}

	Zeny -= 100000;
	#AscensaoStart = gettimetick(2);
	@AscensaoLastMinute = -1;
	deltimer "AscensaoController::OnPlayerTimer";
	close2;
	warp "ascensao", 199, 152;
	end;
}

-	script	AscensaoController	-1,{
OnInit:
	.map$ = "ascensao";
	.npc$ = strnpcinfo(3);
	.kill_goal = 2000;
	.mvp_cooldown = 7200;
	setarray .mob_id[0], 31910, 31911, 31912, 31913;
	setarray .mob_name$[0], "Sentinela da Ascensao", "Basilisco da Ascensao", "Escavador da Ascensao", "Emissario da Ascensao";
	setarray .mob_event$[0], "OnSentinelaDead", "OnBasiliscoDead", "OnEscavadorDead", "OnEmissarioDead";

	.@restore_mvp = $AscensaoMvpAlive;
	for (.@i = 0; .@i < 4; .@i++)
		killmonster .map$, .npc$ + "::" + .mob_event$[.@i];
	killmonster .map$, .npc$ + "::OnMvpDead";
	$AscensaoMvpAlive = 0;
	if (.@restore_mvp)
		$AscensaoMvpPending = 1;

	for (.@i = 0; .@i < 4; .@i++)
		monster .map$, 0, 0, .mob_name$[.@i], .mob_id[.@i], 100, .npc$ + "::" + .mob_event$[.@i];

	// Recria os timers pessoais depois de um reload de scripts.
	getmapunits(BL_PC, .map$, .@units);
	for (.@i = 0; .@i < getarraysize(.@units); .@i++) {
		if (attachrid(.@units[.@i])) {
			if (#AscensaoStart > 0) {
				@AscensaoLastMinute = -1;
				deltimer .npc$ + "::OnPlayerTimer";
				addtimer 1000, .npc$ + "::OnPlayerTimer";
			}
			detachrid;
		}
	}

	initnpctimer;
	if ($AscensaoMvpPending && gettimetick(2) >= $AscensaoMvpNext)
		callsub S_SpawnMvp;
	end;

OnPCLoginEvent:
	// O tempo desconectado nunca e cobrado. O logout normal ja consolidou o uso.
	deltimer .npc$ + "::OnPlayerTimer";
	#AscensaoStart = 0;
	callfunc "F_AscensaoResetDaily";
	end;

OnPCLogoutEvent:
	if (#AscensaoStart > 0) {
		callfunc "F_AscensaoCommitTime";
		#AscensaoStart = 0;
	}
	end;

OnPCLoadMapEvent:
	if (strcharinfo(3) != .map$)
		end;
	if (#AscensaoStart <= 0) {
		dispbottom "[Ascensao] Entrada invalida. Fale com o Guardiao em Prontera.";
		warp "prontera", 163, 247;
		end;
	}
	@AscensaoLastMinute = -1;
	deltimer .npc$ + "::OnPlayerTimer";
	addtimer 1000, .npc$ + "::OnPlayerTimer";
	end;

OnPlayerTimer:
	.@remaining = callfunc("F_AscensaoCommitTime");
	if (strcharinfo(3) != .map$) {
		#AscensaoStart = 0;
		end;
	}
	if (.@remaining <= 0) {
		#AscensaoStart = 0;
		dispbottom "[Ascensao] Suas quatro horas diarias terminaram.";
		warp "prontera", 163, 247;
		end;
	}

	.@remaining_minutes = (.@remaining + 59) / 60;
	.@show_time = 0;
	if (@AscensaoLastMinute < 0)
		.@show_time = 1;
	else if (.@remaining_minutes <= 5 && @AscensaoLastMinute != .@remaining_minutes)
		.@show_time = 1;
	else if (.@remaining_minutes > 5 && .@remaining_minutes <= @AscensaoLastMinute - 10)
		.@show_time = 1;

	if (.@show_time) {
		@AscensaoLastMinute = .@remaining_minutes;
		dispbottom "[Ascensao] Tempo restante: " + sprintf("%02d:%02d", .@remaining_minutes / 60, .@remaining_minutes % 60) + ".";
	}
	addtimer 10000, .npc$ + "::OnPlayerTimer";
	end;

OnSentinelaDead:
	callsub S_NormalDead, 31910, "Sentinela da Ascensao", "OnSentinelaDead";
	end;

OnBasiliscoDead:
	callsub S_NormalDead, 31911, "Basilisco da Ascensao", "OnBasiliscoDead";
	end;

OnEscavadorDead:
	callsub S_NormalDead, 31912, "Escavador da Ascensao", "OnEscavadorDead";
	end;

OnEmissarioDead:
	callsub S_NormalDead, 31913, "Emissario da Ascensao", "OnEmissarioDead";
	end;

S_NormalDead:
	.@mob_id = getarg(0);
	.@mob_name$ = getarg(1);
	.@event$ = getarg(2);
	$AscensaoKills++;

	if (($AscensaoKills % 100) == 0)
		mapannounce .map$, "[Ascensao] Progresso para o MVP: " + $AscensaoKills + "/" + .kill_goal + ".", bc_map;

	if ($AscensaoKills >= .kill_goal) {
		$AscensaoKills = 0;
		if ($AscensaoMvpAlive) {
			if (!$AscensaoMvpPending)
				mapannounce .map$, "[Ascensao] Outra invocacao foi preparada e aguardara o intervalo do MVP.", bc_map;
			$AscensaoMvpPending = 1;
		} else if (gettimetick(2) >= $AscensaoMvpNext) {
			callsub S_SpawnMvp;
		} else {
			$AscensaoMvpPending = 1;
			.@wait = $AscensaoMvpNext - gettimetick(2);
			mapannounce .map$, "[Ascensao] Meta concluida. O MVP aparecera em " + sprintf("%02d:%02d", .@wait / 3600, (.@wait % 3600) / 60) + ".", bc_map;
		}
	}

	sleep 5000;
	monster .map$, 0, 0, .@mob_name$, .@mob_id, 1, .npc$ + "::" + .@event$;
	return;

S_SpawnMvp:
	if ($AscensaoMvpAlive)
		return;
	if (mobcount(.map$, .npc$ + "::OnMvpDead") > 0) {
		$AscensaoMvpAlive = 1;
		$AscensaoMvpPending = 0;
		return;
	}
	monster .map$, 199, 191, "Deusa da Ascensao", 31914, 1, .npc$ + "::OnMvpDead";
	$AscensaoMvpAlive = 1;
	$AscensaoMvpPending = 0;
	mapannounce .map$, "[Ascensao] A Deusa da Ascensao despertou!", bc_map;
	return;

OnMvpDead:
	$AscensaoMvpAlive = 0;
	$AscensaoMvpNext = gettimetick(2) + .mvp_cooldown;
	.@killer$ = strcharinfo(0);
	if (.@killer$ == "")
		.@killer$ = "um aventureiro";
	announce "[Ascensao] A Deusa da Ascensao foi derrotada por " + .@killer$ + "! O intervalo de duas horas foi iniciado.", bc_all;
	callfunc "F_AscensaoMvpRareDrops";
	end;

OnTimer60000:
	if ($AscensaoMvpAlive && mobcount(.map$, .npc$ + "::OnMvpDead") <= 0) {
		$AscensaoMvpAlive = 0;
		$AscensaoMvpNext = gettimetick(2) + .mvp_cooldown;
	}
	if ($AscensaoMvpPending && !$AscensaoMvpAlive && gettimetick(2) >= $AscensaoMvpNext)
		callsub S_SpawnMvp;
	setnpctimer 0;
	end;
}

// Consumivel do MVP: insere uma pedra de classe em um slot oculto livre
// de um dos acessorios equipados, sem substituir cartas ou encantamentos.
function	script	F_AscensaoBrightEnchant	{
	disable_items;

	.@consumable_id = getarg(0, 40118);
	setarray .@scan_part[0], EQI_ACC_L, EQI_ACC_R;
	setarray .@part_name$[0], "Acessorio esquerdo", "Acessorio direito";

	.@menu$ = "";
	.@count = 0;
	for (.@i = 0; .@i < getarraysize(.@scan_part); ++.@i) {
		.@part = .@scan_part[.@i];
		if (!getequipisequiped(.@part))
			continue;

		.@free_slot = -1;
		for (.@slot = 1; .@slot <= 3; ++.@slot) {
			if (getequipcardid(.@part, .@slot) == 0) {
				.@free_slot = .@slot;
				break;
			}
		}
		if (.@free_slot < 0)
			continue;

		.@available_part[.@count] = .@part;
		.@available_slot[.@count] = .@free_slot;
		.@menu$ += .@part_name$[.@i] + " - " + getequipname(.@part) + ":";
		++.@count;
	}

	mes "[Encantamento Brilhante]";
	if (.@count == 0) {
		mes "Equipe ao menos um acessorio com um slot oculto livre.";
		mes "Cartas e encantamentos existentes nao serao substituidos.";
		close;
	}

	mes "Escolha o acessorio que recebera a pedra.";
	next;
	.@pick = select(.@menu$) - 1;
	.@part = .@available_part[.@pick];
	.@target_slot = .@available_slot[.@pick];
	.@equip_id = getequipid(.@part);
	.@equip_refine = getequiprefinerycnt(.@part);
	setarray .@equip_card[0],
		getequipcardid(.@part, 0),
		getequipcardid(.@part, 1),
		getequipcardid(.@part, 2),
		getequipcardid(.@part, 3);

	setarray .@stone_id[0],
		40119, 40120, 40121, 40122, 40123, 40124, 40125, 40126,
		40127, 40128, 40129, 40130, 40131, 40132, 40133, 40134;
	setarray .@job_id[0],
		JOB_SURA, JOB_RANGER, JOB_WARLOCK, JOB_RUNE_KNIGHT,
		JOB_MECHANIC, JOB_ARCH_BISHOP, JOB_GUILLOTINE_CROSS, JOB_ROYAL_GUARD,
		JOB_SORCERER, JOB_MINSTREL, JOB_GENETIC, JOB_STAR_EMPEROR,
		JOB_SOUL_REAPER, JOB_KAGEROU, JOB_REBELLION, JOB_SUMMONER;
	setarray .@class_name$[0],
		"Shura", "Ranger", "Arcano", "Cavaleiro Runico",
		"Mecanico", "Arcebispo", "Sicario", "Guardiao Real",
		"Feiticeiro", "Trovador/Musa", "Bioquimico", "Imperador Estelar",
		"Ceifador de Almas", "Kagerou/Oboro", "Insurgente", "Doram";

	.@class_menu$ = "";
	for (.@i = 0; .@i < getarraysize(.@stone_id); ++.@i)
		.@class_menu$ += .@class_name$[.@i] + ":";

	mes "[Encantamento Brilhante]";
	mes "Acessorio: ^0055FF" + getitemname(.@equip_id) + "^000000";
	mes "Agora escolha a classe da pedra.";
	next;
	.@class_pick = select(.@class_menu$) - 1;
	.@stone = .@stone_id[.@class_pick];
	.@class = .@job_id[.@class_pick];

	// Evita consumir outro cristal para uma pedra que ja esta equipada.
	for (.@i = 0; .@i < getarraysize(.@scan_part); ++.@i) {
		if (!getequipisequiped(.@scan_part[.@i]))
			continue;
		for (.@slot = 0; .@slot <= 3; ++.@slot) {
			if (getequipcardid(.@scan_part[.@i], .@slot) == .@stone) {
				mes "[Encantamento Brilhante]";
				mes "Voce ja possui esta pedra em um acessorio equipado.";
				mes "Duas pedras da mesma classe nao acumulam o bonus.";
				close;
			}
		}
	}

	mes "[Encantamento Brilhante]";
	mes "Pedra: ^00AA00" + getitemname(.@stone) + "^000000";
	mes "Skills afetadas em ^0055FF50%^000000 do cooldown base:";
	for (.@i = 0; .@i < 10; ++.@i) {
		.@skill_name$ = callfunc("F_ShadowSkillData", .@class, .@i, 1);
		if (.@skill_name$ == "")
			break;
		mes "- " + .@skill_name$;
	}
	mes "A pedra sera inserida sem apagar os outros slots.";
	next;
	if (select("Confirmar encantamento:Cancelar") == 2) {
		mes "[Encantamento Brilhante]";
		mes "Operacao cancelada. O cristal nao foi consumido.";
		close;
	}

	if (countitem(.@consumable_id) < 1) {
		mes "[Encantamento Brilhante]";
		mes "O cristal nao foi encontrado no inventario.";
		close;
	}
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) ||
		callfunc("F_IsEquipRefineHack", .@part, .@equip_refine) ||
		callfunc("F_IsEquipCardHack", .@part, .@equip_card[0], .@equip_card[1], .@equip_card[2], .@equip_card[3]) ||
		getequipcardid(.@part, .@target_slot) != 0) {
		mes "[Encantamento Brilhante]";
		mes "O acessorio foi alterado durante o processo.";
		mes "Nenhum item foi consumido.";
		close;
	}

	delitem .@consumable_id, 1;
	setitemcard .@part, .@target_slot, .@stone;
	recalculatestat;
	specialeffect2 EF_ENHANCE;

	mes "[Encantamento Brilhante]";
	mes "A ^00AA00" + getitemname(.@stone) + "^000000 foi inserida com sucesso.";
	mes "Duas copias desta pedra nao acumulam alem de 50%.";
	close;
}

// Cada copia equipada divide o mesmo bonus total. Assim, duas pedras iguais
// continuam fornecendo somente 50% do cooldown base das skills da classe.
function	script	F_AscensaoBrightStoneBonus	{
	.@stone = getarg(0);
	.@class = getarg(1);
	.@copies = isequippedcnt(.@stone);
	if (.@copies < 1)
		return;

	for (.@i = 0; .@i < 10; ++.@i) {
		.@skill_id = callfunc("F_ShadowSkillData", .@class, .@i, 0);
		.@skill_cd = callfunc("F_ShadowSkillData", .@class, .@i, 2);
		if (.@skill_id <= 0 || .@skill_cd <= 0)
			break;
		.@reduction = (.@skill_cd / 2) / .@copies;
		bonus2 bSkillCooldown, .@skill_id, -.@reduction;
	}
	return;
}

// Drops raros controlados pelo evento para permitir anuncio global e log
// somente quando o premio for realmente sorteado. A chance final preserva
// bonus pessoais de drop e a penalidade de nivel Renewal.
function	script	F_AscensaoMvpRareDrops	{
	setarray .@item_id[0], 501000, 9592, 40118;
	setarray .@base_rate[0], 200, 500, 200;
	setarray .@announce_name$[0],
		"Pedra de Grade D",
		"Ticket de Refino Shadow Garantido",
		"Cristal de Encantamento Brilhante";

	.@killer$ = strcharinfo(0);
	if (.@killer$ == "")
		.@killer$ = "um aventureiro";

	for (.@i = 0; .@i < getarraysize(.@item_id); ++.@i) {
		.@final_rate = get_adjusted_drop_rate(31914, .@base_rate[.@i]);
		if (rand(10000) >= .@final_rate)
			continue;

		partylootitem .@item_id[.@i], 1, .@final_rate;
		logmes "ASCENSAO_MVP_RARE_DROP|account_id=" + getcharid(3) +
			"|char_id=" + getcharid(0) +
			"|killer=" + .@killer$ +
			"|item_id=" + .@item_id[.@i] +
			"|item_name=" + getitemname(.@item_id[.@i]) +
			"|base_rate=" + .@base_rate[.@i] +
			"|final_rate=" + .@final_rate;
		announce "[Ascensao] O grupo de " + .@killer$ + " obteve " + .@announce_name$[.@i] + " ao derrotar a Deusa da Ascensao!", bc_all;
	}
	return;
}
