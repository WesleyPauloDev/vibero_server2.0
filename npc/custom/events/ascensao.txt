//============================================================
// Evento Ascensao
// - Entrada: 100.000 Zeny por acesso.
// - Limite: 4 horas diarias por conta.
// - MVP: liberado a cada 2.000 monstros, com 2 horas de intervalo.
//============================================================

function	script	F_AscensaoResetDaily	{
	.@today = gettime(DT_YYYYMMDD);
	if (#AscensaoDay != .@today) {
		#AscensaoDay = .@today;
		#AscensaoUsed = 0;
		if (#AscensaoStart)
			#AscensaoStart = gettimetick(2);
	}
	return;
}

function	script	F_AscensaoCommitTime	{
	callfunc "F_AscensaoResetDaily";
	.@now = gettimetick(2);
	if (#AscensaoStart > 0) {
		.@elapsed = .@now - #AscensaoStart;
		if (.@elapsed > 0)
			#AscensaoUsed += .@elapsed;
		#AscensaoStart = .@now;
	}
	if (#AscensaoUsed > 14400)
		#AscensaoUsed = 14400;
	return 14400 - #AscensaoUsed;
}

prontera,163,244,4	script	Guardiao da Ascensao	4_EP18_MARAM,{
	if (#AscensaoStart > 0) {
		.@remaining = callfunc("F_AscensaoCommitTime");
		#AscensaoStart = 0;
	} else {
		callfunc "F_AscensaoResetDaily";
		.@remaining = 14400 - #AscensaoUsed;
	}

	.@hours = .@remaining / 3600;
	.@minutes = (.@remaining % 3600) / 60;
	mes "[Guardiao da Ascensao]";
	mes "A passagem custa ^FF0000100.000 Zeny^000000 por entrada.";
	mes "Cada conta pode permanecer no mapa por ate quatro horas por dia.";
	mes "Tempo restante hoje: ^0055FF" + sprintf("%02d:%02d", .@hours, .@minutes) + "^000000.";
	if (.@remaining <= 0) {
		mes "Seu limite diario foi esgotado. Volte amanha.";
		close;
	}
	next;
	if (select("Entrar na Ascensao:Agora nao") == 2)
		close;
	if (Zeny < 100000) {
		mes "[Guardiao da Ascensao]";
		mes "Voce precisa de 100.000 Zeny para entrar.";
		close;
	}

	Zeny -= 100000;
	#AscensaoStart = gettimetick(2);
	@AscensaoLastMinute = -1;
	deltimer "AscensaoController::OnPlayerTimer";
	close2;
	warp "ascensao", 199, 152;
	end;
}

-	script	AscensaoController	-1,{
OnInit:
	.map$ = "ascensao";
	.npc$ = strnpcinfo(3);
	.kill_goal = 2000;
	.mvp_cooldown = 7200;
	setarray .mob_id[0], 31910, 31911, 31912, 31913;
	setarray .mob_name$[0], "Sentinela da Ascensao", "Basilisco da Ascensao", "Escavador da Ascensao", "Emissario da Ascensao";
	setarray .mob_event$[0], "OnSentinelaDead", "OnBasiliscoDead", "OnEscavadorDead", "OnEmissarioDead";

	.@restore_mvp = $AscensaoMvpAlive;
	for (.@i = 0; .@i < 4; .@i++)
		killmonster .map$, .npc$ + "::" + .mob_event$[.@i];
	killmonster .map$, .npc$ + "::OnMvpDead";
	$AscensaoMvpAlive = 0;
	if (.@restore_mvp)
		$AscensaoMvpPending = 1;

	for (.@i = 0; .@i < 4; .@i++)
		monster .map$, 0, 0, .mob_name$[.@i], .mob_id[.@i], 100, .npc$ + "::" + .mob_event$[.@i];

	// Recria os timers pessoais depois de um reload de scripts.
	getmapunits(BL_PC, .map$, .@units);
	for (.@i = 0; .@i < getarraysize(.@units); .@i++) {
		if (attachrid(.@units[.@i])) {
			if (#AscensaoStart > 0) {
				@AscensaoLastMinute = -1;
				deltimer .npc$ + "::OnPlayerTimer";
				addtimer 1000, .npc$ + "::OnPlayerTimer";
			}
			detachrid;
		}
	}

	initnpctimer;
	if ($AscensaoMvpPending && gettimetick(2) >= $AscensaoMvpNext)
		callsub S_SpawnMvp;
	end;

OnPCLoginEvent:
	// O tempo desconectado nunca e cobrado. O logout normal ja consolidou o uso.
	deltimer .npc$ + "::OnPlayerTimer";
	#AscensaoStart = 0;
	callfunc "F_AscensaoResetDaily";
	end;

OnPCLogoutEvent:
	if (#AscensaoStart > 0) {
		callfunc "F_AscensaoCommitTime";
		#AscensaoStart = 0;
	}
	end;

OnPCLoadMapEvent:
	if (strcharinfo(3) != .map$)
		end;
	if (#AscensaoStart <= 0) {
		dispbottom "[Ascensao] Entrada invalida. Fale com o Guardiao em Prontera.";
		warp "prontera", 163, 247;
		end;
	}
	@AscensaoLastMinute = -1;
	deltimer .npc$ + "::OnPlayerTimer";
	addtimer 1000, .npc$ + "::OnPlayerTimer";
	end;

OnPlayerTimer:
	.@remaining = callfunc("F_AscensaoCommitTime");
	if (strcharinfo(3) != .map$) {
		#AscensaoStart = 0;
		end;
	}
	if (.@remaining <= 0) {
		#AscensaoStart = 0;
		dispbottom "[Ascensao] Suas quatro horas diarias terminaram.";
		warp "prontera", 163, 247;
		end;
	}

	.@remaining_minutes = (.@remaining + 59) / 60;
	if (@AscensaoLastMinute != .@remaining_minutes) {
		@AscensaoLastMinute = .@remaining_minutes;
		dispbottom "[Ascensao] Tempo restante: " + sprintf("%02d:%02d", .@remaining / 3600, (.@remaining % 3600) / 60) + ".";
	}
	addtimer 10000, .npc$ + "::OnPlayerTimer";
	end;

OnSentinelaDead:
	callsub S_NormalDead, 31910, "Sentinela da Ascensao", "OnSentinelaDead";
	end;

OnBasiliscoDead:
	callsub S_NormalDead, 31911, "Basilisco da Ascensao", "OnBasiliscoDead";
	end;

OnEscavadorDead:
	callsub S_NormalDead, 31912, "Escavador da Ascensao", "OnEscavadorDead";
	end;

OnEmissarioDead:
	callsub S_NormalDead, 31913, "Emissario da Ascensao", "OnEmissarioDead";
	end;

S_NormalDead:
	.@mob_id = getarg(0);
	.@mob_name$ = getarg(1);
	.@event$ = getarg(2);
	$AscensaoKills++;

	if (($AscensaoKills % 100) == 0)
		mapannounce .map$, "[Ascensao] Progresso para o MVP: " + $AscensaoKills + "/" + .kill_goal + ".", bc_map;

	if ($AscensaoKills >= .kill_goal) {
		$AscensaoKills = 0;
		if ($AscensaoMvpAlive) {
			if (!$AscensaoMvpPending)
				mapannounce .map$, "[Ascensao] Outra invocacao foi preparada e aguardara o intervalo do MVP.", bc_map;
			$AscensaoMvpPending = 1;
		} else if (gettimetick(2) >= $AscensaoMvpNext) {
			callsub S_SpawnMvp;
		} else {
			$AscensaoMvpPending = 1;
			.@wait = $AscensaoMvpNext - gettimetick(2);
			mapannounce .map$, "[Ascensao] Meta concluida. O MVP aparecera em " + sprintf("%02d:%02d", .@wait / 3600, (.@wait % 3600) / 60) + ".", bc_map;
		}
	}

	sleep 5000;
	monster .map$, 0, 0, .@mob_name$, .@mob_id, 1, .npc$ + "::" + .@event$;
	return;

S_SpawnMvp:
	if ($AscensaoMvpAlive)
		return;
	if (mobcount(.map$, .npc$ + "::OnMvpDead") > 0) {
		$AscensaoMvpAlive = 1;
		$AscensaoMvpPending = 0;
		return;
	}
	monster .map$, 199, 191, "Deusa da Ascensao", 31914, 1, .npc$ + "::OnMvpDead";
	$AscensaoMvpAlive = 1;
	$AscensaoMvpPending = 0;
	mapannounce .map$, "[Ascensao] A Deusa da Ascensao despertou!", bc_map;
	return;

OnMvpDead:
	$AscensaoMvpAlive = 0;
	$AscensaoMvpNext = gettimetick(2) + .mvp_cooldown;
	mapannounce .map$, "[Ascensao] MVP derrotado. O intervalo de duas horas foi iniciado.", bc_map;
	end;

OnTimer60000:
	if ($AscensaoMvpAlive && mobcount(.map$, .npc$ + "::OnMvpDead") <= 0) {
		$AscensaoMvpAlive = 0;
		$AscensaoMvpNext = gettimetick(2) + .mvp_cooldown;
	}
	if ($AscensaoMvpPending && !$AscensaoMvpAlive && gettimetick(2) >= $AscensaoMvpNext)
		callsub S_SpawnMvp;
	setnpctimer 0;
	end;
}
