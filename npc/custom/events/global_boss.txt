//===========================================================
//  Evento Boss Global - Todos os dias as 20:00
//  Comandos: @evento (jogadores), @globalboss e @startbossglobal (GM nivel 99)
//===========================================================

-	script	GlobalBossCtrl	-1,{

OnInit:
    .npc_name$ = strnpcinfo(3);           // guarda o nome do NPC
    .map$      = "boss_global";
    .boss_id   = 21401;
    .boss_x    = 50;
    .boss_y    = 55;
    .min_players = 1;
    .start_minutes = 5;
    .active    = 0;
    if ($GlobalBossLevel < 1)
        $GlobalBossLevel = 1;
    bindatcmd "globalboss", .npc_name$ + "::OnGMStart", 99, 99;
    bindatcmd "startbossglobal", .npc_name$ + "::OnGMStart", 99, 99;
    disablenpc "Recompensa do Boss";
    stopnpctimer;
    killmonsterall .map$;

    // Checa a cada minuto se precisa iniciar
    end;

OnClock2000:
    .start_minutes = 5;
    donpcevent .npc_name$ + "::OnOpenEvent";
    end;

OnGMStart:
    if (.active) {
        dispbottom "O Evento Boss Global ja esta ativo.";
        end;
    }
    .start_minutes = 5;
    if (.@atcmd_numparameters >= 1) {
        .start_minutes = atoi(.@atcmd_parameters$[0]);
        if (.start_minutes < 1)
            .start_minutes = 1;
        if (.start_minutes > 5)
            .start_minutes = 5;
    }
    announce "[GM] " + strcharinfo(0) + " ativou o Evento Boss Global!", bc_all, 0xFF0000;
    donpcevent .npc_name$ + "::OnOpenEvent";
    end;

OnOpenEvent:
    if (.active)
        end;
    stopnpctimer;
    killmonsterall .map$;
    disablenpc "Recompensa do Boss";
    .active = 1;
    initnpctimer;
    setnpctimer 300000 - (.start_minutes * 60000);
    announce "[Evento Boss Global] Comeca em " + .start_minutes + " minuto(s)! Use @evento para participar.", bc_all, 0xFF0000;

    end;

OnTimer60000:
    announce "[Evento Boss Global] Comeca em 4 minutos! Use @evento para participar.", bc_all, 0xFF0000;
    end;

OnTimer120000:
    announce "[Evento Boss Global] Comeca em 3 minutos! Use @evento para participar.", bc_all, 0xFF0000;
    end;

OnTimer180000:
    announce "[Evento Boss Global] Comeca em 2 minutos! Use @evento para participar.", bc_all, 0xFF0000;
    end;

OnTimer240000:
    announce "[Evento Boss Global] Comeca em 1 minuto! Use @evento para participar.", bc_all, 0xFF0000;
    end;

OnTimer300000:
    announce "[Evento Boss Global Comecou]", bc_all, 0xFF0000;
    .active = 2;
    donpcevent .npc_name$ + "::OnStartEvent";
    end;

OnTimer1800000:
    if (.active == 2) {
        announce "[Evento Boss Global] O tempo acabou! O Boss fugiu.", bc_all, 0xFF0000;
        callsub S_EndEvent;
    }
    stopnpctimer;
    end;

OnStartEvent:	
    killmonsterall .map$;
    getmapunits(BL_PC, .map$, .@units);
    .@players = getarraysize(.@units);
    if (.@players < .min_players) {
        debugmes "[GlobalBoss] Evento cancelado: jogadores insuficientes no mapa " + .map$ + " (" + .@players + "/" + .min_players + ").";
        announce "[Evento Boss Global] Nao ha participantes suficientes para iniciar.", bc_all, 0xFF0000;
        callsub S_EndEvent;
        end;
    }

    .boss_level = $GlobalBossLevel;
    .@boss_name$ = "Boss Global Lv. " + .boss_level;
    $@mobid[0] = 0;
    monster .map$, .boss_x, .boss_y, .@boss_name$, .boss_id, 1, .npc_name$ + "::OnBossDead";
    .boss_gid = $@mobid[0];
    if (.boss_gid < 1) {
        debugmes "[GlobalBoss] Falha ao criar o boss " + .@boss_name$ + " mob_id " + .boss_id + " no mapa " + .map$ + " em " + .boss_x + "," + .boss_y + " com " + .@players + " jogador(es).";
        announce "[Evento Boss Global] Nao foi possivel iniciar o evento agora.", bc_all, 0xFF0000;
        callsub S_EndEvent;
        end;
    }

    // O nivel funciona como multiplicador dos atributos base do monstro.
    getunitdata .boss_gid, .@status;
    setunitdata .boss_gid, UMOB_LEVEL, .boss_level;
    setunitdata .boss_gid, UMOB_MAXHP, min(.@status[UMOB_MAXHP] * .boss_level, 2000000000);
    setunitdata .boss_gid, UMOB_HP, min(.@status[UMOB_MAXHP] * .boss_level, 2000000000);
    setunitdata .boss_gid, UMOB_STR, min(.@status[UMOB_STR] * .boss_level, 65535);
    setunitdata .boss_gid, UMOB_AGI, min(.@status[UMOB_AGI] * .boss_level, 65535);
    setunitdata .boss_gid, UMOB_VIT, min(.@status[UMOB_VIT] * .boss_level, 65535);
    setunitdata .boss_gid, UMOB_INT, min(.@status[UMOB_INT] * .boss_level, 65535);
    setunitdata .boss_gid, UMOB_DEX, min(.@status[UMOB_DEX] * .boss_level, 65535);
    setunitdata .boss_gid, UMOB_LUK, min(.@status[UMOB_LUK] * .boss_level, 65535);
    setunitdata .boss_gid, UMOB_ATKMIN, min(.@status[UMOB_ATKMIN] * .boss_level, 65535);
    setunitdata .boss_gid, UMOB_ATKMAX, min(.@status[UMOB_ATKMAX] * .boss_level, 65535);
    setunitdata .boss_gid, UMOB_MATKMIN, min(.@status[UMOB_MATKMIN] * .boss_level, 65535);
    setunitdata .boss_gid, UMOB_MATKMAX, min(.@status[UMOB_MATKMAX] * .boss_level, 65535);
    setunitdata .boss_gid, UMOB_DEF, min(.@status[UMOB_DEF] * .boss_level, 32767);
    setunitdata .boss_gid, UMOB_MDEF, min(.@status[UMOB_MDEF] * .boss_level, 32767);

    mapannounce .map$, "[Evento] " + .@boss_name$ + " apareceu!", bc_all;
    end;

OnBossDead:
    .@winner$ = strcharinfo(0);
    getitem 9679, 1;
    announce "[Boss Global Lv. " + .boss_level + "] " + .@winner$ + " Finalizou o Boss Global!", bc_all, 0xFF0000;
    enablenpc "Recompensa do Boss";
    getmapunits(BL_PC, .map$, .@units);
    for (.@i = 0; .@i < getarraysize(.@units); .@i++) {
        if (attachrid(.@units[.@i])) {
            callfunc("F_BattlePassRegisterWeekly", "global_boss");
            detachrid;
        }
    }
    $GlobalBossLevel++;
    announce "[Evento] Boss derrotado! O proximo Boss Global sera Lv. " + $GlobalBossLevel + ".", bc_all, 0xFF0000;

    .boss_gid = 0;
    .active = 0;
    stopnpctimer;
    end;

OnClock2030:
    if (.active == 2) {
        announce "[Evento Boss Global] O tempo acabou! O Boss fugiu.", bc_all, 0xFF0000;
        callsub S_EndEvent;
    }
    end;

S_EndEvent:
    .active = 0;
    disablenpc "Recompensa do Boss";

    getmapunits(BL_PC, .map$, .@units);
    for (.@i = 0; .@i < getarraysize(.@units); .@i++) {
        if (attachrid(.@units[.@i])) {
            warp "SavePoint", 0, 0;
            detachrid;
        }
    }

    sleep 1000;
    killmonsterall .map$;
    return;
}

//-----------------------------------------------------------
//  Comando @evento ira Teleporta para o mapa se o evento ativo
//-----------------------------------------------------------
-	script	cmd_evento	-1,{
OnInit:
    bindatcmd "evento", strnpcinfo(3)+"::OnEnterEvent";
    end;

OnEnterEvent:
    if (getvariableofnpc(.active, "GlobalBossCtrl") == 1) {
        warp getvariableofnpc(.map$, "GlobalBossCtrl"), 36, 38;
    } else {
        dispbottom "Nenhum convite de evento ativo no momento.";
    }
    end;
}

boss_global,50,50,4	script	Recompensa do Boss	10403,{
    // Mensagem de parabenizacao
    mes "[Parabens!]";
    mes "Voce venceu o Evento Boss Global!";
    
    // Da a recompensa. Chance define o peso do sorteio; amount define a quantidade entregue.
    setarray .@rewards[0], 12259,12210,22818,12224,12223,7621,7776,9617,9642,9199, 9198;
    setarray .@chance[0],   30,   20,   10,   10,   10,  10,   5,   5,   5,  10,    10;
    setarray .@amount[0],   10,    3,    3,    5,    5,   5,   1,   1,   3,   3,    10;

    .@total_chance = 0;
    for (.@i = 0; .@i < getarraysize(.@chance); .@i++)
        .@total_chance += .@chance[.@i];

    .@rand = rand(.@total_chance);
    .@sum = 0;
    .@rewardID = .@rewards[0];

    for (.@i = 0; .@i < getarraysize(.@rewards); .@i++) {
        .@sum += .@chance[.@i];
        if (.@rand < .@sum) {
            .@rewardID = .@rewards[.@i];
            break;
        }
    }
    getitem 9608, 20;
    getitem .@rewardID, .@amount[.@i];
    callfunc("F_DiaryRefreshDaily");
    callfunc("F_DiaryRefreshWeekly");
    if (#evento == 1)
        #evento = 2;
    if (#boss == 1)
        #boss = 2;
    
    // Teleporta o jogador para Prontera
    close2;
    warp "prontera", 154, 186;
    
    end;
}
