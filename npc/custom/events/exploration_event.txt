function	script	F_Fenda_MapIndex	{
	.@map$ = getarg(0);
	for (.@i = 0; .@i < 4; ++.@i) {
		if ('Fenda_Map$[.@i] == .@map$)
			return .@i;
		if (instance_mapname($Fenda_AllMap$['Fenda_MapSource[.@i]]) == .@map$)
			return .@i;
		if ($Fenda_AllMap$['Fenda_MapSource[.@i]] == .@map$)
			return .@i;
	}
	return -1;
}

function	script	F_Fenda_Announce	{
	.@message$ = getarg(0);
	for (.@i = 0; .@i < 4; ++.@i)
		if ('Fenda_Map$[.@i] != "")
			mapannounce 'Fenda_Map$[.@i], .@message$, bc_map;
	return;
}

function	script	F_Fenda_Log	{
	.@event$ = getarg(0);
	.@level = getarg(1, 0);
	.@leader$ = getarg(2, "");
	.@clear_seconds = getarg(3, 0);
	.@clear_minutes = getarg(4, 0);
	if (.@level < 1)
		.@level = 'Fenda_Level;
	if (.@leader$ == "")
		.@leader$ = 'Fenda_LeaderName$;
	.@iid = instance_id();
	if (!.@iid)
		.@iid = instance_id(IM_PARTY);
	.@party_id = getcharid(1);
	.@party_name$ = (.@party_id > 0) ? getpartyname(.@party_id) : "";
	getmapxy .@map$, .@x, .@y, BL_PC;

	query_sql("INSERT INTO `fenda_log` (`event_type`, `account_id`, `char_id`, `char_name`, `party_id`, `party_name`, `instance_id`, `fenda_level`, `leader_name`, `clear_seconds`, `clear_minutes`, `map`, `x`, `y`, `created_at`) VALUES ('"
		+ escape_sql(.@event$) + "', " + getcharid(3) + ", " + getcharid(0) + ", '" + escape_sql(strcharinfo(0)) + "', "
		+ .@party_id + ", '" + escape_sql(.@party_name$) + "', " + .@iid + ", " + .@level + ", '"
		+ escape_sql(.@leader$) + "', " + .@clear_seconds + ", " + .@clear_minutes + ", '"
		+ escape_sql(.@map$) + "', " + .@x + ", " + .@y + ", NOW())");
	return;
}

function	script	F_Fenda_PartyMemberCount	{
	.@party_id = getcharid(1);
	if (!.@party_id)
		return 1;

	getpartymember .@party_id, 1, .@char_id;
	getpartymember .@party_id, 2, .@account_id;

	.@count = 0;
	for (.@i = 0; .@i < $@partymembercount; ++.@i) {
		if (!isloggedin(.@account_id[.@i], .@char_id[.@i]))
			continue;

		.@map$ = strcharinfo(3, .@char_id[.@i]);
		for (.@j = 0; .@j < 4; ++.@j) {
			if ('Fenda_Map$[.@j] != "" && .@map$ == 'Fenda_Map$[.@j]) {
				.@count++;
				break;
			}
		}
	}

	if (.@count < 1)
		.@count = 1;
	return .@count;
}

function	script	F_Fenda_ResetDailyEntryCost	{
	.@today = atoi(gettimestr("%Y%m%d", 2));
	if (#FendaEntryDate == .@today)
		return;

	#FendaEntryDate = .@today;
	deletearray #FendaEntryCount[0], getarraysize(#FendaEntryCount);
	return;
}

function	script	F_Fenda_Cost	{
	callfunc "F_Fenda_ResetDailyEntryCost";
	.@level = getarg(0, 1);
	.@type = getarg(1, 0);
	.@base = 0;

	if (.@level <= 1)
		.@base = (.@type == 0) ? 50 : 0;
	if (.@level == 2)
		.@base = (.@type == 0) ? 100 : 0;
	if (.@level == 3) {
		if (.@type == 0) .@base = 100;
		else if (.@type == 1) .@base = 50;
	}
	if (.@level == 4) {
		if (.@type == 0) .@base = 100;
		else if (.@type == 1) .@base = 100;
	}
	if (.@level == 5) {
		if (.@type == 1) .@base = 100;
		else if (.@type == 2) .@base = 50;
	}
	if (.@level == 6) {
		if (.@type == 1) .@base = 100;
		else if (.@type == 2) .@base = 100;
	}
	if (.@level == 7) {
		if (.@type == 1) .@base = 100;
		else if (.@type == 3) .@base = 100;
	}

	if (.@level >= 8 && (.@type == 1 || .@type == 3))
		.@base = 150 + ((.@level - 8) * 50);

	if (.@base <= 0)
		return 0;

	.@repeat = #FendaEntryCount[.@level];
	if (.@repeat < 0)
		.@repeat = 0;
	return (.@base * (100 + (.@repeat * 20))) / 100;
}

function	script	F_Fenda_CostText$	{
	.@level = getarg(0, 1);
	setarray .@name$[0],
		"moeda cheffenia normal", "moeda cheffenia hard",
		"moeda bio normal", "moeda bio hard";
	.@text$ = "";
	for (.@i = 0; .@i < 4; ++.@i) {
		.@amount = callfunc("F_Fenda_Cost", .@level, .@i);
		if (.@amount <= 0)
			continue;
		if (.@text$ != "")
			.@text$ += " + ";
		.@text$ += .@amount + " " + .@name$[.@i];
	}
	return .@text$;
}

function	script	F_Fenda_HasCost	{
	.@level = getarg(0, 1);
	setarray .@item[0], 9646, 9647, 9648, 9649;
	for (.@i = 0; .@i < 4; ++.@i) {
		.@amount = callfunc("F_Fenda_Cost", .@level, .@i);
		if (.@amount > 0 && countitem(.@item[.@i]) < .@amount)
			return 0;
	}
	return 1;
}

	return;
}

function	script	F_Fenda_ScaleMob	{
	.@gid = getarg(0);
	.@level = getarg(1, 1);
	.@mult = 1;
	for (.@i = 1; .@i < .@level; ++.@i)
		.@mult *= 2;
	if (.@gid <= 0 || .@mult <= 1)
		return;

	getunitdata .@gid, .@data;
	.@hp = .@data[UMOB_MAXHP] * .@mult;
	if (.@hp > 2000000000)
		.@hp = 2000000000;
	setunitdata .@gid, UMOB_MAXHP, .@hp;
	setunitdata .@gid, UMOB_HP, .@hp;
	setunitdata .@gid, UMOB_ATKMIN, .@data[UMOB_ATKMIN] * .@mult;
	setunitdata .@gid, UMOB_ATKMAX, .@data[UMOB_ATKMAX] * .@mult;
	setunitdata .@gid, UMOB_MATKMIN, .@data[UMOB_MATKMIN] * .@mult;
	setunitdata .@gid, UMOB_MATKMAX, .@data[UMOB_MATKMAX] * .@mult;
	setunitdata .@gid, UMOB_DEF, min(.@data[UMOB_DEF] * .@mult, 32767);
	setunitdata .@gid, UMOB_MDEF, min(.@data[UMOB_MDEF] * .@mult, 32767);
	setunitdata .@gid, UMOB_RES, min(.@data[UMOB_RES] * .@mult, 32767);
	setunitdata .@gid, UMOB_MRES, min(.@data[UMOB_MRES] * .@mult, 32767);
	setunitdata .@gid, UMOB_STR, min(.@data[UMOB_STR] * .@mult, 32767);
	setunitdata .@gid, UMOB_VIT, min(.@data[UMOB_VIT] * .@mult, 32767);
	setunitdata .@gid, UMOB_INT, min(.@data[UMOB_INT] * .@mult, 32767);
	setunitdata .@gid, UMOB_DEX, min(.@data[UMOB_DEX] * .@mult, 32767);
	return;
}

	for (.@i = 0; .@i < 6; ++.@i) {
		.@map$ = instance_mapname($Fenda_AllMap$[.@i]);
		killmonster .@map$, instance_npcname("FendaController") + "::OnNormalDead";
		killmonster .@map$, instance_npcname("FendaController") + "::OnMiniBossDead";
		killmonster .@map$, instance_npcname("FendaController") + "::OnRareDead";
		disablenpc instance_npcname($Fenda_PortalNpc$[.@i]);
		disablenpc instance_npcname($Fenda_RewardNpc$[.@i]);
	}

	if ('Fenda_Active)
		for (.@i = 0; .@i < 4; ++.@i)
			if ('Fenda_Map$[.@i] != "")
				mapwarp 'Fenda_Map$[.@i], "prontera", 150, 180;

	'Fenda_Active = 0;
	'Fenda_Points = 0;
	'Fenda_NextAnnounce = 20;
	'Fenda_BossPending = 0;
	'Fenda_BossSpawned = 0;
	'Fenda_BossDead = 0;
	'Fenda_RewardPending = 0;
	'Fenda_RewardSource = -1;
	'Fenda_SpecialSpawned = 0;
	'Fenda_Level = 1;
	'Fenda_StartTick = 0;
	'Fenda_ClearSeconds = 0;
	'Fenda_ClearMinutes = 0;
	'Fenda_LeaderName$ = "";
	'Fenda_EntryMap$ = "";
	'Fenda_EntryX = 0;
	'Fenda_EntryY = 0;
	'Fenda_SpawnStage = 0;
	'Fenda_SpawnDone = 0;
	'Fenda_SpawnPending = 0;
	deletearray 'Fenda_NormalCount[0], getarraysize('Fenda_NormalCount);
	deletearray 'Fenda_MiniSpawned[0], getarraysize('Fenda_MiniSpawned);
	for (.@i = 0; .@i < 4; ++.@i) {
		'Fenda_Map$[.@i] = "";
		'Fenda_MiniCount[.@i] = 0;
	}
	return;
}

function	script	F_Fenda_Start	{
	if ('Fenda_Active)
		return;

	'Fenda_Active = 1;
	'Fenda_Points = 0;
	'Fenda_NextAnnounce = 20;
	'Fenda_BossPending = 0;
	'Fenda_BossSpawned = 0;
	'Fenda_BossDead = 0;
	'Fenda_RewardPending = 0;
	'Fenda_RewardSource = -1;
	'Fenda_SpecialSpawned = 0;
	'Fenda_SpawnStage = 0;
	'Fenda_SpawnDone = 0;
	'Fenda_SpawnPending = 0;
	deletearray 'Fenda_NormalCount[0], getarraysize('Fenda_NormalCount);
	deletearray 'Fenda_MiniSpawned[0], getarraysize('Fenda_MiniSpawned);
	if (!'Fenda_StartTick)
		'Fenda_StartTick = gettimetick(2);
	if ('Fenda_LeaderName$ == "")
		'Fenda_LeaderName$ = strcharinfo(0);
	if ('Fenda_Level < 1)
		'Fenda_Level = 1;

	setarray .@pool[0], 0, 1, 2, 3, 4, 5;
	for (.@i = 0; .@i < 4; ++.@i) {
		.@pick = rand(.@i, 5);
		.@swap = .@pool[.@i];
		.@pool[.@i] = .@pool[.@pick];
		.@pool[.@pick] = .@swap;
		.@source = .@pool[.@i];
		'Fenda_Map$[.@i] = instance_mapname($Fenda_AllMap$[.@source]);
		'Fenda_SpawnX[.@i] = $Fenda_AllSpawnX[.@source];
		'Fenda_SpawnY[.@i] = $Fenda_AllSpawnY[.@source];
		'Fenda_MapSource[.@i] = .@source;
		'Fenda_MiniCount[.@i] = 1;
	}
	'Fenda_EntryMap$ = 'Fenda_Map$[0];
	'Fenda_EntryX = 'Fenda_SpawnX[0];
	'Fenda_EntryY = 'Fenda_SpawnY[0];

	.@remaining = 6;
	while (.@remaining > 0) {
		.@stage = rand(4);
		if ('Fenda_MiniCount[.@stage] >= 3)
			continue;
		'Fenda_MiniCount[.@stage]++;
		.@remaining--;
	}

	for (.@i = 0; .@i < 4; ++.@i) {
		.@normal_count = 150;
		if ('Fenda_MapSource[.@i] == 0)
			.@normal_count = 600;
		else if ('Fenda_MapSource[.@i] == 1)
			.@normal_count = 450;
		else if ('Fenda_MapSource[.@i] == 2)
			.@normal_count = 350;
		else if ('Fenda_MapSource[.@i] == 3)
			.@normal_count = 450;
		else if ('Fenda_MapSource[.@i] == 4)
			.@normal_count = 350;
		else if ('Fenda_MapSource[.@i] == 5)
			.@normal_count = 350;
		'Fenda_NormalCount[.@i] = .@normal_count;
		'Fenda_MiniSpawned[.@i] = 0;
	}
	'Fenda_SpawnPending = 1;
	donpcevent instance_npcname("FendaController") + "::OnSpawnBatch";

	for (.@i = 0; .@i < 6; ++.@i)
		disablenpc instance_npcname($Fenda_PortalNpc$[.@i]);
	for (.@i = 0; .@i < 3; ++.@i) {
		.@source = 'Fenda_MapSource[.@i];
		enablenpc instance_npcname($Fenda_PortalNpc$[.@source]);
		mapannounce 'Fenda_Map$[.@i],
			"[Fenda] Portal para a proxima area em " + $Fenda_AllPortalX[.@source] + "," + $Fenda_AllPortalY[.@source] + ".",
			bc_map;
	}

	callfunc "F_Fenda_Announce", "[Fenda] A expedicao foi aberta para este grupo.";
	return;
}

function	script	F_Fenda_Portal	{
	if (!instance_id()) {
		dispbottom "[Fenda] Este portal pertence a uma instancia.";
		return;
	}
	if (!'Fenda_Active) {
		dispbottom "[Fenda] O portal esta inativo.";
		return;
	}

	.@stage = callfunc("F_Fenda_MapIndex", strcharinfo(3));
	if (.@stage < 0 || .@stage >= 3) {
		dispbottom "[Fenda] Nao ha outra passagem neste mapa.";
		return;
	}

	warp 'Fenda_Map$[.@stage + 1], 'Fenda_SpawnX[.@stage + 1], 'Fenda_SpawnY[.@stage + 1];
	return;
}

function	script	F_Fenda_EntryPortal	{
	if (!instance_id()) {
		dispbottom "[Fenda] Este portal pertence a uma instancia.";
		return;
	}
	if (!'Fenda_Active)
		callfunc "F_Fenda_Start";

	if ('Fenda_EntryMap$ == "") {
		dispbottom "[Fenda] A expedicao ainda nao foi preparada.";
		return;
	}

	dispbottom "[Fenda] Primeiro mapa sorteado: " + 'Fenda_EntryMap$ + ".";
	.@source = 'Fenda_MapSource[0];
	dispbottom "[Fenda] Portal deste mapa: " + $Fenda_AllPortalX[.@source] + "," + $Fenda_AllPortalY[.@source] + ".";
	warp 'Fenda_EntryMap$, 'Fenda_EntryX, 'Fenda_EntryY;
	return;
}

function	script	F_Fenda_Reward	{
	if (!instance_id() || !'Fenda_Active || !'Fenda_BossDead) {
		mes "[Fenda]";
		mes "A recompensa ainda nao esta disponivel.";
		close;
	}
	if (FendaJoinedInstance != instance_id()) {
		mes "[Fenda]";
		mes "Voce nao participou desta expedicao.";
		close;
	}
	if (FendaRewardInstance == instance_id()) {
		mes "[Fenda]";
		mes "Voce ja recebeu a recompensa desta expedicao.";
		close;
	}
	if (!checkweight(9608, 50) || !checkweight(9198, 5)) {
		mes "[Fenda]";
		mes "Libere peso no inventario antes de receber os premios.";
		close;
	}

	FendaRewardInstance = instance_id();
	.@base_fenda_points = 100 + ('Fenda_Level * 10);
	.@party_count = callfunc("F_Fenda_PartyMemberCount");
	.@reward_percent = 100 - ((.@party_count - 1) * 20);
	if (.@reward_percent < 50)
		.@reward_percent = 50;
	.@fenda_points = (.@base_fenda_points * .@reward_percent) / 100;
	if (.@fenda_points < 50)
		.@fenda_points = 50;
	#FendaShopPoints += .@fenda_points;
	if (FendaMaxLevel <= 'Fenda_Level)
		FendaMaxLevel = 'Fenda_Level + 1;
	getitem 9608, 50;
	getitem 9198, 5;
	announce "[Fenda] " + strcharinfo(0) + " concluiu a Fenda nivel " + 'Fenda_Level + ".", bc_all;
	mes "[Fenda]";
	mes "Sua recompensa foi entregue.";
	mes "Voce recebeu 50 moedas de instancia e 5 tickets de minerios.";
	mes "Grupo na Fenda: " + .@party_count + " membro(s).";
	mes "Pontos base: " + .@base_fenda_points + ". Reducao aplicada: " + (100 - .@reward_percent) + "%.";
	mes "Voce recebeu " + .@fenda_points + " pontos da Fenda.";
	mes "Saldo atual: " + #FendaShopPoints + " pontos.";
	mes "Voce sera levado de volta ao seu ponto salvo.";
	callfunc "F_Fenda_Log", "reward_exit", 'Fenda_Level, 'Fenda_LeaderName$, 'Fenda_ClearSeconds, 'Fenda_ClearMinutes;
	close2;
	warp "SavePoint", 0, 0;
	end;
}

function	script	F_FendaHatEnchant	{
	mes "[Encanto da Fenda]";

	mes "O item de encanto precisa estar no inventario e sera consumido.";
	next;

	.@enchant_choice = select("Feitico da Benevolencia:Feitico da Maldade:Cancelar");
	if (.@enchant_choice == 3)
		close;

	if (.@enchant_choice == 1) {
		.@enchant_id = 910300;
		if (countitem(910300) > 0)
			.@use_item_id = 910300;
		else if (countitem(312014) > 0)
			.@use_item_id = 312014;
		else
			.@use_item_id = 910300;
	} else {
		.@enchant_id = 910301;
		if (countitem(910301) > 0)
			.@use_item_id = 910301;
		else if (countitem(312021) > 0)
			.@use_item_id = 312021;
		else
			.@use_item_id = 910301;
	}

	if (countitem(.@use_item_id) < 1) {
		mes "[Encanto da Fenda]";
		mes "Voce nao possui ^FF0000" + getitemname(.@use_item_id) + "^000000 no inventario.";
		close;
	}

	setarray .@hat_ids[0],
		910200,910201,910202,910203,910204,910205,910206,910207,910208,910209,910210,
		910211,910212,910213,910214,910215,910216,910217,910218,910219,910220,910221;

	.@part = EQI_HEAD_TOP;
	.@item_id = getequipid(.@part);
	if (inarray(.@hat_ids[0], .@item_id) < 0) {
		mes "[Encanto da Fenda]";
		mes "Equipe no topo o hat da Fenda que deseja encantar.";
		close;
	}

	setarray .@card[0],
		getequipcardid(.@part,0),
		getequipcardid(.@part,1),
		getequipcardid(.@part,2),
		getequipcardid(.@part,3);
	.@refine = getequiprefinerycnt(.@part);

	mes "[Encanto da Fenda]";
	mes "Hat equipado:";
	mes "^00AAFF" + getitemname(.@item_id) + " [+" + .@refine + "]^000000";

		mes "O slot 2 ja possui:";
		mes "^FF0000" + getitemname(.@card[1]) + "^000000";
		mes "Deseja substituir por ^00AAFF" + getitemname(.@enchant_id) + "^000000?";
	} else {
		mes "O encanto sera aplicado no slot 2.";
	}
	next;

	if (select("Confirmar:Cancelar") == 2)
		close;

	if (countitem(.@use_item_id) < 1) {
		mes "[Encanto da Fenda]";
		mes "Voce nao possui mais ^FF0000" + getitemname(.@use_item_id) + "^000000 no inventario.";
		close;
	}

	delitem .@use_item_id, 1;
	setitemcard .@part, 1, .@enchant_id;

	mes "[Encanto da Fenda]";
	mes "Encanto aplicado com sucesso no slot 2.";
	close;
}

		.@menu$ += "Entrar na Fenda:";

			mes "[Fenda]";

			close;
		}
		if (!is_party_leader()) {
			mes "[Fenda]";
			mes "Somente o lider do grupo pode criar a Fenda.";
			close;

			mes "[Fenda]";
			mes "Seu grupo ja possui uma Fenda aberta.";
			close;
		}

				.@level_menu$ += " (" + #FendaEntryCount[.@i] + "x)";
			.@level_menu$ += ":";
		}
		.@level_menu$ += "Cancelar";
		mes "[Fenda]";
		mes "Escolha o nivel da Fenda.";
		next;
		.@level = select(.@level_menu$);
		if (.@level > FendaMaxLevel)
			close;
		if (!callfunc("F_Fenda_HasCost", .@level)) {
			mes "[Fenda]";
			mes "Para abrir o nivel " + .@level + ", voce precisa de:";
			mes callfunc("F_Fenda_CostText$", .@level) + ".";
			close;
		}

		if (.@iid < 0) {
			mes "[Fenda]";
			mes "Falha ao criar a Fenda.";
			close;
		}
		callfunc "F_Fenda_DelCost", .@level;
		#FendaEntryCount[.@level]++;
		setinstancevar('Fenda_Level, .@level, .@iid);
		setinstancevar('Fenda_LeaderName$, strcharinfo(0), .@iid);

		mes "[Fenda]";
		mes "A Fenda nivel " + .@level + " do seu grupo foi aberta.";
		next;
		if (select("Entrar agora:Fechar") == 2)
			close;
		.@enter_fenda = 1;
	}

	if (.@enter_fenda || (.@has_instance && .@menu == 2)) {
		.@iid = instance_id(IM_PARTY);
		if (!.@iid) {
			mes "[Fenda]";
			mes "Seu grupo ainda nao criou uma Fenda.";
			close;
		}
		if (getinstancevar('Fenda_BossDead, .@iid)) {
			mes "[Fenda]";
			mes "Esta Fenda ja foi concluida.";
			mes "Crie uma nova expedicao para continuar.";
			close;

				mes "Seu grupo ainda nao criou uma Fenda.";
				close;
			case IE_NOMEMBER:
				mes "[Fenda]";
				mes "Voce nao pertence ao grupo desta Fenda.";
				close;
			default:
				mes "[Fenda]";
				mes "Nao foi possivel entrar na Fenda agora.";
				close;
		}
	}

	if ((.@has_instance && .@menu == 3) || (!.@has_instance && .@menu == 2))
		callfunc "F_Fenda_Shop";
	close;
}

		end;
	}

	if (!'Fenda_MiniSpawned[.@stage]) {
		monster 'Fenda_Map$[.@stage], 0, 0, "Guardiao da Fenda", 31901, 'Fenda_MiniCount[.@stage], .@mini_label$;
		if ('Fenda_Level > 1)
			for (.@i = 0; .@i < 'Fenda_MiniCount[.@stage]; ++.@i)
				callfunc "F_Fenda_ScaleMob", $@mobid[.@i], 'Fenda_Level;
		'Fenda_MiniSpawned[.@stage] = 1;
	}

	'Fenda_SpawnStage++;
	'Fenda_SpawnDone = 0;
	sleep 10;
	donpcevent instance_npcname("FendaController") + "::OnSpawnBatch";
	end;

OnNormalDead:
	if (killedrid == 31902) {
		callfunc "F_Fenda_MvpDead";
		end;
	}
	if (killedrid == 31900) {
		.@drop_rate = get_adjusted_drop_rate(31900, 1);
		if (rand(10000) < .@drop_rate)
			partylootitem 9198, 1, .@drop_rate;
	}
	callsub S_AddPoints, 1;
	end;

OnMiniBossDead:
	if ($Fenda_MvpDropItem > 0) {
		.@drop_rate = get_adjusted_drop_rate(31901, 1);
		if (rand(10000) < .@drop_rate) {
			if (rand(2) == 0) {
				partylootitem $Fenda_MvpDropItem, 1, .@drop_rate;
				announce "[Fenda] O grupo de " + strcharinfo(0) + " obteve Caixa de Encanto da Fenda ao derrotar um Guardiao da Fenda!", bc_all;
			} else {
				partylootitem 9695, 1, .@drop_rate;
				announce "[Fenda] O grupo de " + strcharinfo(0) + " obteve Encanto da Fenda (Botas) ao derrotar um Guardiao da Fenda!", bc_all;
			}
		}
	}
	.@drop_rate = get_adjusted_drop_rate(31901, 5000);
	if (rand(10000) < .@drop_rate)
		partylootitem 9198, 1, .@drop_rate;
	callsub S_AddPoints, 50;
	end;

OnRareDead:
	.@drop_rate = get_adjusted_drop_rate(31903, 2000);
	if (rand(10000) < .@drop_rate) {
		partylootitem 9718, 1, .@drop_rate;
		announce "[Fenda] O grupo de " + strcharinfo(0) + " obteve Caixa do Passe de Batalha 19 ao derrotar o mob especial da Fenda!", bc_all;
	}
	end;

S_AddPoints:
	if (!'Fenda_Active || 'Fenda_BossPending || 'Fenda_BossSpawned)
		return;
	if (callfunc("F_Fenda_MapIndex", strcharinfo(3)) < 0)
		return;

	'Fenda_Points += getarg(0);
	if ('Fenda_Points >= 'Fenda_NextAnnounce) {
		callfunc "F_Fenda_Announce",
			"[Fenda] A expedicao possui " + 'Fenda_Points + " pontos.";
		'Fenda_NextAnnounce = (('Fenda_Points / 20) + 1) * 20;
	}

	if (!'Fenda_SpecialSpawned && 'Fenda_Points >= 500 && 'Fenda_Points <= 600 && rand(10000) < 10) {
		'Fenda_SpecialSpawned = 1;
		getmapxy .@rare_map$, .@rare_x, .@rare_y, BL_PC;
		monster .@rare_map$, 0, 0, "Especial da Fenda", 31903, 1,
			instance_npcname("FendaController") + "::OnRareDead";
		mapannounce .@rare_map$, "[Fenda] Mob especial apareceu neste mapa!", bc_map;
	}

	if ('Fenda_Points < 1000)
		return;

	'Fenda_BossPending = 1;
	getmapxy 'Fenda_BossMap$, 'Fenda_BossX, 'Fenda_BossY, BL_PC;
	callfunc "F_Fenda_Announce",
		"[Fenda] 1000 pontos! O Senhor da Fenda surgira em 5 segundos!";

	for (.@i = 0; .@i < 4; ++.@i) {
		killmonster 'Fenda_Map$[.@i], instance_npcname("FendaController") + "::OnNormalDead";
		killmonster 'Fenda_Map$[.@i], instance_npcname("FendaController") + "::OnMiniBossDead";
	}
	for (.@i = 0; .@i < 6; ++.@i)
		disablenpc instance_npcname($Fenda_PortalNpc$[.@i]);
	stopnpctimer;
	initnpctimer;
	return;

OnTimer5000:
	if (!'Fenda_Active)
		end;
	if ('Fenda_BossPending && !'Fenda_BossSpawned) {
		'Fenda_BossPending = 0;
		'Fenda_BossSpawned = 1;
		for (.@i = 0; .@i < 4; ++.@i)
			mapwarp 'Fenda_Map$[.@i], 'Fenda_BossMap$, 'Fenda_BossX, 'Fenda_BossY;
		monster 'Fenda_BossMap$, 'Fenda_BossX, 'Fenda_BossY,
			"Senhor da Fenda", 31902, 1, instance_npcname("FendaController") + "::OnNormalDead";
		callfunc "F_Fenda_ScaleMob", $@mobid[0], 'Fenda_Level;
		mapannounce 'Fenda_BossMap$, "[Fenda] O Senhor da Fenda apareceu!", bc_map;
		end;
	}
	if ('Fenda_RewardPending) {
		'Fenda_RewardPending = 0;
		if ('Fenda_RewardSource >= 0) {
			movenpc instance_npcname($Fenda_RewardNpc$['Fenda_RewardSource]), 'Fenda_RewardX, 'Fenda_RewardY;
			enablenpc instance_npcname($Fenda_RewardNpc$['Fenda_RewardSource]);
		}
		callfunc "F_Fenda_Announce", "[Fenda] O emissario de premios apareceu.";
	}
	end;

OnTimer600000:
	if ('Fenda_BossDead) {
		callfunc "F_Fenda_Announce",
			"[Fenda] O emissario partiu. A expedicao foi encerrada.";
		instance_destroy;
	}
	stopnpctimer;
	end;
}
