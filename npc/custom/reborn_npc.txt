//=============================================================
// NPC: Reborn Master/// Função: Resetar Stats/Skills, reduzir EXP e aumentar Drop
//=============================================================
prontera,170,226,4	script	Reborn Master	4_M_GUNSLINGER,{
	// ===== Configurações =====
	.@req_base = 175;        // Nível base mínimo
	.@req_job  = 70;        // Nível de classe mínimo
	.@eac = eaclass();
	.@base_eac = .@eac&(~EAJL_BABY);
	.@is_doram = (.@base_eac == EAJ_SUMMONER || .@base_eac == EAJ_SPIRIT_HANDLER);
	if (.@is_doram)
		.@req_job = 60;
	.@exp_penalty = 10;  // -10% EXP
    .@drop_bonus = 20;   // +20% DROP
	
	mes "[Reborn Master]";	

	if (reborn_count < 0) set reborn_count, 0;
	if (reborn_count > 5) set reborn_count, 5;
	set_reputation_points(7, reborn_count * 1000);

	if(reborn_count < 5) {
		mes "Eu posso guiá-lo pelo renascimento.";
		mes "Você perderá seus status e habilidades, voltara ao nível 1, mas ganhará ^00FF00"
			+.@drop_bonus + "%^000000 de taxa de drop e reduzira ^FF0000"
			+.@exp_penalty + "%^000000 da ganho de experiência.";
		next;

		if (BaseLevel < .@req_base || JobLevel < .@req_job) {
			mes "Você precisa ter pelo menos Nível Base "+.@req_base+
				" e Nível de Classe "+.@req_job+" para renascer.";
			close;
		}

		if (select("Quero renascer","Cancelar") == 2) close;

		// Confirmação
		mes "[Reborn Master]";
		mes "Tem certeza? Isso não pode ser desfeito.";
		if (select("Sim, estou pronto","Não, cancelar") == 2) close;

		.@was_second_visual = (getlook(LOOK_BODY2) >= 4332 && getlook(LOOK_BODY2) <= 4344);

		.@mail_sender$ = "Reborn Master";
		.@mail_title$ = "Recompensa de Reborn";
		.@mail_body$ = "Parabens pelo renascimento. Seus tickets foram enviados como anexo deste RodEx.";
		setarray .@mail_item[0], 9616;
		setarray .@mail_amount[0], 1;
		if (.@was_second_visual) {
			set .@mail_item[1], 6959; // devolve ticket caso o jogador estivesse usando visual 2ND
			set .@mail_amount[1], 1;
		}
		mail getcharid(0), .@mail_sender$, .@mail_title$, .@mail_body$, 0, .@mail_item, .@mail_amount;

		// ===== Aplicar renascimento =====
		// Mesmo fluxo do Job Master:
		// salva a classe atual para manter a progressão linear no Job Master.
		lastJob = Class;
		if (.@is_doram)
			jobchange Job_Summoner;
		else
			jobchange Job_Novice_High;
		resetlvl(1);
		if (.@is_doram)
			StatusPoint = 100;
		setpcblock(0,1); // evita movimentos durante ajustes

		// Guarda em variáveis de personagem		
		
		set reborn_count, reborn_count + 1;
		if (reborn_count < 0) set reborn_count, 0;
		if (reborn_count > 5) set reborn_count, 5;
		set_reputation_points(7, reborn_count * 1000);
		
		.@achievemente_reborn = reborn_count * 1000;
		callfunc("F_AddRep", 7, .@achievemente_reborn, 110008, 5000);

		donpcevent "Reputation_Login::OnPCLoginEvent";

		specialeffect2 EF_ANGEL2;
		specialeffect2 EF_ELECTRIC;

		setpcblock(0,0);
		mes "[Reborn Master]";
		mes "Parabéns, você renasceu!";
		close;
	} else {
		mes "Você já atingiu máximo de Reborn!";
		close;
	}
}
