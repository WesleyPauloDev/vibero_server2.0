	// Bloqueia os chapeus da Nightmare Biolab neste sistema de encantamento.
	if (.@equip_id == 18971 || .@equip_id == 18972 || .@equip_id == 18973 || .@equip_id == 18974
	 || .@equip_id == 18975 || .@equip_id == 18976 || .@equip_id == 18977 || .@equip_id == 18978
	 || .@equip_id == 18979 || .@equip_id == 18980 || .@equip_id == 18981 || .@equip_id == 18982
	 || .@equip_id == 18983 || .@equip_id == 18984) {
		mes " ";
		mes "Esse chapeu memoravel da Nightmare Biolab nao pode ser encantado neste sistema.";
		next;
		mes "[Artesao Aprendiz]";
		mes "Use o NPC especifico da Biolab para encantar, melhorar ou resetar esse item.";

	}
	set .@slot1, getequipcardid(.@part,0);
	set .@slot2, getequipcardid(.@part,1);
	set .@slot3, getequipcardid(.@part,2);

	mes "[Encantador]";
	mes "Voce selecionou: ^0055FF" + getequipname(.@part) + "^000000.";
	next;
	
	mes "[Artesao Aprendiz]";
	.@menu = select("Remover encantamento:Encantar equipamento:Melhorar encantamento:Sair");
	
	switch(.@menu) {
		case 1:
			
			.@hasEnchant = 0;
			if ((.@slot2 > 0) || 
				(.@slot3 > 0) || 
				(.@slot4 > 0))
				.@hasEnchant = 1;

			if (!.@hasEnchant) {
				mes "Este equipamento nao possui encantamento.";
				close;
			}

			if (.@equip_id == 15388 || .@equip_id == 15389
			 || .@equip_id == 32228 || .@equip_id == 32229 || .@equip_id == 32230
			 || .@equip_id == 32231 || .@equip_id == 32232 || .@equip_id == 32233) {
				mes "[Artesao Aprendiz]";
				mes "Encantamentos Schmidt nao podem ser removidos por este NPC.";
				close;
			}

			mes "[Artesao Aprendiz]";
			mes "Quer remover todos os encantamentos, ou apenas de um equipamento?";
			next;
			.@type = select("Remover um encantamento:Remover todos:Sair");
			if (.@type == 3) {
				close;
				end;
			}

			if (.@type == 1) {
				mes "[Artesao Aprendiz]";
				mes "Qual encantamento deseja remover o encantamento?";
				next;
				.@menu$ = "";
				set .@slotMapCount, 0;
				if (.@slot2 > 0) {
					.@menu$ = .@menu$ + "Slot 2 (" + getitemname(.@slot2) + "):";
					set .@slotMap[.@slotMapCount], 1;
					set .@slotMapCount, .@slotMapCount + 1;
				}
				if (.@slot3 > 0) {
					.@menu$ = .@menu$ + "Slot 3 (" + getitemname(.@slot3) + "):";
					set .@slotMap[.@slotMapCount], 2;
					set .@slotMapCount, .@slotMapCount + 1;
				}
				if (.@slot4 > 0) {
					.@menu$ = .@menu$ + "Slot 4 (" + getitemname(.@slot4) + "):";
					set .@slotMap[.@slotMapCount], 3;
					set .@slotMapCount, .@slotMapCount + 1;
				}
				.@menu$ = .@menu$ + "Cancelar";

				if (.@menu$ == "Cancelar") {
					mes "[Encantador Aprendiz]";
					mes "Este equipamento nao possui encantamentos para remover.";
					close;
				}
				
				.@slot_choice = select(.@menu$);
				if (.@slot_choice > .@slotMapCount) close; // Cancelar

				// Determina qual slot real foi escolhido
				.@realSlot = .@slotMap[.@slot_choice - 1];

				// Limpa o slot escolhido
				switch (.@realSlot) {
					case 1:
						getitem callfunc("get_base_enchant", .@slot2),1;
						break;
					case 2:
						getitem callfunc("get_base_enchant", .@slot3),1;
						break;
					case 3:
						getitem callfunc("get_base_enchant", .@slot4),1;
						break;
				}

				// Seta o item no slot do item equipado
				setitemcard(.@part, .@realSlot, 0);						

				mes "[Artesao Aprendiz]";
				mes "O encantamento do slot " + (.@slot_choice + 1) + " foi removido.";
				close;
			}

			// =========================================
			// Remove TODOS os slots de encantamento
			// =========================================
			if (.@type == 2) {

				// Zera os tres ultimos slots
				if (.@slot2 > 0) getitem callfunc("get_base_enchant", .@slot2),1;
				if (.@slot3 > 0) getitem callfunc("get_base_enchant", .@slot3),1;
				if (.@slot4 > 0) getitem callfunc("get_base_enchant", .@slot4),1;

				// Seta o item no slot do item equipado
				setitemcard(.@part, 1, 0);
				setitemcard(.@part, 2, 0);
				setitemcard(.@part, 3, 0);				

				mes "[Artesao Aprendiz]";
				mes "Todos os encantamentos foram removidos deste equipamento.";
				close;
			}
			end;

		case 2:
			// O jogador ja escolheu o equipamento e agora escolhe o slot
			mes "[Artesao Aprendiz]";
			mes "Qual slot deseja encantar?";
			next;
			.@slot_choice = select("Slot 2:Slot 3:Slot 4:Cancelar");
			if (.@slot_choice == 4) close;

			// -------------------------------------------
			// 1?? Verifica se ja existe encantamento no slot escolhido
			// -------------------------------------------
			switch (.@slot_choice) {
				case 1:
					if (.@slot2) {
						mes "[Artesao Aprendiz]";
						mes "Este slot nao esta disponivel!";
						close;
					}
					break;
				case 2:
					if (.@slot3) {
						mes "[Artesao Aprendiz]";
						mes "Este slot ja possui um encantamento!";
						close;
					}
					break;
				case 3:
					if (.@slot4) {
						mes "[Artesao Aprendiz]";
						mes "Este slot ja possui um encantamento!";
						close;
					}
					break;
			}

			// 2?? Define a faixa de IDs do slot
			// -------------------------------------------
			switch (.@slot_choice) {
				case 1: setarray .@range[0], 8100, 8159; break; // Slot 2
				case 2: setarray .@range[0], 8000, 8059; break; // Slot 3
				case 3: setarray .@range[0], 4700, 4759; break; // Slot 4
			}

			// -------------------------------------------
			// Verifica todos os encantamentos disponiveis na faixa e guarda em uma lista
			// -------------------------------------------
			deletearray .@hasEnchantList;
			deletearray .@menu$;

			for (.@i = .@range[0]; .@i <= .@range[1]; .@i++) {
				.@count = countitem(.@i);
				if (.@count > 0) {
					// Adiciona o ID e nome do item na lista
					.@hasEnchantList[getarraysize(.@hasEnchantList)] = .@i;
					.@menu$[getarraysize(.@menu$)] = getitemname(.@i) + " [x" + .@count + "]";
				}
			}

			// Se a lista estiver vazia
			if (!getarraysize(.@hasEnchantList)) {
				mes "[Artesao Aprendiz]";
				mes "Voce nao possui nenhum encantamento apropriado para este slot.";
				close;
			}

			// -------------------------------------------
			// Exibe menu de escolha
			// -------------------------------------------
			mes "[Artesao Aprendiz]";
			mes "Escolha o encantamento que deseja usar:";
			next;

			// Cria a string do menu separada por :
			.@menu_text$ = implode(.@menu$, ":");
			.@s = select(.@menu_text$);

			// Se o jogador fechou/cancelou o menu
			if (.@s <= 0) {
				mes "[Artesao Aprendiz]";
				mes "Operacao cancelada.";
				close;
			}

			// Obtem o ID do encantamento escolhido
			.@selectedEnchant = .@hasEnchantList[.@s - 1];

			// Confirmacao
			mes "[Artesao Aprendiz]";
			mes "Voce escolheu ^0055FF" + getitemname(.@selectedEnchant) + "^000000.";
			mes "Deseja realmente aplicar este encantamento?";
			next;
			if (select("Sim:Cancelar") == 2) close;	
			
			specialeffect2 EF_MAPPILLAR; // Efeito visual de pilar de luz

			// Barra de progresso para simular o processo
			progressbar "ffff00",3;
			specialeffect2 EF_REFINEOK;

			// Definir no slot correspondente:
			switch (.@slot_choice) {
				case 1: set .@slot2, .@selectedEnchant; break;
				case 2: set .@slot3, .@selectedEnchant; break;
				case 3: set .@slot4, .@selectedEnchant; break;
			}			
			
			// Seta o item no slot do item equipado
			setitemcard(.@part, .@slot_choice, .@selectedEnchant);

			next;

			.@hasEnchant = 0;
			if ((.@slot2 > 0) || (.@slot3 > 0) || (.@slot4 > 0))

				close;
			}

			// Escolher qual slot melhorar
			mes "[Artesao Aprendiz]";

			.@slotMapCount = 0;

				.@menu$ = .@menu$ + "Slot 2 (" + getitemname(.@slot2) + "):";
				set .@slotMap[.@slotMapCount], 1;
				.@slotMapCount++;
			}
			if (.@slot3 > 0) {
				.@menu$ = .@menu$ + "Slot 3 (" + getitemname(.@slot3) + "):";
				set .@slotMap[.@slotMapCount], 2;
				.@slotMapCount++;
			}
			if (.@slot4 > 0) {
				.@menu$ = .@menu$ + "Slot 4 (" + getitemname(.@slot4) + "):";
				set .@slotMap[.@slotMapCount], 3;
				.@slotMapCount++;
			}

			.@menu$ = .@menu$ + "Cancelar";
			if (.@menu$ == "Cancelar") {
				mes "[Artesao Aprendiz]";
				mes "Nenhum encantamento pode ser melhorado.";
				close;
			}

			.@slot_choice = select(.@menu$);
			if (.@slot_choice > .@slotMapCount) close; // Cancelar

			// Descobre qual slot real o jogador escolheu
			.@realSlot = .@slotMap[.@slot_choice - 1];

			// Pega o ID atual do encantamento
			switch (.@realSlot) {
				case 1: set .@old_id, .@slot2; break;
				case 2: set .@old_id, .@slot3; break;
				case 3: set .@old_id, .@slot4; break;
			}			

			// Calcula quantos niveis ainda podem ser aprimorados.
			.@max_upgrades = 0;
			.@next_id = .@old_id;
			while (1) {
				.@next_id = callfunc("get_next_enchant", .@next_id);
				if (!.@next_id)
					break;
				.@max_upgrades++;
			}

			if (!.@max_upgrades) {
				mes "[Artesao Aprendiz]";
				mes "Este encantamento ja esta no nivel maximo!";
				close;
			}

			// Calcula o nivel atual (baseado no ID)
			.@level = (.@old_id % 10) + 1;

			// Define o custo base de cada nivel conforme o slot.
			switch (.@realSlot) {
				case 3: // Slot 4 (encantamento comum)
					.@cost_per_level = 20000;
					break;
				case 2: // Slot 3 (encantamento intermediario)
					.@cost_per_level = 40000;
					break;
				case 1: // Slot 2 (encantamento premium)
					.@cost_per_level = 80000;
					break;
			}

			// Descobre quantos niveis o jogador consegue pagar no momento.
			.@affordable_upgrades = 0;
			.@affordable_cost = 0;
			for (.@i = 0; .@i < .@max_upgrades; .@i++) {
				.@next_cost = .@cost_per_level * (.@level + .@i);
				if (.@affordable_cost + .@next_cost > Zeny)
					break;
				.@affordable_cost += .@next_cost;
				.@affordable_upgrades++;
			}

			mes "[Artesao Aprendiz]";
			mes "Nivel atual: ^0055FF" + .@level + "^000000.";
			mes "Niveis disponiveis: ^0055FF" + .@max_upgrades + "^000000.";
			mes "Com seu Zeny, voce pode subir ^0055FF" + .@affordable_upgrades + " nivel(is)^000000 agora.";
			next;
			.@upgrade_option = select("Subir 1 nivel:Subir varios niveis:Cancelar");
			if (.@upgrade_option == 3)
				close;

			if (.@upgrade_option == 1) {
				.@upgrade_count = 1;
			} else {
				if (.@max_upgrades < 2) {
					mes "[Artesao Aprendiz]";
					mes "Este encantamento so pode subir mais 1 nivel.";
					close;
				}
				if (.@affordable_upgrades < 2) {
					mes "[Artesao Aprendiz]";
					mes "Seu Zeny nao e suficiente para subir 2 ou mais niveis.";
					close;
				}

				mes "[Artesao Aprendiz]";
				mes "Digite quantos niveis deseja subir.";
				mes "Minimo: ^0055FF2^000000 / Maximo que pode pagar: ^0055FF" + .@affordable_upgrades + "^000000.";
				input .@upgrade_count, 2, .@affordable_upgrades;
			}

			// Soma o preco de cada nivel e encontra o ID final.
			// Ex.: nivel 3 +5 no slot 4 = 60k+80k+100k+120k+140k.
			.@cost = 0;
			.@new_id = .@old_id;
			for (.@i = 0; .@i < .@upgrade_count; .@i++) {
				.@cost += .@cost_per_level * (.@level + .@i);
				.@new_id = callfunc("get_next_enchant", .@new_id);
			}

			mes "[Artesao Aprendiz]";
			mes "O encantamento passara do nivel ^0055FF" + .@level + "^000000 para o nivel ^0055FF" + (.@level + .@upgrade_count) + "^000000.";
			mes "Custo total: ^FF0000" + F_InsertComma(.@cost) + " Zeny^000000.";
			mes "Ainda assim gostaria de prosseguir?";
			if (select("Sim:Cancelar") == 2) {
				mes "Tudo bem, volte quando estiver pronto.";
				close;
			}

			// Verifica o zeny
			if (Zeny < .@cost) {
				mes "Voce precisa de " + F_InsertComma(.@cost) + " Zeny para realizar este upgrade.";
				close;
			}

			specialeffect2 EF_MAPPILLAR; // Efeito visual de pilar de luz

			// Barra de progresso para simular o processo
			progressbar "ffff00",3;

			// Tudo certo, cobra o zeny
			Zeny -= .@cost;			

			// Aplica o novo ID no slot correto
			switch (.@realSlot) {
				case 1: set .@slot2, .@new_id; break;
				case 2: set .@slot3, .@new_id; break;
				case 3: set .@slot4, .@new_id; break;
			}	

			// Seta o item no slot do item equipado
			setitemcard(.@part, .@realSlot, .@new_id);

			mes "[Artesao Aprendiz]";
			mes "O encantamento foi aprimorado com sucesso!";
			specialeffect2 EF_REPAIRWEAPON;
			close;


		case 4:
			mes "Volte quando estiver pronto.";
			close;
			end;
	}

	end;
}

	setarray .@post_ids[0],8100,8101,8102,8103,8104,8105,8106,8107,8108,8109; // Pos-conjuracao
	setarray .@cast_ids[0],8110,8111,8112,8113,8114,8115,8116,8117,8118,8119; // Conjuracao variavel

	// =====================================================
	// Busca e devolve o proximo ID da linha correspondente
	// =====================================================

		if (getarg(0) == .@str_ids[.@i])  return .@str_ids[.@i+1];
		if (getarg(0) == .@int_ids[.@i])  return .@int_ids[.@i+1];

		if (getarg(0) == .@agi_ids[.@i])  return .@agi_ids[.@i+1];
		if (getarg(0) == .@vit_ids[.@i])  return .@vit_ids[.@i+1];
		if (getarg(0) == .@luk_ids[.@i])  return .@luk_ids[.@i+1];

		if (getarg(0) == .@atk_ids[.@i])  return .@atk_ids[.@i+1];
		if (getarg(0) == .@hp_ids[.@i])   return .@hp_ids[.@i+1];
