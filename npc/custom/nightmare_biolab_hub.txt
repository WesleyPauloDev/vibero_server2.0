L_Make:
	.@reward_id = getarg(0);
	.@equipment_req = getarg(1);
	.@item_misc_req = getarg(2);
	if (getarg(3,0)) {
		setarray .@string$[0], "chill", "cloak";
		setarray .@material[0],20,100;
	} else {
		setarray .@string$[0], "soul", "hat";
		setarray .@material[0],1,20;
	}
	mes "[Sombra do Remorso]";
	mes "Para criar ^006400" + getitemname(.@reward_id) + "^000000, eu preciso dos materiais corretos.";
	next;
	mes "[Sombra do Remorso]";
	mes "Preciso de 1 " + getitemname(.@equipment_req) + ", " + .@material[0] + " " + getitemname(.@item_misc_req) + " e " + .@material[1] + " " + getitemname(6820) + ".";
	mes "Se reunir 5 vezes essa quantidade de energias e " + .@string$[0] + "s, a criacao nunca falhara.";
	next;
	switch (select("Voltar","Usar quantidade normal (20% de chance)","Usar quantidade x5 (100% de chance)")) {
	case 1:
		return;
	case 2:
		.@rate = 1;
		break;
	case 3:
		.@rate = 5;
		break;
	}
	if (countitem(.@equipment_req) < 1 || countitem(.@item_misc_req) < (.@material[0] * .@rate) || countitem(6820) < (.@material[1] * .@rate)) {
		mes "[Sombra do Remorso]";
		mes "Voce nao possui materiais suficientes para criar ^006400" + getitemname(.@reward_id) + "^000000.";
		return;
	}
	progressbar "000000",5;
	percentheal -50,-50;
	specialeffect2 EF_POISONSMOKE;
	specialeffect2 EF_NPC_BURNT;
	delitem .@equipment_req,1;
	delitem .@item_misc_req, (.@material[0] * .@rate);
	delitem 6820, (.@material[1] * .@rate);
	if ((20 * .@rate) > rand(100)) {
		specialeffect EF_BLACKBODY;
		specialeffect EF_GUMGANG4;
		specialeffect EF_DARKCASTING2;
		specialeffect EF_RG_COIN7;
		getitem .@reward_id,1;
		mes "[Sombra do Remorso]";
		mes "Aqui esta seu ^006400" + getitemname(.@reward_id) + "^000000.";
	} else {
		specialeffect EF_POISONSMOKE;
		specialeffect EF_GUMGANG4;
		mes "[Sombra do Remorso]";
		mes "A criacao de ^006400" + getitemname(.@reward_id) + "^000000 falhou.";
	}
	next;
	mes "- Recebi dano critico da poderosa onda mental que invadiu meu corpo. -";
	return;
}
