// Centraliza trocas, encantamentos e reformas dos equipamentos Varmundt.
//==================================================

moc_para01,52,26,4	script	Administrador Varmundt#bio	4_M_BARMUND,{
	disable_items;

	mes "[Administrador Varmundt]";
	mes "Eu organizo os equipamentos e materiais da Biosphere.";
	mes "Escolha o servico desejado.";
	next;

	switch (select(
		"Trocar fragmentos de runa:" +
		"Criar equipamentos Engraved:" +
		"Encantar Varmundt base:" +
		"Encantar Rune Equipment:" +
		"Reformar/upgrade Varmundt:" +
		"Adquirir pedras de evolucao:" +
		"Encantar aneis Varmundt:" +
		"Encantar aneis melhorados:" +
		"Cancelar"
	)) {
	case 1:
		mes "[Administrador Varmundt]";
		mes "Troco fragmentos por runas e essencias por aguas magicas.";
		close2;
		callshop "barter_varmundt_fragments";
		end;
	case 2:
		mes "[Administrador Varmundt]";
		mes "Aqui voce grava os equipamentos classicos usando as runas da Biosphere.";
		close2;
		callshop "barter_varmundt_engraved";
		end;
	case 3:
		callsub L_EnchantBase;
		end;
	case 4:
		callsub L_EnchantRune;
		end;
	case 5:
		callsub L_Reform;
		end;
	case 6:
		callsub L_EvolutionStones;
		end;
	case 7:
		callsub L_EnchantRingBase;
		end;
	case 8:
		callsub L_EnchantRingUpgraded;
		end;
	default:
		close;
	}
	end;

L_EvolutionStones:
	mes "[Administrador Varmundt]";
	mes "Tambem posso intermediar as pedras usadas nas reformas.";
	mes "Para Varmundt, voce normalmente vai precisar de:";
	mes "- Armor Stone 4 para armadura, bota e capa.";
	mes "- Accessory Stone 4 para os aneis.";
	next;

	switch (select("Melhorar pedras menores:Comprar pedras iniciais:Cancelar")) {
	// Opcao indisponivel enquanto os minerios nao possuem fonte acessivel.
	// case 1:
	// 	mes "[Administrador Varmundt]";
	// 	mes "Troco minerios por pedras de evolucao.";
	// 	close2;
	// 	callshop "barter_rgsr_in_1";
	// 	end;
	case 1:
		mes "[Administrador Varmundt]";
		mes "Troco pedras menores por pedras superiores.";
		close2;
		callshop "barter_rgsr_in_2";
		end;
	case 2:
		mes "[Administrador Varmundt]";
		mes "Troco Shadowdecon pelas pedras iniciais.";
		close2;
		callshop "barter_rgsr_in_3";
		end;
	default:
		close;
	}
	end;

L_EnchantBase:
	mes "[Administrador Varmundt]";
	mes "Encantamento dos conjuntos Varmundt base.";
	mes "Escolha as informacoes ou o tipo de conjunto que deseja encantar.";
	next;

	switch (select("Informacoes:Glade:Fire:Ice:Death:Cancelar")) {
	case 1:
		mes "[Administrador Varmundt]";
		mes "Requisitos para encantar Varmundt base:";
		mes "- O equipamento precisa estar +7 ou maior.";
		mes "- O equipamento precisa estar no inventario e desequipado.";
		mes "- Voce precisa estar com menos de 70% de peso.";
		next;
		mes "[Administrador Varmundt]";
		mes "Custos por tentativa:";
		mes "- Primeiro encanto: 50.000 zeny + 1 runa do conjunto.";
		mes "- Segundo encanto: 100.000 zeny + 5 runas do conjunto.";
		mes "- Reset: 1.000.000 zeny + 1 runa do conjunto.";
		next;
		mes "[Administrador Varmundt]";
		mes "Runas usadas por conjunto:";
		mes "- Glade: Glade Rune.";
		mes "- Fire: Fire Rune.";
		mes "- Ice: Ice Rune.";
		mes "- Death: Death Rune.";
		next;
		goto L_EnchantBase;
	case 2:
		.@enchant = 16;
		break;
	case 3:
		.@enchant = 17;
		break;
	case 4:
		.@enchant = 18;
		break;
	case 5:
		.@enchant = 19;
		break;
	default:
		close;
	}

	switch (select("Encantar:Resetar encantamentos:Cancelar")) {
	case 1:
		callsub L_EnchantBaseManual, .@enchant;
		break;
	case 2:
		callsub L_ResetBaseManual, .@enchant;
		break;
	default:
		close;
	}
	end;

L_SetBaseEnchantData:
	.@enchant = getarg(0);

	deletearray .base_target[0], getarraysize(.base_target);

	switch (.@enchant) {
	case 16:
		.base_set_name$ = "Glade";
		.base_material = 1000640; // Plain_Barmund_Rune2
		setarray .base_target[0], 450201, 480145, 470108, 450211; // Barmund_Plain_Armor, Barmund_Plain_Hood, Barmund_Plain_Boots, Guardian_Mail
		.base_target_count = 4;
		break;
	case 17:
		.base_set_name$ = "Fire";
		.base_material = 1000641; // Flame_Barmund_Rune2
		setarray .base_target[0], 450200, 480146, 470109; // Barmund_Flame_Armor, Barmund_Flame_Hood, Barmund_Flame_Boots
		.base_target_count = 3;
		break;
	case 18:
		.base_set_name$ = "Ice";
		.base_material = 1000642; // Ice_Barmund_Rune2
		setarray .base_target[0], 450203, 480148, 470111; // Barmund_Ice_Armor, Barmund_Ice_Hood, Barmund_Ice_Boots
		.base_target_count = 3;
		break;
	case 19:
		.base_set_name$ = "Death";
		.base_material = 1000643; // Death_Barmund_Rune2
		setarray .base_target[0], 450202, 480147, 470110; // Barmund_Death_Armor, Barmund_Death_Hood, Barmund_Death_Boots
		.base_target_count = 3;
		break;
	default:
		return 0;
	}
	return 1;

L_EnchantBaseManual:
	.@enchant = getarg(0);

	if (!callsub(L_SetBaseEnchantData, .@enchant))
		close;

	if (Weight * 100 >= MaxWeight * 70) {
		mes "[Administrador Varmundt]";
		mes "Voce precisa estar com menos de 70% de peso para encantar.";
		close;
	}

	getinventorylist;

	.@has_target = false;
	.@has_equipped = false;
	.@has_broken = false;
	.@has_low_refine = false;
	.@has_full_enchants = false;
	.@has_ready_target = false;
	.@menu$ = "";
	.@choice_count = 0;

	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (inarray(.base_target[0], @inventorylist_id[.@i]) == -1)
			continue;

		.@has_target = true;

		if (@inventorylist_equip[.@i] != 0) {
			.@has_equipped = true;
			continue;
		}
		if (@inventorylist_attribute[.@i] != 0) {
			.@has_broken = true;
			continue;
		}
		if (@inventorylist_refine[.@i] < 7) {
			.@has_low_refine = true;
			continue;
		}
		if (@inventorylist_card4[.@i] != 0 && @inventorylist_card3[.@i] != 0) {
			.@has_full_enchants = true;
			continue;
		}

		.@has_ready_target = true;
		if (@inventorylist_card4[.@i] == 0) {
			.@slot[.@choice_count] = 4;
			.@cost_zeny[.@choice_count] = 50000;
			.@cost_amount[.@choice_count] = 1;
		} else {
			.@slot[.@choice_count] = 3;
			.@cost_zeny[.@choice_count] = 100000;
			.@cost_amount[.@choice_count] = 5;
		}

		.@candidate[.@choice_count] = .@i;
		.@menu$ += getitemname(@inventorylist_id[.@i]) + " [+" + @inventorylist_refine[.@i] + "] - slot " + .@slot[.@choice_count] + ":";
		.@choice_count++;
	}

	if (!.@has_target) {
		mes "[Administrador Varmundt]";
		mes "Nao encontrei nenhum equipamento do conjunto " + .base_set_name$ + " no seu inventario.";
		mes "Itens validos:";
		for (.@i = 0; .@i < .base_target_count; .@i++)
			mes "- " + mesitemlink(.base_target[.@i], false);
		next;
		mes "[Administrador Varmundt]";
		mes "Se voce esta tentando encantar o " + mesitemlink(450199, false) + ", ele nao pertence a este grupo de encantamento.";
		close;
	}

	if (!.@has_ready_target) {
		mes "[Administrador Varmundt]";
		mes "Encontrei equipamento do conjunto " + .base_set_name$ + ", mas ele ainda nao pode ser encantado.";
		if (.@has_equipped)
			mes "- Desequipe o equipamento primeiro.";
		if (.@has_broken)
			mes "- Repare o equipamento antes de encantar.";
		if (.@has_low_refine)
			mes "- O equipamento precisa estar +7 ou maior.";
		if (.@has_full_enchants)
			mes "- Os dois slots de encanto ja estao ocupados. Use o reset antes.";
		close;
	}

	mes "[Administrador Varmundt]";
	mes "Escolha o equipamento que deseja encantar.";
	next;

	.@selected = select(.@menu$ + "Cancelar") - 1;
	if (.@selected == .@choice_count)
		close;

	.@inv_i = .@candidate[.@selected];
	.@item_id = @inventorylist_id[.@inv_i];
	.@identify = @inventorylist_identify[.@inv_i];
	.@refine = @inventorylist_refine[.@inv_i];
	.@attribute = @inventorylist_attribute[.@inv_i];
	.@bound = @inventorylist_bound[.@inv_i];
	.@grade = @inventorylist_enchantgrade[.@inv_i];
	setarray .@card[0], @inventorylist_card1[.@inv_i], @inventorylist_card2[.@inv_i], @inventorylist_card3[.@inv_i], @inventorylist_card4[.@inv_i];

	for (.@i = 0; .@i < 5; .@i++) {
		.@opt_id[.@i] = getd("@inventorylist_option_id" + (.@i + 1) + "[" + .@inv_i + "]");
		.@opt_val[.@i] = getd("@inventorylist_option_value" + (.@i + 1) + "[" + .@inv_i + "]");
		.@opt_param[.@i] = getd("@inventorylist_option_parameter" + (.@i + 1) + "[" + .@inv_i + "]");
	}

	.@slot = .@slot[.@selected];
	.@need_zeny = .@cost_zeny[.@selected];
	.@need_amount = .@cost_amount[.@selected];

	mes "[Administrador Varmundt]";
	mes "Equipamento: " + mesitemlink(.@item_id, false) + " [+" + .@refine + "]";
	mes "Custo: " + .@need_zeny + " zeny + " + .@need_amount + "x " + mesitemlink(.base_material, false) + ".";
	next;
	if (select("Encantar:Cancelar") == 2)
		close;

	if (Zeny < .@need_zeny) {
		mes "[Administrador Varmundt]";
		mes "Zeny insuficiente.";
		mes "Necessario: " + .@need_zeny + " zeny.";
		close;
	}

	if (countitem(.base_material) < .@need_amount) {
		mes "[Administrador Varmundt]";
		mes "Material insuficiente.";
		mes "Necessario: " + .@need_amount + "x " + mesitemlink(.base_material, false) + ".";
		mes "Use a opcao de troca de fragmentos caso voce ainda tenha apenas os fragmentos.";
		close;
	}

	if (countitem4(.@item_id, .@identify, .@refine, .@attribute, .@card[0], .@card[1], .@card[2], .@card[3], .@grade, .@opt_id, .@opt_val, .@opt_param) < 1) {
		mes "[Administrador Varmundt]";
		mes "O equipamento selecionado nao esta mais disponivel.";
		close;
	}

	if (.@slot == 4) {
		setarray .@pool[0], 4702,4712,4732,4742,4721,4703,4713,4733,4743,4722,4714,4734,4704,4744,4723;
		setarray .@chance[0], 15000,15000,15000,15000,15000,4500,4500,4500,4500,4500,500,500,500,500,500;
		.@pool_size = 15;
	} else {
		setarray .@pool[0], 310920,310923,310926,310929,310933,310936,310921,310924,310927,310930,310934,310937,310922,310925,310928,310932,310935,310938;
		setarray .@chance[0], 14000,14000,14000,14000,14000,14000,2400,2400,2400,2400,2400,2400,300,200,200,300,300,300;
		.@pool_size = 18;
	}

	.@total = 0;
	for (.@i = 0; .@i < .@pool_size; .@i++)
		.@total += .@chance[.@i];

	.@roll = rand(.@total) + 1;
	.@sum = 0;
	for (.@i = 0; .@i < .@pool_size; .@i++) {
		.@sum += .@chance[.@i];
		if (.@roll <= .@sum) {
			.@enchant_item = .@pool[.@i];
			break;
		}
	}

	delitem .base_material, .@need_amount;
	Zeny -= .@need_zeny;
	delitem4 .@item_id, 1, .@identify, .@refine, .@attribute, .@card[0], .@card[1], .@card[2], .@card[3], .@grade, .@opt_id, .@opt_val, .@opt_param;

	.@card[.@slot - 1] = .@enchant_item;

	if (.@bound > 0)
		getitembound4 .@item_id, 1, .@identify, .@refine, .@attribute, .@card[0], .@card[1], .@card[2], .@card[3], .@bound, .@grade, .@opt_id, .@opt_val, .@opt_param;
	else
		getitem4 .@item_id, 1, .@identify, .@refine, .@attribute, .@card[0], .@card[1], .@card[2], .@card[3], .@grade, .@opt_id, .@opt_val, .@opt_param;

	mes "[Administrador Varmundt]";
	mes "Encantamento concluido.";
	mes "Resultado: " + mesitemlink(.@enchant_item, false) + ".";
	close;

L_ResetBaseManual:
	.@enchant = getarg(0);

	if (!callsub(L_SetBaseEnchantData, .@enchant))
		close;

	if (Weight * 100 >= MaxWeight * 70) {
		mes "[Administrador Varmundt]";
		mes "Voce precisa estar com menos de 70% de peso para resetar.";
		close;
	}

	getinventorylist;

	.@has_target = false;
	.@has_equipped = false;
	.@has_broken = false;
	.@has_low_refine = false;
	.@has_enchanted = false;
	.@menu$ = "";
	.@choice_count = 0;

	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (inarray(.base_target[0], @inventorylist_id[.@i]) == -1)
			continue;

		.@has_target = true;

		if (@inventorylist_equip[.@i] != 0) {
			.@has_equipped = true;
			continue;
		}
		if (@inventorylist_attribute[.@i] != 0) {
			.@has_broken = true;
			continue;
		}
		if (@inventorylist_refine[.@i] < 7) {
			.@has_low_refine = true;
			continue;
		}
		if (@inventorylist_card2[.@i] == 0 && @inventorylist_card3[.@i] == 0 && @inventorylist_card4[.@i] == 0)
			continue;

		.@has_enchanted = true;
		.@candidate[.@choice_count] = .@i;
		.@menu$ += getitemname(@inventorylist_id[.@i]) + " [+" + @inventorylist_refine[.@i] + "]:";
		.@choice_count++;
	}

	if (!.@has_target) {
		mes "[Administrador Varmundt]";
		mes "Nao encontrei nenhum equipamento do conjunto " + .base_set_name$ + " no seu inventario.";
		mes "Itens validos:";
		for (.@i = 0; .@i < .base_target_count; .@i++)
			mes "- " + mesitemlink(.base_target[.@i], false);
		close;
	}

	if (!.@has_enchanted) {
		mes "[Administrador Varmundt]";
		mes "Nao encontrei nenhum equipamento encantado do conjunto " + .base_set_name$ + " que possa ser resetado.";
		if (.@has_equipped)
			mes "- Desequipe o equipamento primeiro.";
		if (.@has_broken)
			mes "- Repare o equipamento antes de resetar.";
		if (.@has_low_refine)
			mes "- O equipamento precisa estar +7 ou maior.";
		close;
	}

	mes "[Administrador Varmundt]";
	mes "Escolha o equipamento para remover todos os encantamentos.";
	next;

	.@selected = select(.@menu$ + "Cancelar") - 1;
	if (.@selected == .@choice_count)
		close;

	.@inv_i = .@candidate[.@selected];
	.@item_id = @inventorylist_id[.@inv_i];
	.@identify = @inventorylist_identify[.@inv_i];
	.@refine = @inventorylist_refine[.@inv_i];
	.@attribute = @inventorylist_attribute[.@inv_i];
	.@bound = @inventorylist_bound[.@inv_i];
	.@grade = @inventorylist_enchantgrade[.@inv_i];
	setarray .@card[0], @inventorylist_card1[.@inv_i], @inventorylist_card2[.@inv_i], @inventorylist_card3[.@inv_i], @inventorylist_card4[.@inv_i];

	for (.@i = 0; .@i < 5; .@i++) {
		.@opt_id[.@i] = getd("@inventorylist_option_id" + (.@i + 1) + "[" + .@inv_i + "]");
		.@opt_val[.@i] = getd("@inventorylist_option_value" + (.@i + 1) + "[" + .@inv_i + "]");
		.@opt_param[.@i] = getd("@inventorylist_option_parameter" + (.@i + 1) + "[" + .@inv_i + "]");
	}

	mes "[Administrador Varmundt]";
	mes "Equipamento: " + mesitemlink(.@item_id, false) + " [+" + .@refine + "]";
	mes "Todos os encantamentos serao removidos.";
	mes "Custo: gratuito.";
	next;
	if (select("Resetar:Cancelar") == 2)
		close;

	if (countitem4(.@item_id, .@identify, .@refine, .@attribute, .@card[0], .@card[1], .@card[2], .@card[3], .@grade, .@opt_id, .@opt_val, .@opt_param) < 1) {
		mes "[Administrador Varmundt]";
		mes "O equipamento selecionado nao esta mais disponivel.";
		close;
	}

	delitem4 .@item_id, 1, .@identify, .@refine, .@attribute, .@card[0], .@card[1], .@card[2], .@card[3], .@grade, .@opt_id, .@opt_val, .@opt_param;

	.@card[1] = 0;
	.@card[2] = 0;
	.@card[3] = 0;

	if (.@bound > 0)
		getitembound4 .@item_id, 1, .@identify, .@refine, .@attribute, .@card[0], .@card[1], .@card[2], .@card[3], .@bound, .@grade, .@opt_id, .@opt_val, .@opt_param;
	else
		getitem4 .@item_id, 1, .@identify, .@refine, .@attribute, .@card[0], .@card[1], .@card[2], .@card[3], .@grade, .@opt_id, .@opt_val, .@opt_param;

	mes "[Administrador Varmundt]";
	mes "Todos os encantamentos foram removidos.";
	close;

L_EnchantRune:
	mes "[Administrador Varmundt]";
	mes "Encantamento dos equipamentos Rune melhorados.";
	mes "Escolha o tipo de conjunto.";
	next;

	switch (select("Plain:Flame:Ice:Death:Cancelar")) {
	case 1:
		.@enchant = 52;
		break;
	case 2:
		.@enchant = 53;
		break;
	case 3:
		.@enchant = 54;
		break;
	case 4:
		.@enchant = 55;
		break;
	default:
		close;
	}

	switch (select("Encantar:Resetar encantamentos:Cancelar")) {
	case 1:
		callsub L_EnchantRuneManual, .@enchant;
		break;
	case 2:
		callsub L_ResetRuneManual, .@enchant;
		break;
	default:
		close;
	}
	end;

	deletearray .rune_target[0], getarraysize(.rune_target);

	switch (.@enchant) {

		.rune_set_name$ = "Plain Rune Equipment";

		.rune_essence = 1001138; // Barmund_Plain_Essence

		.rune_target_count = 3;
		break;
	default:

	}
	return 1;

L_EnchantRuneManual:
	.@enchant = getarg(0);

	if (!callsub(L_SetRuneEnchantData, .@enchant))
		close;

	if (Weight * 100 >= MaxWeight * 70) {
		mes "[Administrador Varmundt]";
		mes "Voce precisa estar com menos de 70% de peso para encantar.";
		close;
	}

	getinventorylist;

	.@has_target = false;
	.@has_equipped = false;
	.@has_broken = false;
	.@has_low_refine = false;
	.@has_full_enchants = false;
	.@has_ready_target = false;
	.@menu$ = "";
	.@choice_count = 0;

	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (inarray(.rune_target[0], @inventorylist_id[.@i]) == -1)
			continue;

		.@has_target = true;

		if (@inventorylist_equip[.@i] != 0) {
			.@has_equipped = true;

	.@refine = @inventorylist_refine[.@inv_i];
	.@attribute = @inventorylist_attribute[.@inv_i];
	.@bound = @inventorylist_bound[.@inv_i];
	.@grade = @inventorylist_enchantgrade[.@inv_i];
	setarray .@card[0], @inventorylist_card1[.@inv_i], @inventorylist_card2[.@inv_i], @inventorylist_card3[.@inv_i], @inventorylist_card4[.@inv_i];

	for (.@i = 0; .@i < 5; .@i++) {
		.@opt_id[.@i] = getd("@inventorylist_option_id" + (.@i + 1) + "[" + .@inv_i + "]");
		.@opt_val[.@i] = getd("@inventorylist_option_value" + (.@i + 1) + "[" + .@inv_i + "]");
		.@opt_param[.@i] = getd("@inventorylist_option_parameter" + (.@i + 1) + "[" + .@inv_i + "]");
	}

	mes "[Administrador Varmundt]";
	mes "Equipamento: " + mesitemlink(.@item_id, false) + " [+" + .@refine + "]";
	mes "Resultado: " + mesitemlink(.@result_id, false);
	mes "Materiais:";
	for (.@i = 0; .@i < getarraysize(.reform_mat); .@i++)
		mes "- " + .reform_amt[.@i] + "x " + mesitemlink(.reform_mat[.@i], false) + " (" + countitem(.reform_mat[.@i]) + ")";
	next;
	if (select("Reformar:Cancelar") == 2)
		close;

	for (.@i = 0; .@i < getarraysize(.reform_mat); .@i++) {
		if (countitem(.reform_mat[.@i]) < .reform_amt[.@i]) {
			mes "[Administrador Varmundt]";
			mes "Material insuficiente:";
			mes .reform_amt[.@i] + "x " + mesitemlink(.reform_mat[.@i], false) + ".";
			close;
		}
	}

	if (countitem4(.@item_id, .@identify, .@refine, .@attribute, .@card[0], .@card[1], .@card[2], .@card[3], .@grade, .@opt_id, .@opt_val, .@opt_param) < 1) {
		mes "[Administrador Varmundt]";
		mes "O equipamento selecionado nao esta mais disponivel.";
		close;
	}

	for (.@i = 0; .@i < getarraysize(.reform_mat); .@i++)
		delitem .reform_mat[.@i], .reform_amt[.@i];

	delitem4 .@item_id, 1, .@identify, .@refine, .@attribute, .@card[0], .@card[1], .@card[2], .@card[3], .@grade, .@opt_id, .@opt_val, .@opt_param;

	if (.@bound > 0)
		getitembound4 .@result_id, 1, .@identify, .@refine, .@attribute, .@card[0], .@card[1], .@card[2], .@card[3], .@bound, .@grade, .@opt_id, .@opt_val, .@opt_param;
	else
		getitem4 .@result_id, 1, .@identify, .@refine, .@attribute, .@card[0], .@card[1], .@card[2], .@card[3], .@grade, .@opt_id, .@opt_val, .@opt_param;

	mes "[Administrador Varmundt]";
	mes "Reforma concluida.";
	mes "Resultado: " + mesitemlink(.@result_id, false) + " [+" + .@refine + "].";
	close;

OnInit:
	setunittitle(getnpcid(0), "Biosphere");
	end;
}
