// ======================================================================
// Sistema Unificado de AI / Automacao para Personagem Real (PlayerBotAI)
// - Arquitetura Unificada de NPC unico (elimina erros de cross-event)
// - 100% Silencioso: Sem unittalk para nao expor o bot a outros jogadores
// - Atributos Reais e Balanceados: Usa statusup respeitando a tabela de pontos do nivel
// - Promocao automatica para 1a Classe no inicio com Habilidades Ofensivas reais
// - SavePoint configurado no Centro de Prontera (156, 170)
// - Retorno automatico ao SavePoint e revivificacao completa ao morrer (OnPCDieEvent)
// - Caminhada Organica em Prontera: Sorteia quantidade (3 a 6) e coordenadas aleatorias
//   na praca central da fonte com sistema anti-travamento integrado
// - Eden em 2 pontos (30,14 e 28,26) com caminhada real
// - Modo Teste: 5s em Prontera, 5s no Eden, 1 minuto exato de cacada
// - Modo Normal: cada parada em Prontera/Eden sorteia de 1 a 10 minutos inteiros
// - Combate Completo e Agressivo:
//     * Localizacao com getareaunits (raio 25 celulas, nunca teleporta se houver mobs na tela)
//     * Ignora completamente Plantas e Cogumelos
//     * Auto-Pocao SP Azul (505) acionada SEMPRE que o SP estiver abaixo de 80%
//     * Auto-Pocao HP Branca (504) acionada quando HP < 70%
//     * Auto-Ataque fisico basico ininterrupto + Habilidades Ofensivas instantaneas (ignore_range)
//     * Auto-Buffs nativos diretos (sem doevent para evitar enfileiramento)
//     * OnCheckBuffs:
//     if (!@BOT_AI_ACTIVE || @BOT_AI_STAGE != 3) return;
//     * OnApplyBuffs:
//     if (!@BOT_AI_ACTIVE || @BOT_AI_STAGE != 3) return;
//     * OnCheckSpeedPotion:
//     if (!@BOT_AI_ACTIVE || @BOT_AI_STAGE != 3) return;
//     * 30.000 de cada no armazem e @restock de 5 unidades das pocoes de ASPD
//     * Auto-Teleporte aleatorio no mapa APENAS se a tela estiver vazia de monstros
// - Respostas de comandos do GM enviadas exclusivamente para o chat do GM
// - Comandos GM para Controle:
//     @bot test [nome]   -> Inicia o MODO TESTE (5s pausas, 1 min cacada)
//     @bot start [nome]  -> Inicia o MODO NORMAL (2 min pausas, 5 min cacada)
//     @bot stop [nome]   -> Pausa a AI do personagem
//     @bot status [nome] -> Exibe informacoes para o GM
//     @bot               -> Menu Interativo do GM
// - Comando para o proprio jogador:
//     @iniciante         -> Ativa/Desativa a AI no proprio personagem
// Arquivo: npc/custom/iniciante_autobot.txt
// ======================================================================

-	script	PlayerBotAI	-1,{

OnInit:
	bindatcmd("bot", "PlayerBotAI::OnGMCommand", 60, 99);
	bindatcmd("iniciante", "PlayerBotAI::OnPlayerCommand", 0, 99);
	.boot_restore_done = 0;

	// Inicializa variaveis globais de agendamento (se ainda nao definidas)
	if ($BOT_CONFIG_INIT == 0) {
		$BOT_CONFIG_INIT = 1;
		$BOT_AUTO_SCHEDULE = 1; // 1 = Agendamento Ativo, 0 = Manual/24h
		$BOT_SCHED_START_H = 8; // 08:00h
		$BOT_SCHED_START_M = 0;
		$BOT_SCHED_END_H = 23;  // 23:00h
		$BOT_SCHED_END_M = 0;
		$BOT_AUTO_REBOOT = 1;   // 1 = Auto-recuperar no boot do servidor
		$BOT_REST_MAP$ = "prt_in"; // Local discreto padrao: Quarto da Estalagem de Prontera
		$BOT_REST_X = 241;
		$BOT_REST_Y = 166;
	}
	if ($BOT_REST_MAP$ == "") {
		$BOT_REST_MAP$ = "prt_in";
		$BOT_REST_X = 241;
		$BOT_REST_Y = 166;
	}
	// Simulacao individual de instancias: cada bot elegivel entra em um
	// intervalo diferente para evitar anuncios simultaneos.
	if ($BOT_INSTANCE_MIN_MINUTES <= 0) $BOT_INSTANCE_MIN_MINUTES = 30;
	if ($BOT_INSTANCE_MAX_MINUTES < $BOT_INSTANCE_MIN_MINUTES) $BOT_INSTANCE_MAX_MINUTES = 60;
	if ($BOT_INSTANCE_STAY_SECONDS <= 0) $BOT_INSTANCE_STAY_SECONDS = 300;
	// Cacada automatica de MVP abandonado: verificacao leve, a cada 5 minutos.
	if ($BOT_MVP_CHECK_SECONDS <= 0) $BOT_MVP_CHECK_SECONDS = 300;
	if ($BOT_MVP_MIN_ALIVE_SECONDS <= 0) $BOT_MVP_MIN_ALIVE_SECONDS = 3600;
	if ($BOT_MVP_MAX_LEVEL <= 0) $BOT_MVP_MAX_LEVEL = 150;

	initnpctimer;
	end;

// ======================================================================
// RESTAURACAO AUTOMATICA DOS BOTS APOS REBOOT
// ======================================================================
OnTimer5000:
	if ($BOT_AUTO_REBOOT && !.boot_restore_done) {
		.boot_restore_done = 1;
		callsub OnRestoreRegisteredBots;
	}
	end;

OnRestoreRegisteredBots:
	query_sql "SELECT c.`account_id`, c.`char_id`, IF(l.`sex` = 'F', 0, 1) FROM `char` c INNER JOIN `login` l ON l.`account_id` = c.`account_id` INNER JOIN `char_reg_num` r ON c.`char_id` = r.`char_id` WHERE r.`key` = 'BOT_REGISTERED' AND r.`value` = '1' ORDER BY c.`char_id`", .@restore_aids, .@restore_cids, .@restore_sexes;
	.@requested = 0;
	for (.@i = 0; .@i < getarraysize(.@restore_cids); ++.@i) {
		if (!isloggedin(.@restore_aids[.@i]))
			.@requested += restorebot(.@restore_aids[.@i], .@restore_cids[.@i], .@restore_sexes[.@i]);
	}
	return .@requested;

// ======================================================================
// AGENDADOR AUTOMATICO (Verifica a cada 60 segundos)
// ======================================================================
OnTimer60000:
	stopnpctimer;
	initnpctimer;
	callsub OnCheckScheduleAll;
	end;

// Persiste imediatamente o horario individual. Nao depende de logout normal,
// portanto tambem funciona com @kick, offlinebot e quedas do map-server.
OnPersistBotSchedule:
	.@sched_cid = getarg(0);
	.@sched_sh = getarg(1);
	.@sched_sm = getarg(2);
	.@sched_eh = getarg(3);
	.@sched_em = getarg(4);
	query_sql "INSERT INTO `char_reg_num` (`char_id`, `key`, `index`, `value`) VALUES (" + .@sched_cid + ", 'BOT_SCHED_CUSTOM', 0, '1') ON DUPLICATE KEY UPDATE `value` = '1'";
	query_sql "INSERT INTO `char_reg_num` (`char_id`, `key`, `index`, `value`) VALUES (" + .@sched_cid + ", 'BOT_SCHED_START_H', 0, '" + .@sched_sh + "') ON DUPLICATE KEY UPDATE `value` = '" + .@sched_sh + "'";
	query_sql "INSERT INTO `char_reg_num` (`char_id`, `key`, `index`, `value`) VALUES (" + .@sched_cid + ", 'BOT_SCHED_START_M', 0, '" + .@sched_sm + "') ON DUPLICATE KEY UPDATE `value` = '" + .@sched_sm + "'";
	query_sql "INSERT INTO `char_reg_num` (`char_id`, `key`, `index`, `value`) VALUES (" + .@sched_cid + ", 'BOT_SCHED_END_H', 0, '" + .@sched_eh + "') ON DUPLICATE KEY UPDATE `value` = '" + .@sched_eh + "'";
	query_sql "INSERT INTO `char_reg_num` (`char_id`, `key`, `index`, `value`) VALUES (" + .@sched_cid + ", 'BOT_SCHED_END_M', 0, '" + .@sched_em + "') ON DUPLICATE KEY UPDATE `value` = '" + .@sched_em + "'";
	query_sql "INSERT INTO `char_reg_num` (`char_id`, `key`, `index`, `value`) VALUES (" + .@sched_cid + ", 'BOT_REGISTERED', 0, '1') ON DUPLICATE KEY UPDATE `value` = '1'";
	query_sql "INSERT INTO `char_reg_num` (`char_id`, `key`, `index`, `value`) VALUES (" + .@sched_cid + ", 'BOT_AUTO_ACTIVE', 0, '1') ON DUPLICATE KEY UPDATE `value` = '1'";
	return;

OnCheckScheduleAll:
	if ($BOT_AUTO_SCHEDULE == 0) return;

	.@now_h = gettime(DT_HOUR);
	.@now_m = gettime(DT_MINUTE);
	.@now_mins = .@now_h * 60 + .@now_m;

	// Busca todos os bots registrados
	query_sql "SELECT c.`account_id`, c.`char_id`, COALESCE(sc.`value`,0), COALESCE(sh.`value`,0), COALESCE(sm.`value`,0), COALESCE(eh.`value`,0), COALESCE(em.`value`,0) FROM `char` c INNER JOIN `char_reg_num` r ON c.`char_id` = r.`char_id` LEFT JOIN `char_reg_num` sc ON sc.`char_id`=c.`char_id` AND sc.`key`='BOT_SCHED_CUSTOM' LEFT JOIN `char_reg_num` sh ON sh.`char_id`=c.`char_id` AND sh.`key`='BOT_SCHED_START_H' LEFT JOIN `char_reg_num` sm ON sm.`char_id`=c.`char_id` AND sm.`key`='BOT_SCHED_START_M' LEFT JOIN `char_reg_num` eh ON eh.`char_id`=c.`char_id` AND eh.`key`='BOT_SCHED_END_H' LEFT JOIN `char_reg_num` em ON em.`char_id`=c.`char_id` AND em.`key`='BOT_SCHED_END_M' WHERE r.`key`='BOT_REGISTERED' AND r.`value`='1'", .@r_aids, .@r_cids, .@r_custom, .@r_sh, .@r_sm, .@r_eh, .@r_em;

	for (.@i = 0; .@i < getarraysize(.@r_aids); .@i++) {
		.@b_aid = .@r_aids[.@i];
		if (.@b_aid > 0 && isloggedin(.@b_aid)) {
			if (attachrid(.@b_aid) && getcharid(0) == .@r_cids[.@i]) {
				// Verifica se o bot possui horario individual configurado
				if (.@r_custom[.@i] > 0) {
					.@s_h = .@r_sh[.@i];
					.@s_m = .@r_sm[.@i];
					.@e_h = .@r_eh[.@i];
					.@e_m = .@r_em[.@i];
				} else {
					.@s_h = $BOT_SCHED_START_H;
					.@s_m = $BOT_SCHED_START_M;
					.@e_h = $BOT_SCHED_END_H;
					.@e_m = $BOT_SCHED_END_M;
				}

				.@start_mins = .@s_h * 60 + .@s_m;
				.@end_mins = .@e_h * 60 + .@e_m;

				.@bot_in_sched = 0;
				if (.@start_mins <= .@end_mins) {
					if (.@now_mins >= .@start_mins && .@now_mins < .@end_mins) .@bot_in_sched = 1;
				} else {
					// Janela que passa da meia-noite (ex: 22h as 06h)
					if (.@now_mins >= .@start_mins || .@now_mins < .@end_mins) .@bot_in_sched = 1;
				}

				if (.@bot_in_sched && !@BOT_AI_ACTIVE && BOT_AUTO_ACTIVE > 0) {
					callsub OnStartSingleBot, BOT_SAVED_TEST;
				} else if (!.@bot_in_sched && @BOT_AI_ACTIVE) {
					// Pausa causada pelo horario: mantem o bot habilitado para
					// que o agendador possa inicia-lo novamente na proxima janela.
					callsub OnPauseSingleBot, 1;
				} else if (!.@bot_in_sched && !@BOT_AI_ACTIVE && !Sitting) {
					// Se ja estava inativo (login/reload/manual), garante repouso sentado.
					callsub OnPauseSingleBot, 1;
				}
			}
		}
	}
	return;

// ======================================================================
// SUB-ROTINAS UNIVERSAIS DE CONTROLE DE BOT
// ======================================================================
// Inicia ou retoma o bot no RID atualmente anexado
// getarg(0): 1 = Modo Teste, 0 = Modo Normal
OnStartSingleBot:
	.@is_test = getarg(0, 0);
	@BOT_AI_ACTIVE = 1;
	@BOT_TEST_MODE = .@is_test;
	BOT_REGISTERED = 1;
	BOT_AUTO_ACTIVE = 1;
	BOT_SAVED_TEST = .@is_test;

	// Todo inicio vai direto para a cacada. Isso distribui os bots pelos
	// mapas de up e evita que todos aparecam juntos em Prontera apos reboot.
	@BOT_AI_STAGE = 3;
	BOT_SAVED_STAGE = 3;

	// BOT_FINAL_JOB guarda a terceira classe escolhida; @BOT_TARGET_JOB continua
	// representando apenas o arquetipo usado pelos atributos e pela IA.
	callsub OnDetectTargetJob;

	@BOT_WP_INDEX = 0;
	@BOT_HUNT_COUNT = 0;
	@BOT_CURRENT_TARGET = 0;
	@BOT_TARGET_TRACK_ID = 0;
	@BOT_TARGET_LAST_HP = 0;
	@BOT_TARGET_PROGRESS_TICK = 0;
	@BOT_BLOCKED_TARGET = 0;
	@BOT_BLOCKED_UNTIL = 0;
	@BOT_NEXT_MVP_CHECK = gettimetick(2) + rand(10, $BOT_MVP_CHECK_SECONDS);
	@BOT_PRT_MAX = 0;
	@BOT_PRT_COUNT = 0;
	@BOT_LAST_TX = 0;
	@BOT_LAST_TY = 0;
	@BOT_DIR_VX = 0;
	@BOT_DIR_VY = 0;
	@BOT_EXPLORE_STEPS = 0;
	@BOT_WALL_HIT = 0;
	@BOT_SKILL_ROT = 0;
	@BOT_STUCK_X = 0;
	@BOT_STUCK_Y = 0;
	@BOT_STUCK_TICKS = 0;
	@BOT_WAIT_END = 0;
	@BOT_INSTANCE_ACTIVE = 0;
	@BOT_INSTANCE_END_TICK = 0;
	@BOT_INSTANCE_NAME$ = "";
	@BOT_INSTANCE_MAP$ = "";
	@BOT_NEXT_INSTANCE_TICK = gettimetick(2) + rand($BOT_INSTANCE_MIN_MINUTES, $BOT_INSTANCE_MAX_MINUTES) * 60;
	// Atualiza somente os slots que estao equipados neste inicio. Slots vazios
	// preservam o conjunto salvo para que Strip/quebra nao apaguem a referencia.
	callsub OnCaptureBotEquipment;
	stand;

	if (@BOT_AI_STAGE == 3) {
		callsub OnStartHuntTimer;
		callsub OnGetHuntMap;
		warp @hunt_map$, 0, 0;
	} else {
		@BOT_HUNT_START = 0;
		savepoint "prontera", 156, 191;
		warp "prontera", 156, 191;
		callsub OnQueuePronteraBuff;
	}

	// Injeta pocoes e suprimentos
	query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 504, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
	query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 505, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
	query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 645, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
	query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 656, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
	query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 657, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
	if (countitem(504) < 50) getitem 504, 50;
	if (countitem(505) < 50) getitem 505, 50;

	atcommand "@autoloot 100";
	atcommand "@restock 504 100";
	atcommand "@restock 505 100";
	atcommand "@restock 645 5";
	atcommand "@restock 656 5";
	atcommand "@restock 657 5";

	callsub OnCheckJobChange;
	callsub OnDistributeStats;
	callsub OnDistributeSkills;
	callsub OnApplyBuffs;
	callsub OnCheckMounts;
	callsub OnCheckSpeedPotion;

	deltimer "PlayerBotAI::OnStepTimer";
	addtimer 1000, "PlayerBotAI::OnStepTimer";
	return;

// Inicia um ciclo com duracao fixa, sorteada apenas uma vez.
// Teste: 1 minuto. Normal: 5, 6, 7, 8, 9 ou 10 minutos.
OnStartHuntTimer:
	@BOT_HUNT_START = gettimetick(2);
	@BOT_HUNT_DURATION = (@BOT_TEST_MODE ? 60 : rand(5, 10) * 60);
	return;

// Salva o conjunto escolhido manualmente para este personagem.
OnCaptureBotEquipment:
	for (.@slot = EQI_ACC_L; .@slot < EQI_MAX; .@slot++) {
		.@item_id = getequipid(.@slot);
		if (.@item_id > 0) {
			BOT_EQUIP_ID[.@slot] = .@item_id;
			BOT_EQUIP_UID$[.@slot] = getequipuniqueid(.@slot);
		}
	}
	return;

// Repara e reequipa o conjunto salvo caso um monstro quebre ou remova uma peca.
OnMaintainBotEquipment:
	if (gettimetick(2) < @BOT_NEXT_EQUIP_CHECK) return;
	@BOT_NEXT_EQUIP_CHECK = gettimetick(2) + 600;

	.@missing = 0;
	for (.@slot = EQI_ACC_L; .@slot < EQI_MAX; .@slot++) {
		if (BOT_EQUIP_ID[.@slot] > 0 && getequipid(.@slot) <= 0) {
			.@missing = 1;
			break;
		}
	}
	if (!.@missing) return;

	// Strip impede reequipar enquanto o estado estiver ativo.
	sc_end SC_STRIPWEAPON;
	sc_end SC_STRIPSHIELD;
	sc_end SC_STRIPARMOR;
	sc_end SC_STRIPHELM;
	sc_end SC__STRIPACCESSORY;
	sc_end SC_SHADOW_STRIP;
	repairall;

	for (.@slot = EQI_ACC_L; .@slot < EQI_MAX; .@slot++) {
		if (BOT_EQUIP_ID[.@slot] <= 0 || getequipid(.@slot) > 0) continue;

		.@equipped = 0;
		if (BOT_EQUIP_UID$[.@slot] != "" && BOT_EQUIP_UID$[.@slot] != "0")
			.@equipped = equipuniqueid(BOT_EQUIP_UID$[.@slot]);
		if (!.@equipped && countitem(BOT_EQUIP_ID[.@slot]) > 0)
			equip BOT_EQUIP_ID[.@slot];
	}
	return;

// Pausa o bot e salva o estado atual
OnPauseSingleBot:
	// getarg(0): 1 = pausa automatica do agendador (preserva a habilitacao)
	.@keep_auto_active = getarg(0, 0);
	BOT_SAVED_STAGE = @BOT_AI_STAGE;
	BOT_SAVED_JOB = @BOT_TARGET_JOB;
	BOT_SAVED_TEST = @BOT_TEST_MODE;
	if (!.@keep_auto_active) BOT_AUTO_ACTIVE = 0;

	@BOT_AI_ACTIVE = 0;
	@BOT_TEST_MODE = 0;
	@BOT_HUNT_START = 0;
	@BOT_HUNT_DURATION = 0;
	@BOT_CURRENT_TARGET = 0;
	@BOT_TARGET_TRACK_ID = 0;
	@BOT_TARGET_LAST_HP = 0;
	@BOT_TARGET_PROGRESS_TICK = 0;
	@BOT_BLOCKED_TARGET = 0;
	@BOT_BLOCKED_UNTIL = 0;
	@BOT_NEXT_MVP_CHECK = 0;
	@BOT_PRT_MAX = 0;
	@BOT_PRT_COUNT = 0;
	@BOT_STUCK_X = 0;
	@BOT_STUCK_Y = 0;
	@BOT_STUCK_TICKS = 0;
	@BOT_NEXT_EQUIP_CHECK = 0;
	@BOT_INSTANCE_ACTIVE = 0;
	@BOT_INSTANCE_END_TICK = 0;
	@BOT_INSTANCE_NAME$ = "";
	@BOT_INSTANCE_MAP$ = "";
	deltimer "PlayerBotAI::OnStepTimer";
	unitstopwalk getcharid(3), USW_FORCE_STOP;
	unitstopattack getcharid(3);

	// Teleporta discretamente para o local de repouso configurado (individual ou global)
	.@rmap$ = (BOT_REST_MAP$ != "" ? BOT_REST_MAP$ : $BOT_REST_MAP$);
	.@rx = (BOT_REST_MAP$ != "" ? BOT_REST_X : $BOT_REST_X);
	.@ry = (BOT_REST_MAP$ != "" ? BOT_REST_Y : $BOT_REST_Y);

	if (.@rmap$ != "") {
		.@ox = (.@rx > 0 ? .@rx + rand(-1, 1) : 0);
		.@oy = (.@ry > 0 ? .@ry + rand(-1, 1) : 0);
		warp .@rmap$, .@ox, .@oy;
	}

	sit; // Coloca o bot sentado em repouso
	return;

// Agenda @buff exatamente 2 segundos depois de um teleporte ativo para Prontera.
OnQueuePronteraBuff:
	if (!@BOT_AI_ACTIVE) return;
	deltimer "PlayerBotAI::OnPronteraBuffTimer";
	addtimer 2000, "PlayerBotAI::OnPronteraBuffTimer";
	return;

OnPronteraBuffTimer:
	if (!@BOT_AI_ACTIVE) end;
	getmapxy(.@buff_map$, .@buff_x, .@buff_y, BL_PC);
	if (.@buff_map$ != "prontera") end;
	// Equivalente direto ao @buff para nao depender do evento BuffVIP,
	// que pode ser recusado quando a fila de eventos do bot esta cheia.
	@last_buff_time = gettimetick(2);
	repairall;
	specialeffect2 EF_INCAGILITY;
	sc_start SC_INCREASEAGI, 600000, 10;
	specialeffect2 EF_BLESSING;
	sc_start SC_BLESSING, 600000, 10;
	end;

// ======================================================================
// COMANDO DO JOGADOR: @iniciante (para rodar no proprio char)
// ======================================================================
OnPlayerCommand:
	.@subcmd$ = strtolower(.@atcmd_parameters$[0]);

	// @iniciante test / @iniciante teste
	if (.@subcmd$ == "test" || .@subcmd$ == "teste") {
		callsub OnStartSingleBot, 1;
		dispbottom "[Iniciante AI]: MODO TESTE ativado direto na cacada (" + @hunt_map$ + ")!", 0x55FF55;
		end;
	}

	// @iniciante at / @iniciante offline
	if (.@subcmd$ == "at" || .@subcmd$ == "offline" || .@subcmd$ == "autotrade") {
		if (!@BOT_AI_ACTIVE) {
			callsub OnStartSingleBot, 0;
		}
		offlinebot;
		end;
	}

	if (@BOT_AI_ACTIVE) {
		callsub OnPauseSingleBot;
		dispbottom "[Iniciante AI]: Sistema desativado com sucesso.", 0xFF5555;
	} else {
		callsub OnStartSingleBot, 0;
		dispbottom "[Iniciante AI]: Sistema ativado no seu personagem!", 0x55FF55;
	}
	end;

// ======================================================================
// COMANDOS DO GM: @bot <acao> [nome | all]
// ======================================================================
OnGMCommand:
	.@gm_aid = getcharid(3);

	if (getgmlevel() < 60) {
		dispbottom "[Bot Manager]: Voce nao tem permissao para usar este comando.", 0xFF5555;
		end;
	}

	.@cmd$ = strtolower(.@atcmd_parameters$[0]);
	.@arg1$ = strtolower(.@atcmd_parameters$[1]);
	.@arg2$ = strtolower(.@atcmd_parameters$[2]);

	// Junta o nome do alvo caso tenha espacos
	.@target_name$ = "";
	if (.@atcmd_numparameters > 1) {
		.@target_name$ = .@atcmd_parameters$[1];
		for (.@i = 2; .@i < .@atcmd_numparameters; ++.@i) {
			.@target_name$ = .@target_name$ + " " + .@atcmd_parameters$[.@i];
		}
	}

	// ==================================================================
	// COMANDOS DE CONFIGURACAO E LISTAGEM
	// ==================================================================
	// 1. @bot list -> Lista todos os bots cadastrados
	if (.@cmd$ == "list" || .@cmd$ == "lista") {
		query_sql "SELECT c.`name`, c.`account_id`, c.`char_id`, c.`base_level`, c.`job_level`, c.`last_map`, COALESCE(sc.`value`,0), COALESCE(sh.`value`,0), COALESCE(sm.`value`,0), COALESCE(eh.`value`,0), COALESCE(em.`value`,0) FROM `char` c INNER JOIN `char_reg_num` r ON c.`char_id` = r.`char_id` LEFT JOIN `char_reg_num` sc ON sc.`char_id`=c.`char_id` AND sc.`key`='BOT_SCHED_CUSTOM' LEFT JOIN `char_reg_num` sh ON sh.`char_id`=c.`char_id` AND sh.`key`='BOT_SCHED_START_H' LEFT JOIN `char_reg_num` sm ON sm.`char_id`=c.`char_id` AND sm.`key`='BOT_SCHED_START_M' LEFT JOIN `char_reg_num` eh ON eh.`char_id`=c.`char_id` AND eh.`key`='BOT_SCHED_END_H' LEFT JOIN `char_reg_num` em ON em.`char_id`=c.`char_id` AND em.`key`='BOT_SCHED_END_M' WHERE r.`key`='BOT_REGISTERED' AND r.`value`='1' ORDER BY c.`name` ASC", .@b_names$, .@b_aids, .@b_cids, .@b_blvls, .@b_jlvls, .@b_maps$, .@b_custom, .@b_sh, .@b_sm, .@b_eh, .@b_em;

		dispbottom "==================================================", 0x00FFFF;
		dispbottom "   LISTA DE BOTS CADASTRADOS NO SERVIDOR (" + getarraysize(.@b_names$) + ")", 0xFFFF00;
		dispbottom "==================================================", 0x00FFFF;

		if (getarraysize(.@b_names$) == 0) {
			dispbottom "Nenhum bot cadastrado. Use '@bot register <nome>' para cadastrar.", 0xFFAAAA;
		} else {
			for (.@i = 0; .@i < getarraysize(.@b_names$); .@i++) {
				.@is_on = 0;
				.@bot_st$ = "^FF0000[OFFLINE]^000000";
				.@stage_name$ = "-";
				.@sched_str$ = "[Global: " + ($BOT_SCHED_START_H < 10 ? "0" : "") + $BOT_SCHED_START_H + ":" + ($BOT_SCHED_START_M < 10 ? "0" : "") + $BOT_SCHED_START_M + "-" + ($BOT_SCHED_END_H < 10 ? "0" : "") + $BOT_SCHED_END_H + ":" + ($BOT_SCHED_END_M < 10 ? "0" : "") + $BOT_SCHED_END_M + "]";
				if (.@b_custom[.@i])
					.@sched_str$ = "[Indiv: " + (.@b_sh[.@i] < 10 ? "0" : "") + .@b_sh[.@i] + ":" + (.@b_sm[.@i] < 10 ? "0" : "") + .@b_sm[.@i] + "-" + (.@b_eh[.@i] < 10 ? "0" : "") + .@b_eh[.@i] + ":" + (.@b_em[.@i] < 10 ? "0" : "") + .@b_em[.@i] + "]";

				if (isloggedin(.@b_aids[.@i]) && attachrid(.@b_aids[.@i]) && getcharid(0) == .@b_cids[.@i]) {
					.@is_on = 1;
					if (@BOT_AI_ACTIVE) {
						.@bot_st$ = "^00FF00[ATIVO]^000000";
						if (@BOT_AI_STAGE == 0) .@stage_name$ = "Prontera";
						else if (@BOT_AI_STAGE == 1) .@stage_name$ = "Fonte";
						else if (@BOT_AI_STAGE == 2) .@stage_name$ = "Eden";
						else if (@BOT_AI_STAGE == 3) .@stage_name$ = "Cacada (" + strcharinfo(3) + ")";
					} else {
						.@bot_st$ = "^FFFF00[PARADO]^000000";
					}
				}
				attachrid(.@gm_aid);

				dispbottom (.@i + 1) + ". " + .@b_names$[.@i] + " (Nv " + .@b_blvls[.@i] + "/" + .@b_jlvls[.@i] + ") " + .@bot_st$ + " " + .@sched_str$ + " | " + .@stage_name$, 0xFFFFFF;
			}
		}
		dispbottom "==================================================", 0x00FFFF;
		end;
	}

	// 2. @bot schedule [nome | all] [hora_ini min_ini hora_fim min_fim | hora_ini:min_ini hora_fim:min_fim | clear | on | off] -> Agendador
	if (.@cmd$ == "schedule" || .@cmd$ == "horario" || .@cmd$ == "agendamento") {
		if (.@arg1$ == "on" || .@arg1$ == "ativar" || .@arg1$ == "1") {
			$BOT_AUTO_SCHEDULE = 1;
			callsub OnCheckScheduleAll;
			dispbottom "[Bot Manager]: Agendamento automatico ATIVADO. Horario Global: " + ($BOT_SCHED_START_H < 10 ? "0" : "") + $BOT_SCHED_START_H + ":" + ($BOT_SCHED_START_M < 10 ? "0" : "") + $BOT_SCHED_START_M + " as " + ($BOT_SCHED_END_H < 10 ? "0" : "") + $BOT_SCHED_END_H + ":" + ($BOT_SCHED_END_M < 10 ? "0" : "") + $BOT_SCHED_END_M + ".", 0x55FF55;
			end;
		}
		if (.@arg1$ == "off" || .@arg1$ == "desativar" || .@arg1$ == "0") {
			$BOT_AUTO_SCHEDULE = 0;
			dispbottom "[Bot Manager]: Agendamento automatico DESATIVADO. Bots operam em modo manual/24h.", 0xFFAAAA;
			end;
		}

		// Verifica se o 1o parametro e o nome de um bot registrado
		.@t_name$ = .@atcmd_parameters$[1];
		.@t_aid = 0;
		.@t_cid = 0;
		query_sql "SELECT `account_id`, `char_id` FROM `char` WHERE `name` = '" + escape_sql(.@t_name$) + "' LIMIT 1", .@t_aid, .@t_cid;

		if (.@t_cid > 0) {
			// Opcao: @bot schedule <nome> clear
			if (.@arg2$ == "clear" || .@arg2$ == "padrao" || .@arg2$ == "default") {
				if (isloggedin(.@t_aid) && attachrid(.@t_aid) && getcharid(0) == .@t_cid) {
					BOT_SCHED_CUSTOM = 0;
				}
				query_sql "DELETE FROM `char_reg_num` WHERE `char_id` = " + .@t_cid + " AND `key` IN ('BOT_SCHED_CUSTOM', 'BOT_SCHED_START_H', 'BOT_SCHED_START_M', 'BOT_SCHED_END_H', 'BOT_SCHED_END_M')";
				attachrid(.@gm_aid);
				callsub OnCheckScheduleAll;
				dispbottom "[Bot Manager]: Horario individual de '" + .@t_name$ + "' removido. Agora obedece o horario GLOBAL.", 0x55FF55;
				end;
			}

			// Formato A: @bot schedule <nome> 01 10 01 12 (5 parametros)
			if (.@atcmd_numparameters >= 6) {
				.@b_sh = atoi(.@atcmd_parameters$[2]);
				.@b_sm = atoi(.@atcmd_parameters$[3]);
				.@b_eh = atoi(.@atcmd_parameters$[4]);
				.@b_em = atoi(.@atcmd_parameters$[5]);
				if (isloggedin(.@t_aid) && attachrid(.@t_aid) && getcharid(0) == .@t_cid) {
					BOT_SCHED_CUSTOM = 1;
					BOT_SCHED_START_H = .@b_sh;
					BOT_SCHED_START_M = .@b_sm;
					BOT_SCHED_END_H   = .@b_eh;
					BOT_SCHED_END_M   = .@b_em;
					// Configurar um horario individual tambem cadastra e rearma o bot.
					BOT_REGISTERED = 1;
					BOT_AUTO_ACTIVE = 1;
				}
				callsub OnPersistBotSchedule, .@t_cid, .@b_sh, .@b_sm, .@b_eh, .@b_em;
				attachrid(.@gm_aid);
				callsub OnCheckScheduleAll;
				dispbottom "[Bot Manager]: Horario individual do bot '" + .@t_name$ + "' configurado: " + (.@b_sh < 10 ? "0" : "") + .@b_sh + ":" + (.@b_sm < 10 ? "0" : "") + .@b_sm + " as " + (.@b_eh < 10 ? "0" : "") + .@b_eh + ":" + (.@b_em < 10 ? "0" : "") + .@b_em + "!", 0x55FF55;
				end;
			}

			// Formato B: @bot schedule <nome> 01:10 01:12 (3 parametros)
			if (.@atcmd_numparameters >= 4) {
				explode(.@t_start$, .@atcmd_parameters$[2], ":");
				explode(.@t_end$, .@atcmd_parameters$[3], ":");
				.@b_sh = atoi(.@t_start$[0]);
				.@b_sm = (getarraysize(.@t_start$) > 1 ? atoi(.@t_start$[1]) : 0);
				.@b_eh = atoi(.@t_end$[0]);
				.@b_em = (getarraysize(.@t_end$) > 1 ? atoi(.@t_end$[1]) : 0);
				if (isloggedin(.@t_aid) && attachrid(.@t_aid) && getcharid(0) == .@t_cid) {
					BOT_SCHED_CUSTOM = 1;
					BOT_SCHED_START_H = .@b_sh;
					BOT_SCHED_START_M = .@b_sm;
					BOT_SCHED_END_H   = .@b_eh;
					BOT_SCHED_END_M   = .@b_em;
					// Configurar um horario individual tambem cadastra e rearma o bot.
					BOT_REGISTERED = 1;
					BOT_AUTO_ACTIVE = 1;
				}
				callsub OnPersistBotSchedule, .@t_cid, .@b_sh, .@b_sm, .@b_eh, .@b_em;
				attachrid(.@gm_aid);
				callsub OnCheckScheduleAll;
				dispbottom "[Bot Manager]: Horario individual do bot '" + .@t_name$ + "' configurado: " + (.@b_sh < 10 ? "0" : "") + .@b_sh + ":" + (.@b_sm < 10 ? "0" : "") + .@b_sm + " as " + (.@b_eh < 10 ? "0" : "") + .@b_eh + ":" + (.@b_em < 10 ? "0" : "") + .@b_em + "!", 0x55FF55;
				end;
			}
		}

		// Configuracao GLOBAL:
		// Formato 1: @bot schedule 1 10 1 12 ou @bot schedule 01 10 01 12 (4 parametros)
		if (.@atcmd_numparameters >= 5) {
			$BOT_SCHED_START_H = atoi(.@atcmd_parameters$[1]);
			$BOT_SCHED_START_M = atoi(.@atcmd_parameters$[2]);
			$BOT_SCHED_END_H   = atoi(.@atcmd_parameters$[3]);
			$BOT_SCHED_END_M   = atoi(.@atcmd_parameters$[4]);
			$BOT_AUTO_SCHEDULE = 1;
			callsub OnCheckScheduleAll;
			dispbottom "[Bot Manager]: Horario GLOBAL configurado: " + ($BOT_SCHED_START_H < 10 ? "0" : "") + $BOT_SCHED_START_H + ":" + ($BOT_SCHED_START_M < 10 ? "0" : "") + $BOT_SCHED_START_M + " as " + ($BOT_SCHED_END_H < 10 ? "0" : "") + $BOT_SCHED_END_H + ":" + ($BOT_SCHED_END_M < 10 ? "0" : "") + $BOT_SCHED_END_M + " (Agendador ATIVO).", 0x55FF55;
			end;

		if (.@atcmd_numparameters >= 3) {
			explode(.@t_start$, .@atcmd_parameters$[1], ":");
			explode(.@t_end$, .@atcmd_parameters$[2], ":");

			if (getarraysize(.@t_start$) >= 2) {
				$BOT_SCHED_START_H = atoi(.@t_start$[0]);
				$BOT_SCHED_START_M = atoi(.@t_start$[1]);
			} else {
				$BOT_SCHED_START_H = atoi(.@atcmd_parameters$[1]);
				$BOT_SCHED_START_M = 0;
			}

			if (getarraysize(.@t_end$) >= 2) {
				$BOT_SCHED_END_H = atoi(.@t_end$[0]);
				$BOT_SCHED_END_M = atoi(.@t_end$[1]);
			} else {
				$BOT_SCHED_END_H = atoi(.@atcmd_parameters$[2]);
				$BOT_SCHED_END_M = 0;
			}

			$BOT_AUTO_SCHEDULE = 1;
			callsub OnCheckScheduleAll;
			dispbottom "[Bot Manager]: Horario GLOBAL configurado: " + ($BOT_SCHED_START_H < 10 ? "0" : "") + $BOT_SCHED_START_H + ":" + ($BOT_SCHED_START_M < 10 ? "0" : "") + $BOT_SCHED_START_M + " as " + ($BOT_SCHED_END_H < 10 ? "0" : "") + $BOT_SCHED_END_H + ":" + ($BOT_SCHED_END_M < 10 ? "0" : "") + $BOT_SCHED_END_M + " (Agendador ATIVO).", 0x55FF55;
			end;
		}

		.@cur_h = gettime(DT_HOUR);
		.@cur_m = gettime(DT_MINUTE);
		dispbottom "=== Configuracao de Agendamento ===", 0x00FFFF;
		dispbottom "Hora Atual do Servidor: " + (.@cur_h < 10 ? "0" : "") + .@cur_h + ":" + (.@cur_m < 10 ? "0" : "") + .@cur_m, 0xFFFF00;
		dispbottom "Status: " + ($BOT_AUTO_SCHEDULE ? "^00FF00ATIVADO^000000" : "^FF0000DESATIVADO^000000"), 0xFFFFFF;
		dispbottom "Horario Global: " + ($BOT_SCHED_START_H < 10 ? "0" : "") + $BOT_SCHED_START_H + ":" + ($BOT_SCHED_START_M < 10 ? "0" : "") + $BOT_SCHED_START_M + " as " + ($BOT_SCHED_END_H < 10 ? "0" : "") + $BOT_SCHED_END_H + ":" + ($BOT_SCHED_END_M < 10 ? "0" : "") + $BOT_SCHED_END_M, 0xFFFFFF;
		dispbottom "Auto-Retomada no Reboot: " + ($BOT_AUTO_REBOOT ? "SIM" : "NAO"), 0xFFFFFF;
		dispbottom "Exemplos de Uso:", 0x55FF55;
		dispbottom "  @bot schedule <nome> 01:10 01:12  (Define horario para bot especifico)", 0xFFFFFF;
		dispbottom "  @bot schedule <nome> clear        (Faz o bot usar o horario global)", 0xFFFFFF;
		dispbottom "  @bot schedule 01:10 01:12         (Define horario GLOBAL para todos)", 0xFFFFFF;
		dispbottom "  @bot schedule on | off            (Liga/Desliga agendamento)", 0xFFFFFF;
		end;
	}

	// 3. @bot restpos [nome | all] [mapa x y | here | clear] -> Local de Descanso
	if (.@cmd$ == "restpos" || .@cmd$ == "descanso" || .@cmd$ == "repouso" || .@cmd$ == "rest") {
		// Se digitou apenas: @bot restpos (ou @bot restpos here / aqui) -> Define o padrao global para a posicao atual
		if (.@atcmd_numparameters == 1 || .@arg1$ == "here" || .@arg1$ == "aqui") {
			getmapxy($BOT_REST_MAP$, $BOT_REST_X, $BOT_REST_Y, BL_PC);
			dispbottom "[Bot Manager]: Local de repouso GLOBAL dos bots definido para sua posicao: " + $BOT_REST_MAP$ + " (" + $BOT_REST_X + "," + $BOT_REST_Y + ")!", 0x55FF55;
			end;
		}

		// Se digitou: @bot restpos all <mapa> <x> <y> ou @bot restpos all here
		if (.@arg1$ == "all" || .@arg1$ == "todos") {
			if (.@arg2$ == "here" || .@arg2$ == "aqui" || .@atcmd_numparameters == 2) {
				getmapxy($BOT_REST_MAP$, $BOT_REST_X, $BOT_REST_Y, BL_PC);
				dispbottom "[Bot Manager]: Local de repouso GLOBAL dos bots definido para sua posicao: " + $BOT_REST_MAP$ + " (" + $BOT_REST_X + "," + $BOT_REST_Y + ")!", 0x55FF55;
				end;
			}
			if (.@atcmd_numparameters >= 5) {
				$BOT_REST_MAP$ = .@atcmd_parameters$[2];
				$BOT_REST_X = atoi(.@atcmd_parameters$[3]);
				$BOT_REST_Y = atoi(.@atcmd_parameters$[4]);
				dispbottom "[Bot Manager]: Local de repouso GLOBAL configurado para: " + $BOT_REST_MAP$ + " (" + $BOT_REST_X + "," + $BOT_REST_Y + ").", 0x55FF55;
				end;
			}
		}

		// Se o 1o parametro for um mapa com 3 coordenadas: @bot restpos prt_in 241 166 (configura global)
		if (.@atcmd_numparameters == 4 && (.@arg1$ == "prt_in" || .@arg1$ == "prontera" || .@arg1$ == "moc_para01" || .@arg1$ == "geffen" || .@arg1$ == "payon" || .@arg1$ == "morocc" || .@arg1$ == "alberta" || .@arg1$ == "izlude" || .@arg1$ == "aldebaran" || .@arg1$ == "yuno" || .@arg1$ == "einbroch" || .@arg1$ == "lighthalzen" || .@arg1$ == "hugel" || .@arg1$ == "rachel" || .@arg1$ == "veins")) {
			$BOT_REST_MAP$ = .@atcmd_parameters$[1];
			$BOT_REST_X = atoi(.@atcmd_parameters$[2]);
			$BOT_REST_Y = atoi(.@atcmd_parameters$[3]);
			dispbottom "[Bot Manager]: Local de repouso GLOBAL configurado para: " + $BOT_REST_MAP$ + " (" + $BOT_REST_X + "," + $BOT_REST_Y + ").", 0x55FF55;
			end;
		}

		// Se especificou um bot:
		// Exemplo A: @bot restpos <nome> here (ou apenas @bot restpos <nome>)
		// Exemplo B: @bot restpos <nome> <mapa> <x> <y>
		// Exemplo C: @bot restpos <nome> clear (remove individual e volta pro global)
		.@t_name$ = .@atcmd_parameters$[1];
		.@t_aid = 0;
		.@t_cid = 0;
		query_sql "SELECT `account_id`, `char_id` FROM `char` WHERE `name` = '" + escape_sql(.@t_name$) + "' LIMIT 1", .@t_aid, .@t_cid;

		if (.@t_cid > 0) {
			// Opcao Clear:
			if (.@arg2$ == "clear" || .@arg2$ == "padrao" || .@arg2$ == "default") {
				if (isloggedin(.@t_aid) && attachrid(.@t_aid) && getcharid(0) == .@t_cid) {
					BOT_REST_MAP$ = "";
					BOT_REST_X = 0;
					BOT_REST_Y = 0;
				} else {
					query_sql "DELETE FROM `char_reg_str` WHERE `char_id` = " + .@t_cid + " AND `key` = 'BOT_REST_MAP$'";
					query_sql "DELETE FROM `char_reg_num` WHERE `char_id` = " + .@t_cid + " AND `key` IN ('BOT_REST_X', 'BOT_REST_Y')";
				}
				attachrid(.@gm_aid);
				dispbottom "[Bot Manager]: Local individual do bot '" + .@t_name$ + "' removido. Ele usara o padrao global (" + $BOT_REST_MAP$ + ").", 0x55FF55;
				end;
			}

			// Opcao Here / Posicao Atual:
			if (.@atcmd_numparameters == 2 || .@arg2$ == "here" || .@arg2$ == "aqui") {
				getmapxy(.@gm_map$, .@gm_x, .@gm_y, BL_PC);
				if (isloggedin(.@t_aid) && attachrid(.@t_aid) && getcharid(0) == .@t_cid) {
					BOT_REST_MAP$ = .@gm_map$;
					BOT_REST_X = .@gm_x;
					BOT_REST_Y = .@gm_y;
				} else {
					query_sql "INSERT INTO `char_reg_str` (`char_id`, `key`, `index`, `value`) VALUES (" + .@t_cid + ", 'BOT_REST_MAP$', 0, '" + escape_sql(.@gm_map$) + "') ON DUPLICATE KEY UPDATE `value` = '" + escape_sql(.@gm_map$) + "'";
					query_sql "INSERT INTO `char_reg_num` (`char_id`, `key`, `index`, `value`) VALUES (" + .@t_cid + ", 'BOT_REST_X', 0, '" + .@gm_x + "') ON DUPLICATE KEY UPDATE `value` = '" + .@gm_x + "'";
					query_sql "INSERT INTO `char_reg_num` (`char_id`, `key`, `index`, `value`) VALUES (" + .@t_cid + ", 'BOT_REST_Y', 0, '" + .@gm_y + "') ON DUPLICATE KEY UPDATE `value` = '" + .@gm_y + "'";
				}
				attachrid(.@gm_aid);
				dispbottom "[Bot Manager]: Local de repouso individual do bot '" + .@t_name$ + "' configurado para sua posicao: " + .@gm_map$ + " (" + .@gm_x + "," + .@gm_y + ")!", 0x55FF55;
				end;
			}

			// Opcao Mapa X Y especificados: @bot restpos <nome> <mapa> <x> <y>
			if (.@atcmd_numparameters >= 5) {
				.@c_map$ = .@atcmd_parameters$[2];
				.@c_x = atoi(.@atcmd_parameters$[3]);
				.@c_y = atoi(.@atcmd_parameters$[4]);
				if (isloggedin(.@t_aid) && attachrid(.@t_aid) && getcharid(0) == .@t_cid) {
					BOT_REST_MAP$ = .@c_map$;
					BOT_REST_X = .@c_x;
					BOT_REST_Y = .@c_y;
				} else {
					query_sql "INSERT INTO `char_reg_str` (`char_id`, `key`, `index`, `value`) VALUES (" + .@t_cid + ", 'BOT_REST_MAP$', 0, '" + escape_sql(.@c_map$) + "') ON DUPLICATE KEY UPDATE `value` = '" + escape_sql(.@c_map$) + "'";
					query_sql "INSERT INTO `char_reg_num` (`char_id`, `key`, `index`, `value`) VALUES (" + .@t_cid + ", 'BOT_REST_X', 0, '" + .@c_x + "') ON DUPLICATE KEY UPDATE `value` = '" + .@c_x + "'";
					query_sql "INSERT INTO `char_reg_num` (`char_id`, `key`, `index`, `value`) VALUES (" + .@t_cid + ", 'BOT_REST_Y', 0, '" + .@c_y + "') ON DUPLICATE KEY UPDATE `value` = '" + .@c_y + "'";
				}
				attachrid(.@gm_aid);
				dispbottom "[Bot Manager]: Local de repouso individual do bot '" + .@t_name$ + "' configurado para: " + .@c_map$ + " (" + .@c_x + "," + .@c_y + ")!", 0x55FF55;
				end;
			}
		}

		dispbottom "=== Local de Repouso dos Bots Inativos ===", 0x00FFFF;
		dispbottom "Padrao Global Atual: " + $BOT_REST_MAP$ + " (" + $BOT_REST_X + "," + $BOT_REST_Y + ")", 0xFFFFFF;
		dispbottom "Comandos de Configuracao:", 0x55FF55;
		dispbottom "  @bot restpos <nome>                - Define sua posicao atual para o bot especificado", 0xFFFFFF;
		dispbottom "  @bot restpos <nome> <mapa> <x> <y> - Define mapa/x/y para o bot especificado", 0xFFFFFF;
		dispbottom "  @bot restpos <nome> clear          - Remove local individual e usa padrao global", 0xFFFFFF;
		dispbottom "  @bot restpos all <mapa> <x> <y>    - Define padrao global para todos os bots", 0xFFFFFF;
		dispbottom "  @bot restpos                       - Define padrao global para sua posicao atual", 0xFFFFFF;
		dispbottom "Exemplos:", 0x55FFFF;
		dispbottom "  @bot restpos BotMecanico           (Coloca o BotMecanico onde seu GM esta pisando)", 0xFFFF00;
		dispbottom "  @bot restpos BotBruxo prt_in 241 166 (Coloca o BotBruxo no quarto 1)", 0xFFFF00;
		dispbottom "  @bot restpos BotAlgoz prt_in 250 170 (Coloca o BotAlgoz no quarto 2)", 0xFFFF00;
		end;
	}

	// 4. @bot register [nome] -> Cadastra o personagem como bot
	if (.@cmd$ == "register" || .@cmd$ == "cadastrar") {
		.@t_aid = 0;
		.@t_cid = 0;

OnPCLoginEvent:
	// Se for um bot registrado no sistema
	if (BOT_REGISTERED > 0) {
		.@now_h = gettime(DT_HOUR);
		.@now_m = gettime(DT_MINUTE);
		.@now_mins = .@now_h * 60 + .@now_m;

		if (BOT_SCHED_CUSTOM > 0) {
			.@s_h = BOT_SCHED_START_H;
			.@s_m = BOT_SCHED_START_M;
			.@e_h = BOT_SCHED_END_H;
			.@e_m = BOT_SCHED_END_M;
		} else {
			.@s_h = $BOT_SCHED_START_H;
			.@s_m = $BOT_SCHED_START_M;
			.@e_h = $BOT_SCHED_END_H;
			.@e_m = $BOT_SCHED_END_M;
		}

		.@start_mins = .@s_h * 60 + .@s_m;
		.@end_mins = .@e_h * 60 + .@e_m;

		.@in_sched = 0;
		if ($BOT_AUTO_SCHEDULE == 0) {
			.@in_sched = (BOT_AUTO_ACTIVE > 0 || $BOT_AUTO_REBOOT > 0);
		} else if (.@start_mins <= .@end_mins) {
			if (.@now_mins >= .@start_mins && .@now_mins < .@end_mins) .@in_sched = 1;
		} else {
			if (.@now_mins >= .@start_mins || .@now_mins < .@end_mins) .@in_sched = 1;
		}

		if (.@in_sched) {
			callsub OnStartSingleBot, BOT_SAVED_TEST;
		} else {
			// Bot que conecta fora do horario vai direto ao descanso e fica sentado.
			callsub OnPauseSingleBot, 1;
		}
	}
	end;

OnStepTimer:
	if (!@BOT_AI_ACTIVE) end;
	stand; // Bot NUNCA senta, permanece sempre em pe e ativo
	// O estagio 4 e reservado para a permanencia simulada em instancia.
	if (@BOT_AI_STAGE == 4) {
		callsub OnHandleBotInstance;
		end;
	}
	callsub OnMaintainBotEquipment;

	// Bots ativos de nivel 130+ procuram, no maximo a cada 5 minutos, um
	// MVP abaixo do nivel 150 que esteja vivo ha uma hora em mapa sem jogadores.
	if (BaseLevel >= 130 && gettimetick(2) >= @BOT_NEXT_MVP_CHECK) {
		@BOT_NEXT_MVP_CHECK = gettimetick(2) + $BOT_MVP_CHECK_SECONDS;
		if (botmvphunt($BOT_MVP_MIN_ALIVE_SECONDS, $BOT_MVP_MAX_LEVEL)) {
			@BOT_CURRENT_TARGET = 0;
			@BOT_TARGET_TRACK_ID = 0;
			@BOT_TARGET_LAST_HP = 0;
			@BOT_TARGET_PROGRESS_TICK = 0;
			@BOT_LAST_TX = 0;
			@BOT_LAST_TY = 0;
			addtimer 1500, "PlayerBotAI::OnStepTimer";
			end;
		}
	}

	// Limpeza de inventario se pesado
	if ((Weight * 100 / MaxWeight) >= 80) {
		callsub OnCleanHeavyInventory;
	}

	// Pontos, classes e pocoes de cura
	if (StatusPoint > 0) callsub OnDistributeStats;
	if (SkillPoint > 0 || (BaseClass == Job_Swordman && BaseLevel >= 99 && getskilllv("LG_BANISHINGPOINT") < 10)) callsub OnDistributeSkills;
	callsub OnCheckJobChange;
	callsub OnCheckPotions;
	callsub OnCheckMounts;
	if (callsub(OnTryBotInstance)) end;

	// =========================================================
	// =========================================================
	// ESTAGIO 0: Caminhada Organica e Dinamica em Prontera
	// Caminhada fluida com parada obrigatoria de no minimo 1 minuto em cada ponto
	// =========================================================
	if (@BOT_AI_STAGE == 0) {
		getmapxy(.@m$, .@cx, .@cy, BL_PC);
		if (.@m$ != "prontera") {
			stand;
			warp "prontera", 156, 191;
			callsub OnQueuePronteraBuff;
			@BOT_PRT_MAX = 0;
			@BOT_PRT_COUNT = 0;
			@BOT_WP_INDEX = 0;
			@BOT_LAST_TX = 0;
			@BOT_LAST_TY = 0;
			@BOT_STUCK_TICKS = 0;
			addtimer 1500, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Sub-estagio 99: retorno recente da cacada; @buff roda em timer proprio de 2s.
		if (@BOT_WP_INDEX == 99) {
			if (gettimetick(2) >= @BOT_WAIT_END) {
				@BOT_WP_INDEX = 0;
				@BOT_WAIT_END = 0;
				callsub OnApplyBuffs;
				callsub OnCheckMounts;
				callsub OnCheckSpeedPotion;
				addtimer 1000, "PlayerBotAI::OnStepTimer";
				end;
			}
			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Piscina ampla de mais de 60 coordenadas abertas para total variedade e liberdade de travessia
		setarray .@pool_x[0],
			// Setor Sul (Y 177 ~ 190)
			148, 152, 156, 160, 164,
			148, 152, 156, 160, 164,
			148, 152, 156, 160, 164,
			148, 154, 158, 164,
			// Setor Oeste / Diagonal Esquerda (148,177 -> 124,202 -> 121,206 -> 146,230)
			144, 140, 136, 132, 128, 124,
			121, 121, 124, 128, 134, 140,
			124, 128, 134, 140,
			124, 128, 132, 136, 140, 146,
			// Setor Norte (Y 214 ~ 230)
			146, 150, 154, 158, 162, 165,
			148, 152, 156, 160, 164,
			148, 152, 156, 160, 164,
			148, 152, 156, 160, 164,
			// Setor Leste / Diagonal Direita (165,230 -> 188,207 -> 186,200 -> 161,177)
			168, 172, 176, 180, 184, 188,
			186, 186, 182, 178, 172, 168,
			182, 178, 172, 168,
			186, 182, 178, 174, 170, 165, 161;

		setarray .@pool_y[0],
			// Setor Sul
			178, 178, 178, 178, 178,
			182, 182, 182, 182, 182,
			186, 186, 186, 186, 186,
			190, 190, 190, 190,
			// Setor Oeste
			179, 183, 187, 191, 195, 202,
			204, 208, 206, 206, 206, 206,
			210, 210, 210, 210,
			214, 218, 222, 225, 228, 230,
			// Setor Norte
			230, 230, 230, 230, 230, 230,
			226, 226, 226, 226, 226,
			222, 222, 222, 222, 222,
			216, 216, 216, 216, 216,
			// Setor Leste
			228, 225, 221, 217, 213, 207,
			204, 208, 206, 206, 206, 206,
			210, 210, 210, 210,
			200, 196, 192, 188, 184, 180, 178;
		.@pool_size = getarraysize(.@pool_x);

		// Sorteia quantos pontos visitara nesta passagem por Prontera:
		if (@BOT_PRT_MAX <= 0) {
			@BOT_PRT_MAX = rand(3, 6);
			@BOT_PRT_COUNT = 0;
			@BOT_WP_INDEX = 0;
			@BOT_LAST_TX = 0;
			@BOT_LAST_TY = 0;
			@BOT_STUCK_TICKS = 0;
		}

		// Sub-estagio 0: Sorteia o proximo ponto e inicia o passo suave
		if (@BOT_WP_INDEX == 0) {
			.@r = rand(0, .@pool_size - 1);
			@BOT_LAST_TX = .@pool_x[.@r];
			@BOT_LAST_TY = .@pool_y[.@r];
			@BOT_WP_INDEX = 1; // Transiciona para caminhando
			@BOT_STUCK_TICKS = 0;
			@BOT_STUCK_X = .@cx;
			@BOT_STUCK_Y = .@cy;
			stand; // Garante que esta de pe para andar
			unitstopwalk getcharid(3), USW_FORCE_STOP;
			unitwalk getcharid(3), @BOT_LAST_TX, @BOT_LAST_TY; // Envia o caminho suave uma unica vez
			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Sub-estagio 1: Caminhando suavemente ate o ponto de destino
		if (@BOT_WP_INDEX == 1) {
			.@dist = distance(.@cx, .@cy, @BOT_LAST_TX, @BOT_LAST_TY);

			// Chegou perto do ponto de destino (<= 2 celulas):
			if (.@dist <= 2) {
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				@BOT_PRT_COUNT++;
				@BOT_WP_INDEX = 2; // Transiciona para parado no ponto
				@BOT_STUCK_TICKS = 0;
				// Sorteia um tempo inteiro entre 1 e 10 minutos para esta parada.
				@BOT_WAIT_END = gettimetick(2) + (@BOT_TEST_MODE ? 5 : rand(1, 10) * 60);

				// Permanece em pe observando o movimento (sem sentar):
				stand;

				addtimer 1000, "PlayerBotAI::OnStepTimer";
				end;
			}

			// Anti-travamento inteligente caso colida com obstaculo por mais de 5s:
			if (@BOT_STUCK_X == .@cx && @BOT_STUCK_Y == .@cy) {
				@BOT_STUCK_TICKS++;
				if (@BOT_STUCK_TICKS >= 5) {
					// Sorteia outro ponto aberto a pe
					@BOT_WP_INDEX = 0;
					@BOT_LAST_TX = 0;
					@BOT_LAST_TY = 0;
					@BOT_STUCK_TICKS = 0;
					unitstopwalk getcharid(3), USW_FORCE_STOP;
					addtimer 500, "PlayerBotAI::OnStepTimer";
					end;
				}
			} else {
				@BOT_STUCK_TICKS = 0;
			}

			@BOT_STUCK_X = .@cx;
			@BOT_STUCK_Y = .@cy;

			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Sub-estagio 2: Parado no ponto descansando/observando (Minimo 1 minuto)
		if (@BOT_WP_INDEX == 2) {
			if (gettimetick(2) >= @BOT_WAIT_END) {
				stand; // Levanta do chao
				// Se ja visitou todos os pontos sorteados desta passagem -> Vai para o Eden
				if (@BOT_PRT_COUNT >= @BOT_PRT_MAX) {
					@BOT_AI_STAGE = 2; // Segue para o Grupo Eden
					@BOT_WP_INDEX = 0;
					@BOT_PRT_MAX = 0;
					@BOT_PRT_COUNT = 0;
					@BOT_LAST_TX = 0;
					@BOT_LAST_TY = 0;
					@BOT_STUCK_TICKS = 0;
					@BOT_WAIT_END = 0;
					unitstopwalk getcharid(3), USW_FORCE_STOP;
					warp "moc_para01", 31, 14;
					addtimer 1500, "PlayerBotAI::OnStepTimer";
					end;
				}

				// Ainda tem pontos a visitar: sorteia o proximo a partir de onde esta agora
				@BOT_WP_INDEX = 0;
				@BOT_LAST_TX = 0;
				@BOT_LAST_TY = 0;
				@BOT_STUCK_TICKS = 0;
				@BOT_WAIT_END = 0;
				addtimer 500, "PlayerBotAI::OnStepTimer";
				end;
			}

			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}
	}

	// =========================================================
	// ESTAGIO 1: Descanso Distribuido em Prontera (Minimo 1 Minuto no Modo Start)
	// =========================================================
	if (@BOT_AI_STAGE == 1) {
		getmapxy(.@m$, .@cx, .@cy, BL_PC);
		if (.@m$ != "prontera") {
			stand;
			warp "prontera", 156, 191;
			callsub OnQueuePronteraBuff;
			@BOT_WP_INDEX = 0;
			@BOT_LAST_TX = 0;
			@BOT_LAST_TY = 0;
			@BOT_STUCK_TICKS = 0;
			addtimer 1500, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Grade ampla de mais de 30 pontos de descanso distribuidos (todos em Y >= 177)
		setarray .@p_rest_x[0],
			156, 148, 164, 152, 160, 150, 162,
			132, 126, 138, 122, 142, 130, 136,
			156, 148, 164, 152, 160, 150, 162,
			180, 186, 174, 188, 170, 182, 176,
			146, 165, 126, 184, 138, 172, 144, 168;

		setarray .@p_rest_y[0],
			178, 178, 178, 184, 184, 188, 188,
			206, 206, 206, 208, 212, 216, 222,
			224, 224, 224, 220, 220, 216, 216,
			206, 206, 206, 208, 212, 216, 220,
			230, 230, 212, 212, 225, 222, 180, 180;
		.@p_rest_size = getarraysize(.@p_rest_x);

		// Sorteia o ponto de descanso de forma 100% uniforme:
		if (@BOT_LAST_TX == 0 || @BOT_LAST_TY == 0) {
			.@r_idx = rand(0, .@p_rest_size - 1);
			@BOT_LAST_TX = .@p_rest_x[.@r_idx];
			@BOT_LAST_TY = .@p_rest_y[.@r_idx];
			@BOT_STUCK_TICKS = 0;
		}

		// Sub-estagio 0: Caminhando ate o ponto de descanso sorteado
		if (@BOT_WP_INDEX == 0) {
			.@dist = distance(.@cx, .@cy, @BOT_LAST_TX, @BOT_LAST_TY);

			if (.@dist <= 2) {
				@BOT_WP_INDEX = 1; // Em descanso em pe
				@BOT_STUCK_TICKS = 0;
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				stand; // Permanece em pe no ponto escolhido

				if (@BOT_TEST_MODE) {
					@BOT_WAIT_END = gettimetick(2) + 5;
				} else {
					// No modo Start: sorteia 1, 2, 3, ... ou 10 minutos.
					@BOT_WAIT_END = gettimetick(2) + rand(1, 10) * 60;
				}
				addtimer 1000, "PlayerBotAI::OnStepTimer";
				end;
			}

			// Inicia caminhada ou anti-travamento para o ponto de descanso (100% a pe):
			if (@BOT_STUCK_X == .@cx && @BOT_STUCK_Y == .@cy) {
				@BOT_STUCK_TICKS++;
				if (@BOT_STUCK_TICKS >= 3) {
					// Se encontrou obstaculo no caminho, sorteia outro ponto aberto e continua a pe
					.@r_idx = rand(0, .@p_rest_size - 1);
					@BOT_LAST_TX = .@p_rest_x[.@r_idx];
					@BOT_LAST_TY = .@p_rest_y[.@r_idx];
					@BOT_STUCK_TICKS = 0;
					unitstopwalk getcharid(3), USW_FORCE_STOP;
					unitwalk getcharid(3), @BOT_LAST_TX, @BOT_LAST_TY;
					addtimer 1000, "PlayerBotAI::OnStepTimer";
					end;
				}
			} else {
				@BOT_STUCK_TICKS = 0;
			}

			unitstopwalk getcharid(3), USW_FORCE_STOP;
			unitwalk getcharid(3), @BOT_LAST_TX, @BOT_LAST_TY;

			@BOT_STUCK_X = .@cx;
			@BOT_STUCK_Y = .@cy;

			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Sub-estagio 1: Em pe descansando em seu ponto
		if (@BOT_WP_INDEX == 1) {
			if (gettimetick(2) >= @BOT_WAIT_END) {
				stand;
				@BOT_AI_STAGE = 2;
				@BOT_WP_INDEX = 0;
				@BOT_LAST_TX = 0;
				@BOT_LAST_TY = 0;
				@BOT_STUCK_TICKS = 0;
				@BOT_WAIT_END = 0;
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				warp "moc_para01", 31, 14;
				addtimer 1500, "PlayerBotAI::OnStepTimer";
				end;
			}
			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}
	}

	// =========================================================
	// ESTAGIO 2: Grupo Eden (1 a 10 minutos inteiros por parada) - 100% a pe
	// =========================================================
	if (@BOT_AI_STAGE == 2) {
		stand;
		getmapxy(.@m$, .@cx, .@cy, BL_PC);
		if (.@m$ != "moc_para01") {
			stand;
			warp "moc_para01", 31, 14;
			@BOT_WP_INDEX = 0;
			@BOT_LAST_TX = 0;
			@BOT_LAST_TY = 0;
			@BOT_STUCK_TICKS = 0;
			addtimer 1500, "PlayerBotAI::OnStepTimer";
			end;

		if (@BOT_WP_INDEX == 0) {
			@BOT_WP_INDEX = 1; // Aguardando tempo na entrada
			@BOT_LAST_TX = 0;
			@BOT_LAST_TY = 0;
			@BOT_STUCK_TICKS = 0;

			if (@BOT_TEST_MODE) {
				@BOT_WAIT_END = gettimetick(2) + 5;
			} else {
				// No modo Start: sorteia 1, 2, 3, ... ou 10 minutos.
				@BOT_WAIT_END = gettimetick(2) + rand(1, 10) * 60;
			}
			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;

		if (@BOT_WP_INDEX == 1) {
			if (gettimetick(2) >= @BOT_WAIT_END) {
				@BOT_WP_INDEX = 2; // Vai para uma mesa a pe
				@BOT_LAST_TX = 0;
				@BOT_LAST_TY = 0;
				@BOT_STUCK_TICKS = 0;
				@BOT_WAIT_END = 0;
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				addtimer 500, "PlayerBotAI::OnStepTimer";
				end;
			}
			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Sub-estagio 2: Caminhando da entrada (31,14) para uma mesa de descanso sorteada
		if (@BOT_WP_INDEX == 2) {
			// Grade de mesas/assentos distribuidos no Eden
			setarray .@eden_table_x[0], 28, 26, 30, 28, 26, 30, 32, 24, 22, 34, 28, 30;
			setarray .@eden_table_y[0], 26, 26, 26, 28, 28, 28, 26, 26, 28, 28, 30, 30;
			.@eden_size = getarraysize(.@eden_table_x);

				.@e_idx = rand(0, .@eden_size - 1);
				@BOT_LAST_TX = .@eden_table_x[.@e_idx];
				@BOT_LAST_TY = .@eden_table_y[.@e_idx];
				@BOT_STUCK_TICKS = 0;
			}

			.@dist = distance(.@cx, .@cy, @BOT_LAST_TX, @BOT_LAST_TY);

			// Chegou perto da mesa de descanso sorteada (<= 2 celulas)
			if (.@dist <= 2) {
				@BOT_WP_INDEX = 3; // Em descanso em pe na mesa
				@BOT_STUCK_TICKS = 0;
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				stand; // Permanece em pe na mesa sorteada

				if (@BOT_TEST_MODE) {
					@BOT_WAIT_END = gettimetick(2) + 5;
				} else {
					// No modo Start: sorteia 1, 2, 3, ... ou 10 minutos.
					@BOT_WAIT_END = gettimetick(2) + rand(1, 10) * 60;
				}
				addtimer 1000, "PlayerBotAI::OnStepTimer";
				end;
			}

			// Inicia caminhada continua para a mesa sorteada (100% a pe, sem teleporte):
			if (@BOT_STUCK_X == .@cx && @BOT_STUCK_Y == .@cy) {
				@BOT_STUCK_TICKS++;
				if (@BOT_STUCK_TICKS >= 4) {
					@BOT_WP_INDEX = 3;
					@BOT_LAST_TX = 0;
					@BOT_LAST_TY = 0;
					@BOT_STUCK_TICKS = 0;
					unitstopwalk getcharid(3), USW_FORCE_STOP;
					stand;
					@BOT_WAIT_END = gettimetick(2) + (@BOT_TEST_MODE ? 5 : rand(1, 10) * 60);
					addtimer 1000, "PlayerBotAI::OnStepTimer";
					end;
				}
			} else {
				@BOT_STUCK_TICKS = 0;
			}

			unitstopwalk getcharid(3), USW_FORCE_STOP;
			unitwalk getcharid(3), @BOT_LAST_TX, @BOT_LAST_TY;

			@BOT_STUCK_X = .@cx;
			@BOT_STUCK_Y = .@cy;

			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Sub-estagio 3: Descansando sentado na mesa do Eden
		if (@BOT_WP_INDEX == 3) {
			if (gettimetick(2) >= @BOT_WAIT_END) {
				stand; // Levanta da mesa
				@BOT_WP_INDEX = 4; // Vai para a saida a pe
				@BOT_LAST_TX = 0;
				@BOT_LAST_TY = 0;
				@BOT_STUCK_TICKS = 0;
				@BOT_WAIT_END = 0;
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				addtimer 800, "PlayerBotAI::OnStepTimer";
				end;
			}
			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Sub-estagio 4: Apos descanso, caminhando de volta para a saida (31,14)
		if (@BOT_WP_INDEX == 4) {
			.@tx = 31; .@ty = 14;
			.@dist = distance(.@cx, .@cy, .@tx, .@ty);

			// Chegou perto da saida (<= 2 celulas) -> Vai para a cacada
			if (.@dist <= 2) {
				@BOT_AI_STAGE = 3;
				@BOT_WP_INDEX = 0;
				@BOT_HUNT_COUNT = 0;
				@BOT_CURRENT_TARGET = 0;
				@BOT_LAST_TX = 0;
				@BOT_LAST_TY = 0;
				@BOT_DIR_VX = 0;
				@BOT_DIR_VY = 0;
				@BOT_EXPLORE_STEPS = 0;
				@BOT_WALL_HIT = 0;
				@BOT_STUCK_X = 0;
				@BOT_STUCK_Y = 0;
				@BOT_STUCK_TICKS = 0;
				callsub OnStartHuntTimer;
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				callsub OnGetHuntMap;
				warp @hunt_map$, 0, 0;
				addtimer 1500, "PlayerBotAI::OnStepTimer";
				end;
			}

			// Inicia caminhada continua para 31,14 (100% a pe, sem teleporte):
			if (@BOT_LAST_TX != 31 || @BOT_LAST_TY != 14 || (@BOT_STUCK_X == .@cx && @BOT_STUCK_Y == .@cy)) {
				if (@BOT_STUCK_X == .@cx && @BOT_STUCK_Y == .@cy) {
					@BOT_STUCK_TICKS++;
					if (@BOT_STUCK_TICKS >= 4) {
						@BOT_AI_STAGE = 3;
						@BOT_WP_INDEX = 0;
						@BOT_HUNT_COUNT = 0;
						@BOT_CURRENT_TARGET = 0;
						@BOT_LAST_TX = 0;
						@BOT_LAST_TY = 0;
						@BOT_STUCK_TICKS = 0;
						callsub OnStartHuntTimer;
						unitstopwalk getcharid(3), USW_FORCE_STOP;
						callsub OnGetHuntMap;
						warp @hunt_map$, 0, 0;
						addtimer 1500, "PlayerBotAI::OnStepTimer";
						end;
					}
				} else {
					@BOT_STUCK_TICKS = 0;
				}
				@BOT_LAST_TX = 31;
				@BOT_LAST_TY = 14;
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				unitwalk getcharid(3), 31, 14;
			}

			@BOT_STUCK_X = .@cx;
			@BOT_STUCK_Y = .@cy;

			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}
	}

	// =========================================================
	// ESTAGIO 3: Cacada Inteligente e Agressiva (Busca por getareaunits)
	// =========================================================
	if (@BOT_AI_STAGE == 3) {
		stand;
		callsub OnCheckBuffs;
		callsub OnCheckSpeedPotion;
		callsub OnCheckPotions;
		callsub OnGetHuntMap;
		getmapxy(.@m$, .@cx, .@cy, BL_PC);

		if (.@m$ != @hunt_map$) {
			stand;
			warp @hunt_map$, 0, 0;
			@BOT_CURRENT_TARGET = 0;
			@BOT_TARGET_TRACK_ID = 0;
			@BOT_TARGET_LAST_HP = 0;
			@BOT_TARGET_PROGRESS_TICK = 0;
			@BOT_BLOCKED_TARGET = 0;
			@BOT_BLOCKED_UNTIL = 0;
			@BOT_LAST_TX = 0;
			@BOT_LAST_TY = 0;
			@BOT_STUCK_TICKS = 0;
			callsub OnStartHuntTimer;
			addtimer 1500, "PlayerBotAI::OnStepTimer";
			end;
		}

		if (@BOT_HUNT_START == 0 || @BOT_HUNT_DURATION <= 0)
			callsub OnStartHuntTimer;

		// A duracao foi fixada no inicio deste ciclo para nao mudar a cada tick.
		if ((gettimetick(2) - @BOT_HUNT_START) >= @BOT_HUNT_DURATION) {
			@BOT_HUNT_START = 0;
			@BOT_HUNT_DURATION = 0;
			@BOT_AI_STAGE = 0; // Retorna para o Estagio 0 (Passeio e descanso em Prontera)
			@BOT_WP_INDEX = 99; // Sub-estagio 99: aguarda 2s em Prontera enquanto o @buff e agendado.
			@BOT_CURRENT_TARGET = 0;
			@BOT_PRT_MAX = 0;
			@BOT_PRT_COUNT = 0;
			@BOT_LAST_TX = 0;
			@BOT_LAST_TY = 0;
			@BOT_DIR_VX = 0;
			@BOT_DIR_VY = 0;
			@BOT_EXPLORE_STEPS = 0;
			@BOT_WALL_HIT = 0;
			@BOT_STUCK_X = 0;
			@BOT_STUCK_Y = 0;
			@BOT_STUCK_TICKS = 0;
			unitstopattack getcharid(3);
			unitstopwalk getcharid(3), USW_FORCE_STOP;
			savepoint "prontera", 156, 191;
			warp "prontera", 156, 191;
			callsub OnQueuePronteraBuff;
			@BOT_WAIT_END = gettimetick(2) + 2;
			addtimer 2000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Libera o monstro bloqueado depois do tempo de seguranca.
		if (@BOT_BLOCKED_TARGET > 0 && gettimetick(2) >= @BOT_BLOCKED_UNTIL) {
			@BOT_BLOCKED_TARGET = 0;
			@BOT_BLOCKED_UNTIL = 0;
		}

		// 1. Verifica se o monstro alvo anterior ainda existe, esta vivo, valido e nao esta sendo atacado por outro (Anti-KS)
		if (@BOT_CURRENT_TARGET > 0) {
			if (!unitexists(@BOT_CURRENT_TARGET)) {
				@BOT_CURRENT_TARGET = 0;
				@BOT_TARGET_TRACK_ID = 0;
				@BOT_TARGET_LAST_HP = 0;
				@BOT_TARGET_PROGRESS_TICK = 0;
				@BOT_LAST_TX = 0;
				@BOT_LAST_TY = 0;
			} else {
				getunitdata @BOT_CURRENT_TARGET, .@cur_data;
				.@cur_class = .@cur_data[UMOB_CLASS];
				.@cur_target_id = .@cur_data[UMOB_TARGETID];
				.@d = distance(.@cx, .@cy, .@cur_data[UMOB_X], .@cur_data[UMOB_Y]);

				// Se o monstro morreu, e planta/cogumelo, se afastou mais de 14 celulas ou foi engatado por outro jogador (Anti-KS):
				if (.@cur_data[UMOB_HP] <= 0 || (.@cur_class >= 1078 && .@cur_class <= 1085) || (.@cur_class >= 1930 && .@cur_class <= 1933) || .@cur_class == 1064 ||
				    .@d > 14 || (.@cur_data[UMOB_HP] < .@cur_data[UMOB_MAXHP] && .@cur_target_id > 0 && .@cur_target_id != getcharid(3))) {
					@BOT_CURRENT_TARGET = 0;
					@BOT_TARGET_TRACK_ID = 0;
					@BOT_TARGET_LAST_HP = 0;
					@BOT_TARGET_PROGRESS_TICK = 0;
					@BOT_LAST_TX = 0;
					@BOT_LAST_TY = 0;
				} else {
					.@mx = .@cur_data[UMOB_X];
					.@my = .@cur_data[UMOB_Y];

					// Se o bot ja iniciou o ataque e o HP nao diminui por 8 segundos,
					// existe provavelmente uma parede ou obstaculo entre os dois.
					if (@BOT_TARGET_TRACK_ID == @BOT_CURRENT_TARGET) {
						if (.@cur_data[UMOB_HP] < @BOT_TARGET_LAST_HP) {
							@BOT_TARGET_LAST_HP = .@cur_data[UMOB_HP];
							@BOT_TARGET_PROGRESS_TICK = gettimetick(2);
						} else if ((gettimetick(2) - @BOT_TARGET_PROGRESS_TICK) >= 8) {
							@BOT_BLOCKED_TARGET = @BOT_CURRENT_TARGET;
							@BOT_BLOCKED_UNTIL = gettimetick(2) + 20;
							@BOT_CURRENT_TARGET = 0;
							@BOT_TARGET_TRACK_ID = 0;
							@BOT_TARGET_LAST_HP = 0;
							@BOT_TARGET_PROGRESS_TICK = 0;
							@BOT_LAST_TX = 0;
							@BOT_LAST_TY = 0;
							unitstopattack getcharid(3);
							unitstopwalk getcharid(3), USW_FORCE_STOP;
						} else if (.@cur_data[UMOB_HP] > @BOT_TARGET_LAST_HP) {
							@BOT_TARGET_LAST_HP = .@cur_data[UMOB_HP];
						}
					}
				}
			}
		}

		if (@BOT_CURRENT_TARGET == 0) {
			.@num_mobs = getareaunits(BL_MOB, strcharinfo(3), .@cx - 14, .@cy - 14, .@cx + 14, .@cy + 14, .@mob_list[0]);
			.@best_score = -999999;
			.@best_mob = 0;

			for (.@i = 0; .@i < .@num_mobs; .@i++) {
				if (unitexists(.@mob_list[.@i])) {
					// Nao seleciona imediatamente o alvo inacessivel que acabou de travar a IA.
					if (.@mob_list[.@i] == @BOT_BLOCKED_TARGET && gettimetick(2) < @BOT_BLOCKED_UNTIL)
						continue;
					getunitdata .@mob_list[.@i], .@t_data;
					.@m_class = .@t_data[UMOB_CLASS];
					.@m_lvl = .@t_data[UMOB_LEVEL];

					// Ignora Plantas e Cogumelos
					if ((.@m_class >= 1078 && .@m_class <= 1085) || (.@m_class >= 1930 && .@m_class <= 1933) || .@m_class == 1064) {
						continue;
					}

					// Anti-KS: Se o monstro ja estiver com HP reduzido e lutando contra outro jogador, ignora!
					.@m_target = .@t_data[UMOB_TARGETID];
					if (.@t_data[UMOB_HP] < .@t_data[UMOB_MAXHP] && .@m_target > 0 && .@m_target != getcharid(3)) {
						continue;
					}

					if (.@t_data[UMOB_HP] > 0) {
						.@d = distance(.@cx, .@cy, .@t_data[UMOB_X], .@t_data[UMOB_Y]);
						if (.@d <= 14) {
							// Pontuacao: prioriza monstros com cerca de 10 a 20 niveis a mais para ganho acelerado de EXP
							.@lvl_diff = .@m_lvl - BaseLevel;
							.@score = (100 - .@d * 4);
							if (.@lvl_diff >= 8 && .@lvl_diff <= 25) {
								.@score += 60; // Maxima prioridade para monstros de nivel ideal (+10 a +15 niveis)
							} else if (.@lvl_diff > 0) {
								.@score += 30;
							}
							if (.@score > .@best_score) {
								.@best_score = .@score;
								.@best_mob = .@mob_list[.@i];
								.@mx = .@t_data[UMOB_X];
								.@my = .@t_data[UMOB_Y];
							}
						}
					}
				}
			}

			if (.@best_mob > 0) {
				@BOT_CURRENT_TARGET = .@best_mob;
			}
		}

		if (@BOT_CURRENT_TARGET == 0) {
			if ((gettimetick(2) - @BOT_LAST_TELEPORT) >= 5) {
				@BOT_LAST_TELEPORT = gettimetick(2);
				@BOT_LAST_TX = 0;
				@BOT_LAST_TY = 0;
				@BOT_STUCK_TICKS = 0;
				unitstopattack getcharid(3);
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				warp @hunt_map$, 0, 0;
				addtimer 1000, "PlayerBotAI::OnStepTimer";
				end;
			} else {
				// Enquanto aguarda o delay de 5 segundos de teleporte, caminha e explora ao redor
				if (@BOT_LAST_TX == 0 || @BOT_LAST_TY == 0 || (@BOT_STUCK_X == .@cx && @BOT_STUCK_Y == .@cy)) {
					@BOT_LAST_TX = .@cx + rand(-8, 8);
					@BOT_LAST_TY = .@cy + rand(-8, 8);
					unitstopwalk getcharid(3), USW_FORCE_STOP;
					unitwalk getcharid(3), @BOT_LAST_TX, @BOT_LAST_TY;
				}
				@BOT_STUCK_X = .@cx;
				@BOT_STUCK_Y = .@cy;
				addtimer 600, "PlayerBotAI::OnStepTimer";
				end;
			}
		}

			.@dist = distance(.@cx, .@cy, .@mx, .@my);
			.@atk_range = 2; // Padrao corpo a corpo

			// Classes ou builds com habilidades de longo alcance podem iniciar o combate a 7-8 celulas:
			if (BaseClass == Job_Archer || BaseClass == Job_Mage || BaseClass == Job_Ninja || Class == Job_Kagerou || Class == Job_Oboro || Class == Job_Super_Novice || Class == Job_Super_Novice_E || Class == Job_Gunslinger || Class == Job_Rebellion || BaseClass == Job_Summoner || Class == Job_Summoner || Class == Job_Baby_Summoner || Class == Job_Spirit_Handler || Class == 4218 || Class == 4220 || Class == 4308 ||
			    (BaseClass == Job_Merchant && (Class == Job_Mechanic || Class == Job_Mechanic_T || Class == Job_Mechanic2 || Class == Job_Mechanic_T2 || Class == Job_Meister || Class == Job_Meister2 || Class == Job_Genetic || Class == Job_Genetic_T || Class == Job_Genetic2 || Class == Job_Genetic_T2 || Class == Job_Biolo || Class == Job_Biolo2 || getskilllv("NC_ARMSCANNON") > 0 || getskilllv("GN_CARTCANNON") > 0)) ||
			    (BaseClass == Job_Swordman && (getskilllv("LG_BANISHINGPOINT") > 0 || getskilllv("LG_PINPOINTATTACK") > 0 || getskilllv("LG_CANNONSPEAR") > 0 || getskilllv("RK_DRAGONBREATH") > 0 || getskilllv("RK_SONICWAVE") > 0))) {
				.@atk_range = 7; // Pode iniciar disparo a distancia
			}
			// Fire Dance atinge a area ao redor do personagem; com revolver o
			// Rebellion se aproxima o suficiente para a habilidade acertar o alvo.
			if (Class == Job_Rebellion && getequipid(EQI_HAND_R) > 0 && getItemInfo(getequipid(EQI_HAND_R), ITEMINFO_SUBTYPE) == W_REVOLVER)
				.@atk_range = 3;

			// Se ainda esta fora do alcance de ataque: anda em direcao ao monstro de forma fluida
			if (.@dist > .@atk_range) {
				// Sistema Anti-Travamento (Tolerancia de 8 ticks / aprox. 4 segundos antes de resetar)
				if (@BOT_STUCK_X == .@cx && @BOT_STUCK_Y == .@cy) {
					@BOT_STUCK_TICKS++;
					if (@BOT_STUCK_TICKS >= 8) {
						@BOT_CURRENT_TARGET = 0;
						@BOT_LAST_TX = 0;
						@BOT_LAST_TY = 0;
						@BOT_STUCK_TICKS = 0;
						if ((gettimetick(2) - @BOT_LAST_TELEPORT) >= 5) {
							@BOT_LAST_TELEPORT = gettimetick(2);
							unitstopwalk getcharid(3), USW_FORCE_STOP;
							warp @hunt_map$, 0, 0;
							addtimer 1000, "PlayerBotAI::OnStepTimer";
							end;
						}
					}
				} else {
					@BOT_STUCK_TICKS = 0;
				}

				// Envia o comando de andar apenas se o monstro mudou de posicao:
				if (@BOT_LAST_TX != .@mx || @BOT_LAST_TY != .@my) {
					@BOT_LAST_TX = .@mx;
					@BOT_LAST_TY = .@my;
					unitstopwalk getcharid(3), USW_FORCE_STOP;
					unitwalk getcharid(3), .@mx, .@my;
				}

				@BOT_STUCK_X = .@cx;
				@BOT_STUCK_Y = .@cy;

				addtimer 400, "PlayerBotAI::OnStepTimer";
				end;
			}

			// Alcancou o alcance de ataque! Para de andar suavemente
			if (@BOT_LAST_TX != 0 || @BOT_LAST_TY != 0) {
				@BOT_LAST_TX = 0;
				@BOT_LAST_TY = 0;
				unitstopwalk getcharid(3), USW_FORCE_STOP;
			}
			@BOT_STUCK_TICKS = 0;
			@BOT_STUCK_X = .@cx;
			@BOT_STUCK_Y = .@cy;

			// Garante SP cheio antes de executar as habilidades!
			callsub OnCheckPotions;

			// A partir daqui o bot realmente comecou a atacar. O HP deste alvo
			// passa a ser acompanhado para detectar parede/linha de visao bloqueada.
			if (@BOT_TARGET_TRACK_ID != @BOT_CURRENT_TARGET) {
				getunitdata @BOT_CURRENT_TARGET, .@attack_data;
				@BOT_TARGET_TRACK_ID = @BOT_CURRENT_TARGET;
				@BOT_TARGET_LAST_HP = .@attack_data[UMOB_HP];
				@BOT_TARGET_PROGRESS_TICK = gettimetick(2);
			}

			// Engata ataque basico continuo
			unitattack getcharid(3), @BOT_CURRENT_TARGET, 1;
			@BOT_SKILL_ROT++;

			// Calcula o atraso de ataque e pós-conjuração baseado na ASPD real do personagem
			.@cur_aspd = readparam(bAspd);

// mantendo Mestre Taekwon e Espiritualista fora da selecao.
OnCanSimulateInstance:
	if (eaclass() & (EAJL_THIRD | EAJL_FOURTH)) return 1;
	if (Class == Job_Super_Novice_E || Class == Job_Super_Baby_E ||
	    Class == Job_Kagerou || Class == Job_Oboro || Class == Job_Rebellion ||
	    Class == Job_Baby_Kagerou || Class == Job_Baby_Oboro || Class == Job_Baby_Rebellion)
		return 1;
	return 0;

OnScheduleNextInstance:
	@BOT_NEXT_INSTANCE_TICK = gettimetick(2) + rand($BOT_INSTANCE_MIN_MINUTES, $BOT_INSTANCE_MAX_MINUTES) * 60;
	return;

OnTryBotInstance:
	if (!@BOT_AI_ACTIVE || @BOT_AI_STAGE == 4 || @BOT_INSTANCE_ACTIVE) return 0;
	if (@BOT_NEXT_INSTANCE_TICK <= 0) {
		callsub OnScheduleNextInstance;
		return 0;
	}
	if (gettimetick(2) < @BOT_NEXT_INSTANCE_TICK) return 0;
	if (!callsub(OnCanSimulateInstance)) {
		// Reavalia depois: o personagem pode evoluir para uma classe elegivel.
		@BOT_NEXT_INSTANCE_TICK = gettimetick(2) + 600;
		return 0;
	}

	// ate Labirinto de Oz. Torre Sem Fim foi intencionalmente removida.

		}

OnCheckJobChange:
	callsub OnDetectTargetJob;

	// Novico segue para a primeira classe correspondente. Novico T. sempre
	// segue para a primeira classe trans da mesma arvore.
	if (Class == Job_Novice || Class == Job_Novice_High) {
		if (BOT_FINAL_JOB == 117) {
			// Super Novico exige nivel de base 45 e nao possui renascimento.
			if (BaseLevel < 45) return;
			jobchange Job_Super_Novice;
		} else if (BOT_FINAL_JOB == 118) {
			jobchange Job_Gunslinger;
			if (countitem(13200) < 500) getitem 13200, 2000;
			if (getequipid(EQI_AMMO) <= 0) equip 13200;
			atcommand "@restock 13200 1000";
			query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 13200, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
		} else if (BOT_FINAL_JOB == 113) {
			jobchange Job_Summoner;
		} else if (BOT_FINAL_JOB == 114 || BOT_FINAL_JOB == 115) {
			jobchange Job_Taekwon;
		} else if (BOT_FINAL_JOB == 116) {
			jobchange Job_Ninja;
		} else switch(@BOT_TARGET_JOB) {
			case 1: 
				jobchange (Class == Job_Novice_High ? Job_Swordman_High : Job_Swordman);
				if (countitem(1124) == 0 && countitem(1104) == 0) getitem 1124, 1;
				break;
			case 2: 
				jobchange (Class == Job_Novice_High ? Job_Mage_High : Job_Mage);
				if (countitem(1601) == 0) getitem 1601, 1;
				break;
			case 3: 
				jobchange (Class == Job_Novice_High ? Job_Archer_High : Job_Archer);
				if (countitem(1704) == 0) getitem 1704, 1;
				if (countitem(1750) < 500) getitem 1750, 2000;
				atcommand "@restock 1750 1000";
				query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 1750, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
				break;
			case 4: 
				jobchange (Class == Job_Novice_High ? Job_Thief_High : Job_Thief);
				if (countitem(1207) == 0) getitem 1207, 1;
				break;
			case 5: 
				jobchange (Class == Job_Novice_High ? Job_Acolyte_High : Job_Acolyte);
				if (countitem(1504) == 0) getitem 1504, 1;
				break;
			case 6:
				jobchange (Class == Job_Novice_High ? Job_Merchant_High : Job_Merchant);
				if (countitem(1304) == 0) getitem 1304, 1;
				// Nao troca a arma que ja foi entregue/equipada no personagem.
				if (getequipid(EQI_HAND_R) <= 0) equip 1304;
				if (Zeny < 5000000) Zeny += 10000000;
				break;
		}
		callsub OnFinalizeBotJobChange;
		return;
	}

	// Doram nao possui renascimento/transclasse: evolucao direta normal.
	if (BOT_FINAL_JOB == 113 && Class == Job_Summoner && BaseLevel >= 200 && JobLevel >= 60) {
		jobchange Job_Spirit_Handler;
		callsub OnFinalizeBotJobChange;
		return;
	}

	// Taekwon tambem nao renasce. Primeiro escolhe Mestre Taekwon ou
	// Espiritualista; depois evolui diretamente para a classe expandida final.
	if ((BOT_FINAL_JOB == 114 || BOT_FINAL_JOB == 115) && Class == Job_Taekwon && JobLevel >= 50) {
		jobchange (BOT_FINAL_JOB == 114 ? Job_Star_Gladiator : Job_Soul_Linker);
		callsub OnFinalizeBotJobChange;
		return;
	}
	if (BOT_FINAL_JOB == 114 && (Class == Job_Star_Gladiator || Class == Job_Star_Gladiator2) && BaseLevel >= 99 && JobLevel >= 50) {
		jobchange Job_Star_Emperor;
		callsub OnFinalizeBotJobChange;
		return;
	}
	if (BOT_FINAL_JOB == 115 && Class == Job_Soul_Linker && BaseLevel >= 99 && JobLevel >= 50) {
		jobchange Job_Soul_Reaper;
		callsub OnFinalizeBotJobChange;
		return;
	}

	// Ninja evolui diretamente, sem renascimento, conforme o sexo do personagem.
	if (BOT_FINAL_JOB == 116 && Class == Job_Ninja && BaseLevel >= 99 && JobLevel >= 70) {
		jobchange (Sex == SEX_MALE ? Job_Kagerou : Job_Oboro);
		callsub OnFinalizeBotJobChange;
		return;
	}

	// Super Novico e Justiceiro tambem seguem as evolucoes expandidas sem renascer.
	if (BOT_FINAL_JOB == 117 && Class == Job_Super_Novice && BaseLevel >= 99 && JobLevel >= 99) {
		jobchange Job_Super_Novice_E;
		callsub OnFinalizeBotJobChange;
		return;
	}
	if (BOT_FINAL_JOB == 118 && Class == Job_Gunslinger && BaseLevel >= 99 && JobLevel >= 70) {
		jobchange Job_Rebellion;
		if (countitem(13200) < 500) getitem 13200, 2000;
		if (getequipid(EQI_AMMO) <= 0) equip 13200;
		atcommand "@restock 13200 1000";
		query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 13200, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
		callsub OnFinalizeBotJobChange;
		return;
	}

	// Toda segunda classe normal das arvores classicas deve renascer em 99/50.
	if (BaseLevel >= 99 && JobLevel >= 50 && Upper == 0 &&
	   (Class == Job_Knight || Class == Job_Crusader || Class == Job_Wizard || Class == Job_Sage ||
	    Class == Job_Hunter || Class == Job_Bard || Class == Job_Dancer || Class == Job_Assassin ||
	    Class == Job_Rogue || Class == Job_Priest || Class == Job_Monk || Class == Job_Blacksmith ||
	    Class == Job_Alchemist)) {
		jobchange Job_Novice_High;
		resetlvl 1;
		resetskill;
		specialeffect EF_ANGEL;
		callsub OnCheckJobChange;
		return;
	}

	// Primeira classe normal escolhe a ramificacao definida pelo GM.
	if (JobLevel >= 50 && Upper == 0) {
		.@changed = 1;
		if (Class == Job_Swordman) jobchange (BOT_FINAL_JOB == 102 ? Job_Crusader : Job_Knight);
		else if (Class == Job_Mage) jobchange (BOT_FINAL_JOB == 104 ? Job_Sage : Job_Wizard);
		else if (Class == Job_Archer) {
			if (BOT_FINAL_JOB == 106) jobchange (Sex == SEX_MALE ? Job_Bard : Job_Dancer);
			else jobchange Job_Hunter;
		}
		else if (Class == Job_Thief) jobchange (BOT_FINAL_JOB == 108 ? Job_Rogue : Job_Assassin);
		else if (Class == Job_Acolyte) jobchange (BOT_FINAL_JOB == 110 ? Job_Monk : Job_Priest);
		else if (Class == Job_Merchant) jobchange (BOT_FINAL_JOB == 112 ? Job_Alchemist : Job_Blacksmith);
		else .@changed = 0;

		if (.@changed) {
			callsub OnFinalizeBotJobChange;
			return;
		}
	}

	// Espadachim T./Mago T./etc. escolhe a transclasse da ramificacao final.
	if (JobLevel >= 50 && Upper == 1) {
		.@changed = 1;
		if (Class == Job_Swordman_High) jobchange (BOT_FINAL_JOB == 102 ? Job_Paladin : Job_Lord_Knight);
		else if (Class == Job_Mage_High) jobchange (BOT_FINAL_JOB == 104 ? Job_Professor : Job_High_Wizard);
		else if (Class == Job_Archer_High) {
			if (BOT_FINAL_JOB == 106) jobchange (Sex == SEX_MALE ? Job_Clown : Job_Gypsy);
			else jobchange Job_Sniper;
		}
		else if (Class == Job_Thief_High) jobchange (BOT_FINAL_JOB == 108 ? Job_Stalker : Job_Assassin_Cross);
		else if (Class == Job_Acolyte_High) jobchange (BOT_FINAL_JOB == 110 ? Job_Champion : Job_High_Priest);
		else if (Class == Job_Merchant_High) jobchange (BOT_FINAL_JOB == 112 ? Job_Creator : Job_Whitesmith);
		else .@changed = 0;

		if (.@changed) {
			callsub OnFinalizeBotJobChange;
			return;
		}
	}

	// A transclasse evolui em 99/70 para a terceira classe escolhida.
	if (BaseLevel >= 99 && JobLevel >= 70 && Upper == 1 &&
	   (Class == Job_Lord_Knight || Class == Job_Paladin || Class == Job_High_Wizard || Class == Job_Professor ||
	    Class == Job_Sniper || Class == Job_Clown || Class == Job_Gypsy || Class == Job_Assassin_Cross ||
	    Class == Job_Stalker || Class == Job_High_Priest || Class == Job_Champion || Class == Job_Whitesmith ||
	    Class == Job_Creator)) {
		if (BOT_FINAL_JOB == 101) jobchange Job_Rune_Knight;
		else if (BOT_FINAL_JOB == 102) jobchange Job_Royal_Guard;
		else if (BOT_FINAL_JOB == 103) jobchange Job_Warlock;
		else if (BOT_FINAL_JOB == 104) jobchange Job_Sorcerer;
		else if (BOT_FINAL_JOB == 105) jobchange Job_Ranger;
		else if (BOT_FINAL_JOB == 106) jobchange (Sex == SEX_MALE ? Job_Minstrel : Job_Wanderer);
		else if (BOT_FINAL_JOB == 107) jobchange Job_Guillotine_Cross;
		else if (BOT_FINAL_JOB == 108) jobchange Job_Shadow_Chaser;
		else if (BOT_FINAL_JOB == 109) jobchange Job_Arch_Bishop;
		else if (BOT_FINAL_JOB == 110) jobchange Job_Sura;
		else if (BOT_FINAL_JOB == 111) jobchange Job_Mechanic;
		else if (BOT_FINAL_JOB == 112) jobchange Job_Genetic;
		else return;

		if (BOT_FINAL_JOB == 111 || BOT_FINAL_JOB == 112) {
			if (countitem(18000) < 500) getitem 18000, 2000;
			if (getequipid(EQI_AMMO) <= 0) equip 18000;
			atcommand "@restock 18000 1000";
			query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 18000, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
		}
		callsub OnFinalizeBotJobChange;
		return;
	}
	return;

OnFinalizeBotJobChange:
	specialeffect EF_ANGEL;
	resetskill;
	callsub OnDistributeSkills;
	callsub OnDistributeStats;
	return;
}
