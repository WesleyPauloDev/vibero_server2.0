// ======================================================================
// Sistema Unificado de AI / Automacao para Personagem Real (PlayerBotAI)
// - Arquitetura Unificada de NPC unico (elimina erros de cross-event)
// - 100% Silencioso: Sem unittalk para nao expor o bot a outros jogadores
// - Atributos Reais e Balanceados: Usa statusup respeitando a tabela de pontos do nivel
// - Promocao automatica para 1a Classe no inicio com Habilidades Ofensivas reais
// - SavePoint configurado no Centro de Prontera (156, 170)
// - Retorno automatico ao SavePoint e revivificacao completa ao morrer (OnPCDieEvent)
// - Caminhada Organica em Prontera: Sorteia quantidade (3 a 6) e coordenadas aleatorias
//   na praca central da fonte com sistema anti-travamento integrado
// - Eden em 2 pontos (30,14 e 28,26) com caminhada real
// - Modo Teste: 5s em Prontera, 5s no Eden, 1 minuto exato de cacada
// - Combate Completo e Agressivo:
//     * Localizacao com getareaunits (raio 25 celulas, nunca teleporta se houver mobs na tela)
//     * Ignora completamente Plantas e Cogumelos
//     * Auto-Pocao SP Azul (505) acionada SEMPRE que o SP estiver abaixo de 80%
//     * Auto-Pocao HP Branca (504) acionada quando HP < 70%
//     * Auto-Ataque fisico basico ininterrupto + Habilidades Ofensivas instantaneas (ignore_range)
//     * Auto-Buffs nativos diretos (sem doevent para evitar enfileiramento)
//     * OnCheckBuffs:
//     if (!#BOT_AI_ACTIVE || #BOT_AI_STAGE != 3) return;
//     * OnApplyBuffs:
//     if (!#BOT_AI_ACTIVE || #BOT_AI_STAGE != 3) return;
//     * OnCheckSpeedPotion:
//     if (!#BOT_AI_ACTIVE || #BOT_AI_STAGE != 3) return;
//     * 30.000 de cada no armazem e @restock de 5 unidades das pocoes de ASPD
//     * Auto-Teleporte aleatorio no mapa APENAS se a tela estiver vazia de monstros
// - Respostas de comandos do GM enviadas exclusivamente para o chat do GM
// - Comandos GM para Controle:
//     @bot test [nome]   -> Inicia o MODO TESTE (5s pausas, 1 min cacada)
//     @bot start [nome]  -> Inicia o MODO NORMAL (2 min pausas, 5 min cacada)
//     @bot stop [nome]   -> Pausa a AI do personagem
//     @bot status [nome] -> Exibe informacoes para o GM
//     @bot               -> Menu Interativo do GM
// - Comando para o proprio jogador:
//     @iniciante         -> Ativa/Desativa a AI no proprio personagem
// Arquivo: npc/custom/iniciante_autobot.txt
// ======================================================================

-	script	PlayerBotAI	-1,{

OnInit:
	bindatcmd("bot", "PlayerBotAI::OnGMCommand", 60, 99);
	bindatcmd("iniciante", "PlayerBotAI::OnPlayerCommand", 0, 99);
	end;

// ======================================================================
// COMANDO DO JOGADOR: @iniciante (para rodar no proprio char)
// ======================================================================
	// @iniciante test / @iniciante teste
	if (.@subcmd$ == "test" || .@subcmd$ == "teste") {
		#BOT_AI_ACTIVE = 1;
		#BOT_TEST_MODE = 1;
		#BOT_AI_STAGE = 3; // Inicia direto na Cacada
		#BOT_HUNT_START = gettimetick(2);

		resetskill;
		callsub OnCheckJobChange;
		callsub OnDistributeSkills;
		callsub OnDistributeStats;
		callsub OnApplyBuffs;
		callsub OnCheckMounts;
		callsub OnCheckSpeedPotion;

		addtimer 1000, "PlayerBotAI::OnStepTimer";
		end;
	}

	// @iniciante at / @iniciante offline
	if (.@subcmd$ == "at" || .@subcmd$ == "offline" || .@subcmd$ == "autotrade") {
		if (!#BOT_AI_ACTIVE) {
			#BOT_AI_ACTIVE = 1;
			#BOT_TEST_MODE = 0;

			#BOT_HUNT_START = 0;

			savepoint "prontera", 156, 178;

			resetskill;
			callsub OnCheckJobChange;

		#BOT_TEST_MODE = 0;
		#BOT_HUNT_START = 0;
		#BOT_CURRENT_TARGET = 0;
		#BOT_PRT_MAX = 0;

		#BOT_STUCK_X = 0;
		#BOT_STUCK_Y = 0;
		#BOT_STUCK_TICKS = 0;
		deltimer "PlayerBotAI::OnStepTimer";
		unitstopwalk getcharid(3), USW_FORCE_STOP;
		unitstopattack getcharid(3);
		dispbottom "[Iniciante AI]: Sistema desativado com sucesso.", 0xFF5555;
	} else {
		#BOT_AI_ACTIVE = 1;
		#BOT_TEST_MODE = 0; // Padrao normal
		#BOT_AI_STAGE = 0;  // Inicia em Prontera
		#BOT_HUNT_START = 0;
		#BOT_WP_INDEX = 0;
		#BOT_HUNT_COUNT = 0;
		#BOT_CURRENT_TARGET = 0;
		#BOT_PRT_MAX = 0;
		#BOT_PRT_COUNT = 0;
		#BOT_LAST_TX = 0;
		#BOT_LAST_TY = 0;
		#BOT_DIR_VX = 0;
		#BOT_DIR_VY = 0;
		#BOT_EXPLORE_STEPS = 0;
		#BOT_WALL_HIT = 0;
		#BOT_SKILL_ROT = 0;
		#BOT_STUCK_X = 0;
		#BOT_STUCK_Y = 0;
		#BOT_STUCK_TICKS = 0;
		#BOT_WAIT_END = 0;
		stand;
		callsub OnDetectTargetJob;

		savepoint "prontera", 156, 178;
		warp "prontera", 156, 178;

		// Injeta 30.000 de cada pocao no armazem da Kafra
		query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 504, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
		query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 505, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
		query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 645, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
		query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 656, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
		query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 657, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";

		if (countitem(505) < 50) getitem 505, 50;

		atcommand "@autoloot 100";
		atcommand "@restock 504 100";
		atcommand "@restock 505 100";
		atcommand "@restock 645 5";
		atcommand "@restock 656 5";
		atcommand "@restock 657 5";
		resetstatus;
		callsub OnCheckJobChange;
		dispbottom "[Iniciante AI]: Sistema ativado no seu personagem!", 0x55FF55;

// ======================================================================
// COMANDOS DO GM: @bot <acao> [nome] (Respostas vao para o GM)
// ======================================================================
OnGMCommand:
	.@gm_aid = getcharid(3);

	if (getgmlevel() < 60) {
		dispbottom "[Bot Manager]: Voce nao tem permissao para usar este comando.", 0xFF5555;
		end;
	}

	.@cmd$ = strtolower(.@atcmd_parameters$[0]);
	.@target_name$ = "";
	if (.@atcmd_numparameters > 1) {
		.@target_name$ = .@atcmd_parameters$[1];
		for (.@i = 2; .@i < .@atcmd_numparameters; ++.@i) {
			.@target_name$ = .@target_name$ + " " + .@atcmd_parameters$[.@i];
		}
	}

	if (.@target_name$ == "" && .@cmd$ != "") {
		.@target_name$ = strcharinfo(0);
		.@target_aid = getcharid(3);
	} else if (.@target_name$ != "") {
		.@target_aid = 0;
		query_sql "SELECT `account_id` FROM `char` WHERE `name` = '" + escape_sql(.@target_name$) + "' LIMIT 1", .@target_aid;
		if (.@target_aid > 0 && !isloggedin(.@target_aid)) {
			.@target_aid = 0;
		}
	}

	// Menu visual se digitou apenas @bot sem parametros
	if (.@atcmd_numparameters == 0) {
		mes "^0055FF[Gerenciador de AI de Personagem - GM]^000000";
		mes "Digite o nome do personagem (ou deixe em branco para usar o seu):";
		next;
		input .@target_name$;
		if (.@target_name$ == "") {
			.@target_name$ = strcharinfo(0);
			.@target_aid = getcharid(3);
		} else {
			.@target_aid = 0;
			query_sql "SELECT `account_id` FROM `char` WHERE `name` = '" + escape_sql(.@target_name$) + "' LIMIT 1", .@target_aid;
			if (.@target_aid > 0 && !isloggedin(.@target_aid)) {
				.@target_aid = 0;
			}
		}

		if (.@target_aid == 0) {
			mes "^FF0000Erro:^000000 O personagem '" + .@target_name$ + "' nao foi encontrado online.";
			close;
		}

		mes "Escolha a acao:";
		next;

		               "2. Iniciar MODO NORMAL (Pausas Realistas / 5 min Cacada):" +
		               "3. Colocar em MODO OFFLINE / Deslogar Cliente (@at):" +
		               "4. Parar AI (Pausar Acoes):" +
		               "5. Configurar Classe Alvo:" +
		               "6. Ver Status no Chat:" +
		               "7. Teleportar ate o Bot (@jumpto):" +
		               "Cancelar");

					#BOT_AI_ACTIVE = 1;

					#BOT_WP_INDEX = 0;
					#BOT_HUNT_COUNT = 0;
					#BOT_CURRENT_TARGET = 0;
					#BOT_PRT_MAX = 0;
					#BOT_PRT_COUNT = 0;
					#BOT_LAST_TX = 0;
					#BOT_LAST_TY = 0;
					#BOT_DIR_VX = 0;
					#BOT_DIR_VY = 0;
					#BOT_EXPLORE_STEPS = 0;

					warp @hunt_map$, 0, 0;

					query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 504, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
					query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 505, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
					query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 645, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
					query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 656, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
					query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 657, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
					if (countitem(504) < 50) getitem 504, 50;
					if (countitem(505) < 50) getitem 505, 50;

					atcommand "@autoloot 100";
					atcommand "@restock 504 100";
					atcommand "@restock 505 100";
					atcommand "@restock 645 5";
					atcommand "@restock 656 5";
					atcommand "@restock 657 5";
					resetstatus;
					resetskill;
					callsub OnCheckJobChange;
					callsub OnDistributeStats;
					callsub OnDistributeSkills;
					callsub OnApplyBuffs;
					callsub OnCheckMounts;
					callsub OnCheckSpeedPotion;
					addtimer 1000, "PlayerBotAI::OnStepTimer";
				}
				attachrid(.@gm_aid);
				dispbottom "[Bot Manager]: MODO TESTE (Iniciando direto na Cacada: " + @hunt_map$ + ") ativado em '" + .@target_name$ + "'.", 0x55FF55;
				break;

			case 2: // MODO NORMAL
				if (attachrid(.@target_aid)) {
					#BOT_AI_ACTIVE = 1;
					#BOT_TEST_MODE = 0;
					#BOT_AI_STAGE = 0; // Inicia em Prontera
					#BOT_HUNT_START = 0;
					#BOT_WP_INDEX = 0;
					#BOT_HUNT_COUNT = 0;
					#BOT_CURRENT_TARGET = 0;
					#BOT_PRT_MAX = 0;
					#BOT_PRT_COUNT = 0;
					#BOT_LAST_TX = 0;
					#BOT_LAST_TY = 0;
					#BOT_DIR_VX = 0;
					#BOT_DIR_VY = 0;
					#BOT_EXPLORE_STEPS = 0;
					#BOT_WALL_HIT = 0;
					#BOT_SKILL_ROT = 0;
					#BOT_STUCK_X = 0;
					#BOT_STUCK_Y = 0;
					#BOT_STUCK_TICKS = 0;
					#BOT_WAIT_END = 0;
					stand;
					callsub OnDetectTargetJob;

					savepoint "prontera", 156, 178;
					warp "prontera", 156, 178;

					query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 504, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
					query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 505, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
					query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 645, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
					query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 656, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
					query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 657, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
					if (countitem(504) < 50) getitem 504, 50;
					if (countitem(505) < 50) getitem 505, 50;

					atcommand "@autoloot 100";
					atcommand "@restock 504 100";
					atcommand "@restock 505 100";
					atcommand "@restock 645 5";
					atcommand "@restock 656 5";
					atcommand "@restock 657 5";
					resetstatus;
					resetskill;
					callsub OnCheckJobChange;
					addtimer 1000, "PlayerBotAI::OnStepTimer";
				}
				attachrid(.@gm_aid);
				dispbottom "[Bot Manager]: MODO NORMAL iniciado com sucesso em '" + .@target_name$ + "'.", 0x55FF55;
				break;

			case 3: // MODO OFFLINE (@at)
				if (attachrid(.@target_aid)) {
					if (!#BOT_AI_ACTIVE) {
						#BOT_AI_ACTIVE = 1;
						#BOT_TEST_MODE = 0;
						#BOT_AI_STAGE = 0;
						#BOT_WP_INDEX = 0;
						#BOT_HUNT_START = 0;
						callsub OnDetectTargetJob;
						savepoint "prontera", 156, 178;
						warp "prontera", 156, 178;
						resetstatus;
						callsub OnCheckJobChange;
						addtimer 1000, "PlayerBotAI::OnStepTimer";
					}
					offlinebot;
				}
				attachrid(.@gm_aid);
				dispbottom "[Bot Manager]: MODO OFFLINE ativado em '" + .@target_name$ + "'! Cliente desconectado e bot ativo no servidor.", 0x55FF55;
				break;

			case 4: // PARAR AI
				if (attachrid(.@target_aid)) {
					#BOT_AI_ACTIVE = 0;
					#BOT_TEST_MODE = 0;
					#BOT_HUNT_START = 0;
					#BOT_CURRENT_TARGET = 0;
					#BOT_PRT_MAX = 0;
					#BOT_PRT_COUNT = 0;
					#BOT_STUCK_X = 0;
					#BOT_STUCK_Y = 0;
					#BOT_STUCK_TICKS = 0;
					deltimer "PlayerBotAI::OnStepTimer";
					unitstopwalk getcharid(3), USW_FORCE_STOP;
					unitstopattack getcharid(3);
				}
				attachrid(.@gm_aid);
				dispbottom "[Bot Manager]: AI PAUSADA no personagem '" + .@target_name$ + "'.", 0xFFAAAA;
				break;

			case 5: // CONFIGURAR CLASSE
				mes "^0055FF[Escolha da Classe Alvo]^000000";
				.@jchoice = select("Espadachim -> Cavaleiro / Lorde:" +
				                   "Mago -> Bruxo / Arquimago:" +
				                   "Arqueiro -> Cacador / Atirador de Elite:" +
				                   "Gatuno -> Mercenario / Algoz:" +
				                   "Novico -> Sacerdote / Sumo Sacerdote:" +
				                   "Mercador -> Ferreiro / Mestre-Ferreiro");
				if (attachrid(.@target_aid)) {
					#BOT_TARGET_JOB = .@jchoice;
					callsub OnCheckJobChange;
				}
				attachrid(.@gm_aid);
				dispbottom "[Bot Manager]: Classe alvo atualizada para o personagem '" + .@target_name$ + "'.", 0x55FF55;
				break;

			case 6: // STATUS
				callsub OnShowGMStatus, .@target_aid, .@target_name$;
				break;

			case 7: // TELEPORTAR ATE O BOT (@jumpto)
				if (attachrid(.@target_aid)) {
					getmapxy(.@t_map$, .@t_x, .@t_y, BL_PC);
				}
				attachrid(.@gm_aid);
				if (.@t_map$ != "") {
					warp .@t_map$, .@t_x, .@t_y;
					dispbottom "[Bot Manager]: Teleportado ate '" + .@target_name$ + "' em " + .@t_map$ + " (" + .@t_x + "," + .@t_y + ").", 0x55FF55;
				} else {
					dispbottom "[Bot Manager]: Nao foi possivel obter a localizacao do personagem.", 0xFF5555;
				}
				break;
		}
		close;
	}

	// Comandos por texto: @bot goto [nome] ou @bot jumpto [nome] (TELEPORTE DO GM)
	if (.@cmd$ == "goto" || .@cmd$ == "jumpto" || .@cmd$ == "tp" || .@cmd$ == "warpto") {
		if (.@target_aid == 0) {
			dispbottom "[Bot Manager]: Personagem '" + .@target_name$ + "' nao encontrado ou offline.", 0xFF5555;
			end;
		}
		if (attachrid(.@target_aid)) {
			getmapxy(.@t_map$, .@t_x, .@t_y, BL_PC);
		}
		attachrid(.@gm_aid);
		if (.@t_map$ != "") {
			warp .@t_map$, .@t_x, .@t_y;
			dispbottom "[Bot Manager]: Teleportado ate '" + .@target_name$ + "' em " + .@t_map$ + " (" + .@t_x + "," + .@t_y + ").", 0x55FF55;
		} else {
			dispbottom "[Bot Manager]: Nao foi possivel obter a localizacao de '" + .@target_name$ + "'.", 0xFF5555;
		}
		end;
	}

	// Comandos por texto: @bot at [nome] ou @bot offline [nome] (MODO OFFLINE)
	if (.@cmd$ == "at" || .@cmd$ == "offline" || .@cmd$ == "autotrade") {
		if (.@target_aid == 0) {
			dispbottom "[Bot Manager]: Personagem '" + .@target_name$ + "' nao encontrado ou offline.", 0xFF5555;
			end;
		}
		if (attachrid(.@target_aid)) {
			if (!#BOT_AI_ACTIVE) {
				#BOT_AI_ACTIVE = 1;
				#BOT_TEST_MODE = 0;
				#BOT_AI_STAGE = 0;
				#BOT_WP_INDEX = 0;
				#BOT_HUNT_START = 0;
				callsub OnDetectTargetJob;
				savepoint "prontera", 156, 178;
				warp "prontera", 156, 178;
				resetstatus;
				callsub OnCheckJobChange;
				addtimer 1000, "PlayerBotAI::OnStepTimer";
			}
			offlinebot;
		}
		attachrid(.@gm_aid);
		dispbottom "[Bot Manager]: MODO OFFLINE ATIVADO em '" + .@target_name$ + "'! Jogo deslogado e AI agindo no servidor.", 0x55FF55;
		end;
	}

	// Comandos por texto: @bot test [nome] (MODO TESTE - Inicia direto na Cacada)
	if (.@cmd$ == "test" || .@cmd$ == "teste") {
		if (.@target_aid == 0) {
			dispbottom "[Bot Manager]: Personagem '" + .@target_name$ + "' nao encontrado ou offline.", 0xFF5555;
			end;
		}
		if (attachrid(.@target_aid)) {
			#BOT_AI_ACTIVE = 1;
			#BOT_TEST_MODE = 1;
			#BOT_AI_STAGE = 3; // Inicia direto na Cacada
			#BOT_HUNT_START = gettimetick(2);
			#BOT_WP_INDEX = 0;
			#BOT_HUNT_COUNT = 0;
			#BOT_CURRENT_TARGET = 0;
			#BOT_PRT_MAX = 0;
			#BOT_PRT_COUNT = 0;
			#BOT_LAST_TX = 0;
			#BOT_LAST_TY = 0;
			#BOT_DIR_VX = 0;
			#BOT_DIR_VY = 0;
			#BOT_EXPLORE_STEPS = 0;
			#BOT_WALL_HIT = 0;
			#BOT_SKILL_ROT = 0;
			#BOT_STUCK_X = 0;
			#BOT_STUCK_Y = 0;
			#BOT_STUCK_TICKS = 0;
			#BOT_WAIT_END = 0;
			stand;
			callsub OnDetectTargetJob;

OnStepTimer:
	if (!#BOT_AI_ACTIVE) end;
	stand; // Bot NUNCA senta, permanece sempre em pe e ativo

	// Limpeza de inventario se pesado
	if ((Weight * 100 / MaxWeight) >= 80) {
		callsub OnCleanHeavyInventory;
	}

	// Pontos, classes e pocoes de cura
	if (StatusPoint > 0) callsub OnDistributeStats;
	if (SkillPoint > 0 || (BaseClass == Job_Swordman && BaseLevel >= 99 && getskilllv("LG_BANISHINGPOINT") < 10)) callsub OnDistributeSkills;
	callsub OnCheckJobChange;
	callsub OnCheckPotions;
	callsub OnCheckMounts;

	// =========================================================
	// =========================================================
	// ESTAGIO 0: Caminhada Organica e Dinamica em Prontera
	// Caminhada fluida com parada obrigatoria de no minimo 1 minuto em cada ponto
	// =========================================================
	if (#BOT_AI_STAGE == 0) {
		getmapxy(.@m$, .@cx, .@cy, BL_PC);
		if (.@m$ != "prontera") {
			stand;
			warp "prontera", 156, 178;
			#BOT_PRT_MAX = 0;
			#BOT_PRT_COUNT = 0;
			#BOT_WP_INDEX = 0;
			#BOT_LAST_TX = 0;
			#BOT_LAST_TY = 0;
			#BOT_STUCK_TICKS = 0;
			addtimer 1500, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Piscina ampla de mais de 60 coordenadas abertas para total variedade e liberdade de travessia
		setarray .@pool_x[0],
			// Setor Sul (Y 177 ~ 190)
			148, 152, 156, 160, 164,
			148, 152, 156, 160, 164,
			148, 152, 156, 160, 164,
			148, 154, 158, 164,
			// Setor Oeste / Diagonal Esquerda (148,177 -> 124,202 -> 121,206 -> 146,230)
			144, 140, 136, 132, 128, 124,
			121, 121, 124, 128, 134, 140,
			124, 128, 134, 140,
			124, 128, 132, 136, 140, 146,
			// Setor Norte (Y 214 ~ 230)
			146, 150, 154, 158, 162, 165,
			148, 152, 156, 160, 164,
			148, 152, 156, 160, 164,
			148, 152, 156, 160, 164,
			// Setor Leste / Diagonal Direita (165,230 -> 188,207 -> 186,200 -> 161,177)
			168, 172, 176, 180, 184, 188,
			186, 186, 182, 178, 172, 168,
			182, 178, 172, 168,
			186, 182, 178, 174, 170, 165, 161;

		setarray .@pool_y[0],
			// Setor Sul
			178, 178, 178, 178, 178,
			182, 182, 182, 182, 182,
			186, 186, 186, 186, 186,
			190, 190, 190, 190,
			// Setor Oeste
			179, 183, 187, 191, 195, 202,
			204, 208, 206, 206, 206, 206,
			210, 210, 210, 210,
			214, 218, 222, 225, 228, 230,
			// Setor Norte
			230, 230, 230, 230, 230, 230,
			226, 226, 226, 226, 226,
			222, 222, 222, 222, 222,
			216, 216, 216, 216, 216,
			// Setor Leste
			228, 225, 221, 217, 213, 207,
			204, 208, 206, 206, 206, 206,
			210, 210, 210, 210,
			200, 196, 192, 188, 184, 180, 178;
		.@pool_size = getarraysize(.@pool_x);

		// Sorteia quantos pontos visitara nesta passagem por Prontera:
		if (#BOT_PRT_MAX <= 0) {
			#BOT_PRT_MAX = rand(3, 6);
			#BOT_PRT_COUNT = 0;
			#BOT_WP_INDEX = 0;
			#BOT_LAST_TX = 0;
			#BOT_LAST_TY = 0;
			#BOT_STUCK_TICKS = 0;
		}

		// Sub-estagio 0: Sorteia o proximo ponto e inicia o passo suave
		if (#BOT_WP_INDEX == 0) {
			.@r = rand(0, .@pool_size - 1);
			#BOT_LAST_TX = .@pool_x[.@r];
			#BOT_LAST_TY = .@pool_y[.@r];
			#BOT_WP_INDEX = 1; // Transiciona para caminhando
			#BOT_STUCK_TICKS = 0;
			#BOT_STUCK_X = .@cx;
			#BOT_STUCK_Y = .@cy;
			stand; // Garante que esta de pe para andar
			unitstopwalk getcharid(3), USW_FORCE_STOP;
			unitwalk getcharid(3), #BOT_LAST_TX, #BOT_LAST_TY; // Envia o caminho suave uma unica vez
			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Sub-estagio 1: Caminhando suavemente ate o ponto de destino
		if (#BOT_WP_INDEX == 1) {
			.@dist = distance(.@cx, .@cy, #BOT_LAST_TX, #BOT_LAST_TY);

			// Chegou perto do ponto de destino (<= 2 celulas):
			if (.@dist <= 2) {
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				#BOT_PRT_COUNT++;
				#BOT_WP_INDEX = 2; // Transiciona para parado no ponto
				#BOT_STUCK_TICKS = 0;
				// Parada obrigatoria no ponto: Minimo 1 minuto (60 a 180s no modo start | 5s no teste)
				#BOT_WAIT_END = gettimetick(2) + (#BOT_TEST_MODE ? 5 : rand(60, 180));

				// Permanece em pe observando o movimento (sem sentar):
				stand;

				addtimer 1000, "PlayerBotAI::OnStepTimer";
				end;
			}

			// Anti-travamento inteligente caso colida com obstaculo por mais de 5s:
			if (#BOT_STUCK_X == .@cx && #BOT_STUCK_Y == .@cy) {
				#BOT_STUCK_TICKS++;
				if (#BOT_STUCK_TICKS >= 5) {
					// Sorteia outro ponto aberto a pe
					#BOT_WP_INDEX = 0;
					#BOT_LAST_TX = 0;
					#BOT_LAST_TY = 0;
					#BOT_STUCK_TICKS = 0;
					unitstopwalk getcharid(3), USW_FORCE_STOP;
					addtimer 500, "PlayerBotAI::OnStepTimer";
					end;
				}
			} else {
				#BOT_STUCK_TICKS = 0;
			}

			#BOT_STUCK_X = .@cx;
			#BOT_STUCK_Y = .@cy;

			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Sub-estagio 2: Parado no ponto descansando/observando (Minimo 1 minuto)
		if (#BOT_WP_INDEX == 2) {
			if (gettimetick(2) >= #BOT_WAIT_END) {
				stand; // Levanta do chao
				// Se ja visitou todos os pontos sorteados desta passagem -> Vai para o Eden
				if (#BOT_PRT_COUNT >= #BOT_PRT_MAX) {
					#BOT_AI_STAGE = 2; // Segue para o Grupo Eden
					#BOT_WP_INDEX = 0;
					#BOT_PRT_MAX = 0;
					#BOT_PRT_COUNT = 0;
					#BOT_LAST_TX = 0;
					#BOT_LAST_TY = 0;
					#BOT_STUCK_TICKS = 0;
					#BOT_WAIT_END = 0;
					unitstopwalk getcharid(3), USW_FORCE_STOP;
					warp "moc_para01", 31, 14;
					addtimer 1500, "PlayerBotAI::OnStepTimer";
					end;
				}

				// Ainda tem pontos a visitar: sorteia o proximo a partir de onde esta agora
				#BOT_WP_INDEX = 0;
				#BOT_LAST_TX = 0;
				#BOT_LAST_TY = 0;
				#BOT_STUCK_TICKS = 0;
				#BOT_WAIT_END = 0;
				addtimer 500, "PlayerBotAI::OnStepTimer";
				end;
			}

			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}
	}

	// =========================================================
	// ESTAGIO 1: Descanso Distribuido em Prontera (Minimo 1 Minuto no Modo Start)
	// =========================================================
	if (#BOT_AI_STAGE == 1) {
		getmapxy(.@m$, .@cx, .@cy, BL_PC);
		if (.@m$ != "prontera") {
			stand;
			warp "prontera", 156, 178;
			#BOT_WP_INDEX = 0;
			#BOT_LAST_TX = 0;
			#BOT_LAST_TY = 0;
			#BOT_STUCK_TICKS = 0;
			addtimer 1500, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Grade ampla de mais de 30 pontos de descanso distribuidos (todos em Y >= 177)
		setarray .@p_rest_x[0],
			156, 148, 164, 152, 160, 150, 162,
			132, 126, 138, 122, 142, 130, 136,
			156, 148, 164, 152, 160, 150, 162,
			180, 186, 174, 188, 170, 182, 176,
			146, 165, 126, 184, 138, 172, 144, 168;

		setarray .@p_rest_y[0],
			178, 178, 178, 184, 184, 188, 188,
			206, 206, 206, 208, 212, 216, 222,
			224, 224, 224, 220, 220, 216, 216,
			206, 206, 206, 208, 212, 216, 220,
			230, 230, 212, 212, 225, 222, 180, 180;
		.@p_rest_size = getarraysize(.@p_rest_x);

		// Sorteia o ponto de descanso de forma 100% uniforme:
		if (#BOT_LAST_TX == 0 || #BOT_LAST_TY == 0) {
			.@r_idx = rand(0, .@p_rest_size - 1);
			#BOT_LAST_TX = .@p_rest_x[.@r_idx];
			#BOT_LAST_TY = .@p_rest_y[.@r_idx];
			#BOT_STUCK_TICKS = 0;
		}

		// Sub-estagio 0: Caminhando ate o ponto de descanso sorteado
		if (#BOT_WP_INDEX == 0) {
			.@dist = distance(.@cx, .@cy, #BOT_LAST_TX, #BOT_LAST_TY);

			if (.@dist <= 2) {
				#BOT_WP_INDEX = 1; // Em descanso em pe
				#BOT_STUCK_TICKS = 0;
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				stand; // Permanece em pe no ponto escolhido

				if (#BOT_TEST_MODE) {
					#BOT_WAIT_END = gettimetick(2) + 5;
				} else {
					// No modo Start: Minimo de 1 minuto e Maximo de 5 minutos (60 a 300 segundos)
					#BOT_WAIT_END = gettimetick(2) + rand(60, 300);
				}
				addtimer 1000, "PlayerBotAI::OnStepTimer";
				end;
			}

			// Inicia caminhada ou anti-travamento para o ponto de descanso (100% a pe):
			if (#BOT_STUCK_X == .@cx && #BOT_STUCK_Y == .@cy) {
				#BOT_STUCK_TICKS++;
				if (#BOT_STUCK_TICKS >= 3) {
					// Se encontrou obstaculo no caminho, sorteia outro ponto aberto e continua a pe
					.@r_idx = rand(0, .@p_rest_size - 1);
					#BOT_LAST_TX = .@p_rest_x[.@r_idx];
					#BOT_LAST_TY = .@p_rest_y[.@r_idx];
					#BOT_STUCK_TICKS = 0;
					unitstopwalk getcharid(3), USW_FORCE_STOP;
					unitwalk getcharid(3), #BOT_LAST_TX, #BOT_LAST_TY;
					addtimer 1000, "PlayerBotAI::OnStepTimer";
					end;
				}
			} else {
				#BOT_STUCK_TICKS = 0;
			}

			unitstopwalk getcharid(3), USW_FORCE_STOP;
			unitwalk getcharid(3), #BOT_LAST_TX, #BOT_LAST_TY;

			#BOT_STUCK_X = .@cx;
			#BOT_STUCK_Y = .@cy;

			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Sub-estagio 1: Em pe descansando em seu ponto
		if (#BOT_WP_INDEX == 1) {
			if (gettimetick(2) >= #BOT_WAIT_END) {
				stand;
				#BOT_AI_STAGE = 2;
				#BOT_WP_INDEX = 0;
				#BOT_LAST_TX = 0;
				#BOT_LAST_TY = 0;
				#BOT_STUCK_TICKS = 0;
				#BOT_WAIT_END = 0;
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				warp "moc_para01", 31, 14;
				addtimer 1500, "PlayerBotAI::OnStepTimer";
				end;
			}
			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}
	}

	// =========================================================
	// ESTAGIO 2: Grupo Eden (Minimo 1 Minuto e Maximo 5 Minutos) - 100% a pe
	// =========================================================
	if (#BOT_AI_STAGE == 2) {
		stand;
		getmapxy(.@m$, .@cx, .@cy, BL_PC);
		if (.@m$ != "moc_para01") {
			stand;
			warp "moc_para01", 31, 14;
			#BOT_WP_INDEX = 0;
			#BOT_LAST_TX = 0;
			#BOT_LAST_TY = 0;
			#BOT_STUCK_TICKS = 0;
			addtimer 1500, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Sub-estagio 0: Chegou na entrada (31,14) -> Tempo de espera na entrada
		if (#BOT_WP_INDEX == 0) {
			#BOT_WP_INDEX = 1; // Aguardando tempo na entrada
			#BOT_LAST_TX = 0;
			#BOT_LAST_TY = 0;
			#BOT_STUCK_TICKS = 0;
			unitstopwalk getcharid(3), USW_FORCE_STOP;

			if (#BOT_TEST_MODE) {
				#BOT_WAIT_END = gettimetick(2) + 5;
			} else {
				// No modo Start: 1 a 3 minutos (60 a 180 segundos)
				#BOT_WAIT_END = gettimetick(2) + rand(60, 180);
			}
			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Sub-estagio 1: Aguardando tempo na entrada do Eden
		if (#BOT_WP_INDEX == 1) {
			if (gettimetick(2) >= #BOT_WAIT_END) {
				#BOT_WP_INDEX = 2; // Vai para uma mesa a pe
				#BOT_LAST_TX = 0;
				#BOT_LAST_TY = 0;
				#BOT_STUCK_TICKS = 0;
				#BOT_WAIT_END = 0;
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				addtimer 500, "PlayerBotAI::OnStepTimer";
				end;
			}
			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Sub-estagio 2: Caminhando da entrada (31,14) para uma mesa de descanso sorteada
		if (#BOT_WP_INDEX == 2) {
			// Grade de mesas/assentos distribuidos no Eden
			setarray .@eden_table_x[0], 28, 26, 30, 28, 26, 30, 32, 24, 22, 34, 28, 30;
			setarray .@eden_table_y[0], 26, 26, 26, 28, 28, 28, 26, 26, 28, 28, 30, 30;
			.@eden_size = getarraysize(.@eden_table_x);

			if (#BOT_LAST_TX == 0 || #BOT_LAST_TY == 0) {
				.@e_idx = rand(0, .@eden_size - 1);
				#BOT_LAST_TX = .@eden_table_x[.@e_idx];
				#BOT_LAST_TY = .@eden_table_y[.@e_idx];
				#BOT_STUCK_TICKS = 0;
			}

			.@dist = distance(.@cx, .@cy, #BOT_LAST_TX, #BOT_LAST_TY);

			// Chegou perto da mesa de descanso sorteada (<= 2 celulas)
			if (.@dist <= 2) {
				#BOT_WP_INDEX = 3; // Em descanso em pe na mesa
				#BOT_STUCK_TICKS = 0;
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				stand; // Permanece em pe na mesa sorteada

				if (#BOT_TEST_MODE) {
					#BOT_WAIT_END = gettimetick(2) + 5;
				} else {
					// No modo Start: Minimo de 1 minuto e Maximo de 5 minutos (60 a 300 segundos)
					#BOT_WAIT_END = gettimetick(2) + rand(60, 300);
				}
				addtimer 1000, "PlayerBotAI::OnStepTimer";
				end;
			}

			// Inicia caminhada continua para a mesa sorteada (100% a pe, sem teleporte):
			if (#BOT_STUCK_X == .@cx && #BOT_STUCK_Y == .@cy) {
				#BOT_STUCK_TICKS++;
				if (#BOT_STUCK_TICKS >= 4) {
					#BOT_WP_INDEX = 3;
					#BOT_LAST_TX = 0;
					#BOT_LAST_TY = 0;
					#BOT_STUCK_TICKS = 0;
					unitstopwalk getcharid(3), USW_FORCE_STOP;
					stand;
					#BOT_WAIT_END = gettimetick(2) + (#BOT_TEST_MODE ? 5 : rand(60, 300));
					addtimer 1000, "PlayerBotAI::OnStepTimer";
					end;
				}
			} else {
				#BOT_STUCK_TICKS = 0;
			}

			unitstopwalk getcharid(3), USW_FORCE_STOP;
			unitwalk getcharid(3), #BOT_LAST_TX, #BOT_LAST_TY;

			#BOT_STUCK_X = .@cx;
			#BOT_STUCK_Y = .@cy;

			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Sub-estagio 3: Descansando sentado na mesa do Eden
		if (#BOT_WP_INDEX == 3) {
			if (gettimetick(2) >= #BOT_WAIT_END) {
				stand; // Levanta da mesa
				#BOT_WP_INDEX = 4; // Vai para a saida a pe
				#BOT_LAST_TX = 0;
				#BOT_LAST_TY = 0;
				#BOT_STUCK_TICKS = 0;
				#BOT_WAIT_END = 0;
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				addtimer 800, "PlayerBotAI::OnStepTimer";
				end;
			}
			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// Sub-estagio 4: Apos descanso, caminhando de volta para a saida (31,14)
		if (#BOT_WP_INDEX == 4) {
			.@tx = 31; .@ty = 14;
			.@dist = distance(.@cx, .@cy, .@tx, .@ty);

			// Chegou perto da saida (<= 2 celulas) -> Vai para a cacada
			if (.@dist <= 2) {
				#BOT_AI_STAGE = 3;
				#BOT_WP_INDEX = 0;
				#BOT_HUNT_COUNT = 0;
				#BOT_CURRENT_TARGET = 0;
				#BOT_LAST_TX = 0;
				#BOT_LAST_TY = 0;
				#BOT_DIR_VX = 0;
				#BOT_DIR_VY = 0;
				#BOT_EXPLORE_STEPS = 0;
				#BOT_WALL_HIT = 0;
				#BOT_STUCK_X = 0;
				#BOT_STUCK_Y = 0;
				#BOT_STUCK_TICKS = 0;
				#BOT_HUNT_START = gettimetick(2);
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				callsub OnGetHuntMap;
				warp @hunt_map$, 0, 0;
				addtimer 1500, "PlayerBotAI::OnStepTimer";
				end;
			}

			// Inicia caminhada continua para 31,14 (100% a pe, sem teleporte):
			if (#BOT_LAST_TX != 31 || #BOT_LAST_TY != 14 || (#BOT_STUCK_X == .@cx && #BOT_STUCK_Y == .@cy)) {
				if (#BOT_STUCK_X == .@cx && #BOT_STUCK_Y == .@cy) {
					#BOT_STUCK_TICKS++;
					if (#BOT_STUCK_TICKS >= 4) {
						#BOT_AI_STAGE = 3;
						#BOT_WP_INDEX = 0;
						#BOT_HUNT_COUNT = 0;
						#BOT_CURRENT_TARGET = 0;
						#BOT_LAST_TX = 0;
						#BOT_LAST_TY = 0;
						#BOT_STUCK_TICKS = 0;
						#BOT_HUNT_START = gettimetick(2);
						unitstopwalk getcharid(3), USW_FORCE_STOP;
						callsub OnGetHuntMap;
						warp @hunt_map$, 0, 0;
						addtimer 1500, "PlayerBotAI::OnStepTimer";
						end;
					}
				} else {
					#BOT_STUCK_TICKS = 0;
				}
				#BOT_LAST_TX = 31;
				#BOT_LAST_TY = 14;
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				unitwalk getcharid(3), 31, 14;
			}

			#BOT_STUCK_X = .@cx;
			#BOT_STUCK_Y = .@cy;

			addtimer 1000, "PlayerBotAI::OnStepTimer";
			end;
		}
	}

	// =========================================================
	// ESTAGIO 3: Cacada Inteligente e Agressiva (Busca por getareaunits)
	// =========================================================
	if (#BOT_AI_STAGE == 3) {
		stand;
		callsub OnCheckBuffs;
		callsub OnCheckSpeedPotion;
		callsub OnCheckPotions;
		callsub OnGetHuntMap;
		getmapxy(.@m$, .@cx, .@cy, BL_PC);

		if (.@m$ != @hunt_map$) {
			stand;
			warp @hunt_map$, 0, 0;
			#BOT_CURRENT_TARGET = 0;
			#BOT_LAST_TX = 0;
			#BOT_LAST_TY = 0;
			#BOT_STUCK_TICKS = 0;
			#BOT_HUNT_START = gettimetick(2);
			addtimer 1500, "PlayerBotAI::OnStepTimer";
			end;
		}

		if (#BOT_HUNT_START == 0) {
			#BOT_HUNT_START = gettimetick(2);
		}

		// Duracao da Cacada: 1 minuto (60s) no Modo Teste | 5 minutos (300s) no Modo Normal
		.@hunt_duration = (#BOT_TEST_MODE ? 60 : 300);
		if ((gettimetick(2) - #BOT_HUNT_START) >= .@hunt_duration) {
			#BOT_HUNT_START = 0;
			#BOT_AI_STAGE = 0; // Retorna para o Estagio 0 (Passeio e descanso em Prontera)
			#BOT_WP_INDEX = 0;
			#BOT_CURRENT_TARGET = 0;
			#BOT_PRT_MAX = 0;
			#BOT_PRT_COUNT = 0;
			#BOT_LAST_TX = 0;
			#BOT_LAST_TY = 0;
			#BOT_DIR_VX = 0;
			#BOT_DIR_VY = 0;
			#BOT_EXPLORE_STEPS = 0;
			#BOT_WALL_HIT = 0;
			#BOT_STUCK_X = 0;
			#BOT_STUCK_Y = 0;
			#BOT_STUCK_TICKS = 0;
			unitstopattack getcharid(3);
			unitstopwalk getcharid(3), USW_FORCE_STOP;
			warp "prontera", 156, 178;
			addtimer 2000, "PlayerBotAI::OnStepTimer";
			end;
		}

		// 1. Verifica se o monstro alvo anterior ainda existe, esta vivo, valido e nao esta sendo atacado por outro (Anti-KS)
		if (#BOT_CURRENT_TARGET > 0) {
			if (!unitexists(#BOT_CURRENT_TARGET)) {
				#BOT_CURRENT_TARGET = 0;
				#BOT_LAST_TX = 0;
				#BOT_LAST_TY = 0;
			} else {
				getunitdata #BOT_CURRENT_TARGET, .@cur_data;
				.@cur_class = .@cur_data[UMOB_CLASS];
				.@cur_target_id = .@cur_data[UMOB_TARGETID];
				.@d = distance(.@cx, .@cy, .@cur_data[UMOB_X], .@cur_data[UMOB_Y]);

				// Se o monstro morreu, e planta/cogumelo, se afastou mais de 14 celulas ou foi engatado por outro jogador (Anti-KS):
				if (.@cur_data[UMOB_HP] <= 0 || (.@cur_class >= 1078 && .@cur_class <= 1085) || (.@cur_class >= 1930 && .@cur_class <= 1933) || .@cur_class == 1064 ||
				    .@d > 14 || (.@cur_data[UMOB_HP] < .@cur_data[UMOB_MAXHP] && .@cur_target_id > 0 && .@cur_target_id != getcharid(3))) {
					#BOT_CURRENT_TARGET = 0;
					#BOT_LAST_TX = 0;
					#BOT_LAST_TY = 0;
				} else {
					.@mx = .@cur_data[UMOB_X];
					.@my = .@cur_data[UMOB_Y];
				}
			}
		}

		// 2. Se nao tem alvo valido, procura o monstro vivo com melhor EXP (+10/+15 niveis) em um raio de 14 celulas
		if (#BOT_CURRENT_TARGET == 0) {
			.@num_mobs = getareaunits(BL_MOB, strcharinfo(3), .@cx - 14, .@cy - 14, .@cx + 14, .@cy + 14, .@mob_list[0]);
			.@best_score = -999999;
			.@best_mob = 0;

			for (.@i = 0; .@i < .@num_mobs; .@i++) {
				if (unitexists(.@mob_list[.@i])) {
					getunitdata .@mob_list[.@i], .@t_data;
					.@m_class = .@t_data[UMOB_CLASS];
					.@m_lvl = .@t_data[UMOB_LEVEL];

					// Ignora Plantas e Cogumelos
					if ((.@m_class >= 1078 && .@m_class <= 1085) || (.@m_class >= 1930 && .@m_class <= 1933) || .@m_class == 1064) {
						continue;
					}

					// Anti-KS: Se o monstro ja estiver com HP reduzido e lutando contra outro jogador, ignora!
					.@m_target = .@t_data[UMOB_TARGETID];
					if (.@t_data[UMOB_HP] < .@t_data[UMOB_MAXHP] && .@m_target > 0 && .@m_target != getcharid(3)) {
						continue;
					}

					if (.@t_data[UMOB_HP] > 0) {
						.@d = distance(.@cx, .@cy, .@t_data[UMOB_X], .@t_data[UMOB_Y]);
						if (.@d <= 14) {
							// Pontuacao: prioriza monstros com cerca de 10 a 20 niveis a mais para ganho acelerado de EXP
							.@lvl_diff = .@m_lvl - BaseLevel;
							.@score = (100 - .@d * 4);
							if (.@lvl_diff >= 8 && .@lvl_diff <= 25) {
								.@score += 60; // Maxima prioridade para monstros de nivel ideal (+10 a +15 niveis)
							} else if (.@lvl_diff > 0) {
								.@score += 30;
							}
							if (.@score > .@best_score) {
								.@best_score = .@score;
								.@best_mob = .@mob_list[.@i];
								.@mx = .@t_data[UMOB_X];
								.@my = .@t_data[UMOB_Y];
							}
						}
					}
				}
			}

			if (.@best_mob > 0) {
				#BOT_CURRENT_TARGET = .@best_mob;
			}
		}

		// 3. Regra de Teleporte: Se NAO houver nenhum inimigo valido em 14 celulas -> Teleporta respeitando delay de 5 segundos!
		if (#BOT_CURRENT_TARGET == 0) {
			if ((gettimetick(2) - #BOT_LAST_TELEPORT) >= 5) {
				#BOT_LAST_TELEPORT = gettimetick(2);
				#BOT_LAST_TX = 0;
				#BOT_LAST_TY = 0;
				#BOT_STUCK_TICKS = 0;
				unitstopattack getcharid(3);
				unitstopwalk getcharid(3), USW_FORCE_STOP;
				warp @hunt_map$, 0, 0;
				addtimer 1000, "PlayerBotAI::OnStepTimer";
				end;
			} else {
				// Enquanto aguarda o delay de 5 segundos de teleporte, caminha e explora ao redor
				if (#BOT_LAST_TX == 0 || #BOT_LAST_TY == 0 || (#BOT_STUCK_X == .@cx && #BOT_STUCK_Y == .@cy)) {
					#BOT_LAST_TX = .@cx + rand(-8, 8);
					#BOT_LAST_TY = .@cy + rand(-8, 8);
					unitstopwalk getcharid(3), USW_FORCE_STOP;
					unitwalk getcharid(3), #BOT_LAST_TX, #BOT_LAST_TY;
				}
				#BOT_STUCK_X = .@cx;
				#BOT_STUCK_Y = .@cy;
				addtimer 600, "PlayerBotAI::OnStepTimer";
				end;
			}
		}

		// 4. Se encontrou um monstro por perto:
		if (#BOT_CURRENT_TARGET > 0) {
			.@dist = distance(.@cx, .@cy, .@mx, .@my);
			.@atk_range = 2; // Padrao corpo a corpo
			if (Class == Job_Archer || Class == Job_Archer_High || Class == Job_Hunter || Class == Job_Sniper || Class == Job_Ranger ||
			    Class == Job_Mage || Class == Job_Mage_High || Class == Job_Wizard || Class == Job_High_Wizard || Class == Job_Warlock ||
			    Class == Job_Mechanic || Class == Job_Mechanic_T || Class == Job_Mechanic2 || Class == Job_Mechanic_T2 || Class >= Job_Mechanic ||
			    Class == Job_Genetic || Class == Job_Genetic_T || Class == Job_Genetic2 || Class == Job_Genetic_T2 ||
			    getskilllv("NC_ARMSCANNON") > 0 || getskilllv("GN_CARTCANNON") > 0 ||
			    getskilllv("SN_SHARPSHOOTING") > 0 || getskilllv("RA_AIMEDBOLT") > 0 || getskilllv("RA_ARROWSTORM") > 0) {
				.@atk_range = 8; // Ranged (Ataca de longe a 8-9 celulas sem se aproximar)
			}

			// Se ainda esta fora do alcance de ataque: anda em direcao ao monstro de forma fluida
			if (.@dist > .@atk_range) {
				// Sistema Anti-Travamento (Tolerancia de 8 ticks / aprox. 4 segundos antes de resetar)
				if (#BOT_STUCK_X == .@cx && #BOT_STUCK_Y == .@cy) {
					#BOT_STUCK_TICKS++;
					if (#BOT_STUCK_TICKS >= 8) {
						#BOT_CURRENT_TARGET = 0;
						#BOT_LAST_TX = 0;
						#BOT_LAST_TY = 0;
						#BOT_STUCK_TICKS = 0;
						if ((gettimetick(2) - #BOT_LAST_TELEPORT) >= 5) {
							#BOT_LAST_TELEPORT = gettimetick(2);
							unitstopwalk getcharid(3), USW_FORCE_STOP;
							warp @hunt_map$, 0, 0;
							addtimer 1000, "PlayerBotAI::OnStepTimer";
							end;
						}
					}
				} else {
					#BOT_STUCK_TICKS = 0;
				}

				// Envia o comando de andar apenas se o monstro mudou de posicao:
				if (#BOT_LAST_TX != .@mx || #BOT_LAST_TY != .@my) {
					#BOT_LAST_TX = .@mx;
					#BOT_LAST_TY = .@my;
					unitstopwalk getcharid(3), USW_FORCE_STOP;
					unitwalk getcharid(3), .@mx, .@my;
				}

				#BOT_STUCK_X = .@cx;
				#BOT_STUCK_Y = .@cy;

				addtimer 400, "PlayerBotAI::OnStepTimer";
				end;
			}

			// Alcancou o monstro! Para de andar suavemente
			if (#BOT_LAST_TX != 0 || #BOT_LAST_TY != 0) {
				#BOT_LAST_TX = 0;
				#BOT_LAST_TY = 0;
				unitstopwalk getcharid(3), USW_FORCE_STOP;
			}
			#BOT_STUCK_TICKS = 0;
			#BOT_STUCK_X = .@cx;
			#BOT_STUCK_Y = .@cy;

			// Garante SP cheio antes de executar as habilidades!
			callsub OnCheckPotions;

			// Engata ataque basico continuo
			unitattack getcharid(3), #BOT_CURRENT_TARGET, 1;
			#BOT_SKILL_ROT++;

			// Calcula o atraso de ataque e pós-conjuração baseado na ASPD real do personagem
			.@cur_aspd = readparam(bAspd);
			if (.@cur_aspd < 100) .@cur_aspd = 100;
			if (.@cur_aspd > 195) .@cur_aspd = 195;
			.@aspd_delay = (200 - .@cur_aspd) * 20; // 159 ASPD = 820ms, 175 ASPD = 500ms, 185 ASPD = 300ms
			.@skill_delay = .@aspd_delay;

			// =========================================================
			// 1. ESPADACHIM / CAVALEIRO / GUARDIAO REAL
			// =========================================================
			if (BaseClass == Job_Swordman) {
				// Guardiao Real (3rd Class):
				if (getskilllv("LG_BANISHINGPOINT") > 0 || getskilllv("LG_PINPOINTATTACK") > 0 || getskilllv("LG_SHIELDPRESS") > 0 || getskilllv("LG_RAYOFGENESIS") > 0 || getskilllv("LG_CANNONSPEAR") > 0 || getskilllv("LG_OVERBRAND") > 0 || Class >= Job_Royal_Guard) {
					.@rg_step = #BOT_SKILL_ROT % 5;

					// Alterna Golpe Fulminante com as habilidades de 3ª classe:
					if (.@rg_step == 0) {
						// 1. Golpe Fulminante (SM_BASH)
						unitskilluseid getcharid(3), "SM_BASH", 10, #BOT_CURRENT_TARGET, 0, 0, 0, 1;
						.@skill_delay = (.@aspd_delay > 400 ? .@aspd_delay : 400);
					} else if (.@rg_step == 1 && getskilllv("LG_BANISHINGPOINT") > 0 && Sp >= 20) {
						// 2. Toque do Oblívio (LG_BANISHINGPOINT)
						unitskilluseid getcharid(3), "LG_BANISHINGPOINT", (getskilllv("LG_BANISHINGPOINT") > 0 ? getskilllv("LG_BANISHINGPOINT") : 10), #BOT_CURRENT_TARGET, 0, 0, 0, 1;
						.@skill_delay = (.@aspd_delay > 350 ? .@aspd_delay : 350);
					} else if (.@rg_step == 2 && getskilllv("LG_SHIELDPRESS") > 0 && Sp >= 15) {
						// 3. Escudo Compressor (LG_SHIELDPRESS)
						unitskilluseid getcharid(3), "LG_SHIELDPRESS", (getskilllv("LG_SHIELDPRESS") > 0 ? getskilllv("LG_SHIELDPRESS") : 10), #BOT_CURRENT_TARGET, 0, 0, 0, 1;
						.@skill_delay = 700;
					} else if (.@rg_step == 3 && getskilllv("LG_PINPOINTATTACK") > 0 && Sp >= 50 && (gettimetick(2) - #BOT_LAST_PINPOINT) >= 5) {
						// 4. Lança do Destino (LG_PINPOINTATTACK)
						#BOT_LAST_PINPOINT = gettimetick(2);
						unitskilluseid getcharid(3), "LG_PINPOINTATTACK", (getskilllv("LG_PINPOINTATTACK") > 0 ? getskilllv("LG_PINPOINTATTACK") : 5), #BOT_CURRENT_TARGET, 0, 0, 0, 1;
						.@skill_delay = 800;
					} else if (.@rg_step == 4 && getskilllv("LG_RAYOFGENESIS") > 0 && Sp >= 55 && (gettimetick(2) - #BOT_LAST_GENESIS) >= 3 && (getstatus(SC_INSPIRATION) || getstatus(SC_BANDING))) {
						// 5. Luz da Criação (LG_RAYOFGENESIS)
						#BOT_LAST_GENESIS = gettimetick(2);
						unitskilluseid getcharid(3), "LG_RAYOFGENESIS", (getskilllv("LG_RAYOFGENESIS") > 0 ? getskilllv("LG_RAYOFGENESIS") : 10), getcharid(3), 0, 0, 0, 1;
						.@skill_delay = 1000;
					} else {
						// Fallback imediato: Golpe Fulminante ou Toque do Oblívio
						if (getskilllv("LG_BANISHINGPOINT") > 0 && Sp >= 20) {
							unitskilluseid getcharid(3), "LG_BANISHINGPOINT", (getskilllv("LG_BANISHINGPOINT") > 0 ? getskilllv("LG_BANISHINGPOINT") : 10), #BOT_CURRENT_TARGET, 0, 0, 0, 1;
							.@skill_delay = (.@aspd_delay > 350 ? .@aspd_delay : 350);
						} else {
							unitskilluseid getcharid(3), "SM_BASH", 10, #BOT_CURRENT_TARGET, 0, 0, 0, 1;
							.@skill_delay = (.@aspd_delay > 400 ? .@aspd_delay : 400);
						}
					}

		} else {
			.@pot_id = 656;
		}
	} else if (BaseLevel >= 40 && Class != Job_Novice && Class != Job_Novice_High) {
		.@pot_id = 656;
	}

		delitem .@pot_id, 1;
		if (.@pot_id == 657) sc_start SC_ASPDPOTION2, 1800000, 0;
		else if (.@pot_id == 656) sc_start SC_ASPDPOTION1, 1800000, 0;

		atcommand "@restock";
		if (countitem(.@pot_id) == 0) {
			getitem .@pot_id, 5;
		}
		if (countitem(.@pot_id) > 0) {
			delitem .@pot_id, 1;
			if (.@pot_id == 657) sc_start SC_ASPDPOTION2, 1800000, 0;
			else if (.@pot_id == 656) sc_start SC_ASPDPOTION1, 1800000, 0;

OnCheckPotions:
	// HP < 85% -> Pocao Branca (504) em loop rapido ate atingir vida segura
	.@hp_loops = 0;
	while ((Hp * 100 / MaxHp) < 85 && .@hp_loops < 10) {
		.@hp_loops++;
		if (countitem(504) == 0) {
			atcommand "@restock 504 100";
			atcommand "@restock";
			if (countitem(504) == 0) getitem 504, 50;
		}
		if (countitem(504) > 0) {
			delitem 504, 1;
			itemheal rand(325, 405), 0;
			specialeffect EF_POTION_WHITE;
		} else {
			percentheal 30, 0;
			break;
		}
	}

	// SP < 80% -> Pocao Azul (505) em loop rapido ate restaurar
	.@sp_loops = 0;
	while ((Sp * 100 / MaxSp) < 80 && .@sp_loops < 10) {
		.@sp_loops++;
		if (countitem(505) == 0) {
			atcommand "@restock 505 100";
			atcommand "@restock";
			if (countitem(505) == 0) getitem 505, 50;
		}
		if (countitem(505) > 0) {
			delitem 505, 1;
			itemheal 0, rand(40, 65);
			specialeffect EF_POTION_BLUE;
		} else {
			percentheal 0, 50;
			break;
		}
	}
	return;

OnDetectTargetJob:
	if (BaseClass == Job_Swordman) {
		#BOT_TARGET_JOB = 1;
	} else if (BaseClass == Job_Mage) {
		#BOT_TARGET_JOB = 2;
	} else if (BaseClass == Job_Archer) {
		#BOT_TARGET_JOB = 3;
	} else if (BaseClass == Job_Thief) {
		#BOT_TARGET_JOB = 4;
	} else if (BaseClass == Job_Acolyte) {
		#BOT_TARGET_JOB = 5;
	} else if (BaseClass == Job_Merchant) {

OnDistributeStats:
	callsub OnDetectTargetJob;
	while (StatusPoint > 0) {
		.@old_points = StatusPoint;
		switch(#BOT_TARGET_JOB) {
			case 1: // Espadachim (STR > AGI > VIT > DEX)
				if (readparam(bStr) < 90 && StatusPoint >= 1) statusup bStr;
				else if (readparam(bAgi) < 80 && StatusPoint >= 1) statusup bAgi;
				else if (readparam(bDex) < 50 && StatusPoint >= 1) statusup bDex;
				else if (readparam(bVit) < 60 && StatusPoint >= 1) statusup bVit;
				else if (readparam(bStr) < 120 && StatusPoint >= 1) statusup bStr;
				else statusup bAgi;
				break;
			case 2: // Mago (INT > DEX > VIT) - Zero STR!
				if (readparam(bInt) < 99 && StatusPoint >= 1) statusup bInt;
				else if (readparam(bDex) < 90 && StatusPoint >= 1) statusup bDex;
				else if (readparam(bVit) < 60 && StatusPoint >= 1) statusup bVit;
				else if (readparam(bInt) < 120 && StatusPoint >= 1) statusup bInt;
				else statusup bDex;
				break;
			case 3: // Arqueiro (DEX > AGI > INT > VIT) - Zero STR!
				if (readparam(bDex) < 99 && StatusPoint >= 1) statusup bDex;
				else if (readparam(bAgi) < 90 && StatusPoint >= 1) statusup bAgi;
				else if (readparam(bVit) < 50 && StatusPoint >= 1) statusup bVit;
				else if (readparam(bInt) < 40 && StatusPoint >= 1) statusup bInt;
				else if (readparam(bDex) < 120 && StatusPoint >= 1) statusup bDex;
				else statusup bAgi;
				break;
			case 4: // Gatuno (AGI > STR > DEX > VIT)
				if (readparam(bAgi) < 90 && StatusPoint >= 1) statusup bAgi;
				else if (readparam(bStr) < 80 && StatusPoint >= 1) statusup bStr;
				else if (readparam(bDex) < 50 && StatusPoint >= 1) statusup bDex;
				else if (readparam(bVit) < 50 && StatusPoint >= 1) statusup bVit;
				else if (readparam(bStr) < 120 && StatusPoint >= 1) statusup bStr;
				else statusup bAgi;
				break;
			case 5: // Novico / Sacerdote (INT > VIT > DEX) vs Monge / Shura (STR > AGI > DEX > VIT > INT)
				if (Class == Job_Monk || Class == Job_Champion || Class >= Job_Sura || getskilllv("MO_TRIPLEATTACK") > 0 || getskilllv("MO_INVESTIGATE") > 0) {
					if (readparam(bStr) < 90 && StatusPoint >= 1) statusup bStr;
					else if (readparam(bAgi) < 80 && StatusPoint >= 1) statusup bAgi;
					else if (readparam(bDex) < 60 && StatusPoint >= 1) statusup bDex;
					else if (readparam(bVit) < 50 && StatusPoint >= 1) statusup bVit;
					else if (readparam(bStr) < 120 && StatusPoint >= 1) statusup bStr;
					else if (readparam(bInt) < 40 && StatusPoint >= 1) statusup bInt;
					else statusup bAgi;
				} else {
					if (readparam(bInt) < 90 && StatusPoint >= 1) statusup bInt;
					else if (readparam(bVit) < 70 && StatusPoint >= 1) statusup bVit;
					else if (readparam(bDex) < 60 && StatusPoint >= 1) statusup bDex;
					else if (readparam(bInt) < 120 && StatusPoint >= 1) statusup bInt;
					else statusup bVit;
				}
				break;
			default: // Mercador (STR > AGI > DEX > VIT)
				if (readparam(bStr) < 90 && StatusPoint >= 1) statusup bStr;
				else if (readparam(bAgi) < 80 && StatusPoint >= 1) statusup bAgi;
				else if (readparam(bDex) < 50 && StatusPoint >= 1) statusup bDex;
				else if (readparam(bVit) < 60 && StatusPoint >= 1) statusup bVit;
				else if (readparam(bStr) < 120 && StatusPoint >= 1) statusup bStr;
				else statusup bAgi;
				break;
		}
		if (StatusPoint == .@old_points) break;
	}
	return;

		skill "SR_FLASHCOMBO", 5, 1;
	}

	// =========================================================
	// 6. MERCADOR / FERREIRO / MESTRE-FERREIRO / MECANICO
	// =========================================================
	// 1a Classe (Mercador) - Apenas skills de 1a classe!
	if (Class == Job_Merchant || Class == Job_Merchant_High || Class == Job_Blacksmith || Class == Job_Whitesmith || Class == Job_Mechanic) {
		skill "MC_INCCARRY", 10, 1;
		skill "MC_DISCOUNT", 10, 1;
		skill "MC_OVERCHARGE", 10, 1;
		skill "MC_PUSHCART", 10, 1;
		skill "MC_MAMMONITE", 10, 1;
		skill "MC_CARTREVOLUTION", 1, 1;
		skill "MC_CHANGECART", 1, 1;
		skill "MC_CARTDECORATE", 1, 1;
		skill "MC_LOUD", 1, 1;
		skill "MC_IDENTIFY", 1, 1;
		skill "MC_VENDING", 10, 1;
		if (!checkcart()) setcart;
	}
	// 2a Classe (Ferreiro)
	if (Class == Job_Blacksmith || Class == Job_Whitesmith || Class == Job_Mechanic) {
		skill "BS_HILTBINDING", 1, 1;
		skill "BS_WEAPONRESEARCH", 10, 1;
		skill "BS_REPAIRWEAPON", 1, 1;

	if (Class == Job_Whitesmith || Class == Job_Mechanic) {

OnCheckJobChange:
	if (Class == Job_Novice || Class == Job_Novice_High) {
		switch(#BOT_TARGET_JOB) {
			case 1: 
				jobchange (Class == Job_Novice_High ? Job_Swordman_High : Job_Swordman);
				if (countitem(1124) == 0 && countitem(1104) == 0) getitem 1124, 1;
				break;
			case 2: 
				jobchange (Class == Job_Novice_High ? Job_Mage_High : Job_Mage);
				if (countitem(1601) == 0) getitem 1601, 1;
				break;
			case 3: 
				jobchange (Class == Job_Novice_High ? Job_Archer_High : Job_Archer);
				if (countitem(1704) == 0) getitem 1704, 1;
				if (countitem(1750) < 500) getitem 1750, 2000;
				atcommand "@restock 1750 1000";
				query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 1750, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
				break;
			case 4: 
				jobchange (Class == Job_Novice_High ? Job_Thief_High : Job_Thief);
				if (countitem(1207) == 0) getitem 1207, 1;
				break;
			case 5: 
				jobchange (Class == Job_Novice_High ? Job_Acolyte_High : Job_Acolyte);
				if (countitem(1504) == 0) getitem 1504, 1;
				break;
			default: 
				jobchange (Class == Job_Novice_High ? Job_Merchant_High : Job_Merchant);
				if (countitem(1304) == 0) getitem 1304, 1;
				equip 1304;
				if (Zeny < 5000000) Zeny += 10000000;
				break;
		}
		specialeffect EF_LEVELUP;
		resetskill;
		callsub OnDistributeSkills;
		callsub OnDistributeStats;
		return;
	}

	if (JobLevel >= 50) {
		if (Class == Job_Swordman) jobchange Job_Knight;
		else if (Class == Job_Swordman_High) jobchange Job_Lord_Knight;
		else if (Class == Job_Mage) jobchange Job_Wizard;
		else if (Class == Job_Mage_High) jobchange Job_High_Wizard;
		else if (Class == Job_Archer) jobchange Job_Hunter;
		else if (Class == Job_Archer_High) jobchange Job_Sniper;
		else if (Class == Job_Thief) jobchange Job_Assassin;
		else if (Class == Job_Thief_High) jobchange Job_Assassin_Cross;
		else if (Class == Job_Acolyte) jobchange Job_Priest;
		else if (Class == Job_Acolyte_High) jobchange Job_High_Priest;
		else if (Class == Job_Merchant) jobchange Job_Blacksmith;
		else if (Class == Job_Merchant_High) jobchange Job_Whitesmith;
		else return;

		specialeffect EF_ANGEL;
		resetskill;
		callsub OnDistributeSkills;
		callsub OnDistributeStats;
		return;
	}

	if (BaseLevel >= 99 && JobLevel >= 50 && Upper == 0) {
		jobchange Job_Novice_High;
		resetlvl 1;
		resetskill;
		specialeffect EF_ANGEL;
		#BOT_AI_STAGE = 0;
		#BOT_WP_INDEX = 0;
		callsub OnCheckJobChange;
		return;
	}

	if (BaseLevel >= 99 && JobLevel >= 70 && Upper == 1) {
		if (Class == Job_Lord_Knight) jobchange Job_Rune_Knight;
		else if (Class == Job_Paladin) jobchange Job_Royal_Guard;
		else if (Class == Job_High_Wizard) jobchange Job_Warlock;
		else if (Class == Job_Professor) jobchange Job_Sorcerer;
		else if (Class == Job_Sniper) jobchange Job_Ranger;
		else if (Class == Job_Assassin_Cross) jobchange Job_Guillotine_Cross;
		else if (Class == Job_Stalker) jobchange Job_Shadow_Chaser;
		else if (Class == Job_High_Priest) jobchange Job_Arch_Bishop;
		else if (Class == Job_Champion) jobchange Job_Sura;
		else if (Class == Job_Whitesmith) {
			jobchange Job_Mechanic;
			if (countitem(18000) < 500) getitem 18000, 2000;
			equip 18000;
			atcommand "@restock 18000 1000";
			query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 18000, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
		}
		else if (Class == Job_Creator) {
			jobchange Job_Genetic;
			if (countitem(18000) < 500) getitem 18000, 2000;
			equip 18000;
			atcommand "@restock 18000 1000";
			query_sql "INSERT INTO `storage` (`account_id`, `nameid`, `amount`, `identify`) VALUES (" + getcharid(3) + ", 18000, 30000, 1) ON DUPLICATE KEY UPDATE `amount` = 30000;";
		}
		else return;

		specialeffect EF_ANGEL;
		resetskill;
		callsub OnDistributeSkills;
		callsub OnDistributeStats;
		return;
	}
	return;
