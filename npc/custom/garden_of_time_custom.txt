//===== Custom Script ========================================
//= Garden of Time custom exchange hub
//===== Description: =========================================
//= Single NPC for Garden of Time barter, cloak reform and
//= Constellation meteorite material exchange.
//============================================================

moc_para01,48,22,4	script	Jardim do Tempo#custom	624,{
	mes "[Jardim do Tempo]";
	mes "Posso negociar equipamentos do Jardim do Tempo e materiais da Torre da Constelacao.";
	next;

L_Menu:
	switch (select("Comprar acessorios:Comprar capas:Reformar capas:Trocar meteoritos:Encantar Nebula:Encantar Signet Star:Encantar Vivatus:Evoluir Adulter para Vivatus:Cancelar")) {
	case 1:
		callshop("barter_t_garden_4");
		end;
	case 2:
		callshop("barter_t_garden_5");
		end;
	case 3:
		callshop("barter_t_garden_6");
		end;
	case 4:
		callsub S_Meteorite;
		end;
	case 5:
		callsub S_EnchantNebula;
		end;
	case 6:
		callsub S_EnchantSignetStar;
		end;
	case 7:
		callsub S_EnchantVivatus;
		end;
	case 8:
		callsub S_ReformVivatus;
		end;
	default:
		close;
	}

S_ReformVivatus:
	mes "[Jardim do Tempo]";
	mes "O Ritual de Purificacao transforma uma arma Adulter Fides em sua versao Vivatus Fides.";
	mes " ";
	mes "Requisitos:";
	mes "- Arma Adulter Fides ^0055ff+9 ou superior^000000.";
	mes "- 10 projetos especificos da arma.";
	mes "- 50 Oleos Sagrados de Purificacao.";
	mes "- 50 Aguas Sagradas de Purificacao.";
	mes "- 50 Mortalhas do Santuario.";
	mes "- 200 Fragmentos de Ametista.";
	mes " ";
	mes "^ff0000A arma resultante perdera 5 niveis de refino.^000000";
	next;
	if (select("Abrir Ritual de Purificacao:Voltar") == 2)
		goto L_Menu;
	disable_items;
	close2;
	item_reform("Consecrate_Weapon_1");
	end;

S_EnchantVivatus:
	mes "[Jardim do Tempo]";
	mes "Este servico aplica encantamentos nos slots das armas Vivatus Fides.";
	mes "Requisitos e custos:";
	mes "- Arma refinada em ^0055ff+7 ou superior^000000.";
	mes "- Fragmentos de Ametista e Zeny.";
	mes "- Encantamento e reset com 100% de sucesso.";
	mes "- Os dois slots possuem encantamento perfeito.";
	mes "- O segundo recebe os bonus finais de skills do Gray Wolf.";
	mes " ";
	mes "^777777As opcoes aleatorias das pedras Pontifex sao um sistema separado.^000000";
	next;
	if (select("Abrir encantamento de slots:Voltar") == 2)
		goto L_Menu;
	close2;
	item_enchant(164);
	end;

S_EnchantNebula:
	mes "[Jardim do Tempo]";
	mes "Escolha a armadura Nebula que deseja encantar.";
	next;

	.@choice = select("Power:Stamina:Concentration:Creative:Spell:Wisdom:Voltar");
	if (.@choice == 7)
		goto L_Menu;

	setarray .@enchant_ids[0],7,8,9,10,11,12;
	setarray .@item_ids[0],450169,450170,450171,450172,450173,450174;
	setarray .@dust_ids[0],1000398,1000399,1000400,1000401,1000402,1000403;
	setarray .@fragment_ids[0],1000442,1000443,1000444,1000445,1000446,1000447;
	.@idx = .@choice - 1;

	mes "[Jardim do Tempo]";
	mes "Este encantamento aceita:";
	mes "- "+mesitemlink(.@item_ids[.@idx], false)+" ^777777+9 ou superior^000000";
	mes " ";
	mes "Custos principais:";
	mes "- Slot 3: 50.000z + "+mesitemlink(1000372, false)+" x5";
	mes "- Slot 2: 100.000z + "+mesitemlink(.@dust_ids[.@idx], false)+" x5";
	mes "- Slot 1: 1.000.000z + "+mesitemlink(.@fragment_ids[.@idx], false)+" x5";
	mes " ";
	mes "^ff0000O equipamento precisa estar no inventario, nao equipado.^000000";
	next;

	getinventorylist;
	.@found = 0;
	.@found_low_refine = 0;
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_id[.@i] != .@item_ids[.@idx])
			continue;
		if (@inventorylist_refine[.@i] < 9) {
			.@found_low_refine = 1;
			continue;
		}
		.@found = 1;
		break;
	}

	if (!.@found) {
		mes "[Jardim do Tempo]";
		if (.@found_low_refine)
			mes "Encontrei "+mesitemlink(.@item_ids[.@idx], false)+", mas precisa estar +9 ou superior.";
		else
			mes "Voce nao possui "+mesitemlink(.@item_ids[.@idx], false)+" no inventario.";
		close;
	}

	item_enchant(.@enchant_ids[.@idx]);
	end;

S_EnchantSignetStar:
	mes "[Jardim do Tempo]";
	mes "Este encantamento aceita os acessorios Signet of Star:";
	mes "- "+mesitemlink(490132, false);
	mes "- "+mesitemlink(490133, false);
	mes "- "+mesitemlink(490134, false);
	mes "- "+mesitemlink(490135, false);
	mes "- "+mesitemlink(490136, false);
	mes "- "+mesitemlink(490137, false);
	mes " ";
	mes "Custos principais:";
	mes "- Slot 3: 300.000z + "+mesitemlink(1000373, false)+" x5";
	mes "- Slot 2: 500.000z + "+mesitemlink(1000373, false)+" x10";
	mes " ";
	mes "^ff0000O equipamento precisa estar no inventario, nao equipado.^000000";
	next;

	getinventorylist;
	.@found = 0;
	setarray .@signet_ids[0],490132,490133,490134,490135,490136,490137;
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (inarray(.@signet_ids[0], @inventorylist_id[.@i]) == -1)
			continue;
		.@found = 1;
		break;
	}

	if (!.@found) {
		mes "[Jardim do Tempo]";
		mes "Voce nao possui nenhum acessorio Signet of Star aceito no inventario.";
		close;
	}

	item_enchant(13);
	end;

S_Meteorite:
	.@need_item = 0;
	.@need_amount = 0;
	.@need_item2 = 0;
	.@need_amount2 = 0;
	.@need_item3 = 0;
	.@need_amount3 = 0;
	.@need_zeny = 0;
	.@give_item = 0;
	.@give_amount = 0;

	mes "[Jardim do Tempo]";
	mes "Escolha a troca desejada.";
	next;

	switch (select("Po de Meteorito:Fragmento de Meteorito:Verdade da Circulacao:Voltar")) {
	case 1:
		.@exchange_type = select("Po de Meteorito x5 por fragmento do mesmo atributo:Po de Meteorito x5 por outro po de atributo:Voltar");
		if (.@exchange_type == 3)
			goto L_Menu;
		.@from_choice = select("Usar Po de Meteorito do Poder:Usar Po de Meteorito do Vigor:Usar Po de Meteorito da Agilidade:Usar Po de Meteorito da Sorte:Usar Po de Meteorito da Magia:Usar Po de Meteorito da Sabedoria:Voltar");
		if (.@from_choice == 7)
			goto L_Menu;
		.@need_item = 1000397 + .@from_choice;
		if (.@exchange_type == 1) {
			.@need_amount = 5;
			.@give_item = 1000441 + .@from_choice;
		} else {
			.@dust_choice = select("Po de Meteorito do Poder:Po de Meteorito do Vigor:Po de Meteorito da Agilidade:Po de Meteorito da Sorte:Po de Meteorito da Magia:Po de Meteorito da Sabedoria:Voltar");
			if (.@dust_choice == 7)
				goto L_Menu;
			.@need_amount = 5;
			.@give_item = 1000397 + .@dust_choice;
		}
		.@give_amount = 1;
		break;
	case 2:
		.@from_choice = select("Usar Fragmento de Meteorito:Usar Fragmento de Meteorito do Poder:Usar Fragmento de Meteorito do Vigor:Usar Fragmento de Meteorito da Agilidade:Usar Fragmento de Meteorito da Sorte:Usar Fragmento de Meteorito da Magia:Usar Fragmento de Meteorito da Sabedoria:Voltar");
		if (.@from_choice == 8)
			goto L_Menu;
		.@fragment_choice = select("Fragmento de Meteorito do Poder:Fragmento de Meteorito do Vigor:Fragmento de Meteorito da Agilidade:Fragmento de Meteorito da Sorte:Fragmento de Meteorito da Magia:Fragmento de Meteorito da Sabedoria:Voltar");
		if (.@fragment_choice == 7)
			goto L_Menu;
		if (.@from_choice == 1)
			.@need_item = 1000373;
		else
			.@need_item = 1000440 + .@from_choice;
		.@need_amount = 5;
		.@give_item = 1000441 + .@fragment_choice;
		.@give_amount = 1;
		break;
	case 3:
		.@need_item = 1000373;
		.@need_amount = 50;
		.@need_item2 = 1000402;
		.@need_amount2 = 20;
		.@need_item3 = 1000445;
		.@need_amount3 = 10;
		.@need_zeny = 5000000;
		.@give_item = 102514;
		.@give_amount = 1;
		break;
	default:
		goto L_Menu;
	}

	if (.@need_item <= 0 || .@need_amount <= 0 || .@give_item <= 0 || .@give_amount <= 0) {
		mes "[Jardim do Tempo]";
		mes "Esta troca esta invalida. Tente novamente.";
		close;
	}

	mes "[Jardim do Tempo]";
	mes "Voce recebera:";
	mes "^0000ff"+getitemname(.@give_item)+" x"+.@give_amount+"^000000";
	mes " ";
	mes "Custo:";
	mes getitemname(.@need_item)+" x"+.@need_amount;
	if (.@need_item2)
		mes getitemname(.@need_item2)+" x"+.@need_amount2;
	if (.@need_item3)
		mes getitemname(.@need_item3)+" x"+.@need_amount3;
	if (.@need_zeny)
		mes "Zeny: "+.@need_zeny;
	next;

	if (select("Confirmar:Cancelar") == 2)
		close;

	if (countitem(.@need_item) < .@need_amount) {
		mes "[Jardim do Tempo]";
		mes "Voce nao possui "+getitemname(.@need_item)+" suficiente.";
		close;
	}
	if (.@need_item2) {
		if (countitem(.@need_item2) < .@need_amount2) {
			mes "[Jardim do Tempo]";
			mes "Voce nao possui "+getitemname(.@need_item2)+" suficiente.";
			close;
		}
	}
	if (.@need_item3) {
		if (countitem(.@need_item3) < .@need_amount3) {
			mes "[Jardim do Tempo]";
			mes "Voce nao possui "+getitemname(.@need_item3)+" suficiente.";
			close;
		}
	}
	if (.@need_zeny && Zeny < .@need_zeny) {
		mes "[Jardim do Tempo]";
		mes "Voce nao possui Zeny suficiente.";
		close;
	}
	if (!checkweight(.@give_item,.@give_amount)) {
		mes "[Jardim do Tempo]";
		mes "Voce esta carregando peso ou tipos de itens demais.";
		close;
	}

	delitem .@need_item,.@need_amount;
	if (.@need_item2)
		delitem .@need_item2,.@need_amount2;
	if (.@need_item3)
		delitem .@need_item3,.@need_amount3;
	if (.@need_zeny)
		set Zeny,Zeny-.@need_zeny;
	getitem .@give_item,.@give_amount;

	mes "[Jardim do Tempo]";
	mes "Troca realizada.";
	close;
}
