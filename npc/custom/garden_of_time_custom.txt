//===== Custom Script ========================================
//= Garden of Time custom exchange hub
//===== Description: =========================================
//= Single NPC for Garden of Time barter, cloak reform and
//= Constellation meteorite material exchange.
//============================================================

moc_para01,48,22,4	script	Jardim do Tempo#custom	624,{
	mes "[Jardim do Tempo]";
	mes "Posso negociar equipamentos do Jardim do Tempo e materiais da Torre da Constelacao.";
	next;

L_Menu:
	switch (select("Comprar acessorios:Comprar capas:Reformar capas:Trocar meteoritos:Encantar Nebula:Encantar Signet Star:Encantar Vivatus:Cancelar")) {
	case 1:
		callshop("barter_t_garden_4");
		end;
	case 2:
		callshop("barter_t_garden_5");
		end;
	case 3:
		callshop("barter_t_garden_6");
		end;
	case 4:
		callsub S_Meteorite;
		end;
	case 5:
		callsub S_EnchantNebula;
		end;
	case 6:
		callsub S_EnchantSignetStar;
		end;
	case 7:
		callsub S_EnchantVivatus;
		end;
	default:
		close;
	}

S_EnchantVivatus:
	mes "[Jardim do Tempo]";
	mes "Este servico aplica encantamentos nos slots das armas Vivatus Fides.";
	mes "Requisitos e custos:";
	mes "- Arma refinada em ^0055ff+7 ou superior^000000.";
	mes "- Fragmentos de Ametista e Zeny.";
	mes "- Encantamento e reset com 100% de sucesso.";
	mes "- O segundo slot possui encantamento perfeito.";
	mes " ";
	mes "^777777As opcoes aleatorias das pedras Pontifex sao um sistema separado.^000000";
	next;
	if (select("Abrir encantamento de slots:Voltar") == 2)
		goto L_Menu;
	close2;
	item_enchant(164);
	end;

S_EnchantNebula:
	mes "[Jardim do Tempo]";
	mes "Escolha a armadura Nebula que deseja encantar.";
	next;

	.@choice = select("Power:Stamina:Concentration:Creative:Spell:Wisdom:Voltar");
	if (.@choice == 7)
		goto L_Menu;

	setarray .@enchant_ids[0],7,8,9,10,11,12;
	setarray .@item_ids[0],450169,450170,450171,450172,450173,450174;
	setarray .@dust_ids[0],1000398,1000399,1000400,1000401,1000402,1000403;
	setarray .@fragment_ids[0],1000442,1000443,1000444,1000445,1000446,1000447;
	.@idx = .@choice - 1;

	mes "[Jardim do Tempo]";
	mes "Este encantamento aceita:";
	mes "- "+mesitemlink(.@item_ids[.@idx], false)+" ^777777+9 ou superior^000000";
	mes " ";
	mes "Custos principais:";
	mes "- Slot 3: 50.000z + "+mesitemlink(1000372, false)+" x5";
	mes "- Slot 2: 100.000z + "+mesitemlink(.@dust_ids[.@idx], false)+" x5";
	mes "- Slot 1: 1.000.000z + "+mesitemlink(.@fragment_ids[.@idx], false)+" x5";
	mes " ";
	mes "^ff0000O equipamento precisa estar no inventario, nao equipado.^000000";
	next;

	getinventorylist;
	.@found = 0;
	.@found_low_refine = 0;
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_id[.@i] != .@item_ids[.@idx])
			continue;
		if (@inventorylist_refine[.@i] < 9) {
			.@found_low_refine = 1;
			continue;
		}
		.@found = 1;
		break;
	}

	if (!.@found) {
		mes "[Jardim do Tempo]";
		if (.@found_low_refine)
			mes "Encontrei "+mesitemlink(.@item_ids[.@idx], false)+", mas precisa estar +9 ou superior.";
		else
			mes "Voce nao possui "+mesitemlink(.@item_ids[.@idx], false)+" no inventario.";
		close;
	}

	item_enchant(.@enchant_ids[.@idx]);
	end;

S_EnchantSignetStar:
	mes "[Jardim do Tempo]";
	mes "Este encantamento aceita os acessorios Signet of Star:";
	mes "- "+mesitemlink(490132, false);
	mes "- "+mesitemlink(490133, false);
	mes "- "+mesitemlink(490134, false);
	mes "- "+mesitemlink(490135, false);
	mes "- "+mesitemlink(490136, false);
	mes "- "+mesitemlink(490137, false);
	mes " ";
	mes "Custos principais:";
	mes "- Slot 3: 300.000z + "+mesitemlink(1000373, false)+" x5";
	mes "- Slot 2: 500.000z + "+mesitemlink(1000373, false)+" x10";
	mes " ";
	mes "^ff0000O equipamento precisa estar no inventario, nao equipado.^000000";
	next;

	getinventorylist;
	.@found = 0;
	setarray .@signet_ids[0],490132,490133,490134,490135,490136,490137;
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (inarray(.@signet_ids[0], @inventorylist_id[.@i]) == -1)
			continue;
		.@found = 1;
		break;
	}

	if (!.@found) {
		mes "[Jardim do Tempo]";
		mes "Voce nao possui nenhum acessorio Signet of Star aceito no inventario.";
		close;
	}

	item_enchant(13);
	end;

S_Meteorite:
	mes "[Jardim do Tempo]";
	mes "Escolha a troca desejada.";
	next;

	switch (select("Po de Feitico:Fragmento Sortudo:Verdade da Circulacao:Voltar")) {
	case 1:
		.@need_item = 1000373;
		.@need_amount = 1;
		.@give_item = 1000402;
		.@give_amount = 5;
		break;
	case 2:
		.@need_item = 1000373;
		.@need_amount = 5;
		.@need_item2 = 1000402;
		.@need_amount2 = 10;
		.@give_item = 1000445;
		.@give_amount = 1;
		break;
	case 3:
		.@need_item = 1000373;
		.@need_amount = 50;
		.@need_item2 = 1000402;
		.@need_amount2 = 20;
		.@need_item3 = 1000445;
		.@need_amount3 = 10;
		.@need_zeny = 5000000;
		.@give_item = 102514;
		.@give_amount = 1;
		break;
	default:
		goto L_Menu;
	}

	mes "[Jardim do Tempo]";
	mes "Voce recebera:";
	mes "^0000ff"+getitemname(.@give_item)+" x"+.@give_amount+"^000000";
	mes " ";
	mes "Custo:";
	mes getitemname(.@need_item)+" x"+.@need_amount;
	if (.@need_item2)
		mes getitemname(.@need_item2)+" x"+.@need_amount2;
	if (.@need_item3)
		mes getitemname(.@need_item3)+" x"+.@need_amount3;
	if (.@need_zeny)
		mes "Zeny: "+.@need_zeny;
	next;

	if (select("Confirmar:Cancelar") == 2)
		close;

	if (countitem(.@need_item) < .@need_amount) {
		mes "[Jardim do Tempo]";
		mes "Voce nao possui "+getitemname(.@need_item)+" suficiente.";
		close;
	}
	if (.@need_item2 && countitem(.@need_item2) < .@need_amount2) {
		mes "[Jardim do Tempo]";
		mes "Voce nao possui "+getitemname(.@need_item2)+" suficiente.";
		close;
	}
	if (.@need_item3 && countitem(.@need_item3) < .@need_amount3) {
		mes "[Jardim do Tempo]";
		mes "Voce nao possui "+getitemname(.@need_item3)+" suficiente.";
		close;
	}
	if (.@need_zeny && Zeny < .@need_zeny) {
		mes "[Jardim do Tempo]";
		mes "Voce nao possui Zeny suficiente.";
		close;
	}
	if (!checkweight(.@give_item,.@give_amount)) {
		mes "[Jardim do Tempo]";
		mes "Voce esta carregando peso ou tipos de itens demais.";
		close;
	}

	delitem .@need_item,.@need_amount;
	if (.@need_item2)
		delitem .@need_item2,.@need_amount2;
	if (.@need_item3)
		delitem .@need_item3,.@need_amount3;
	if (.@need_zeny)
		set Zeny,Zeny-.@need_zeny;
	getitem .@give_item,.@give_amount;

	mes "[Jardim do Tempo]";
	mes "Troca realizada.";
	close;
}
