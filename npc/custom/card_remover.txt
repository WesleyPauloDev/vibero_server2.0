//===== rAthena Script =======================================
//= Card Removal NPC
//===== Description: =========================================
//= Removes cards from equipped items with guaranteed success.
//============================================================

-	script	Removedora de Cartas	78,{

	set .percardcost,20000;

	disable_items;
	mes "[Velha Sábia]";
	mes "Ola, jovem! Posso remover cartas dos seus equipamentos com total segurança.";
	next;
	switch(select("Sim, quero remover cartas.:Quanto custa?:Não, obrigado.")) {
	case 1:
		mes "[Velha Sábia]";
		mes "Escolha o equipamento que deseja tratar:";
		next;

		setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
		for (set .@i,1; .@i <= 10; set .@i,.@i+1) {
			if (getequipisequiped(.@indices[.@i]))
				set .@menu$, .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			set .@menu$, .@menu$ + ":";
		}
		set .@part, .@indices[select(.@menu$)];
		if (!getequipisequiped(.@part)) {
			mes "[Velha Sábia]";
			mes "Jovem... Voce não esta usando nada que eu possa analisar.";
			close;
		}

		deletearray .@card_slot[0], getarraysize(.@card_slot);
		deletearray .@card_id[0], getarraysize(.@card_id);
		.@card_menu$ = "";
		.@card_count = 0;
		for (.@i = 0; .@i < 4; .@i++) {
			.@card = getequipcardid(.@part,.@i);
			if (.@card <= 0)
				continue;
			if (callfunc("F_IsCharm", .@card))
				continue;
			.@card_menu$ += "Slot " + (.@i + 1) + " - " + getitemname(.@card) + ":";
			.@card_slot[.@card_count] = .@i;
			.@card_id[.@card_count] = .@card;
			.@card_count++;
		}

		if (.@card_count == 0) {
			mes "[Velha Sábia]";
			mes "Este equipamento não possui cartas removiveis.";
			mes "Se houver apenas encantamentos, eu não removo.";
			close;
		}

		if (!checkweight(.@card_id[0], .@card_count)) {
			mes "[Velha Sábia]";
			mes "Voce esta carregando peso demais. Guarde alguns itens e volte.";
			close;
		}

		.@card_menu$ += "Cancelar";
		mes "[Velha Sábia]";
		mes "Escolha qual carta deseja remover:";
		next;
		.@sel = select(.@card_menu$);
		if (.@sel > .@card_count)
			close;

		.@slot = .@card_slot[.@sel - 1];
		.@card = .@card_id[.@sel - 1];
		if (callfunc("F_IsCharm", .@card)) {
			mes "[Velha Sábia]";
			mes "Isso é um encantamento, não uma carta. Eu não removo encantamentos aqui.";
			close;
		}

		if (Zeny < .percardcost) {
			mes "[Velha Sábia]";
			mes "Voce precisa de " + callfunc("F_InsertComma", .percardcost) + " zeny para remover esta carta.";
			close;
		}

		mes "[Velha Sábia]";
		mes "Vou remover ^0055FF" + getitemname(.@card) + "^000000 do slot " + (.@slot + 1) + ".";
		mes "Custo: ^0055FF" + callfunc("F_InsertComma", .percardcost) + " zeny^000000.";
		next;
		if (select("Confirmar:Cancelar") == 2) {
			mes "[Velha Sábia]";
			mes "Muito bem. Volte se precisar dos meus serviços.";
			close;
		}

		Zeny -= .percardcost;
		progressbar "ffff00",2;
		setitemcard(.@part,.@slot,0);
		getitem .@card,1;
		specialeffect2 EF_REPAIRWEAPON;

		mes "[Velha Sábia]";
		mes "Pronto. Removi a carta com sucesso e aqui esta ela de volta.";
		close;

	case 2:
		mes "[Velha Sábia]";
		mes "Cobro " + callfunc("F_InsertComma", .percardcost) + " zeny por carta removida.";
		mes "Eu removo apenas cartas normais. Encantamentos ficam onde estão.";
		close;

	case 3:
		mes "[Velha Sábia]";
		mes "Muito bem. Volte se precisar dos meus serviços.";
		close;
	}
}
