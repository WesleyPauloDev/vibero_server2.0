L_AutomaticInsert:
	mes "[Operador OS]";
	mes "Escolha primeiro o equipamento Automatic equipado que recebera o modulo.";
	next;

	.@menu$ = "";
	.@equip_count = 0;

	if (getequipid(EQI_GARMENT) == 480020 || getequipid(EQI_GARMENT) == 480021) {
		.@part_list[.@equip_count] = EQI_GARMENT;
		.@menu$ += getequipname(EQI_GARMENT) + " [Engine Wing]:";
		.@equip_count++;
	}
	if (getequipid(EQI_ARMOR) == 450127 || getequipid(EQI_ARMOR) == 450128) {
		.@part_list[.@equip_count] = EQI_ARMOR;
		.@menu$ += getequipname(EQI_ARMOR) + " [Armor]:";
		.@equip_count++;
	}
	if (getequipid(EQI_SHOES) == 470022 || getequipid(EQI_SHOES) == 470023) {
		.@part_list[.@equip_count] = EQI_SHOES;
		.@menu$ += getequipname(EQI_SHOES) + " [Leg]:";
		.@equip_count++;
	}
	if (getequipid(EQI_ACC_R) == 490024 || getequipid(EQI_ACC_R) == 490026) {
		.@part_list[.@equip_count] = EQI_ACC_R;
		.@menu$ += getequipname(EQI_ACC_R) + " [Acessorio Direito]:";
		.@equip_count++;
	}
	if (getequipid(EQI_ACC_L) == 490025 || getequipid(EQI_ACC_L) == 490027) {
		.@part_list[.@equip_count] = EQI_ACC_L;
		.@menu$ += getequipname(EQI_ACC_L) + " [Acessorio Esquerdo]:";
		.@equip_count++;
	}

	if (!.@equip_count) {
		mes "[Operador OS]";
		mes "Voce nao possui nenhum equipamento Automatic elegivel equipado.";
		mes "Equipe Armor, Engine Wing, Leg ou Acessorios Automatic e tente novamente.";
		close;
	}

	.@equip_choice = select(.@menu$ + "Cancelar") - 1;
	if (.@equip_choice >= .@equip_count)
		close;

	.@part = .@part_list[.@equip_choice];
	.@equip_id = getequipid(.@part);
	.@equip_name$ = getequipname(.@part);
	.@refine = getequiprefinerycnt(.@part);
	for (.@i = 0; .@i < 4; ++.@i)
		.@card[.@i] = getequipcardid(.@part,.@i);

	if (.@card[1] > 0) {
		mes "[Operador OS]";
		mes "Esse equipamento Automatic ja esta com a capacidade maxima de modulos.";
		close;
	}

	switch(.@part) {
	case EQI_GARMENT:
		setarray .@module_etc_data[0],1000105,3,1000106,3,1000125,2,1000126,2,1000127,2,1000135,1,1000142,1,1000208,1;
		break;
	case EQI_ARMOR:
		setarray .@module_etc_data[0],1000105,3,1000106,3,1000122,2,1000123,2,1000124,2,1000128,1,1000129,1,1000130,1,1000131,1,1000132,1,1000133,1,1000207,1,1000140,1;
		.@size = getarraysize(.@module_etc_data);
		for (.@i = 1000152; .@i < 1000203; .@i++) {
			setarray .@module_etc_data[.@size], .@i, 2;
			.@size += 2;
		}
		break;
	case EQI_SHOES:
		setarray .@module_etc_data[0],1000105,3,1000106,3,1000119,2,1000120,2,1000121,2,1000133,1,1000134,1,1000141,1,1000144,1,1000145,1,1000146,1,1000147,1,1000148,1,1000149,1;
		break;
	case EQI_ACC_L:
	case EQI_ACC_R:
		setarray .@module_etc_data[0],1000107,3,1000108,3,1000109,3,1000110,3,1000111,3,1000112,3,1000113,2,1000114,2,1000115,1,1000116,1,1000117,1,1000118,1,1000136,1,1000137,1,1000138,1,1000139,1,1000143,1;
		break;
	default:
		close;
	}

	mes "[Operador OS]";
	mes "Escolha o modulo para inserir em:";
	mes "^0055FF" + .@equip_name$ + "^000000";
	next;

	.@size = getarraysize(.@module_etc_data);
	.@module_count = .@size / 2;
	.@page_size = 15;
	.@page = 0;

	while (true) {
		.@start = .@page * .@page_size;
		.@end = .@start + .@page_size;
		if (.@end > .@module_count)
			.@end = .@module_count;

		.@menu$ = "Cancelar:";
		for (.@i = .@start; .@i < .@end; .@i++) {
			.@module_id = .@module_etc_data[.@i * 2];
			if (countitem(.@module_id) < 1)
				.@menu$ += "^808080" + getitemname(.@module_id) + " (0)^000000:";
			else
				.@menu$ += getitemname(.@module_id) + " (" + countitem(.@module_id) + "):";
		}
		if (.@page > 0)
			.@menu$ += "Pagina anterior:";
		if (.@end < .@module_count)
			.@menu$ += "Proxima pagina:";

		.@pick = select(.@menu$) - 1;
		if (.@pick == 0)
			close;

		.@listed = .@end - .@start;
		if (.@pick <= .@listed) {
			.@data_index = (.@start + .@pick - 1) * 2;
			break;
		}

		.@nav = .@pick - .@listed;
		if (.@page > 0 && .@nav == 1) {
			.@page--;
			continue;
		}
		.@page++;
	}

	.@module_id = .@module_etc_data[.@data_index];
	.@max_module = .@module_etc_data[.@data_index + 1];

	if (countitem(.@module_id) < 1) {
		mes "[Operador OS]";
		mes "Voce precisa ter o modulo escolhido no inventario.";
		close;
	}
	if (.@module_id < 1000105 || .@module_id == 1000150 || .@module_id == 1000151 || (.@module_id >= 1000203 && .@module_id != 1000207 && .@module_id != 1000208)) {
		mes "[Operador OS]";
		mes "O modulo selecionado nao e valido para esse sistema.";
		close;
	}

	for (.@i = 1000105; .@i < 1000203; ++.@i) {
		if (.@i == 1000150 || .@i == 1000151)
			continue;
		.@auto_module[.@i] = 310082 + .@size_auto_module;
		.@size_auto_module++;
	}
	.@auto_module[1000207] = 310178;
	.@auto_module[1000208] = 310179;
	.@enchant_id = .@auto_module[.@module_id];

	if (.@enchant_id == 0) {
		mes "[Operador OS]";
		mes "Nao foi possivel converter o modulo selecionado em encantamento.";
		close;
	}
	if (countinarray(.@card[0], .@enchant_id) >= .@max_module) {
		mes "[Operador OS]";
		mes "Esse equipamento ja atingiu o limite desse mesmo modulo.";
		close;
	}

	mes "[Operador OS]";
	mes "Vou inserir " + getitemname(.@module_id) + " em " + .@equip_name$ + ".";
	mes "Esse processo consome o modulo e nao pode ser desfeito por este menu.";
	next;
	if (select("Confirmar insercao:Cancelar") == 2)
		close;

	delitem .@module_id,1;
	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]))
		end;

	for (.@i = 3; .@i > 0; .@i--) {
		if (.@card[.@i] == 0) {
			.@card[.@i] = .@enchant_id;
			break;
		}
	}
	delequip .@part;
	getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
	mes "[Operador OS]";
	mes "Modulo inserido com sucesso.";
	close;
L_AutomaticInfo:
	mes "[Operador OS]";
	mes "O sistema Automatic usa equipamentos da linha Illusion, Barmil Ticket e modulos.";
	next;
	switch(select("Como adquirir equipamentos:Como adquirir modulos:Como inserir modulo:Requisitos e limites:Voltar")) {

		mes "Equipe o item Automatic na posicao correta.";
		mes "Tenha o modulo no inventario.";
		mes "Entre no menu Inserir Modulo e escolha o slot do equipamento.";
		close;
	case 4:
		mes "[Operador OS]";
		mes "Armor, Engine Wing, Leg e Accessories aceitam listas diferentes de modulos.";
		mes "O sistema tambem limita quantas vezes o mesmo modulo pode repetir no mesmo equipamento.";

		mes "[Operador OS]";

		mes "Shoes aceitam ate Legendary.";
		mes "Accessories e Engine seguem as listas proprias do sistema oficial.";
		close;
	default:
		close;
	}
