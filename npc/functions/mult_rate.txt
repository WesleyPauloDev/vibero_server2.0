-	script	RateEscolha#login	-1,{

OnPCLoginEvent:
    // #drop_final = 0;
    // #exp_final = 0;

    // #rate_escolhido = 0;
	// Se já escolheu a taxa, apenas aplica os bônus
	if (#rate_escolhido > 0) end;

	mes "^00AAFFBem-vindo, " + strcharinfo(0) + "!^000000";
	mes "Antes de começar sua jornada, escolha sua taxa de experiência:";
	mes " ";
	mes "Quanto menor a taxa de XP, maior será o bônus de drop!";
	next;

	while (true) {
		mes "Selecione sua taxa de experiência:";
		mes "";
		mes "^00FF00[1x]^000000  XP normal, Taxa de Drop (2x)";
		mes "^88FF88[2x]^000000  XP (2X), Taxa de Drop (1.5x)";
		mes "^FFAA00[5x]^000000  XP (5X), Taxa de Drop (1.2x)";
		mes "^FF0000[10x]^000000 XP (10X), Sem bônus de Drop";
		next;

		switch(select("^00FF001x^000000", "^88FF882x^000000", "^FFAA005x^000000", "^FF000010x^000000")) {
			case 1:
				#rate_escolhido = 1;
				#bonus_drop = 100; // máximo bônus de drop
				.@drop_display$ = "2";
				break;
			case 2:
				#rate_escolhido = 2;
				#bonus_drop = 50;
				.@drop_display$ = "1.5";
				break;
			case 3:
				#rate_escolhido = 5;
				#bonus_drop = 20;
				.@drop_display$ = "1.2";
				break;
			case 4:
				#rate_escolhido = 10;
				#bonus_drop = 0; // sem bônus
				.@drop_display$ = "1";
				break;
		}
        mes "Você escolheu a taxa de ^FFAA00" + #rate_escolhido + "X^000000 de XP.";
        mes "Bônus de Drop: ^00AAFF" + .@drop_display$ + "X^000000.";
        mes " ";
        mes "Sua escolha é permanente!";
        mes "Tem certeza que deseja continuar com esta Rate?";
        if (select("Sim","Não") == 1) break;
	}	
    mes "Rate escolhida foi ^FFAA00" + #rate_escolhido + "x^000000 de XP.";
	close2;

    // #rate_bonus = 0;
    if (!#rate_bonus) {
        if (!#exp_final) #exp_final = 1;
        
        if (#rate_escolhido == 1) {
            #exp_final = 1;
        } else {
            set #exp_final, (#exp_final * #rate_escolhido * 100) - 100;
        }
            
        if (!#drop_final) #drop_final = 0;
        set #drop_final, #drop_final + #bonus_drop;
        #rate_bonus = 1;
    }

	set_reborn_exp(#exp_final);
	set_reborn_drop(#drop_final);

	set #exp_base, #exp_final;
	set #drop_base, #drop_final;

	end;
}
