function	script	F_Cursed_Fragment	{	// 23016
	.@r = rand(1,12);
	if (.@r == 1) {
		showscript "Eu nunca vou te perdoar...!";
		sc_start SC_ATKPOTION,600000,50;
		sc_start SC_MATKPOTION,600000,50;
	}
	else if (.@r == 2) {
		showscript "Alguem, por favor, me ajude...";
		sc_start SC_CURSE,10000,0;
	}
	else if (.@r == 3) {
		showscript "Pare... Isso doi...";
		sc_start SC_BLEEDING,5000,0;
	}
	else if (.@r == 4) {
		showscript "Esta tao frio aqui...";
		sc_start SC_FREEZE,5000,0;
	}
	else if (.@r == 5) {
		showscript "Por favor... Fique comigo...";
		itemskill "AL_TELEPORT",1;
	}
	else if (.@r == 6) {
		showscript "Fique... aqui comigo...";
		itemskill "AL_TELEPORT",1;
	}
	else if (.@r == 7) {
		showscript "Voce nao devia ter vindo aqui...";
		sc_start SC_STUN,5000,0;
	}
	else if (.@r == 8) {
		showscript "Ola... Tem alguem ai...?";
		sc_start SC_BLIND,5000,0;
	}
	else if (.@r == 9) {
		showscript "Estou com tanto medo... tanto medo...";
		sc_start SC_CONFUSION,10000,0;
	}
	else if (.@r == 10) {
		showscript "Querido Odin...";
		sc_start SC_ASSUMPTIO,100000,5;
		specialeffect2 EF_ASSUMPTIO;
	}
	else if (.@r == 11) {
		showscript "Acorde!";
		percentheal 0,20;
	}
	else {
		showscript "Me desculpe... Eu vou embora agora... obrigado...";
		getitem 6820,1;// Energy_Fragment

}
