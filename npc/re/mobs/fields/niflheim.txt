	end;

	stopnpctimer;
	.@boss_gid = callsub(S_FindBoss);

		initnpctimer;
		end;

	.@count = getmapunits(BL_MOB,"niflheim",.@units[0]);
	for (.@i = 0; .@i < .@count; ++.@i) {
		getunitdata .@units[.@i],.@data;
		if (.@data[UMOB_CLASS] == 1373 && .@data[UMOB_HP] > 0)
			return .@units[.@i];
	}
	return 0;
