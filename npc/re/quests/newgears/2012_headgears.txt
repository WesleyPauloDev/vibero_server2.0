OnBuyItem:
	setarray .@shop_items[0],
		18760,18761,18762,18763,18764,18769,18770,18771,
		15068,2183,22015,20710,18776,18772,18774,18775,2956,
		15068,2183,22015,20710,18776,18767,18765,18768,18766,22014,2957,2958;
	setarray .@shop_prices[0],
		100,100,100,100,100,100,100,100,
		100,100,100,100,100,200,200,200,200,
		20,20,20,20,20,100,300,300,300,300,300,300;
	setarray .@shop_crystals[0],
		6623,6623,6623,6623,6623,6623,6623,6623,
		6624,6624,6624,6624,6624,6624,6624,6624,6624,
		6625,6625,6625,6625,6625,6625,6625,6625,6625,6625,6625,6625;

	.@total_zeny = 0;
	.@rough_cost = 0;
	.@purified_cost = 0;
	.@high_cost = 0;

	for (.@i = 0; .@i < getarraysize(@bought_nameid); .@i++) {
		.@found = 0;
		for (.@j = 0; .@j < getarraysize(.@shop_items); .@j++) {
			if (@bought_nameid[.@i] == .@shop_items[.@j] && @bought_price[.@i] == .@shop_prices[.@j]) {
				.@found = 1;
				.@unit_price = .@shop_prices[.@j];
				.@crystal_id = .@shop_crystals[.@j];
				break;
			}
		}
		if (!.@found) {
			dispbottom "[Reno] Erro: item nao esta registrado na loja.";
			end;
		}
		if (!checkweight(@bought_nameid[.@i], @bought_quantity[.@i])) {
			dispbottom "[Reno] Voce nao tem espaco suficiente no inventario.";
			end;
		}

		.@line_cost = .@unit_price * @bought_quantity[.@i];
		.@total_zeny += .@line_cost;
		if (.@crystal_id == 6623) .@rough_cost += .@line_cost;
		else if (.@crystal_id == 6624) .@purified_cost += .@line_cost;
		else if (.@crystal_id == 6625) .@high_cost += .@line_cost;
	}

	if (Zeny < .@total_zeny) {
		dispbottom "[Reno] Zeny insuficiente.";
		end;
	}
	if (countitem(6623) < .@rough_cost) {
		dispbottom "[Reno] Cristais de Energia Brutos insuficientes.";
		end;
	}
	if (countitem(6624) < .@purified_cost) {
		dispbottom "[Reno] Cristais de Energia Purificados insuficientes.";
		end;
	}
	if (countitem(6625) < .@high_cost) {
		dispbottom "[Reno] Cristais de Energia Superiores insuficientes.";
		end;
	}

	Zeny -= .@total_zeny;
	if (.@rough_cost) delitem 6623, .@rough_cost;
	if (.@purified_cost) delitem 6624, .@purified_cost;
	if (.@high_cost) delitem 6625, .@high_cost;

	for (.@i = 0; .@i < getarraysize(@bought_nameid); .@i++)
		getitem @bought_nameid[.@i], @bought_quantity[.@i];

	dispbottom "[Reno] Compra concluida. Foram consumidos " + .@total_zeny + " zeny e os cristais correspondentes.";
	end;

//callsub L_CheckHunting,<quest index>,<start quest>,<number of quests>,"<type>",<reward id>,<reward amount>;
L_CheckHunting:
	for (set .@quest,getarg(0)+1; .@quest<getarg(1)+getarg(2); set .@quest,.@quest+1) {
		if (checkquest(.@quest,HUNTING) == 2) {
			mes "[Reno]";
			mes "Missao de caca da colecao "+getarg(3)+" concluida.";
			next;
			setquest getarg(0);
			erasequest .@quest;
			getitem getarg(4),getarg(5);
			mes "Voce recebe "+getarg(5)+" "+getitemname(getarg(4))+" como recompensa pela colecao "+getarg(3)+".";
			close;
		}
	}
	return;
