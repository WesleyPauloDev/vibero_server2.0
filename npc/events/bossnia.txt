//============================================================

prontera,160,182,4	script	Cheffenia Staff	908,{
	mes "[Cheffenia]";
	mes "Olá aventureiro!";
	mes "Deseja participar da Cheffenia?";
	mes "Isso lhe custará ^FF0000 50.000 zeny ^000000.";
	if (select("Sim, quero participar:Não, obrigado") == 2) close;

	// Verifica zeny
	if (Zeny < 50000) {
		mes "Você precisa de pelo menos ^FF0000 50.000 zeny ^000000 para entrar.";
		close;
	}

	// Escolha do mapa
	mes "Escolha o mapa da Cheffenia:";
	next;
	set .@mapa, select("Mapa 1:Mapa 2");

	// Escolha do ponto de spawn
	mes "Escolha o ponto onde deseja aparecer:";
	next;
	set .@spot, select("Canto Superior Esquerdo:Canto Superior Direito:Canto Inferior Esquerdo:Canto Inferior Direito");

	// Define coordenadas de spawn
	switch (.@spot) {
		case 1: set .@x,31;  set .@y,208; break;
		case 2: set .@x,208; set .@y,208; break;
		case 3: set .@x,31;  set .@y,31;  break;
		case 4: set .@x,208; set .@y,31;  break;
	}	

	// Desconta zeny e teleporta
	set Zeny, Zeny - 50000;
	specialeffect2 EF_MAXPOWER;
	if (.@mapa == 1)
		warp "bossnia_01", .@x, .@y;
	else
		warp "bossnia_02", .@x, .@y;

	end;
}

-	script	MapCustomKills	-1,{
	OnNPCKillEvent:
		// Só processa mobs
		if (strcharinfo(3) == "bossnia_01") {				

			callfunc("CustomMapDrop", "bossnia_01", 674, 10);
			// Adiciona pontos
			// set #mvp_points, #mvp_points + 1;

			// // Mensagem
			// dispbottom "Você matou um MVP e ganhou 1 ponto MVP!";
			// dispbottom "Total de pontos MVP: " + #mvp_points;
		}
		if (strcharinfo(3) == "bossnia_02") {				

			callfunc("CustomMapDrop", "bossnia_02", 674, 10);
			
		}
	end;
}
