// callfunc "F_Mora_Mist",<warp>,<success message>,<fail message>{,<end flag>,<timer>,<mob limit>,<extra warp>...}
// Se <mob limit> for maior que 0, a arvore pode ser cortada quando restarem ate essa quantidade de guardioes.
// A instancia simplificada chama o Dragao Roxo Errante apos 4 arvores cortadas.
function	script	F_Mora_Mist	{
	set .@map$, instance_mapname("1@mist");
	set .@mob_left, mobcount(.@map$,instance_npcname(strnpcinfo(0))+"::OnMyMobDead");
	set .@end_flag, 0;
	set .@timer_flag, 0;
	set .@mob_limit, 0;
	if (getargcount() > 3)
		set .@end_flag, getarg(3);
	if (getargcount() > 4)
		set .@timer_flag, getarg(4);
	if (getargcount() > 5)
		set .@mob_limit, getarg(5);
	if (.@mob_left <= .@mob_limit) {
		mes "Obviamente ninguem esta cuidando dela.";
		mes "Parece que voce pode cortar a arvore do jardim.";
		next;
		if(select("Cortar a arvore do jardim.:Desistir.") == 2)
			close;
		mes "Voce corta a arvore que bloqueava o caminho do labirinto, "+((.@end_flag)?"abrindo o caminho para fora da floresta.":"entao agora voce pode continuar.");
		mapannounce .@map$,getarg(1),bc_map,"0xccffcc"; //FW_NORMAL 12 0 0
		if (getarg(0) != "#to_bif02")
			enablenpc instance_npcname(getarg(0));
		.@summoned_dragon = callfunc("F_HazyForest_TreeProgress");
		for (.@i = 6; .@i < getargcount(); .@i++)
			enablenpc instance_npcname(getarg(.@i));
		if (.@timer_flag)
			initnpctimer;
		if (.@summoned_dragon) {
			next;
			mes "A floresta treme e um caminho se abre para o centro.";
			mes "Seu grupo e levado para perto do Dragao Roxo Errante.";
			close2;
			warpparty .@map$,183,291,getcharid(1),.@map$;
			disablenpc();
			end;
		}
		disablenpc();
		close;
	} else {
		mes "Ainda ha "+(.@mob_left - .@mob_limit)+" monstro(s) protegendo esta arvore.";
		mes "Derrote mais alguns para conseguir corta-la.";
		mapannounce .@map$,((.@end_flag)?getarg(2):getarg(2)+" grita: Ha? Quem esta fazendo coisas ruins com minha arvore?!"),bc_map,"0xccffcc"; //FW_NORMAL 12 0 0
	}
	end;
}

function	script	F_HazyForest_TreeProgress	{
	set .@map$, instance_mapname("1@mist");
	'hazy_tree_count++;
	set .@tree_count, 'hazy_tree_count;

	if ('hazy_dragon_summoned)
		return 0;

	set .@left, 4 - .@tree_count;
	if (.@left > 0) {
		mapannounce .@map$,"Objetivo da Floresta Nebulosa: "+.@tree_count+"/4 arvores cortadas. Faltam "+.@left+" para atrair o Dragao Roxo Errante.",bc_map,"0x99ff66";
		return 0;
	}

	'hazy_dragon_summoned = 1;
	mapannounce .@map$,"Objetivo da Floresta Nebulosa: as arvores suficientes foram cortadas. O grupo foi levado ao centro. Preparem-se e enfrentem o Dragao Roxo Errante ao norte!",bc_map,"0xffff00";
	donpcevent instance_npcname("#Dragon Herder")+"::OnSummon";
	return 1;
}

function	script	F_HazyForest_FinishPlayer	{
	if (!'hazy_finished)
		return 0;

	mes "[Floresta Nebulosa]";
	if (#HazyForest_STATUS == 3) {
		mes "Voce ja concluiu esta instancia.";
		next;
		if (select("Sair da instancia.:Ficar aqui.") == 1) {
			close2;
			warp "SavePoint",0,0;
			end;
		}
		close;
	}

	#HazyForest_STATUS = 2;
	mes "O Dragao Roxo Errante foi derrotado.";
	mes "A Floresta Nebulosa foi concluida.";
	next;
	.@finish_result = callfunc("F_InstanceFinishBase", "Floresta Nebulosa", "#HazyForest_STATUS", 10, 2, "#inst175_name$", "#inst175_status", 1, 10, 110002, 3000, 3, 0, "Mistwood Maze");
	if (.@finish_result == -1) {
		mes "[Floresta Nebulosa]";
		mes "Voce esta com peso demais para receber as recompensas.";
		close;
	}
	mes "[Floresta Nebulosa]";
	mes "Recompensas recebidas.";
	mes "Voce sera enviado para fora da instancia.";
	close2;
	warp "SavePoint",0,0;
	end;
}
