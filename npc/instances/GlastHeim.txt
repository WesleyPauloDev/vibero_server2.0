glast_01,204,128,6	script	OGH_Instance#Custom	709,{

OnStartInstance:

	set .@player_char_id, getattachedrid();

	if (!isloggedin(.@player_char_id)) {
		debugmes "Player com Char ID "+.@player_char_id+" não está online.";
		end;
	}

	attachrid(.@player_char_id);
	// Verificação de Nível
	if (BaseLevel < 50) {
		mes "Sinto muito, mas a Torre Sem Fim é um lugar perigoso. Você precisa ter no mínimo o nível 50 para poder entrar.";
		close;
	}

	// Verificação de Grupo
	set .@party_id, getcharid(1);
	if (.@party_id == 0) {
        mes "Você precisa estar em um grupo para ir para a Torre.";
        close;
	}
	mes "[Glast Heim Guide]";
	mes "Welcome to the Old Glast Heim Instance " + strcharinfo(0) + ".";
	
	// +1 ponto de reputação na categoria “Instâncias”
	add_reputation_points(1, 10);


	end;
}
