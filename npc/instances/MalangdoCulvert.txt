			callsub L_EraseQuest,12261,12262,12263,12264,12265,12266,12272;
		else {
			// Reward: 1x A Grade Coin
			callsub L_CheckQuest,12261,"Deep Sea Marc",6418,1;

	for(set .@j,0; .@j<getargcount(); set .@j,.@j+1) {

	mes "Nao posso entregar pagamento.";

		mes "[Madeca]";
		mes "Voce eliminou ^0000ff"+getarg(1)+"^000000!";
		mes "Aqui esta seu pagamento.";
		erasequest getarg(0);
		specialeffect2 EF_STEAL;
		getitem getarg(2),getarg(3);
		close;
	}
	return;

1@pump,1,1,4	script	Monster Hole#n	-1,{
	end;
OnSpawn:
	set .@i$, charat(strnpcinfo(2),0);
	if (.@i$ == "n") {
		set .@n,3;
		set .@map$, instance_mapname("1@pump");
		if ('culvert_n_done || 'culvert_n_active || 'culvert_n_waiting)
			end;
		set 'culvert_n_active,1;
		set .@cleaned,'culvert_n_cleaned;
	} else if (.@i$ == "h") {
		set .@n,5;
		set .@map$, instance_mapname("2@pump");
		if ('culvert_h_done || 'culvert_h_active || 'culvert_h_waiting)
			end;
		set 'culvert_h_active,1;
		set .@cleaned,'culvert_h_cleaned;
	}
	mapannounce .@map$,"Bueiro aberto. Faltam "+(.@n-.@cleaned)+" para limpar.",bc_map,"0x00ffcc",FW_NORMAL,15;
	donpcevent instance_npcname("#Culvert_"+.@i$+rand(1,.@n))+"::OnEnable";
	end;
OnCleaned:
	set .@i$, charat(strnpcinfo(2),0);
	if (.@i$ == "n") {
		if (!'culvert_n_active)
			end;
		set .@n,3;
		set .@map$, instance_mapname("1@pump");
		set 'culvert_n_active,0;
		set 'culvert_n_cleaned,'culvert_n_cleaned+1;
		set .@cleaned,'culvert_n_cleaned;
	} else if (.@i$ == "h") {
		if (!'culvert_h_active)
			end;
		set .@n,5;
		set .@map$, instance_mapname("2@pump");
		set 'culvert_h_active,0;
		set 'culvert_h_cleaned,'culvert_h_cleaned+1;
		set .@cleaned,'culvert_h_cleaned;
	}
