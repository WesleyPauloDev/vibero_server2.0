moro_cav,50,64,5	script	Semente Amarela#bios	CLEAR_NPC,{
	if (BaseLevel < 160) {
		mes "^ff0000Personagens abaixo do Nivel 160 nao podem entrar nesta instancia.^000000";
		close;
	}
	if (isbegin_quest(15006) == 0) {
		mes "^ff0000Voce precisa falar primeiro com o Velho Andarilho para liberar esta instancia.^000000";
		close;
	}
	.@party_id = getcharid(1);
	if (.@party_id < 1) {
		mes "^ff0000Voce precisa criar ou entrar em um grupo para entrar nesta instancia.^000000";
		close;
	}
	switch( checkquest(15005,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		if (checkquest(15007,PLAYTIME) == 2) {// 5 mins to re-enter
			mes "^ff0000Esta instancia so pode ser acessada 23 horas apos sua ultima visita.^000000";
			close;
		}
		switch( checkquest(15008,PLAYTIME) ) {
		case -1:
			break;
		case 0:
		case 1:
			mes "^0000ffEsta instancia so pode ser acessada 23 horas apos sua ultima visita.^000000";
			close;
		case 2:
			break;
		}
		break;
	case 2:
		erasequest 15005;// Memorial Dungeon: Isle of Bios
		erasequest 15007;// Memorial Dungeon: Isle of Bios
		if (isbegin_quest(15008) > 0)
			erasequest 15008;// Memorial Dungeon: Isle of Bios
		break;
	}
	if (is_party_leader() == true)
		.@create$ = "Criar instancia.";
	switch( select( .@create$, "Entrar na Ilha de Bios.", "Cancelar." ) ) {
	case 1:
		instance_create("Isle of Bios");
		mes "^ff0000Instancia criada.^000000";
		mes "^ff0000Clique novamente para entrar.^000000";
		close;
	case 2:
		switch( instance_enter("Isle of Bios") ) {
		case IE_NOMEMBER:
		case IE_NOINSTANCE:
		case IE_OTHER:
			if (is_party_leader(.@party_id) == false) {
				mes "^ff0000Lembre-se: esta instancia so pode ser acessada 23 horas apos sua ultima visita.^000000";
				next;
			}
			mes "^ff0000Seu lider de grupo ainda nao criou a instancia.^000000";
			close;
		case IE_OK:
			mapannounce "moro_cav", "O grupo " + getpartyname( getcharid(1) ) + " de " + strcharinfo(0) + " esta entrando na Ilha de Bios.",bc_map,"0x00FF99";
			if (isbegin_quest(15005) == 0) {
				setquest 15005;// Memorial Dungeon: Isle of Bios
				setquest 15007;// Memorial Dungeon: Isle of Bios
			}
			end;
		}
	case 3:
		end;
	}
	end;

OnInit:
	while(true) {
		sleep 500;
		specialeffect EF_ENHANCE;
	}
	end;
}
