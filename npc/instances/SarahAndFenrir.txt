function	script	F_SarahFenrirSoloMob	{
	.@gid = getarg(0, 0);
	if (!.@gid)
		return;

	getunitdata .@gid, .@mob;
	.@mode = .@mob[UMOB_MODE];
	.@mode = .@mode & ~(MD_IGNOREMELEE|MD_IGNOREMAGIC|MD_IGNORERANGED|MD_IGNOREMISC);
	setunitdata .@gid, UMOB_MODE, .@mode;
	setunitdata .@gid, UMOB_DMGIMMUNE, 0;
	return;
}
