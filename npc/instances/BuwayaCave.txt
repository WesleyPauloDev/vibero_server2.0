1@ma_c,28,57,0	script	Guarda da Saída#cave_out	570,{

	next;
	if (select("Sim:Não, vou ficar mais um pouco.") == 1) {
		if (#BuwayaCave_STATUS == 2) {
			.@finish_result = callfunc("F_InstanceFinishBase", "Caverna Buwaya", "#BuwayaCave_STATUS", 10, 2, "#inst175_name$", "#inst175_status", 1, 10, 110002, 3000, 3, 0, "Buwaya Cave");

				mes "Voce esta com peso demais para receber as recompensas.";
				close;
			}
		}
		warp "SavePoint",0,0;
		end;
	}
	close;

OnInstanceInit:
	'reward_announced = 0;
	hideonnpc instance_npcname("Guarda da Saída#cave_out");
	disablenpc instance_npcname("Guarda da Saída#cave_out");
	end;
OnEnable:
	hideoffnpc instance_npcname("Guarda da Saída#cave_out");
	enablenpc instance_npcname("Guarda da Saída#cave_out");
	end;
