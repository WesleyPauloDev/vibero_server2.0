1@cash,190,208,0	script	oct_exit_1	45,2,2,{
	end;
OnInstanceInit:
	disablenpc();
	end;
OnTouch:
	if (!'oct_finished) {
		mes "[Portal de Saida]";
		mes "O caminho de saida ainda nao esta seguro.";
		close;
	}
	mes "[Portal de Saida]";
	mes "Deseja receber sua recompensa e voltar ao ponto de salvamento?";
	next;
	if (select("Receber recompensa e sair.:Ficar um pouco mais.") == 2)
		close;

	.@aid = getcharid(3);
	if ('oct_reward_claimed[.@aid]) {
		mes "[Portal de Saida]";
		mes "Voce ja recebeu a recompensa desta execucao.";
		mes "Voltando ao ponto de salvamento.";
		close2;
		warp "SavePoint",0,0;
		end;
	}

	.@finish_result = callfunc("F_InstanceFinishBase", "Caverna do Polvo", "#OctopusCave_STATUS", 10, 2, "", "", 1, 10, 110002, 3000, 3, 0, "Octopus Cave", 1);
	if (.@finish_result == -1) {
		mes "[Portal de Saida]";
		mes "Voce esta carregando peso demais para receber as recompensas.";
		mes "Libere espaco no inventario e fale comigo novamente.";
		close;
	}
	if (.@finish_result == 1) {
		'oct_reward_claimed[.@aid] = 1;
		'oct_finish_announced = 1;
		mes "[Portal de Saida]";
		mes "Recompensa entregue.";
		mes "Voltando ao ponto de salvamento.";
		close2;
		warp "SavePoint",0,0;
		end;
	}

	mes "[Portal de Saida]";
	mes "Nao foi possivel entregar a recompensa agora.";
	mes "Voltando ao ponto de salvamento.";
	close2;
	warp "SavePoint",0,0;
	end;
}
