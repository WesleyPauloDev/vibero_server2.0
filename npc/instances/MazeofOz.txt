		close2;
		warp "SavePoint",0,0;
		end;
	}

	.@announce = (!'oz_finish_announced);
	.@reward = callfunc("F_InstanceFinishBase", "Labirinto de Oz", "#MazeofOz_STATUS", 20, 2, "", "", 1, 20, 110002, 3000, 3, 0, "Labirinto de Oz", .@announce);
	if (.@reward == -1) {
		mes "[Corda de Saida]";
		mes "Voce esta carregando peso demais para receber as recompensas.";
		mes "Libere espaco no inventario e fale comigo novamente.";
		close;
	}
	if (.@reward == 1) {
		'oz_reward_claimed[.@aid] = 1;
		if (.@announce)
			'oz_finish_announced = 1;
		mes "[Corda de Saida]";
		mes "Recompensa entregue.";
		mes "Voltando ao ponto de salvamento.";
		close2;
		warp "SavePoint",0,0;
		end;
	}

	mes "[Corda de Saida]";
	mes "Nao foi possivel entregar a recompensa agora.";
	mes "Voltando ao ponto de salvamento.";
	close2;
	warp "SavePoint",0,0;
	end;

OnInstanceInit:
	end;
}
