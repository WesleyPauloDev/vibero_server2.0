-	script	Nyd_Instance#Custom	709,{



OnStartInstance:

	set .@player_char_id, getattachedrid();



	attachrid(.@player_char_id);



	sleep2 100; // espera 0.1s pra liberar a stack anterior



	mes "[Guardiao do Portao de Yggdrasil]";



	// Verificacao de Nivel

	if (BaseLevel < 65) {

		mes "Sinto muito, mas O Ninho de Nydhog e um lugar perigoso. Voce precisa ter no minimo o nivel 65 para poder entrar.";

		close;

	}



	// Verificacao de Grupo

	set .@party_id, getcharid(1);

	if (.@party_id == 0) {

        mes "Voce precisa estar em um grupo para ir para O Ninho de Nydhog.";

        close;

	}

	

	set .@pid, getcharid(1);	

	if (.@pid <= 0) { mes "Voce precisa estar em um grupo."; close; }

	// Tempo

	set .@hora, gettime(3);

	set .@min, gettime(2);

	set .@seg, gettime(1);

	set .@segundos_do_dia, (.@hora * 3600) + (.@min * 60) + .@seg;

	set .@faltam_reset, 86400 - .@segundos_do_dia;



	if (#NYD_COOLDOWN_4H <= gettimetick(2) && #NYD_COOLDOWN_4H > 0) {

		// Tempo de Instancia acabou

		set #NYD_STATUS, 3;

	}

	

	// set #NYD_STATUS, 0;

	// Cooldown

	set .@restante, #NYD_COOLDOWN - gettimetick(2);

	if (#NYD_STATUS == 3 && .@restante <= 0) {

		// cooldown acabou

		set #NYD_STATUS, 0;

	}	



	set .@live_iid, instance_id(IM_PARTY);

	set .@same_live_instance, (.@live_iid > 0 && instance_live_info(ILI_NAME, .@live_iid) == "Nidhoggur's Nest");



	set .@partyName$, getpartyname(.@pid);



	// Mostrar status atual

	if (#NYD_STATUS == 0) {

		mes "Bem vindo a Instancia Ninho de Nydhog.";

		if (checkquest(3135,PLAYTIME) >= 0) {

			erasequest 3135;



			// limpa progresso do grupo

			.@pid = getcharid(1);

			if (.@pid > 0) {

				setd("$party_nyd"+.@pid, 0);

				setd("$party_current_nyd"+.@pid, "");

			}

		}

	} else if (#NYD_STATUS == 1) {

		mes "Voce ja abriu uma instancia.";

	} else if (#NYD_STATUS == 2) {

		mes "Sua instancia Ninho de Nydhog esta ativa.";

	} else if (#NYD_STATUS == 3) {

		mes "Voce nao pode mais entrar nesta instancia hoje.";

		mes "Volte amanha apos o reset diario.";

		close;


	} else if (#NYD_STATUS == 4) {		

		mes "Parabens, voce ja realizou uma grande conquista hoje no Ninho de Nydhog!";

    	mes "Aguarde o descanso necessario para continuar a sua jornada amanha.";

		set #NYD_STATUS, 3;

		close;

	}



	if (#NYD_STATUS > 0 && !.@same_live_instance) {

		mes "Voce nao pode mais entrar nesta instancia hoje.";

		mes "Volte amanha apos o reset diario.";

		close;

	}



	next;



	// =========================

	// BLOQUEIO POR PARTY + IP

	// =========================

	if (!callfunc("F_CheckInstanceIPPermission")) {

		mes "[Ninho de Nydhog]";

		mes "Jogadores com mesmo IP so podem participar de mais de uma ";

		mes "Instancia simultaneamente , se estiverem em grupos diferentes.";

		close;

	}



	if (#INSTANCE_MENU_RESUME > 0 && #INSTANCE_MENU_RESUME == instance_id(IM_PARTY)) {
		#INSTANCE_MENU_RESUME = 0;
		goto S_Enter;
	}

	// Menu principal

	if (#NYD_STATUS > 0)

		.@menu = select("Entrar na Instancia:Cancelar") + 1;

	else

		.@menu = select("Gerar Instancia:Entrar na Instancia:Cancelar");



	switch(.@menu) {

		case 1: // Gerar

			if (is_party_leader() != true) {

				mes "[Guardiao do Portao de Yggdrasil]";

				mes "Apenas o lider do grupo pode gerar a Instancia.";			

				close;

			}



			if (#NYD_STATUS > 0) {

				mes "[Guardiao do Portao de Yggdrasil]";

				mes "Voce ja possui uma instancia em andamento.";

				close;

			}

			.@instance_check = instance_create("Nidhoggur's Nest");

			if (.@instance_check < 0) {

				mes "[Guardiao do Portao de Yggdrasil]";

				mes "Voce  tem uma Instancia aberta.";

				close;

			}



			// Marca como criada

			set #NYD_STATUS, 1;

			set #NYD_PARTY, getcharid(1); // salva o id do grupo

			

			announce "[Instancias] " + strcharinfo(0) + " criou Ninho de Nidhog para o grupo " + .@partyName$ + ".", bc_all, 0xFF9900;

			callfunc("F_InstanceLogStart", "NYD", "Ninho de Nidhog");



			mes "[Guardiao do Portao de Yggdrasil]";

			mes "Instancia criada! Use ^0000FFEntrar na Instancia^000000 para comecar.";

			next;

			if (select("Entrar na Instancia", "Fechar") == 1)
				goto S_Enter;

			close;



		case 2: // Entrar

		S_Enter:

			.@party_id = getcharid(1);			



			switch(instance_enter("Nidhoggur's Nest")) {

				case IE_OK:

					set #NYD_STATUS, 2;

					callfunc("F_InstanceLogEnter", "NYD", "Ninho de Nidhog");

					

					set etower_timer, gettimetick(2);

					set .@pid, getcharid(1);

					set .@iid, instance_id(IM_PARTY);

					setd("$nyd_pid_by_iid_"+.@iid, .@pid);


					

					if (checkquest(3135,PLAYTIME) < 0) {

						

						setquest 3135;



						set #NYD_COOLDOWN_4H, gettimetick(2) + 14400000;

						set #NYD_COOLDOWN, gettimetick(2) + .@faltam_reset;

						set ins_nyd2, 0;

	

						addtimer 14400000, strnpcinfo(3)+"::OnTimeout_NYD"; // 4h = 14400000 ms					

					}

					

					close2;

					end;



				case IE_NOINSTANCE:

					if (#NYD_STATUS == 0) {

						mes "[Guardiao do Portao de Yggdrasil]";

						mes "Voce nao possui nenhuma instancia criada.";

						close;

					}

					mes "[Guardiao do Portao de Yggdrasil]";

					mes "A instancia nao esta disponivel no momento.";

					mes "Sua tentativa de hoje ja foi registrada, entao sera preciso aguardar o proximo reset.";

					close;

				default:

					mes "Nao foi possivel entrar na instancia.";

					close;

			}



		case 3: // Cancelar

			close;

	}



	// Handler do Timeout da Endless Tower

	OnTimeout_NYD:

    .@iid = instance_id(IM_PARTY);

	if (.@iid > 0) {

		// pega o party_id salvo para essa instancia

		.@pid = getd("$nyd_pid_by_iid_"+.@iid);



		if (.@pid > 0) {

			// limpa o progresso do grupo

			setd("$party_nyd"+.@pid, 0);

			setd("$party_current_nyd"+.@pid, "");

			setd("$nyd_pid_by_iid_"+.@iid, 0);

		}



		erasequest 3135;

		set #NYD_STATUS, 3;



		mapannounce strcharinfo(3),

			"O tempo da instancia terminou! O Ninho de Nydhog foi destruida.",

			bc_map, "0xff0000";

		sleep2 1000; // espera 5 segundos pra evitar bugs

		instance_destroy .@iid;

	}

	end;



}
