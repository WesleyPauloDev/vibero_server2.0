-	script	ET_Instance#Custom	709,{



OnStartInstance:

	set .@player_char_id, getattachedrid();



	if (!isloggedin(.@player_char_id)) {

		end;

	}



	attachrid(.@player_char_id);



	sleep2 100; // espera 0.1s pra liberar a stack anterior



	mes "[Captain Janssen]";



	// Verificacao de Nivel

	if (BaseLevel < 70) {

		mes "Sinto muito, mas a Torre Sem Fim e um lugar perigoso. Voce precisa ter no minimo o nivel 70 para poder entrar.";

		close;

	}

	

	set .@pid, getcharid(1);	

	if (.@pid <= 0) { mes "Voce precisa estar em um grupo."; close; }



	set .@floor, getd("$party_current_floor_"+.@pid);

	set .@current_map$, getd("$party_current_map_"+.@pid);

	if (.@floor <= 0) set .@floor, 1; // default



	// Tempo

	set .@hora, gettime(3);

	set .@min, gettime(2);

	set .@seg, gettime(1);

	set .@segundos_do_dia, (.@hora * 3600) + (.@min * 60) + .@seg;

	set .@faltam_reset, 86400 - .@segundos_do_dia;



	if (#ETOWER_COOLDOWN_4H <= gettimetick(2) && #ETOWER_COOLDOWN_4H > 0) {

		// Tempo de Instancia acabou

		set #ETOWER_STATUS, 3;

	}

	

	// Cooldown

	set .@restante, #ETOWER_COOLDOWN - gettimetick(2);

	if (#ETOWER_STATUS == 3 && .@restante <= 0) {

		// cooldown acabou

		set #ETOWER_STATUS, 0;

	}	



	set .@live_iid, instance_id(IM_PARTY);

	set .@same_live_instance, (.@live_iid > 0 && instance_live_info(ILI_NAME, .@live_iid) == "Endless Tower");



	set .@partyName$, getpartyname(.@pid);



	// Mostrar status atual

	if (#ETOWER_STATUS == 0) {

		mes "Bem vindo a Instancia Torre Sem Fim.";

		if (checkquest(60201,PLAYTIME) >= 0) {

			erasequest 60201;



			// limpa progresso do grupo

			.@pid = getcharid(1);

			if (.@pid > 0) {

				setd("$party_current_floor_"+.@pid, 0);

				setd("$party_current_map_"+.@pid, "");

			}

		}

	} else if (#ETOWER_STATUS == 1) {

		mes "Voce ja abriu uma instancia.";

	} else if (#ETOWER_STATUS == 2) {

		mes "Sua instancia da Torre esta ativa.";

	} else if (#ETOWER_STATUS == 3) {

		mes "Voce nao pode mais entrar nesta instancia hoje.";

		mes "Volte amanha apos o reset diario.";

		close;


	} else if (#ETOWER_STATUS == 4) {		

		mes "Parabens, voce ja realizou uma grande conquista hoje na Torre Sem Fim!";

    	mes "Aguarde o descanso necessario para continuar a sua jornada amanha.";

		set #ETOWER_STATUS, 3;

		close;

	}



	if (#ETOWER_STATUS > 0 && !.@same_live_instance) {

		mes "Voce nao pode mais entrar nesta instancia hoje.";

		mes "Volte amanha apos o reset diario.";

		close;

	}



	next;



	// =========================

	// BLOQUEIO POR PARTY + IP

	// =========================

	if (!callfunc("F_CheckInstanceIPPermission")) {

		mes "[Torre Sem Fim]";

		mes "Jogadores com mesmo IP so podem participar de mais de uma ";

		mes "Instancia simultaneamente , se estiverem em grupos diferentes.";

		close;

	}

	

	if (#INSTANCE_MENU_RESUME > 0 && #INSTANCE_MENU_RESUME == instance_id(IM_PARTY)) {
		#INSTANCE_MENU_RESUME = 0;
		goto S_Enter;
	}

	// .@instance_id = instance_id(IM_PARTY); // ID da instancia

	// if (.@instance_id > 0) {		

	// 	.@ip$         = getcharip();           // IP do jogador

	// 	.@day$        = gettimestr("%Y%m%d", 2); // Data (reboot-safe)

	// 	.@pid         = getcharid(1);           // Party ID CORRETO



	// 	.@permission_key$ = "#IR_" + .@day$ + "_" + .@instance_id + "_" + .@ip$ + "_" + .@pid;



	// 	.@exists = 0;

	// 	query_sql(

	// 		"SELECT 1 FROM acc_reg_num WHERE `key` = '"+ .@permission_key$ +"' LIMIT 1",

	// 		.@exists

	// 	);



	// 	if (.@exists) {		



	// 		.@saved_cid = 0;



	// 		query_sql(

	// 			"SELECT `value` FROM acc_reg_num "+

	// 			"WHERE `key` = '"+ .@permission_key$ +"' "+

	// 			"LIMIT 1",

	// 			.@saved_cid

	// 		);

			

	// 		// Se NAO for o mesmo char ? bloqueia

	// 		if (.@saved_cid != getcharid(0)) {

	// 			mes "[Captain Janssen]";

	// 			mes "Apenas um personagem por IP pode entrar nesta instancia neste grupo.";

	// 			close;

	// 		}

	// 	}



	// 	if (.@saved_cid <= 0) {

	// 		query_sql(

	// 			"INSERT INTO acc_reg_num (`account_id`, `key`, `value`) VALUES ("+

	// 			getcharid(3)+", '"+ .@permission_key$ +"', '"+ getcharid(0) +"')"

	// 		);

	// 	}



	// }



	// Menu principal

	if (#ETOWER_STATUS > 0)

		.@menu = select("Entrar na Instancia:Cancelar") + 1;

	else

		.@menu = select("Gerar Instancia:Entrar na Instancia:Cancelar");



	switch(.@menu) {

		case 1: // Gerar

			if (is_party_leader() != true) {

				mes "[Captain Janssen]";

				mes "Apenas o lider do grupo pode gerar a Instancia.";			

				close;

			}



			if (#ETOWER_STATUS > 0) {

				mes "[Captain Janssen]";

				mes "Voce ja possui uma instancia em andamento.";

				close;

			}



			if (instance_create("Endless Tower") < 0) {

				mes "[Captain Janssen]";

				mes "Voce  tem uma Instancia aberta.";

				close;

			}



			// Marca como criada

			set #ETOWER_STATUS, 1;

			set #ETOWER_PARTY, getcharid(1); // salva o id do grupo

			

			announce "[Instancias] " + strcharinfo(0) + " criou Torre Sem Fim para o grupo " + .@partyName$ + ".", bc_all, 0xFF9900;

			callfunc("F_InstanceLogStart", "ETOWER", "Torre Sem Fim");



			mes "[Captain Janssen]";

			mes "Instancia criada! Use '^0000FFEntrar na Instancia^000000' para comecar.";

			next;

			if (select("Entrar na Instancia", "Fechar") == 1)
				goto S_Enter;

			close;



		case 2: // Entrar

		S_Enter:

			.@party_id = getcharid(1);			



			switch(instance_enter("Endless Tower")) {

				case IE_OK:					

					set #ETOWER_STATUS, 2;

					callfunc("F_InstanceLogEnter", "ETOWER", "Torre Sem Fim");

					

					set etower_timer, gettimetick(2);

					set .@pid, getcharid(1);

					set .@iid, instance_id(IM_PARTY);



					setd("$etower_pid_by_iid_"+.@iid, .@pid);



					// inicializa o andar do grupo, se vazio

					if (getd("$party_current_floor_"+.@pid) <= 0)

						setd("$party_current_floor_"+.@pid, 1);



					set .@saved, getd("$party_current_floor_"+.@pid);



					if (checkquest(60201,PLAYTIME) < 0) {

						

						setquest 60201;



						set #ETOWER_COOLDOWN_4H, gettimetick(2) + 14400000;

						set #ETOWER_COOLDOWN, gettimetick(2) + .@faltam_reset;

	

						addtimer 14400000, strnpcinfo(3)+"::OnTimeout_ET"; // 4h = 14400000 ms					

					}	

					close2;				

					

					end;



				case IE_NOINSTANCE:

					if (#ETOWER_STATUS == 0) {

						mes "[Captain Janssen]";

						mes "Voce nao possui nenhuma instancia criada.";

						close;

					}

					mes "[Captain Janssen]";

					mes "A instancia nao esta disponivel no momento.";

					mes "Sua tentativa de hoje ja foi registrada, entao sera preciso aguardar o proximo reset.";

					close;

				default:

					mes "Nao foi possivel entrar na instancia.";

					close;

			}



		case 3: // Cancelar

			close;

	}



	// Handler do Timeout da Endless Tower

	OnTimeout_ET:

    .@iid = instance_id(IM_PARTY);

	if (.@iid > 0) {

		// pega o party_id salvo para essa instancia

		.@pid = getd("$etower_pid_by_iid_"+.@iid);



		if (.@pid > 0) {

			// limpa o progresso do grupo

			setd("$party_current_floor_"+.@pid, 0);

			setd("$party_current_map_"+.@pid, "");

			setd("$etower_pid_by_iid_"+.@iid, 0);

		}



		erasequest 60201;

		set #ETOWER_STATUS, 3;



		instance_destroy .@iid;

		mapannounce "e_tower",

			"O tempo da instancia terminou! A Torre Sem Fim foi destruida.",

			bc_all, "0xff0000";

	}

	end;



}
