function	script	CT_FinishRun	{
	.@stage$ = getarg(0, "final");
	.@coin_amount = 20;
	.@ticket_amount = 2;
	.@already_finished = (#ConstellationTower_STATUS == 3);

	if (.@stage$ == "andar 50") {
		.@coin_amount = 30;
		.@ticket_amount = 3;
	} else if (.@stage$ == "andar 75") {
		.@coin_amount = 40;
		.@ticket_amount = 4;
	}

	if (#CT_REWARD_INSTANCE != instance_id()) {
		if (.@already_finished)
			#ConstellationTower_STATUS = 2;
		.@finish_result = callfunc("F_InstanceFinishBase", "Constellation Tower", "#ConstellationTower_STATUS", .@coin_amount, .@ticket_amount, "", "", 1, 20, 110002, 3000, 3, 0, "Constellation Tower", !.@already_finished);
		if (.@finish_result == -1) {
			if (.@already_finished)
				#ConstellationTower_STATUS = 3;
			return -1;
		}
		if (!.@finish_result && .@already_finished)
			#ConstellationTower_STATUS = 3;
		if (.@finish_result)
			#CT_REWARD_INSTANCE = instance_id();
	}

	warp "SavePoint",0,0;
	return 1;
}

function	script	last_boss	{
	.@mvp_UID = getarg(0);
	if(!unitexists(.@mvp_UID))return;
	getunitdata .@mvp_UID,.@por_arr;
	
	setunitdata .@mvp_UID,UMOB_MAXHP,.@por_arr[UMOB_MAXHP]+('sum_pt_1*.@por_arr[UMOB_MAXHP]/100);
	setunitdata .@mvp_UID,UMOB_ATKMIN,.@por_arr[UMOB_ATKMIN]+('sum_pt_2)*.@por_arr[UMOB_ATKMIN]/100;
	setunitdata .@mvp_UID,UMOB_ATKMAX,.@por_arr[UMOB_ATKMAX]+('sum_pt_2)*.@por_arr[UMOB_ATKMAX]/100;
	setunitdata .@mvp_UID,UMOB_MATKMIN,.@por_arr[UMOB_MATKMIN]+('sum_pt_2)*.@por_arr[UMOB_MATKMIN]/100;
	setunitdata .@mvp_UID,UMOB_MATKMAX,.@por_arr[UMOB_MATKMAX]+('sum_pt_2)*.@por_arr[UMOB_MATKMAX]/100;
	
	setunitdata .@mvp_UID,UMOB_RES,.@por_arr[UMOB_RES]+('sum_pt_4)*.@por_arr[UMOB_RES]/100;
	setunitdata .@mvp_UID,UMOB_MRES,.@por_arr[UMOB_MRES]+('sum_pt_4)*.@por_arr[UMOB_MRES]/100;	
	return;
}

	setunitdata .@skill_UID,UMOB_DMGIMMUNE, 1;

	return;
