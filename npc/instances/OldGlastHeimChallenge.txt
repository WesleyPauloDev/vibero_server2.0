function	script	F_OGHC_SelectedLevel	{
	.@max = #OGHC_MAX_LEVEL;
	if (.@max < 1)
		.@max = 1;
	if (.@max > 10)
		.@max = 10;

	.@menu$ = "";
	for (.@i = 1; .@i <= .@max; ++.@i)
		.@menu$ += "Nivel de Contaminacao " + .@i + ":";
	.@menu$ = "Cancelar:" + .@menu$;

	.@s = select(.@menu$);
	if (.@s == 1)
		return 0;
	return .@s - 1;
}

function	script	F_OGHC_ApplyMobLevel	{
	.@gid = getarg(0);
	if (.@gid <= 0)
		return;

	.@lv = getarg(1, 1);
	if (.@lv < 1)
		.@lv = 1;
	if (.@lv > 10)
		.@lv = 10;

	getunitdata .@gid, .@mob;
	if (!getarraysize(.@mob))
		return;

	.@hp_mul = 100 + ((.@lv - 1) * 45);
	.@atk_mul = 100 + ((.@lv - 1) * 18);
	.@def_mul = 100 + ((.@lv - 1) * 10);

	.@maxhp = (.@mob[UMOB_MAXHP] * .@hp_mul) / 100;
	.@atkmin = (.@mob[UMOB_ATKMIN] * .@atk_mul) / 100;
	.@atkmax = (.@mob[UMOB_ATKMAX] * .@atk_mul) / 100;
	.@def = (.@mob[UMOB_DEF] * .@def_mul) / 100;
	.@mdef = (.@mob[UMOB_MDEF] * .@def_mul) / 100;

	if (.@maxhp < 1) .@maxhp = 1;
	if (.@atkmin < 1) .@atkmin = 1;
	if (.@atkmax < .@atkmin) .@atkmax = .@atkmin;

	setunitdata .@gid, UMOB_MAXHP, .@maxhp;
	setunitdata .@gid, UMOB_HP, .@maxhp;
	setunitdata .@gid, UMOB_ATKMIN, .@atkmin;
	setunitdata .@gid, UMOB_ATKMAX, .@atkmax;
	setunitdata .@gid, UMOB_DEF, .@def;
	setunitdata .@gid, UMOB_MDEF, .@mdef;
	return;
}

function	script	F_OGHC_SpawnValidArea	{
	.@map$ = getarg(0);
	.@x1 = getarg(1);
	.@y1 = getarg(2);
	.@x2 = getarg(3);
	.@y2 = getarg(4);
	.@name$ = getarg(5);
	.@mob_id = getarg(6);
	.@amount = getarg(7);
	.@label$ = getarg(8);
	.@level = getarg(9, 1);

	for (.@i = 0; .@i < .@amount; ++.@i) {
		.@spawned = 0;
		for (.@try = 0; .@try < 100; ++.@try) {
			.@x = rand(.@x1, .@x2);
			.@y = rand(.@y1, .@y2);
			if (checkcell(.@map$, .@x, .@y, CELL_CHKPASS)) {
				monster .@map$, .@x, .@y, .@name$, .@mob_id, 1, .@label$;
				callfunc("F_OGHC_ApplyMobLevel", $@mobid[0], .@level);
				.@spawned = 1;
				break;
			}
		}

		if (!.@spawned) {
			getfreecell .@map$, .@x, .@y, ((.@x1 + .@x2) / 2), ((.@y1 + .@y2) / 2), ((.@x2 - .@x1) / 2), ((.@y2 - .@y1) / 2), 5;
			if (.@x >= .@x1 && .@x <= .@x2 && .@y >= .@y1 && .@y <= .@y2 && checkcell(.@map$, .@x, .@y, CELL_CHKPASS)) {
				monster .@map$, .@x, .@y, .@name$, .@mob_id, 1, .@label$;
				callfunc("F_OGHC_ApplyMobLevel", $@mobid[0], .@level);
			} else {
				--.@i;
			}
		}
	}
	return;
}

function	script	F_OGHC_DropReward	{
	.@level = getarg(0, 1);
	if (.@level < 1) .@level = 1;
	if (.@level > 10) .@level = 10;

	.@circlet_amount = .@level;
	.@frag_amount = .@level * 2;
	.@map$ = instance_mapname("1@gl_he");
	.@x1 = 146;
	.@y1 = 228;
	.@x2 = 170;
	.@y2 = 250;

	callfunc("F_OGHC_MakeSpreadReward", 25864, .@circlet_amount, .@map$, .@x1, .@y1, .@x2, .@y2);
	callfunc("F_OGHC_MakeSpreadReward", 25867, .@frag_amount, .@map$, .@x1, .@y1, .@x2, .@y2);

	if (.@level >= 6) {
		.@spell_amount = .@frag_amount / 2;
		callfunc("F_OGHC_MakeSpreadReward", 25866, .@spell_amount, .@map$, .@x1, .@y1, .@x2, .@y2);

		.@jewel_chance = .@level * 10;
		if (.@jewel_chance > 100)
			.@jewel_chance = 100;
		if (rand(100) < .@jewel_chance) {
			.@jewel_amount = .@level - 5;
			if (.@jewel_amount < 1)
				.@jewel_amount = 1;
			if (.@jewel_amount > 5)
				.@jewel_amount = 5;
			callfunc("F_OGHC_MakeSpreadReward", 25865, .@jewel_amount, .@map$, .@x1, .@y1, .@x2, .@y2);
		}
	}
	return;
}

function	script	F_OGHC_MakeSpreadReward	{
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@map$ = getarg(2);
	.@x1 = getarg(3);
	.@y1 = getarg(4);
	.@x2 = getarg(5);
	.@y2 = getarg(6);

	for (.@i = 0; .@i < .@amount; ++.@i) {
		.@x = 158;
		.@y = 240;
		for (.@try = 0; .@try < 50; ++.@try) {
			.@tx = rand(.@x1, .@x2);
			.@ty = rand(.@y1, .@y2);
			if (checkcell(.@map$, .@tx, .@ty, CELL_CHKPASS)) {
				.@x = .@tx;
				.@y = .@ty;
				break;
			}
		}
		makeitem .@item_id, 1, .@map$, .@x, .@y;
	}
	return;
}

function	script	F_OGHC_CheckInstance	{
	if (instance_id() > 0)
		return 1;

	mes "[Oscar]";
	mes "Esta distorcao ainda nao foi estabilizada.";
	mes "Entre pela opcao correta da instancia.";
	close;
	return 0;
}

				mes "Somente o lider do grupo pode abrir o portal.";
				close;

			.@iid = instance_create(.@db_name$);

				mes "[" + .@npc_name$ + "]";

				callfunc("F_InstanceLogStart", .@status_var$, .@display_name$);

				setd(.@status_name$, 1);

				$OGHC_LEVEL_PENDING[getcharid(1)] = 0;
				mes "Falha ao criar a instancia.";
			}
			close;

		case 2:
			switch(instance_enter(.@db_name$)) {
			case IE_OK:
				setd(.@cooldown_name$, gettimetick(2) + (86400 * 3));
				setd(.@status_name$, 2);
				callfunc("F_InstanceLogEnter", .@status_var$, .@display_name$);

				end;

				close;
			case IE_NOMEMBER:
				mes "Voce nao faz parte de um grupo valido para esta instancia.";
				close;
			default:
				mes "Ocorreu um erro desconhecido ao tentar entrar.";
				close;
			}
			close;

		case 3:
			close;
		}
	end;

			.@finish = callfunc("F_InstanceFinishBase", "Glast Heim Modo Desafio", "#OldGlastHeimChallenge_STATUS", 40, 4, "", "", 1, 20, 110002, 3000, 3, 0, "Glast Heim Modo Desafio", ('oghc_finish_announced ? 0 : 1));
			if (.@finish == -1) {
				mes "[Oscar]";
				mes "Voce esta carregando peso demais para receber as recompensas.";
				close;
			}

			if (.@finish > 0) {
				#OGHC_FINISH_INSTANCE = instance_id();
				if (#OGHC_MAX_LEVEL < 'oghc_level + 1 && 'oghc_level < 10)
					#OGHC_MAX_LEVEL = 'oghc_level + 1;
				else if (#OGHC_MAX_LEVEL < 'oghc_level)
					#OGHC_MAX_LEVEL = 'oghc_level;
				'oghc_finish_announced = 1;
			}
		}

		close2;
		warp "SavePoint", 0, 0;
		end;
	}
	close;

OnInstanceInit:
	disablenpc instance_npcname(strnpcinfo(0));
	end;
}
