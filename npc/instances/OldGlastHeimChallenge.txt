function	script	F_OGHC_SelectedLevel	{
	.@max = #OGHC_MAX_LEVEL;
	if (.@max < 1)
		.@max = 1;
	if (.@max > 10)
		.@max = 10;

	.@menu$ = "";
	for (.@i = 1; .@i <= .@max; ++.@i)
		.@menu$ += "Nivel de Contaminacao " + .@i + ":";
	.@menu$ = "Cancelar:" + .@menu$;

	.@s = select(.@menu$);
	if (.@s == 1)
		return 0;
	return .@s - 1;
}

function	script	F_OGHC_ApplyMobLevel	{
	.@gid = getarg(0);
	if (.@gid <= 0)
		return;

	.@lv = getarg(1, 1);
	if (.@lv < 1)
		.@lv = 1;
	if (.@lv > 10)
		.@lv = 10;

	getunitdata .@gid, .@mob;
	if (!getarraysize(.@mob))
		return;

	.@hp_mul = 100 + ((.@lv - 1) * 45);
	.@atk_mul = 100 + ((.@lv - 1) * 18);
	.@def_mul = 100 + ((.@lv - 1) * 10);
	.@stat_mul = 100 + ((.@lv - 1) * 35);

	.@maxhp = (.@mob[UMOB_MAXHP] * .@hp_mul) / 100;
	.@atkmin = (.@mob[UMOB_ATKMIN] * .@atk_mul) / 100;
	.@atkmax = (.@mob[UMOB_ATKMAX] * .@atk_mul) / 100;
	.@def = (.@mob[UMOB_DEF] * .@def_mul) / 100;
	.@mdef = (.@mob[UMOB_MDEF] * .@def_mul) / 100;
	.@str = (.@mob[UMOB_STR] * .@stat_mul) / 100;
	.@agi = (.@mob[UMOB_AGI] * .@stat_mul) / 100;
	.@vit = (.@mob[UMOB_VIT] * .@stat_mul) / 100;
	.@int = (.@mob[UMOB_INT] * .@stat_mul) / 100;
	.@dex = (.@mob[UMOB_DEX] * .@stat_mul) / 100;
	.@luk = (.@mob[UMOB_LUK] * .@stat_mul) / 100;

	if (.@maxhp < 1) .@maxhp = 1;
	if (.@atkmin < 1) .@atkmin = 1;
	if (.@atkmax < .@atkmin) .@atkmax = .@atkmin;
	if (.@str < 1) .@str = 1;
	if (.@agi < 1) .@agi = 1;
	if (.@vit < 1) .@vit = 1;
	if (.@int < 1) .@int = 1;
	if (.@dex < 1) .@dex = 1;
	if (.@luk < 1) .@luk = 1;

	setunitdata .@gid, UMOB_MAXHP, .@maxhp;
	setunitdata .@gid, UMOB_HP, .@maxhp;
	setunitdata .@gid, UMOB_ATKMIN, .@atkmin;
	setunitdata .@gid, UMOB_ATKMAX, .@atkmax;
	setunitdata .@gid, UMOB_DEF, .@def;
	setunitdata .@gid, UMOB_MDEF, .@mdef;
	setunitdata .@gid, UMOB_STR, .@str;
	setunitdata .@gid, UMOB_AGI, .@agi;
	setunitdata .@gid, UMOB_VIT, .@vit;
	setunitdata .@gid, UMOB_INT, .@int;
	setunitdata .@gid, UMOB_DEX, .@dex;
	setunitdata .@gid, UMOB_LUK, .@luk;
	return;
}

function	script	F_OGHC_SpawnValidArea	{
	.@map$ = getarg(0);
	.@x1 = getarg(1);
	.@y1 = getarg(2);
	.@x2 = getarg(3);
	.@y2 = getarg(4);
	.@name$ = getarg(5);
	.@mob_id = getarg(6);
	.@amount = getarg(7);
	.@label$ = getarg(8);
	.@level = getarg(9, 1);

	for (.@i = 0; .@i < .@amount; ++.@i) {
		.@spawned = 0;
		for (.@try = 0; .@try < 100; ++.@try) {
			.@x = rand(.@x1, .@x2);
			.@y = rand(.@y1, .@y2);
			if (checkcell(.@map$, .@x, .@y, CELL_CHKPASS)) {
				monster .@map$, .@x, .@y, .@name$, .@mob_id, 1, .@label$;
				callfunc("F_OGHC_ApplyMobLevel", $@mobid[0], .@level);
				.@spawned = 1;
				break;
			}
		}

		if (!.@spawned) {
			getfreecell .@map$, .@x, .@y, ((.@x1 + .@x2) / 2), ((.@y1 + .@y2) / 2), ((.@x2 - .@x1) / 2), ((.@y2 - .@y1) / 2), 5;
			if (.@x >= .@x1 && .@x <= .@x2 && .@y >= .@y1 && .@y <= .@y2 && checkcell(.@map$, .@x, .@y, CELL_CHKPASS)) {
				monster .@map$, .@x, .@y, .@name$, .@mob_id, 1, .@label$;
				callfunc("F_OGHC_ApplyMobLevel", $@mobid[0], .@level);
			} else {
				--.@i;
			}
		}
	}
	return;
}

function	script	F_OGHC_DropReward	{
	.@level = getarg(0, 1);
	if (.@level < 1) .@level = 1;
	if (.@level > 10) .@level = 10;

	.@circlet_amount = .@level;
	.@frag_amount = .@level * 2;
	.@map$ = instance_mapname("1@gl_he");
	.@x1 = 146;
	.@y1 = 228;
	.@x2 = 170;
	.@y2 = 250;

	callfunc("F_OGHC_MakeSpreadReward", 25864, .@circlet_amount, .@map$, .@x1, .@y1, .@x2, .@y2);
	callfunc("F_OGHC_MakeSpreadReward", 25867, .@frag_amount, .@map$, .@x1, .@y1, .@x2, .@y2);

	if (.@level >= 6) {
		.@spell_amount = .@frag_amount / 2;
		callfunc("F_OGHC_MakeSpreadReward", 25866, .@spell_amount, .@map$, .@x1, .@y1, .@x2, .@y2);

		.@jewel_chance = .@level * 10;
		if (.@jewel_chance > 100)
			.@jewel_chance = 100;
		if (rand(100) < .@jewel_chance) {
			.@jewel_amount = .@level - 5;
			if (.@jewel_amount < 1)
				.@jewel_amount = 1;
			if (.@jewel_amount > 5)
				.@jewel_amount = 5;
			callfunc("F_OGHC_MakeSpreadReward", 25865, .@jewel_amount, .@map$, .@x1, .@y1, .@x2, .@y2);
		}
	}
	return;
}

function	script	F_OGHC_MakeSpreadReward	{
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@map$ = getarg(2);
	.@x1 = getarg(3);
	.@y1 = getarg(4);
	.@x2 = getarg(5);
	.@y2 = getarg(6);

	for (.@i = 0; .@i < .@amount; ++.@i) {
		.@x = 158;
		.@y = 240;
		for (.@try = 0; .@try < 50; ++.@try) {
			.@tx = rand(.@x1, .@x2);
			.@ty = rand(.@y1, .@y2);
			if (checkcell(.@map$, .@tx, .@ty, CELL_CHKPASS)) {
				.@x = .@tx;
				.@y = .@ty;
				break;
			}
		}
		makeitem .@item_id, 1, .@map$, .@x, .@y;
	}
	return;
}

function	script	F_OGHC_CheckInstance	{
	if (instance_id() > 0)
		return 1;

	mes "[Oscar]";
	mes "Esta distorcao ainda nao foi estabilizada.";
	mes "Entre pela opcao correta da instancia.";
	close;
	return 0;
}

		callshop "barter_ogh_challenge_weapons",1;

	case 6:

		mes "A janela de encantamento reconhece os Circlets e acessorios temporais no inventario ou equipados.";

L_SelectEnchantEquip:
	setarray .@accessory_ids[0],490211,490219,490238,490257,490258,490259,490260,490262,490263,490269,490281,490284,490561,490575,490580,490586,490590,490598,490600,490603,490611;
	setarray .@scan_part[0],EQI_HEAD_TOP,EQI_ACC_L,EQI_ACC_R;
	setarray .@slot_name$[0],"Topo","Acessorio esquerdo","Acessorio direito";
	.@total = 0;
	.@menu$ = "";

	for (.@i = 0; .@i < getarraysize(.@scan_part); ++.@i) {
		.@part = .@scan_part[.@i];
		.@id = getequipid(.@part);
		if (.@id <= 0)
			continue;

		.@valid = (.@part == EQI_HEAD_TOP)
			? (.@id >= 19474 && .@id <= 19492)
			: (inarray(.@accessory_ids, .@id) != -1);
		if (!.@valid)
			continue;

		.@available_part[.@total] = .@part;
		.@menu$ += getequipname(.@part) + " [" + .@slot_name$[.@i] + "]:";
		.@total++;
	}

	if (.@total == 0) {
		mes "[Oscar]";
		mes "Nao encontrei nenhum equipamento temporal valido.";
		mes "Equipe um Circlet Temporal ou um dos acessorios temporais da loja.";
		close;
	}

	if (.@total == 1)
		return .@available_part[0];

	mes "[Oscar]";
	mes "Encontrei mais de um equipamento temporal.";
	mes "Qual deles deseja usar?";
	next;
	.@choice = select(.@menu$ + "Cancelar") - 1;
	if (.@choice >= .@total)
		close;

	return .@available_part[.@choice];

L_Upgrade:
	.@part = callsub(L_SelectEnchantEquip);
	.@equip_id = getequipid(.@part);

	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@refine = getequiprefinerycnt(.@part);
	setarray .@base[0],29706,29061,29111,29101,29091,29071,29081;
	.@base_id = 0;
	for (.@i = 0; .@i < getarraysize(.@base); ++.@i) {
		if (.@card[1] >= .@base[.@i] && .@card[1] <= .@base[.@i] + 9) {
			.@base_id = .@base[.@i];
			break;
		}
	}
	if (!.@base_id) {
		mes "[Oscar]";
		mes "O 2o slot deste equipamento nao possui um encantamento especial aprimoravel.";
		close;
	}

	.@level = .@card[1] - .@base_id + 1;
	if (.@level >= 10) {
		mes "[Oscar]";
		mes "Este encantamento ja esta no nivel maximo.";
		close;
	}

	setarray .@cost[1],4,6,8,10,12,16,20,28,40;
	setarray .@chance[1],80,70,50,20,20,20,20,20,20;
	.@perfect_cost = .@cost[.@level] * 5;
	.@perfect_zeny = .@level * 5000000;
	mes "[Oscar]";
	mes "Encantamento atual: ^0055FF" + getitemname(.@card[1]) + "^000000.";
	mes "Tentativa para nivel " + (.@level + 1) + ".";
	mes "Normal: ^0055FF" + .@cost[.@level] + "x " + getitemname(25866) + "^000000, chance " + .@chance[.@level] + "%.";
	mes "Perfeito: ^0055FF" + .@perfect_cost + "x " + getitemname(25866) + " + " + .@perfect_zeny + " Zeny^000000, chance 100%.";
	mes "Na tentativa normal, falha reduz 1 nivel se o encantamento atual for nivel 2 ou maior.";
	next;
	.@mode = select("Aprimorar normal:Aprimorar perfeito:Cancelar");
	if (.@mode == 3)
		close;

	.@need_spell = (.@mode == 2) ? .@perfect_cost : .@cost[.@level];
	.@need_zeny = (.@mode == 2) ? .@perfect_zeny : 0;

	if (countitem(25866) < .@need_spell) {
		mes "[Oscar]";
		mes "Voce nao possui " + getitemname(25866) + " suficiente.";
		close;
	}
	if (Zeny < .@need_zeny) {
		mes "[Oscar]";
		mes "Voce nao possui Zeny suficiente.";
		close;
	}

	if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@part, .@refine))
		close;

	delitem 25866, .@need_spell;
	Zeny -= .@need_zeny;
	progressbar "ffff00",2;
	if (.@mode == 2 || rand(100) < .@chance[.@level]) {
		setitemcard(.@part, 1, .@card[1] + 1);
		specialeffect2 EF_ENHANCE;
		mes "[Oscar]";
		if (.@mode == 2)
			mes "Aprimoramento perfeito concluido.";
		else
			mes "Sucesso.";
		mes "O encantamento evoluiu para ^0055FF" + getitemname(.@card[1] + 1) + "^000000.";
	} else {
		if (.@level > 1) {
			setitemcard(.@part, 1, .@card[1] - 1);
			mes "[Oscar]";
			mes "A tentativa falhou. O encantamento voltou para ^FF5555" + getitemname(.@card[1] - 1) + "^000000.";
		} else {
			mes "[Oscar]";
			mes "A tentativa falhou. O encantamento permaneceu no nivel 1.";
		}
	}
	close;

}
